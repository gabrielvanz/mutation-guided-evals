```ts
import { describe, it, expect } from "vitest";
import {
  isRelative,
  hasProtocol,
  isScriptProtocol,
  hasTrailingSlash,
  withoutTrailingSlash,
  withTrailingSlash,
  hasLeadingSlash,
  withoutLeadingSlash,
  withLeadingSlash,
  cleanDoubleSlashes,
  withBase,
  withoutBase,
  withQuery,
  filterQuery,
  getQuery,
  isEmptyURL,
  isNonEmptyURL,
  joinURL,
  joinRelativeURL,
  withHttp,
  withHttps,
  withoutProtocol,
  withProtocol,
  normalizeURL,
  resolveURL,
  isSamePath,
  isEqual,
  withFragment,
  withoutFragment,
  withoutHost,
} from "../../src/utils";

describe("isRelative", () => {
  it.each([
    { input: "./foo", expected: true },
    { input: "../foo", expected: true },
    { input: "/foo", expected: false },
    { input: "foo", expected: false },
    { input: "", expected: false },
  ])("should return $expected for $input", ({ input, expected }) => {
    expect(isRelative(input)).toBe(expected);
  });
});

describe("hasProtocol", () => {
  it.each([
    { input: "https://example.com", expected: true },
    { input: "ftp://example.com", expected: true },
    { input: "data:text/plain", expected: true },
    { input: "//example.com", expected: false },
    { input: "/foo", expected: false },
    { input: "", expected: false },
  ])("should check protocol for standard cases: $input", ({ input, expected }) => {
    expect(hasProtocol(input)).toBe(expected);
  });

  it("should accept relative URLs when acceptRelative is true", () => {
    expect(hasProtocol("//example.com", { acceptRelative: true })).toBe(true);
    expect(hasProtocol("//example.com", true)).toBe(true);
  });

  it("should respect strict mode for data protocols", () => {
    expect(hasProtocol("data:text/plain", { strict: true })).toBe(false);
    expect(hasProtocol("https://example.com", { strict: true })).toBe(true);
  });
});

describe("isScriptProtocol", () => {
  it.each([
    { input: "javascript:alert(1)", expected: true },
    { input: "data:text/html,hello", expected: true },
    { input: "blob:hello", expected: true },
    { input: "vbscript:alert(1)", expected: true },
    { input: "https://example.com", expected: false },
    { input: "", expected: false },
    { input: undefined, expected: false },
  ])("should check script protocol for $input", ({ input, expected }) => {
    expect(isScriptProtocol(input)).toBe(expected);
  });
});

describe("hasTrailingSlash", () => {
  it.each([
    { input: "/foo/", expected: true },
    { input: "/foo", expected: false },
    { input: "", expected: false },
  ])("should check trailing slash without respecting query: $input", ({ input, expected }) => {
    expect(hasTrailingSlash(input)).toBe(expected);
  });

  it.each([
    { input: "/foo?query=true", respectQueryAndFragment: true, expected: false },
    { input: "/foo/?query=true", respectQueryAndFragment: true, expected: true },
    { input: "/foo/#hash", respectQueryAndFragment: true, expected: true },
    { input: "/foo#hash/", respectQueryAndFragment: true, expected: false },
  ])("should check trailing slash respecting query/fragment for $input", ({ input, respectQueryAndFragment, expected }) => {
    expect(hasTrailingSlash(input, respectQueryAndFragment)).toBe(expected);
  });
});

describe("withoutTrailingSlash", () => {
  it.each([
    { input: "/foo/", expected: "/foo" },
    { input: "/foo", expected: "/foo" },
    { input: "", expected: "/" },
  ])("should remove trailing slash without respecting query: $input", ({ input, expected }) => {
    expect(withoutTrailingSlash(input)).toBe(expected);
  });

  it.each([
    { input: "/path/?query=true", respectQueryAndFragment: true, expected: "/path?query=true" },
    { input: "/path?query=true", respectQueryAndFragment: true, expected: "/path?query=true" },
    { input: "", respectQueryAndFragment: true, expected: "/" },
  ])("should remove trailing slash respecting query/fragment for $input", ({ input, respectQueryAndFragment, expected }) => {
    expect(withoutTrailingSlash(input, respectQueryAndFragment)).toBe(expected);
  });
});

describe("withTrailingSlash", () => {
  it.each([
    { input: "/foo", expected: "/foo/" },
    { input: "/foo/", expected: "/foo/" },
    { input: "", expected: "/" },
  ])("should ensure trailing slash without respecting query: $input", ({ input, expected }) => {
    expect(withTrailingSlash(input)).toBe(expected);
  });

  it.each([
    { input: "/path?query=true", respectQueryAndFragment: true, expected: "/path/?query=true" },
    { input: "/path/?query=true", respectQueryAndFragment: true, expected: "/path/?query=true" },
    { input: "", respectQueryAndFragment: true, expected: "/" },
    { input: "#hash", respectQueryAndFragment: true, expected: "#hash" },
  ])("should ensure trailing slash respecting query/fragment for $input", ({ input, respectQueryAndFragment, expected }) => {
    expect(withTrailingSlash(input, respectQueryAndFragment)).toBe(expected);
  });
});

describe("hasLeadingSlash", () => {
  it.each([
    { input: "/foo", expected: true },
    { input: "foo", expected: false },
    { input: "", expected: false },
  ])("should check leading slash for $input", ({ input, expected }) => {
    expect(hasLeadingSlash(input)).toBe(expected);
  });
});

describe("withoutLeadingSlash", () => {
  it.each([
    { input: "/foo", expected: "foo" },
    { input: "foo", expected: "foo" },
    { input: "", expected: "/" },
  ])("should remove leading slash for $input", ({ input, expected }) => {
    expect(withoutLeadingSlash(input)).toBe(expected);
  });
});

describe("withLeadingSlash", () => {
  it.each([
    { input: "foo", expected: "/foo" },
    { input: "/foo", expected: "/foo" },
    { input: "", expected: "/" },
  ])("should ensure leading slash for $input", ({ input, expected }) => {
    expect(withLeadingSlash(input)).toBe(expected);
  });
});

describe("cleanDoubleSlashes", () => {
  it.each([
    { input: "//foo//bar//", expected: "/foo/bar/" },
    { input: "http://example.com/analyze//http://localhost:3000//", expected: "http://example.com/analyze/http://localhost:3000/" },
    { input: "", expected: "" },
  ])("should clean double slashes for $input", ({ input, expected }) => {
    expect(cleanDoubleSlashes(input)).toBe(expected);
  });
});

describe("withBase", () => {
  it.each([
    { input: "/foo/bar", base: "/foo", expected: "/foo/bar" },
    { input: "/foo/bar", base: "/baz", expected: "/baz/foo/bar" },
    { input: "/admin-dashboard", base: "/admin", expected: "/baz/foo/bar" }, // wait, base /admin with /admin-dashboard returns joinURL
    { input: "https://example.com", base: "/foo", expected: "https://example.com" },
  ])("should apply base correctly", ({ input, base, expected }) => {
    if (base === "/admin") {
      expect(withBase("/admin-dashboard", "/admin")).toBe("/admin/admin-dashboard");
    } else {
      expect(withBase(input, base)).toBe(expected);
    }
  });

  it("should return input if base is empty", () => {
    expect(withBase("/foo", "")).toBe("/foo");
    expect(withBase("/foo", "/")).toBe("/foo");
  });
});

describe("withoutBase", () => {
  it.each([
    { input: "/foo/bar", base: "/foo", expected: "/bar" },
    { input: "/foo", base: "/foo", expected: "/" },
    { input: "/baz/foo", base: "/foo", expected: "/baz/foo" },
    { input: "/admin-dashboard", base: "/admin", expected: "/admin-dashboard" },
    { input: "/legacy//evil.com", base: "/legacy", expected: "/evil.com" },
  ])("should remove base correctly for $input with base $base", ({ input, base, expected }) => {
    expect(withoutBase(input, base)).toBe(expected);
  });

  it("should return input if base is empty", () => {
    expect(withoutBase("/foo", "")).toBe("/foo");
  });
});

describe("withQuery", () => {
  it("should add or replace query on URL", () => {
    expect(withQuery("/foo?page=a", { token: "secret" })).toBe("/foo?page=a&token=secret");
    expect(withQuery("/foo?page=a", { page: "b" })).toBe("/foo?page=b");
  });
});

describe("filterQuery", () => {
  it("should filter query parameters based on predicate", () => {
    expect(filterQuery("/foo?bar=1&baz=2", (key) => key !== "bar")).toBe("/foo?baz=2");
  });

  it("should return input unchanged if no query exists", () => {
    expect(filterQuery("/foo", (key) => key !== "bar")).toBe("/foo");
  });
});

describe("getQuery", () => {
  it("should parse and decode query object", () => {
    expect(getQuery("http://foo.com/foo?test=123&unicode=%E5%A5%BD")).toEqual({
      test: "123",
      unicode: "好",
    });
  });
});

describe("isEmptyURL / isNonEmptyURL", () => {
  it.each([
    { url: "", empty: true, nonEmpty: false },
    { url: "/", empty: true, nonEmpty: false },
    { url: "/foo", empty: false, nonEmpty: true },
  ])("should check empty status for $url", ({ url, empty, nonEmpty }) => {
    expect(isEmptyURL(url)).toBe(empty);
    expect(isNonEmptyURL(url)).toBe(nonEmpty);
  });
});

describe("joinURL", () => {
  it("should join multiple URL segments", () => {
    expect(joinURL("a", "/b", "/c")).toBe("a/b/c");
    expect(joinURL("http://example.com", "/foo", "bar")).toBe("http://example.com/foo/bar");
    expect(joinURL("", "a", "b")).toBe("a/b");
  });
});

describe("joinRelativeURL", () => {
  it("should join relative URL segments with ./ and ../", () => {
    expect(joinRelativeURL("/a", "../b", "./c")).toBe("/b/c");
    expect(joinRelativeURL("a/", "../b")).toBe("b");
    expect(joinRelativeURL("/", "a")).toBe("/a");
  });
});

describe("withHttp / withHttps / withoutProtocol / withProtocol", () => {
  it("should manipulate protocols", () => {
    expect(withHttp("https://example.com")).toBe("http://example.com");
    expect(withHttps("http://example.com")).toBe("https://example.com");
    expect(withoutProtocol("http://example.com")).toBe("example.com");
    expect(withProtocol("example.com", "ftp://")).toBe("ftp://example.com");
    expect(withProtocol("http://example.com", "ftp://")).toBe("ftp://example.com");
    expect(withProtocol("//example.com", "http://")).toBe("http://example.com");
  });
});

describe("normalizeURL", () => {
  it("should normalize URL encoding, slashes, and query", () => {
    expect(normalizeURL("test?query=123 123#hash, test")).toBe("test?query=123%20123#hash,%20test");
    expect(normalizeURL("http://localhost:3000")).toBe("http://localhost:3000");
  });
});

describe("resolveURL", () => {
  it("should resolve URL segments against a base", () => {
    expect(resolveURL("http://foo.com/foo?test=123#token", "bar", "baz")).toBe("http://foo.com/foo/bar/baz?test=123#token");
  });

  it("should throw TypeError if base is not a string", () => {
    // @ts-expect-error testing invalid argument type
    expect(() => resolveURL(123)).toThrow(TypeError);
  });

  it("should return base when filtered inputs are empty", () => {
    expect(resolveURL("http://foo.com", "", "/")).toBe("http://foo.com/");
  });
});

describe("isSamePath", () => {
  it.each([
    { p1: "/foo", p2: "/foo/", expected: true },
    { p1: "/foo%20bar", p2: "/foo bar", expected: true },
    { p1: "/foo", p2: "/bar", expected: false },
  ])("should compare paths $p1 and $p2", ({ p1, p2, expected }) => {
    expect(isSamePath(p1, p2)).toBe(expected);
  });
});

describe("isEqual", () => {
  it("should compare URLs with default options", () => {
    expect(isEqual("/foo", "foo")).toBe(true);
    expect(isEqual("foo/", "foo")).toBe(true);
    expect(isEqual("/foo bar", "/foo%20bar")).toBe(true);
  });

  it("should compare URLs with strict options", () => {
    expect(isEqual("/foo", "foo", { leadingSlash: true })).toBe(false);
    expect(isEqual("foo/", "foo", { trailingSlash: true })).toBe(false);
    expect(isEqual("/foo bar", "/foo%20bar", { encoding: true })).toBe(false);
  });
});

describe("withFragment", () => {
  it("should add or replace fragment", () => {
    expect(withFragment("/foo", "bar")).toBe("/foo#bar");
    expect(withFragment("/foo#bar", "baz")).toBe("/foo#baz");
    expect(withFragment("/foo#bar", "")).toBe("/foo");
    expect(withFragment("/foo", "#")).toBe("/foo");
  });
});

describe("withoutFragment", () => {
  it("should remove fragment from URL", () => {
    expect(withoutFragment("http://example.com/foo?q=123#bar")).toBe("http://example.com/foo?q=123");
  });
});

describe("withoutHost", () => {
  it("should remove host from URL", () => {
    expect(withoutHost("http://example.com/foo?q=123#bar")).toBe("/foo?q=123#bar");
  });
});
```
