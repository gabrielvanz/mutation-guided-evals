```ts
import { describe, it, expect } from "vitest";
import {
  sep,
  normalize,
  join,
  resolve,
  normalizeString,
  isAbsolute,
  toNamespacedPath,
  extname,
  relative,
  dirname,
  format,
  basename,
  parse,
  matchesGlob
} from "../../src/_path";

describe("sep", () => {
  it("should be a forward slash", () => {
    expect(sep).toBe("/");
  });
});

describe("normalize", () => {
  it("should normalize empty path to current directory", () => {
    expect(normalize("")).toBe(".");
  });

  it("should normalize standard paths", () => {
    expect(normalize("foo/bar//baz/../asdf")).toBe("foo/bar/asdf");
  });

  it("should normalize Windows paths", () => {
    expect(normalize("foo\\bar\\baz")).toBe("foo/bar/baz");
    expect(normalize("C:\\foo\\bar")).toBe("C:/foo/bar");
  });

  it("should preserve trailing slash", () => {
    expect(normalize("foo/bar/")).toBe("foo/bar/");
  });

  it("should handle drive letter trailing slash insertion", () => {
    expect(normalize("C:")).toBe("C:/");
    expect(normalize("C:\\")).toBe("C:/");
  });

  it("should handle absolute paths correctly with normalize and root conditions", () => {
    expect(normalize("/")).toBe("/");
    expect(normalize("/a/b/..")).toBe("/a");
  });
});

describe("join", () => {
  it("should join multiple segments", () => {
    expect(join("foo", "bar", "baz")).toBe("foo/bar/baz");
  });

  it("should ignore empty segments", () => {
    expect(join("foo", "", "bar")).toBe("foo/bar");
  });

  it("should handle segments with slashes correctly", () => {
    expect(join("/foo", "bar")).toBe("/foo/bar");
    expect(join("foo/", "/bar")).toBe("foo/bar");
  });

  it("should handle both trailing and leading slashes correctly", () => {
    expect(join("foo/", "/bar")).toBe("foo/bar");
  });
});

describe("resolve", () => {
  it("should resolve relative paths", () => {
    expect(resolve("foo", "bar")).toContain("foo/bar");
  });

  it("should resolve absolute paths correctly", () => {
    expect(resolve("/foo/bar", "./baz")).toBe("/foo/bar/baz");
  });

  it("should handle resolved absolute and relative path logical combination", () => {
    expect(resolve("/foo", "bar")).toBe("/foo/bar");
    expect(resolve("foo", "/bar")).toBe("/bar");
  });

  it("should handle empty argument list in resolve", () => {
    expect(resolve()).toBe(process.cwd().replace(/\\/g, "/"));
  });
});

describe("normalizeString", () => {
  it("should normalize string path without allowing above root", () => {
    expect(normalizeString("/foo/../bar", false)).toBe("bar");
  });

  it("should allow above root when specified", () => {
    expect(normalizeString("../foo", true)).toBe("../foo");
  });

  it("should correctly handle complex dot segment combinations and above root with multiple segments", () => {
    expect(normalizeString("/../../foo", true)).toBe("../../foo");
    expect(normalizeString("a/b/../../c", false)).toBe("c");
  });

  it("should handle dot-dot with short or specific length segments", () => {
    expect(normalizeString("a/..b/../c", false)).toBe("a/c");
    expect(normalizeString("..a/..", false)).toBe("");
  });

  it("should handle empty or dot sequences correctly", () => {
    expect(normalizeString("/a/b/./c", false)).toBe("a/b/c");
  });
});

describe("isAbsolute", () => {
  it.each([
    { input: "/foo/bar", expected: true },
    { input: "C:\\foo\\bar", expected: true },
    { input: "foo/bar", expected: false },
    { input: "", expected: false }
  ])("should determine if $input is absolute: $expected", ({ input, expected }) => {
    expect(isAbsolute(input)).toBe(expected);
  });
});

describe("toNamespacedPath", () => {
  it("should normalize windows path for namespacing", () => {
    expect(toNamespacedPath("C:\\foo\\bar")).toBe("C:/foo/bar");
  });
});

describe("extname", () => {
  it.each([
    { input: "index.html", expected: ".html" },
    { input: "index.coffee.md", expected: ".md" },
    { input: "index.", expected: "." },
    { input: "index", expected: "" },
    { input: ".gitignore", expected: "" },
    { input: "..", expected: "" }
  ])("should return extension for $input", ({ input, expected }) => {
    expect(extname(input)).toBe(expected);
  });
});

describe("relative", () => {
  it("should compute relative path between two paths", () => {
    expect(relative("/data/orandea/test/aaa", "/data/orandea/impl/bbb")).toBe("../../impl/bbb");
  });

  it("should handle different drive letters correctly in relative path computation", () => {
    expect(relative("C:\\foo\\bar", "D:\\baz\\qux")).toBe("D:/baz/qux");
  });

  it("should handle different drive letters with logical OR vs AND mutations", () => {
    expect(relative("C:/foo", "D:/bar")).toBe("D:/bar");
    expect(relative("C:/foo", "C:/bar")).toBe("../bar");
  });
});

describe("dirname", () => {
  it.each([
    { input: "/foo/bar/baz", expected: "/foo/bar" },
    { input: "/foo/bar///", expected: "/foo" },
    { input: "foo", expected: "." },
    { input: "C:\\foo\\bar", expected: "C:/foo" },
    { input: "C:\\", expected: "C:/" }
  ])("should return dirname for $input", ({ input, expected }) => {
    expect(dirname(input)).toBe(expected);
  });
});

describe("format", () => {
  it("should format path object to string", () => {
    expect(
      format({
        root: "/",
        dir: "/home/user/dir",
        base: "file.txt",
        ext: ".txt",
        name: "file"
      })
    ).toBe("/home/user/dir/file.txt");
  });

  it("should handle extensions starting or not starting with dot correctly", () => {
    expect(
      format({
        root: "",
        dir: "home",
        base: "file.txt",
        ext: "txt",
        name: "file"
      })
    ).toBe("home/file.txt");
    expect(
      format({
        root: "",
        dir: "home",
        base: "file.txt",
        ext: ".txt",
        name: "file"
      })
    ).toBe("home/file.txt");
  });
});

describe("basename", () => {
  it.each([
    { input: "/foo/bar/baz.html", extension: undefined, expected: "baz.html" },
    { input: "/foo/bar/baz.html", extension: ".html", expected: "baz" },
    { input: "C:\\foo\\bar\\", extension: undefined, expected: "bar" }
  ])("should return basename for $input", ({ input, extension, expected }) => {
    expect(basename(input, extension)).toBe(expected);
  });
});

describe("parse", () => {
  it("should parse path into components", () => {
    const parsed = parse("/home/user/dir/file.txt");
    expect(parsed).toEqual({
      root: "/",
      dir: "/home/user/dir",
      base: "file.txt",
      ext: ".txt",
      name: "file"
    });
  });

  it("should parse root folders and validate root regular expressions strictly", () => {
    expect(parse("/C:")).toEqual({
      root: "/",
      dir: "/",
      base: "C:",
      ext: "",
      name: "C:"
    });
    expect(parse("/C:\\")).toEqual({
      root: "/",
      dir: "/",
      base: "C:",
      ext: "",
      name: "C:"
    });
    expect(parse("/")).toEqual({
      root: "/",
      dir: "/",
      base: "",
      ext: "",
      name: ""
    });
    expect(parse("file.txt")).toEqual({
      root: "",
      dir: "",
      base: "file.txt",
      ext: ".txt",
      name: "file"
    });
  });
});

describe("matchesGlob", () => {
  it("should match path against glob pattern", () => {
    expect(matchesGlob("/foo/bar.js", "*.js")).toBe(false);
    expect(matchesGlob("/foo/bar.js", "**/bar.js")).toBe(true);
  });
});
```
