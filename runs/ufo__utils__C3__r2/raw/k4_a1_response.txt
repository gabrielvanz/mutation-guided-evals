describe("isRelative", () => {
  const cases = [
    { input: "./foo", expected: true },
    { input: "../foo", expected: true },
    { input: "/foo", expected: false },
    { input: "foo", expected: false },
    { input: "", expected: false },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${expected} for input "${input}"`, () => {
      expect(isRelative(input)).toBe(expected);
    });
  }

describe("hasProtocol", () => {
  const cases = [
    { input: "https://example.com", expected: true },
    { input: "ftp://example.com", expected: true },
    { input: "data:text/plain", expected: true },
    { input: "//example.com", expected: false },
    { input: "//example.com", opts: { acceptRelative: true }, expected: true },
    { input: "//example.com", opts: true, expected: true },
    { input: "data:text/plain", opts: { strict: true }, expected: false },
    { input: "https://example.com", opts: { strict: true }, expected: true },
    { input: "/foo/bar", expected: false },
    { input: "Høtp://example.com", expected: false },
  ];

  for (const { input, opts, expected } of cases) {
    it(`should correctly identify protocol for "${input}"`, () => {
      expect(hasProtocol(input, opts as any)).toBe(expected);
    });
  }
});

describe("hasTrailingSlash", () => {

    { input: "/foo", respectQueryAndFragment: false, expected: false },

    { input: "/foo/?query=true", respectQueryAndFragment: true, expected: true },
    { input: "", respectQueryAndFragment: false, expected: false },

  for (const { input, respectQueryAndFragment, expected } of cases) {
    it(`should return ${expected} for "${input}" with respectQueryAndFragment=${respectQueryAndFragment}`, () => {

    });
  }
});

describe("withoutTrailingSlash", () => {

    { input: "/foo", respectQueryAndFragment: false, expected: "/foo" },
    { input: "", respectQueryAndFragment: false, expected: "/" },
    { input: "/#hash", respectQueryAndFragment: true, expected: "/#hash" },

  for (const { input, respectQueryAndFragment, expected } of cases) {
    it(`should return "${expected}" for "${input}"`, () => {

describe("withTrailingSlash", () => {

describe("withoutBase", () => {

describe("joinURL", () => {
  it("should join multiple URL segments", () => {
    expect(joinURL("a", "/b", "/c")).toBe("a/b/c");
  });

  it("should handle empty segments", () => {
    expect(joinURL("a", "", "/b")).toBe("a/b");
  });

  it("should handle empty base with segments", () => {
    expect(joinURL("", "a", "b")).toBe("a/b");
  });
});

describe("joinRelativeURL", () => {
  it("should handle relative segments", () => {
    expect(joinRelativeURL("/a", "../b", "./c")).toBe("/b/c");
  });

  it("should handle navigation parent directories correctly", () => {
    expect(joinRelativeURL("/a/b", "../c")).toBe("/a/c");
  });

  it("should handle single slash / and non-slash conditions", () => {
    expect(joinRelativeURL("/", "a")).toBe("/a");
  });

  it("should handle ./ prefix correctly", () => {
    expect(joinRelativeURL("./a", "b")).toBe("./a/b");
  });

  it("should handle empty inputs", () => {
    expect(joinRelativeURL()).toBe("");
  });

  it("should handle navigating above root with parent references", () => {
    expect(joinRelativeURL("/a", "../..")).toBe("../");
    expect(joinRelativeURL("http://example.com/a", "..")).toBe("http://example.com");
  });

  it("should preserve trailing slash on joinRelativeURL", () => {
    expect(joinRelativeURL("a/", "b/")).toBe("a/b/");
  });
});

describe("withHttp", () => {
  it("should add or replace protocol with http://", () => {
    expect(withHttp("https://example.com")).toBe("http://example.com");
    expect(withHttp("example.com")).toBe("http://example.com");
  });
});

describe("withHttps", () => {
  it("should add or replace protocol with https://", () => {
    expect(withHttps("http://example.com")).toBe("https://example.com");
    expect(withHttps("example.com")).toBe("https://example.com");
  });
});

describe("withoutProtocol", () => {
  it("should remove protocol from input", () => {
    expect(withoutProtocol("http://example.com")).toBe("example.com");
    expect(withoutProtocol("example.com")).toBe("example.com");
  });
});

describe("withProtocol", () => {
  it("should add or replace protocol", () => {
    expect(withProtocol("http://example.com", "ftp://")).toBe("ftp://example.com");
    expect(withProtocol("example.com", "ftp://")).toBe("ftp://example.com");
    expect(withProtocol("//example.com", "http://")).toBe("http://example.com");
  });
});

describe("normalizeURL", () => {
  it("should normalize URL encoding, slashes, and search parts", () => {
    expect(normalizeURL("test?query=123 123#hash, test")).toBe("test?query=123+123#hash,%20test");
    expect(normalizeURL("http://localhost:3000")).toBe("http://localhost:3000");
  });
});

describe("resolveURL", () => {
  it("should resolve multiple URL segments against base", () => {
    expect(resolveURL("http://foo.com/foo?test=123#token", "bar", "baz")).toBe(
      "http://foo.com/foo/bar/baz?test=123#token",
    );
  });

  it("should override hash when segment has hash", () => {
    expect(resolveURL("http://foo.com/foo#old", "bar#new")).toBe("http://foo.com/foo/bar#new");
  });

  it("should combine search queries when both base and segment have search params", () => {
    expect(resolveURL("http://foo.com/foo?a=1", "?b=2")).toBe("http://foo.com/foo?a=1&b=2");
  });

  it("should throw TypeError if base is not a string", () => {
    expect(() => resolveURL(123 as any, "bar")).toThrow(TypeError);
  });

  it("should return base when filtered inputs are empty", () => {
    expect(resolveURL("http://foo.com", "")).toBe("http://foo.com");
  });

  it("should assign search directly if base has no search", () => {
    expect(resolveURL("http://foo.com/foo", "?b=2")).toBe("http://foo.com/foo?b=2");
  });
});

describe("isSamePath", () => {
  it("should return true for paths with different trailing slashes and encoding", () => {
    expect(isSamePath("/foo", "/foo/")).toBe(true);
    expect(isSamePath("/foo%20bar", "/foo bar/")).toBe(true);
  });

  it("should return false for genuinely different paths", () => {
    expect(isSamePath("/foo", "/bar")).toBe(false);
  });
});

describe("isEqual", () => {
  it("should compare paths loosely by default", () => {
    expect(isEqual("/foo", "foo")).toBe(true);
    expect(isEqual("foo/", "foo")).toBe(true);
    expect(isEqual("/foo bar", "/foo%20bar")).toBe(true);
  });

  it("should support strict comparisons via options", () => {
    expect(isEqual("/foo", "foo", { leadingSlash: true })).toBe(false);
    expect(isEqual("foo/", "foo", { trailingSlash: true })).toBe(false);
    expect(isEqual("/foo bar", "/foo%20bar", { encoding: true })).toBe(false);
  });
});

describe("withFragment", () => {
  it("should add or replace the fragment section", () => {
    expect(withFragment("/foo", "bar")).toBe("/foo#bar");
    expect(withFragment("/foo#bar", "baz")).toBe("/foo#baz");
    expect(withFragment("/foo#bar", "")).toBe("/foo#bar");
    expect(withFragment("/foo", "#")).toBe("/foo");
    expect(withFragment("/foo#bar", "")).toBe("/foo#bar");
  });
});

describe("withoutFragment", () => {
  it("should remove the fragment section", () => {
    expect(withoutFragment("http://example.com/foo?q=123#bar")).toBe("http://example.com/foo?q=123");
  });
});

describe("withoutHost", () => {
  it("should remove the host from the URL", () => {
    expect(withoutHost("http://example.com/foo?q=123#bar")).toBe("/foo?q=123#bar");
  });
});
