import { describe, it, expect } from "vitest";
import { normalizeAliases, resolveAlias, reverseResolveAlias, filename } from "../../src/utils";

describe("normalizeAliases", () => {
  it("should return the same object if already normalized", () => {
    const aliases = { "@foo": "/foo" };
    const normalized1 = normalizeAliases(aliases);
    const normalized2 = normalizeAliases(normalized1);
    expect(normalized2).toBe(normalized1);
  });

  it("should sort aliases from specific to general", () => {
    const aliases = {
      "fs": "/node_modules/fs",
      "fs/promises": "/node_modules/fs/promises"
    };
    const result = normalizeAliases(aliases);
    const keys = Object.keys(result);
    expect(keys).toEqual(["fs/promises", "fs"]);
  });

  it("should resolve alias values in relation to each other recursively", () => {
    const aliases = {
      "@base": "/path/to/base",
      "@app": "@base/app"
    };
    const result = normalizeAliases(aliases);
    expect(result["@app"]).toBe("/path/to/base/app");
  });

  it("should not resolve cyclic or self aliases", () => {
    const aliases = {
      "@self": "@self/sub"
    };
    const result = normalizeAliases(aliases);
    expect(result["@self"]).toBe("@self/sub");
  });
});

describe("resolveAlias", () => {
  it("should resolve matching alias to target path", () => {
    const aliases = { "@components": "/src/components" };
    const result = resolveAlias("@components/Button.ts", aliases);
    expect(result).toBe("/src/components/Button.ts");
  });

  it("should return normalized original path if no alias matches", () => {
    const aliases = { "@components": "/src/components" };
    const result = resolveAlias("C:\\src\\index.ts", aliases);
    expect(result).toBe("C:/src/index.ts");
  });

  it("should handle trailing slashes correctly in alias resolution", () => {
    const aliases = { "@utils/": "/src/utils/" };
    const result = resolveAlias("@utils/helpers.ts", aliases);
    expect(result).toBe("/src/utils/helpers.ts");
  });

  it("should handle empty alias mapping and empty path", () => {
    const result = resolveAlias("", {});
    expect(result).toBe("");
  });
});

describe("reverseResolveAlias", () => {
  it("should return array of possible alias resolutions sorted by length", () => {
    const aliases = {
      "u": "/src/utils",
      "app": "/src/utils/app"
    };
    const result = reverseResolveAlias("/src/utils/app/index.ts", aliases);
    expect(Array.isArray(result)).toBe(true);
    expect(result.length).toBeGreaterThan(0);
  });

  it("should return empty array when no reverse alias matches", () => {
    const aliases = { "@lib": "/src/lib" };
    const result = reverseResolveAlias("/other/path.ts", aliases);
    expect(result).toEqual([]);
  });

  it("should handle Windows paths and normalize them in reverse resolution", () => {
    const aliases = { "/src": "C:\\project" };
    const result = reverseResolveAlias("C:\\project\\index.ts", aliases);
    expect(result).toEqual(["/src/index.ts"]);
  });
});

describe("filename", () => {
  const testCases = [
    { input: "/path/to/file.ts", expected: "file" },
    { input: "C:\\path\\to\\image.png", expected: "image" },
    { input: "README.md", expected: "README" },
    { input: ".gitignore", expected: ".gitignore" },
    { input: "file.tar.gz", expected: "file.tar" },
    { input: "no-extension", expected: "no-extension" },
    { input: "/path/to/", expected: undefined },
    { input: "", expected: undefined }
  ];

  for (const { input, expected } of testCases) {
    it(`should extract filename "${expected}" from input "${input}"`, () => {
      expect(filename(input)).toBe(expected);
    });
  }
});
