```ts
import { describe, it, expect } from "vitest";
import {
  normalizeAliases,
  resolveAlias,
  reverseResolveAlias,
  filename
} from "../../src/utils";

describe("normalizeAliases", () => {
  it("should sort aliases from specific to general and resolve nested aliases", () => {
    const input = {
      "~/": "/root/",
      "~/components": "/root/components"
    };
    const result = normalizeAliases(input);
    expect(result).toEqual({
      "~/components": "/root/components",
      "~/": "/root/"
    });
  });

  it("should return the same object if already normalized", () => {
    const input = { "~/": "/root/" };
    const normalized = normalizeAliases(input);
    const resultAgain = normalizeAliases(normalized);
    expect(resultAgain).toBe(normalized);
  });

  it("should handle empty alias objects", () => {
    const result = normalizeAliases({});
    expect(result).toEqual({});
  });
});

describe("resolveAlias", () => {
  it("should resolve a matching alias to its target", () => {
    const aliases = {
      "~": "/app/src"
    };
    const result = resolveAlias("~/components/Button", aliases);
    expect(result).toBe("/app/src/components/Button");
  });

  it("should return the original normalized path if no alias matches", () => {
    const aliases = {
      "~": "/app/src"
    };
    const result = resolveAlias("/other/path", aliases);
    expect(result).toBe("/other/path");
  });

  it("should handle Windows paths correctly", () => {
    const aliases = {
      "~": "C:/app/src"
    };
    const result = resolveAlias("~\\file.ts", aliases);
    expect(result).toBe("C:/app/src/file.ts");
  });

  it("should not partially match an alias without a trailing separator", () => {
    const aliases = {
      "~foo": "/app/foo"
    };
    const result = resolveAlias("~foobar/file.ts", aliases);
    expect(result).toBe("~foobar/file.ts");
  });
});

describe("reverseResolveAlias", () => {
  it("should return matching reverse alias resolutions sorted by length", () => {
    const aliases = {
      "components": "/app/src/components",
      "src": "/app/src"
    };
    const result = reverseResolveAlias("/app/src/components/Button.ts", aliases);
    expect(result).toEqual([
      "src/components/Button.ts",
      "components/Button.ts"
    ]);
  });

  it("should return an empty array if no reverse alias matches", () => {
    const aliases = {
      "components": "/app/src/components"
    };
    const result = reverseResolveAlias("/other/path/file.ts", aliases);
    expect(result).toEqual([]);
  });

  it("should handle Windows paths in reverse resolution", () => {
    const aliases = {
      "components": "C:/app/src/components"
    };
    const result = reverseResolveAlias("C:\\app\\src\\components\\Button.ts", aliases);
    expect(result).toEqual(["components/Button.ts"]);
  });
});

describe("filename", () => {
  const cases = [
    { input: "/path/to/file.ts", expected: "file" },
    { input: "file.tar.gz", expected: "file.tar" },
    { input: "README", expected: "README" },
    { input: ".gitignore", expected: ".gitignore" },
    { input: "C:\\dir\\sub\\script.js", expected: "script" },
    { input: "", expected: undefined },
    { input: "/", expected: undefined },
    { input: "/path/to/", expected: undefined }
  ];

  for (const { input, expected } of cases) {
    it(`should extract filename for ${JSON.stringify(input)}`, () => {
      expect(filename(input)).toBe(expected);
    });
  }
});
```
