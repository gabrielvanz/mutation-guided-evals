```ts
import { describe, it, expect } from "vitest";
import { genString, escapeString, genSafeVariableName } from "../../src/string";

describe("genString", () => {
  it("should generate double quoted string by default", () => {
    const cases = [
      { input: "hello", expected: "\"hello\"" },
      { input: "", expected: "\"\"" },
      { input: "it's a test", expected: "\"it's a test\"" }
    ];

    for (const { input, expected } of cases) {
      expect(genString(input)).toBe(expected);
    }
  });

  it("should generate single quoted string when singleQuotes is true", () => {
    const cases = [
      { input: "hello", expected: "'hello'" },
      { input: "", expected: "''" },
      { input: "it's a test", expected: "'it\\'s a test'" }
    ];

    for (const { input, expected } of cases) {
      expect(genString(input, { singleQuotes: true })).toBe(expected);
    }
  });

  it("should handle special characters and escapes", () => {
    const input = "line1\nline2\r'quoted'";
    expect(genString(input)).toBe("\"line1\\nline2\\r'quoted'\"");
    expect(genString(input, { singleQuotes: true })).toBe("'line1\\nline2\\r\\'quoted\\''");
  });
});

describe("escapeString", () => {
  it("should return the original string if no escape is needed", () => {
    const cases = [
      { input: "hello", expected: "hello" },
      { input: "12345", expected: "12345" },
      { input: "", expected: "" }
    ];

    for (const { input, expected } of cases) {
      expect(escapeString(input)).toBe(expected);
    }
  });

  it("should escape special characters correctly", () => {
    const cases = [
      { input: "line1\nline2", expected: "line1\\nline2" },
      { input: "line1\rline2", expected: "line1\\rline2" },
      { input: "it's", expected: "it\\'s" },
      { input: "path\\to", expected: "path\\\\to" },
      { input: "line\u2028separator", expected: "line\\u2028separator" },
      { input: "line\u2029separator", expected: "line\\u2029separator" }
    ];

    for (const { input, expected } of cases) {
      expect(escapeString(input)).toBe(expected);
    }
  });
});

describe("genSafeVariableName", () => {
  it("should return the original name if it is safe", () => {
    const cases = [
      { input: "myVar", expected: "myVar" },
      { input: "fooBar123", expected: "fooBar123" }
    ];

    for (const { input, expected } of cases) {
      expect(genSafeVariableName(input)).toBe(expected);
    }
  });

  it("should prepend underscore if name starts with a digit", () => {
    const cases = [
      { input: "123var", expected: "_123var" },
      { input: "0foo", expected: "_0foo" }
    ];

    for (const { input, expected } of cases) {
      expect(genSafeVariableName(input)).toBe(expected);
    }
  });

  it("should prepend underscore if name is a reserved JavaScript keyword", () => {
    const cases = [
      { input: "class", expected: "_class" },
      { input: "function", expected: "_function" },
      { input: "const", expected: "_const" },
      { input: "return", expected: "_return" }
    ];

    for (const { input, expected } of cases) {
      expect(genSafeVariableName(input)).toBe(expected);
    }
  });

  it("should replace non-word characters with their char code representation", () => {
    const cases = [
      { input: "foo-bar", expected: "foo_45bar" },
      { input: "foo.bar", expected: "foo_46bar" },
      { input: "foo bar", expected: "foo_32bar" }
    ];

    for (const { input, expected } of cases) {
      expect(genSafeVariableName(input)).toBe(expected);
    }
  });
});
```
