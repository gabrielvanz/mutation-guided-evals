```ts
import { describe, it, expect } from "vitest";
import {
  parseURL,
  parsePath,
  parseAuth,
  parseHost,
  stringifyParsedURL,
  parseFilename
} from "../../src/parse";

describe("parseURL", () => {
  it("should parse absolute URL with protocol, host, pathname, search, and hash", () => {
    const result = parseURL("http://foo.com/foo?test=123#token");
    expect(result).toEqual({
      protocol: "http:",
      auth: "",
      host: "foo.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should parse relative URL without protocol", () => {
    const result = parseURL("foo.com/foo?test=123#token");
    expect(result).toEqual({
      pathname: "foo.com/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should parse relative URL with default protocol", () => {
    const result = parseURL("foo.com/foo?test=123#token", "https://");
    expect(result).toEqual({
      protocol: "https:",
      auth: "",
      host: "foo.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should handle empty input", () => {
    const result = parseURL("");
    expect(result).toEqual({
      pathname: "",
      search: "",
      hash: ""
    });
  });

  it("should parse special protocols like data:", () => {
    const result = parseURL("data:text/plain,hello");
    expect(result).toEqual({
      protocol: "data:",
      pathname: "text/plain,hello",
      href: "data:text/plain,hello",
      auth: "",
      host: "",
      search: "",
      hash: ""
    });
  });

  it("should parse URLs with auth", () => {
    const result = parseURL("http://user:pass@foo.com/bar");
    expect(result).toEqual({
      protocol: "http:",
      auth: "user:pass",
      host: "foo.com",
      pathname: "/bar",
      search: "",
      hash: ""
    });
  });

  it("should handle file protocol paths with drive letters", () => {
    const result = parseURL("file:///C:/foo/bar");
    expect(result).toEqual({
      protocol: "file:",
      auth: "",
      host: "",
      pathname: "/C:/foo/bar",
      search: "",
      hash: ""
    });
  });
});

describe("parsePath", () => {
  it("should parse path with search and hash", () => {
    const result = parsePath("http://foo.com/foo?test=123#token");
    expect(result).toEqual({
      pathname: "http://foo.com/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should handle empty input", () => {
    const result = parsePath("");
    expect(result).toEqual({
      pathname: "",
      search: "",
      hash: ""
    });
  });
});

describe("parseAuth", () => {
  it("should parse username and password", () => {
    const result = parseAuth("user:pass");
    expect(result).toEqual({
      username: "user",
      password: "pass"
    });
  });

  it("should handle empty input", () => {
    const result = parseAuth("");
    expect(result).toEqual({
      username: "",
      password: ""
    });
  });
});

describe("parseHost", () => {
  it("should parse hostname and port", () => {
    const result = parseHost("foo.com:8080");
    expect(result).toEqual({
      hostname: "foo.com",
      port: "8080"
    });
  });

  it("should handle hostname without port", () => {
    const result = parseHost("foo.com");
    expect(result).toEqual({
      hostname: "foo.com",
      port: undefined
    });
  });

  it("should handle empty input", () => {
    const result = parseHost("");
    expect(result).toEqual({
      hostname: "",
      port: undefined
    });
  });
});

describe("stringifyParsedURL", () => {
  it("should stringify a parsed URL object back to a string", () => {
    const parsed = parseURL("http://foo.com/foo?test=123#token");
    parsed.host = "bar.com";
    const result = stringifyParsedURL(parsed);
    expect(result).toBe("http://bar.com/foo?test=123#token");
  });

  it("should handle partial parsed objects", () => {
    const result = stringifyParsedURL({
      pathname: "/foo",
      search: "test=123",
      hash: "#token"
    });
    expect(result).toBe("/foo?test=123#token");
  });
});

describe("parseFilename", () => {
  it("should return the last segment as filename", () => {
    const result = parseFilename("http://example.com/path/to/filename.ext");
    expect(result).toBe("filename.ext");
  });

  it("should respect strict option for files with extensions", () => {
    const strictResult = parseFilename("/path/to/.hidden-file", { strict: true });
    expect(strictResult).toBeUndefined();

    const normalResult = parseFilename("/path/to/filename.ext", { strict: true });
    expect(normalResult).toBe("filename.ext");
  });

  it("should return undefined for empty input", () => {
    const result = parseFilename("");
    expect(result).toBeUndefined();
  });
});
```
