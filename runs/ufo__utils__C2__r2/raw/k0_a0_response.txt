import { describe, it, expect } from "vitest";
import {
  isRelative,
  hasProtocol,
  isScriptProtocol,
  hasTrailingSlash,
  withoutTrailingSlash,
  withTrailingSlash,
  hasLeadingSlash,
  withoutLeadingSlash,
  withLeadingSlash,
  cleanDoubleSlashes,
  withBase,
  withoutBase,
  withQuery,
  filterQuery,
  getQuery,
  isEmptyURL,
  isNonEmptyURL,
  joinURL,
  joinRelativeURL,
  withHttp,
  withHttps,
  withoutProtocol,
  withProtocol,
  normalizeURL,
  resolveURL,
  isSamePath,
  isEqual,
  withFragment,
  withoutFragment,
  withoutHost,
} from "../../src/utils";

describe("isRelative", () => {
  const cases = [
    { input: "./foo", expected: true },
    { input: "../foo", expected: true },
    { input: "/foo", expected: false },
    { input: "foo", expected: false },
    { input: "", expected: false },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${expected} for ${JSON.stringify(input)}`, () => {
      expect(isRelative(input)).toBe(expected);
    });
  }
});

describe("hasProtocol", () => {
  const cases = [
    { input: "https://example.com", opts: undefined, expected: true },
    { input: "//example.com", opts: undefined, expected: false },
    { input: "//example.com", opts: { acceptRelative: true }, expected: true },
    { input: "//example.com", opts: true, expected: true },
    { input: "ftp://example.com", opts: undefined, expected: true },
    { input: "data:text/plain", opts: undefined, expected: true },
    { input: "data:text/plain", opts: { strict: true }, expected: false },
    { input: "https://example.com", opts: { strict: true }, expected: true },
    { input: "/foo/bar", opts: undefined, expected: false },
  ];

  for (const { input, opts, expected } of cases) {
    it(`should return ${expected} for ${JSON.stringify(input)} with opts ${JSON.stringify(opts)}`, () => {
      // @ts-expect-error accepting overload
      expect(hasProtocol(input, opts)).toBe(expected);
    });
  }
});

describe("isScriptProtocol", () => {
  const cases = [
    { input: "javascript:alert(1)", expected: true },
    { input: "data:text/html,hello", expected: true },
    { input: "blob:hello", expected: true },
    { input: "vbscript:alert(1)", expected: true },
    { input: "https://example.com", expected: false },
    { input: undefined, expected: false },
    { input: "", expected: false },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${expected} for ${JSON.stringify(input)}`, () => {
      expect(isScriptProtocol(input)).toBe(expected);
    });
  }
});

describe("hasTrailingSlash", () => {
  const cases = [
    { input: "/foo/", respectQuery: false, expected: true },
    { input: "/foo", respectQuery: false, expected: false },
    { input: "/foo?query=true", respectQuery: true, expected: false },
    { input: "/foo/?query=true", respectQuery: true, expected: true },
    { input: "", respectQuery: false, expected: false },
  ];

  for (const { input, respectQuery, expected } of cases) {
    it(`should return ${expected} for ${JSON.stringify(input)} with respectQuery ${respectQuery}`, () => {
      expect(hasTrailingSlash(input, respectQuery)).toBe(expected);
    });
  }
});

describe("withoutTrailingSlash", () => {
  const cases = [
    { input: "/foo/", respectQuery: false, expected: "/foo" },
    { input: "/foo", respectQuery: false, expected: "/foo" },
    { input: "/path/?query=true", respectQuery: true, expected: "/path?query=true" },
    { input: "", respectQuery: false, expected: "/" },
    { input: "/", respectQuery: false, expected: "/" },
    { input: "/path/#hash", respectQuery: true, expected: "/path#hash" },
  ];

  for (const { input, respectQuery, expected } of cases) {
    it(`should return ${JSON.stringify(expected)} for ${JSON.stringify(input)}`, () => {
      expect(withoutTrailingSlash(input, respectQuery)).toBe(expected);
    });
  }
});

describe("withTrailingSlash", () => {
  const cases = [
    { input: "/foo", respectQuery: false, expected: "/foo/" },
    { input: "/foo/", respectQuery: false, expected: "/foo/" },
    { input: "/path?query=true", respectQuery: true, expected: "/path/?query=true" },
    { input: "", respectQuery: false, expected: "/" },
    { input: "#hash", respectQuery: true, expected: "#hash" },
  ];

  for (const { input, respectQuery, expected } of cases) {
    it(`should return ${JSON.stringify(expected)} for ${JSON.stringify(input)}`, () => {
      expect(withTrailingSlash(input, respectQuery)).toBe(expected);
    });
  }
});

describe("hasLeadingSlash", () => {
  const cases = [
    { input: "/foo", expected: true },
    { input: "foo", expected: false },
    { input: "", expected: false },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${expected} for ${JSON.stringify(input)}`, () => {
      expect(hasLeadingSlash(input)).toBe(expected);
    });
  }
});

describe("withoutLeadingSlash", () => {
  const cases = [
    { input: "/foo", expected: "foo" },
    { input: "foo", expected: "foo" },
    { input: "", expected: "/" },
    { input: "/", expected: "/" },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${JSON.stringify(expected)} for ${JSON.stringify(input)}`, () => {
      expect(withoutLeadingSlash(input)).toBe(expected);
    });
  }
});

describe("withLeadingSlash", () => {
  const cases = [
    { input: "foo", expected: "/foo" },
    { input: "/foo", expected: "/foo" },
    { input: "", expected: "/" },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${JSON.stringify(expected)} for ${JSON.stringify(input)}`, () => {
      expect(withLeadingSlash(input)).toBe(expected);
    });
  }
});

describe("cleanDoubleSlashes", () => {
  const cases = [
    { input: "//foo//bar//", expected: "/foo/bar/" },
    { input: "http://example.com/analyze//http://localhost:3000//", expected: "http://example.com/analyze/http://localhost:3000/" },
    { input: "", expected: "" },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${JSON.stringify(expected)} for ${JSON.stringify(input)}`, () => {
      expect(cleanDoubleSlashes(input)).toBe(expected);
    });
  }
});

describe("withBase", () => {
  const cases = [
    { input: "/foo/bar", base: "/foo", expected: "/foo/bar" },
    { input: "/foo/bar", base: "/baz", expected: "/baz/foo/bar" },
    { input: "/admin-dashboard", base: "/admin/", expected: "/admin/admin-dashboard" },
    { input: "https://example.com", base: "/foo", expected: "https://example.com" },
    { input: "/foo", base: "", expected: "/foo" },
  ];

  for (const { input, base, expected } of cases) {
    it(`should return ${JSON.stringify(expected)} for input ${JSON.stringify(input)} and base ${JSON.stringify(base)}`, () => {
      expect(withBase(input, base)).toBe(expected);
    });
  }
});

describe("withoutBase", () => {
  const cases = [
    { input: "/foo/bar", base: "/foo", expected: "/bar" },
    { input: "/admin-dashboard", base: "/admin/", expected: "/admin-dashboard" },
    { input: "/foo/bar", base: "/baz", expected: "/foo/bar" },
    { input: "/legacy//evil.com", base: "/legacy", expected: "/evil.com" },
    { input: "/foo", base: "", expected: "/foo" },
  ];

  for (const { input, base, expected } of cases) {
    it(`should return ${JSON.stringify(expected)} for input ${JSON.stringify(input)} and base ${JSON.stringify(base)}`, () => {
      expect(withoutBase(input, base)).toBe(expected);
    });
  }
});

describe("withQuery", () => {
  it("should add or replace query parameters", () => {
    expect(withQuery("/foo?page=a", { token: "secret" })).toBe("/foo?page=a&token=secret");
  });
});

describe("filterQuery", () => {
  const cases = [
    { input: "/foo?bar=1&baz=2", predicate: (key: string) => key !== "bar", expected: "/foo?baz=2" },
    { input: "/foo", predicate: () => true, expected: "/foo" },
  ];

  for (const { input, predicate, expected } of cases) {
    it(`should filter query for ${JSON.stringify(input)}`, () => {
      expect(filterQuery(input, predicate)).toBe(expected);
    });
  }
});

describe("getQuery", () => {

    expect(getQuery("http://foo.com/foo?test=123&unicode=%E5%A5%BD")).toEqual({

      unicode: "好",
    });
  });
});

describe("isEmptyURL", () => {
  const cases = [
    { input: "", expected: true },
    { input: "/", expected: true },

  for (const { input, expected } of cases) {
    it(`should return ${expected} for ${JSON.stringify(input)}`, () => {

describe("isNonEmptyURL", () => {
  const cases = [
    { input: "", expected: false },
    { input: "/", expected: false },
    { input: "/foo", expected: true },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${expected} for ${JSON.stringify(input)}`, () => {
      expect(isNonEmptyURL(input)).toBe(expected);

describe("joinURL", () => {
  it("should join multiple URL segments", () => {
    expect(joinURL("a", "/b", "/c")).toBe("a/b/c");
    expect(joinURL("", "b", "c")).toBe("b/c");
  });
});

describe("joinRelativeURL", () => {
  it("should join relative URL segments", () => {
    expect(joinRelativeURL("/a", "../b", "./c")).toBe("/b/c");
  });
});

describe("withHttp", () => {
  it("should add or replace protocol to http://", () => {
    expect(withHttp("https://example.com")).toBe("http://example.com");
    expect(withHttp("example.com")).toBe("http://example.com");
  });
});

describe("withHttps", () => {
  it("should add or replace protocol to https://", () => {
    expect(withHttps("http://example.com")).toBe("https://example.com");
    expect(withHttps("example.com")).toBe("https://example.com");
  });
});

describe("withoutProtocol", () => {
  it("should remove protocol from input", () => {
    expect(withoutProtocol("http://example.com")).toBe("example.com");
    expect(withoutProtocol("example.com")).toBe("example.com");
  });
});

describe("withProtocol", () => {
  it("should add or replace protocol", () => {
    expect(withProtocol("http://example.com", "ftp://")).toBe("ftp://example.com");
    expect(withProtocol("example.com", "http://")).toBe("http://example.com");
    expect(withProtocol("//example.com", "http://")).toBe("http://example.com");
  });
});

describe("normalizeURL", () => {
  it("should normalize URL encoding and slashes", () => {
    expect(normalizeURL("test?query=123 123#hash, test")).toBe("test?query=123%20123#hash,%20test");
    expect(normalizeURL("http://localhost:3000")).toBe("http://localhost:3000");
  });
});

describe("resolveURL", () => {
  it("should resolve multiple URL segments", () => {
    expect(resolveURL("http://foo.com/foo?test=123#token", "bar", "baz")).toBe("http://foo.com/foo/bar/baz?test=123#token");
  });

  it("should throw TypeError for non-string base", () => {
    // @ts-expect-error testing invalid type
    expect(() => resolveURL(123)).toThrow(TypeError);
  });

  it("should return base when no inputs are provided", () => {
    expect(resolveURL("http://foo.com/foo")).toBe("http://foo.com/foo");
  });
});

describe("isSamePath", () => {
  it("should check if two paths are equal after normalization", () => {
    expect(isSamePath("/foo", "/foo/")).toBe(true);
    expect(isSamePath("/foo", "/bar")).toBe(false);
  });
});

describe("isEqual", () => {
  const cases = [
    { a: "/foo", b: "foo", opts: undefined, expected: true },
    { a: "foo/", b: "foo", opts: undefined, expected: true },
    { a: "/foo bar", b: "/foo%20bar", opts: undefined, expected: true },
    { a: "/foo", b: "foo", opts: { leadingSlash: true }, expected: false },
    { a: "foo/", b: "foo", opts: { trailingSlash: true }, expected: false },
    { a: "/foo bar", b: "/foo%20bar", opts: { encoding: true }, expected: false },
  ];

  for (const { a, b, opts, expected } of cases) {
    it(`should return ${expected} for isEqual(${JSON.stringify(a)}, ${JSON.stringify(b)}, ${JSON.stringify(opts)})`, () => {
      expect(isEqual(a, b, opts)).toBe(expected);
    });
  }
});

describe("withFragment", () => {
  const cases = [
    { input: "/foo", hash: "bar", expected: "/foo#bar" },
    { input: "/foo#bar", hash: "baz", expected: "/foo#baz" },
    { input: "/foo#bar", hash: "", expected: "/foo" },
    { input: "/foo", hash: "#", expected: "/foo" },
  ];

  for (const { input, hash, expected } of cases) {
    it(`should return ${JSON.stringify(expected)} for withFragment(${JSON.stringify(input)}, ${JSON.stringify(hash)})`, () => {
      expect(withFragment(input, hash)).toBe(expected);
    });
  }
});

describe("withoutFragment", () => {
  it("should remove fragment section from URL", () => {
    expect(withoutFragment("http://example.com/foo?q=123#bar")).toBe("http://example.com/foo?q=123");
  });
});

describe("withoutHost", () => {
  it("should remove host from URL", () => {
    expect(withoutHost("http://example.com/foo?q=123#bar")).toBe("/foo?q=123#bar");
  });
});
