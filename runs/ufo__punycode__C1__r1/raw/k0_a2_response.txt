```ts
import { describe, it, expect } from "vitest";
import { toASCII } from "../../src/punycode";

describe("punycode toASCII", () => {
  it("passes through ASCII strings unchanged", () => {
    expect(toASCII("example.com")).toBe("example.com");
  });

  it("handles empty strings", () => {
    expect(toASCII("")).toBe("");
  });

  it("converts non-ASCII domain labels to punycode with xn-- prefix", () => {
    expect(toASCII("münchen.de")).toBe("xn--mnchen-3ya.de");
  });

  it("handles emails with non-ASCII domain parts", () => {
    expect(toASCII("user@münchen.de")).toBe("user@xn--mnchen-3ya.de");
  });

  it("handles different types of dots/separators", () => {
    expect(toASCII("example．com")).toBe("example.com");
  });

  it("throws a RangeError on overflow", () => {
    const hugeInput = "a" + "\uffff".repeat(5000);
    expect(() => toASCII(hugeInput)).toThrow(RangeError);
  });
});
```
