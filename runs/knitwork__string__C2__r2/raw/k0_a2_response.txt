```ts
import { describe, it, expect } from "vitest";
import { genString, escapeString, genSafeVariableName } from "../../src/string";

describe("genString", () => {
  it("should generate a double quoted string by default", () => {
    const input = "hello world";
    const result = genString(input);
    expect(result).toBe("\"hello world\"");
  });

  it("should generate a single quoted string when singleQuotes option is true", () => {
    const input = "hello world";
    const result = genString(input, { singleQuotes: true });
    expect(result).toBe("'hello world'");
  });

  it("should handle strings with double quotes when using single quotes", () => {
    const input = "hello \"world\"";
    const result = genString(input, { singleQuotes: true });
    expect(result).toBe("'hello \\"world\\"'");
  });

  it("should handle strings with single quotes when using single quotes", () => {
    const input = "hello 'world'";
    const result = genString(input, { singleQuotes: true });
    expect(result).toBe("'hello \\'world\\''");
  });

  it("should handle empty string", () => {
    const input = "";
    const result = genString(input);
    expect(result).toBe("\"\"");
  });

  it("should handle empty string with single quotes", () => {
    const input = "";
    const result = genString(input, { singleQuotes: true });
    expect(result).toBe("''");
  });
});

describe("escapeString", () => {
  it("should return the same string if no escape is needed", () => {
    const input = "hello world";
    const result = escapeString(input);
    expect(result).toBe("hello world");
  });

  it("should escape backslashes", () => {
    const input = "path\\to\\file";
    const result = escapeString(input);
    expect(result).toBe("path\\\\to\\\\file");
  });

  it("should escape single quotes, newlines and carriage returns", () => {
    const input = "line1\nline2\r'quote'";
    const result = escapeString(input);
    expect(result).toBe("line1\\\nline2\\\r\\'quote\\'");
  });
});

describe("genSafeVariableName", () => {
  it("should return normal variable names as is", () => {
    const input = "myVariable";
    const result = genSafeVariableName(input);
    expect(result).toBe("myVariable");
  });

  it("should prefix with underscore if name starts with a digit", () => {
    const input = "123var";
    const result = genSafeVariableName(input);
    expect(result).toBe("_123var");
  });

  it("should prefix with underscore if name is a reserved keyword", () => {
    const input = "class";
    const result = genSafeVariableName(input);
    expect(result).toBe("_class");
  });

  it("should replace non-word characters with their char code prefixed by underscore", () => {
    const input = "foo-bar";
    const result = genSafeVariableName(input);
    expect(result).toBe("foo_45bar");
  });
});
```
