```ts
import { describe, it, expect } from "vitest";
import defu, { createDefu, defuFn, defuArrayFn } from "../../src/defu";

describe("defu", () => {
  it("should assign default values when properties are missing", () => {
    const base = { a: 1 };
    const defaults = { a: 2, b: 3 };
    const result = defu(base, defaults);

    expect(result).toEqual({ a: 1, b: 3 });
  });

  it("should handle multiple arguments correctly", () => {
    const result = defu({ a: 1 }, { a: 5, b: 2 }, { b: 3, c: 4 });
    expect(result).toEqual({ a: 1, b: 2, c: 4 });
  });

  it("should concatenate arrays", () => {
    const base = { arr: [3, 4] };
    const defaults = { arr: [1, 2] };
    const result = defu(base, defaults);

    expect(result).toEqual({ arr: [3, 4, 1, 2] });
  });

  it("should recursively merge plain objects", () => {
    const base = { obj: { x: 1 } };
    const defaults = { obj: { y: 2, x: 99 } };
    const result = defu(base, defaults);

    expect(result).toEqual({ obj: { x: 1, y: 2 } });
  });

  it("should ignore null and undefined values in base object", () => {
    const base = { a: null, b: undefined, c: 1 };
    const defaults = { a: 10, b: 20, c: 30 };
    const result = defu(base, defaults);

    expect(result).toEqual({ a: 10, b: 20, c: 1 });
  });

  it("should prevent prototype pollution", () => {
    const base = JSON.parse('{"__proto__": {"polluted": true}, "constructor": {"prototype": {"polluted": true}}, "a": 1}');
    const defaults = { a: 2, b: 3 };
    const result = defu(base, defaults);

    expect(result).toEqual({ a: 1, b: 3 });
    expect(({} as any).polluted).toBeUndefined();
  });

  it("should fallback to empty defaults if defaults is not a plain object", () => {
    const base = { a: 1 };
    const result = defu(base, null as any);

    expect(result).toEqual({ a: 1 });
  });

  it("should work with createDefu and custom merger", () => {
    const customDefu = createDefu((object, key, value) => {
      if (key === "custom") {
        object[key] = "merged-" + value;
        return true;
      }
    });

    const result = customDefu({ custom: "default" }, { custom: "base" });
    expect(result).toEqual({ custom: "merged-base" });
  });

  it("should support defuFn for function merging", () => {
    const base = {
      val: (val: number) => val + 1,
    };
    const defaults = {
      val: 5,
    };

    const result = defuFn(base, defaults);
    expect(result.val).toBe(6);
  });

  it("should support defuArrayFn for array-specific function merging", () => {
    const base = {
      items: (arr: number[]) => [...arr, 3],
    };
    const defaults = {
      items: [1, 2],
    };

    const result = defuArrayFn(base, defaults);
    expect(result.items).toEqual([1, 2, 3]);
  });

  it("should default export the same function instance", () => {
    const importedDefu = defu;
    const result = importedDefu({ a: 1 }, { a: 2 });
    expect(result).toEqual({ a: 1 });
  });
});
```
