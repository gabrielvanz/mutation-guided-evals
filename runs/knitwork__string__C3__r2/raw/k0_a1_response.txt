```ts
import { describe, it, expect } from "vitest";
import { genString, escapeString, genSafeVariableName } from "../../src/string";

describe("genString", () => {
  it("should generate double quoted string by default", () => {
    const cases = [
      { input: "hello", expected: "\"hello\"" },
      { input: "hello \"world\"", expected: "\"hello \\\"world\\\"\"" },
      { input: "", expected: "\"\"" }
    ];

    for (const { input, expected } of cases) {
      expect(genString(input)).toBe(expected);
    }
  });

  it("should generate single quoted string when singleQuotes option is true", () => {
    const cases = [
      { input: "hello", expected: "'hello'" },
      { input: "hello 'world'", expected: "'hello \\'world\\''" },
      { input: "hello \"world\"", expected: "'hello \\\"world\\\"'" },
      { input: "", expected: "''" }
    ];

    for (const { input, expected } of cases) {
      expect(genString(input, { singleQuotes: true })).toBe(expected);
    }
  });

  it("should handle special characters and escapes", () => {
    const input = "line1\nline2\r'quote'\\back";
    expect(genString(input)).toBe("\"line1\\nline2\\r'quote'\\\\back\"");
    expect(genString(input, { singleQuotes: true })).toBe("'line1\\nline2\\r\\'quote\\'\\\\back'");
  });
});

describe("escapeString", () => {
  it("should return the original string if no escape needed", () => {
    const cases = [
      { input: "hello", expected: "hello" },
      { input: "foo_bar-123", expected: "foo_bar-123" },
      { input: "", expected: "" }
    ];

    for (const { input, expected } of cases) {
      expect(escapeString(input)).toBe(expected);
    }
  });

  it("should escape special characters", () => {
    const cases = [
      { input: "hello\nworld", expected: "hello\\nworld" },
      { input: "hello\rworld", expected: "hello\\rworld" },
      { input: "hello'world", expected: "hello\\'world" },
      { input: "hello\\world", expected: "hello\\\\world" },
      { input: "hello\u2028world", expected: "hello\\u2028world" },
      { input: "hello\u2029world", expected: "hello\\u2029world" }
    ];

    for (const { input, expected } of cases) {
      expect(escapeString(input)).toBe(expected);
    }
  });
});

describe("genSafeVariableName", () => {
  it("should prepend underscore to reserved names", () => {
    const cases = [
      { input: "class", expected: "_class" },
      { input: "const", expected: "_const" },
      { input: "return", expected: "_return" },
      { input: "default", expected: "_default" }
    ];

    for (const { input, expected } of cases) {
      expect(genSafeVariableName(input)).toBe(expected);
    }
  });

  it("should prepend underscore if name starts with a digit", () => {
    const cases = [
      { input: "123var", expected: "_123var" },
      { input: "0foo", expected: "_0foo" }
    ];

    for (const { input, expected } of cases) {
      expect(genSafeVariableName(input)).toBe(expected);
    }
  });

  it("should replace non-word characters with underscore and char code", () => {
    const cases = [
      { input: "foo-bar", expected: "foo_45bar" },
      { input: "foo.bar", expected: "foo_46bar" },
      { input: "foo bar", expected: "foo_32bar" }
    ];

    for (const { input, expected } of cases) {
      expect(genSafeVariableName(input)).toBe(expected);
    }
  });

  it("should return normal variable names unchanged", () => {
    const cases = [
      { input: "fooBar", expected: "fooBar" },
      { input: "_private", expected: "_private" }
    ];

    for (const { input, expected } of cases) {
      expect(genSafeVariableName(input)).toBe(expected);
    }
  });
});
```
