import { describe, it, expect } from "vitest";
import { normalizeAliases, resolveAlias, reverseResolveAlias, filename } from "../../src/utils";

describe("normalizeAliases", () => {
  it("should return the same object if already normalized", () => {
    const aliases = { "@app": "/src/app" };
    const normalized1 = normalizeAliases(aliases);
    const normalized2 = normalizeAliases(normalized1);
    expect(normalized2).toBe(normalized1);
  });

  it("should sort aliases by specificity and resolve nested aliases", () => {
    const aliases = {
      "@": "/root",
      "@components": "@/components",
    };
    const result = normalizeAliases(aliases);
    expect(result).toEqual({
      "@components": "/root/components",
      "@": "/root",
    });
  });
});

describe("resolveAlias", () => {
  it("should resolve path matching an alias", () => {
    const aliases = { "@app": "/src/app" };
    const result = resolveAlias("@app/components/Button.ts", aliases);
    expect(result).toBe("/src/app/components/Button.ts");
  });

  it("should return original path if no alias matches", () => {
    const aliases = { "@app": "/src/app" };
    const result = resolveAlias("/other/path.ts", aliases);
    expect(result).toBe("/other/path.ts");
  });

  it("should handle Windows path separators in alias resolution", () => {
    const aliases = { "@app": "C:\\src\\app" };
    const result = resolveAlias("@app\\components\\Button.ts", aliases);
    expect(result).toBe("/src/app/components/Button.ts");
  });
});

describe("reverseResolveAlias", () => {
  it("should return sorted array of possible alias resolutions", () => {
    const aliases = {
      "@": "/root",
      "@app": "/root/app",
    };
    const result = reverseResolveAlias("/root/app/index.ts", aliases);
    expect(result).toEqual(["@app/index.ts", "@/app/index.ts"]);
  });

  it("should return empty array when no reverse alias matches", () => {
    const aliases = { "@app": "/src/app" };
    const result = reverseResolveAlias("/other/path.ts", aliases);
    expect(result).toEqual([]);
  });
});

describe("filename", () => {
  const cases = [
    { input: "/path/to/file.ts", expected: "file" },
    { input: "file.tar.gz", expected: "file.tar" },
    { input: "README", expected: "README" },
    { input: ".gitignore", expected: ".gitignore" },
    { input: "/path/to/", expected: undefined },
    { input: "", expected: undefined },
    { input: "C:\\path\\to\\file.js", expected: "file" },
  ];

  for (const { input, expected } of cases) {
    it(`should extract filename for ${JSON.stringify(input)}`, () => {
      expect(filename(input)).toBe(expected);
    });
  }
});
