```ts
import { describe, it, expect } from "vitest";
import {
  parseURL,
  parsePath,
  parseAuth,
  parseHost,
  stringifyParsedURL,
  parseFilename,
} from "../../src/parse";

describe("parse module", () => {
  describe("parseURL", () => {
    it("parses a standard URL correctly", () => {
      const result = parseURL("http://foo.com/foo?test=123#token");
      expect(result.protocol).toBe("http:");
      expect(result.auth).toBe("");
      expect(result.host).toBe("foo.com");
      expect(result.pathname).toBe("/foo");
      expect(result.search).toBe("?test=123");
      expect(result.hash).toBe("#token");
    });

    it("parses a relative path correctly", () => {
      const result = parseURL("foo.com/foo?test=123#token");
      expect(result).toEqual({
        pathname: "foo.com/foo",
        search: "?test=123",
        hash: "#token",
      });
    });

    it("parses a relative path with defaultProto", () => {
      const result = parseURL("foo.com/foo?test=123#token", "https://");
      expect(result.protocol).toBe("https:");
      expect(result.auth).toBe("");
      expect(result.host).toBe("foo.com");
      expect(result.pathname).toBe("/foo");
      expect(result.search).toBe("?test=123");
      expect(result.hash).toBe("#token");
    });

    it("handles special protocols like blob:", () => {
      const result = parseURL("blob:https://example.com/123-456");
      expect(result.protocol).toBe("blob:");
      expect(result.pathname).toBe("https://example.com/123-456");
    });

    it("handles file protocol correctly", () => {
      const result = parseURL("file:///C:/path/to/file");
      expect(result.protocol).toBe("file:");
      expect(result.pathname).toBe("C:/path/to/file");
    });

    it("handles empty input", () => {
      const result = parseURL();
      expect(result).toEqual({
        pathname: "",
        search: "",
        hash: "",
      });
    });
  });

  describe("parsePath", () => {
    it("splits path, search, and hash", () => {
      const result = parsePath("http://foo.com/foo?test=123#token");
      expect(result).toEqual({
        pathname: "http://foo.com/foo",
        search: "?test=123",
        hash: "#token",
      });
    });

    it("handles empty input", () => {
      const result = parsePath();
      expect(result).toEqual({
        pathname: "",
        search: "",
        hash: "",
      });
    });
  });

  describe("parseAuth", () => {
    it("parses and decodes username and password", () => {
      const result = parseAuth("user:pass");
      expect(result).toEqual({
        username: "user",
        password: "pass",
      });
    });

    it("handles empty input", () => {
      const result = parseAuth();
      expect(result).toEqual({
        username: "",
        password: "",
      });
    });
  });

  describe("parseHost", () => {
    it("parses hostname and port", () => {
      const result = parseHost("foo.com:8080");
      expect(result).toEqual({
        hostname: "foo.com",
        port: "8080",
      });
    });

    it("handles hostname without port", () => {
      const result = parseHost("foo.com");
      expect(result).toEqual({
        hostname: "foo.com",
        port: undefined,
      });
    });

    it("handles empty input", () => {
      const result = parseHost();
      expect(result).toEqual({
        hostname: "",
        port: undefined,
      });
    });
  });

  describe("stringifyParsedURL", () => {
    it("stringifies a parsed URL object correctly", () => {
      const parsed = parseURL("http://foo.com/foo?test=123#token");
      parsed.host = "bar.com";
      const result = stringifyParsedURL(parsed);
      expect(result).toBe("http://bar.com/foo?test=123#token");
    });

    it("stringifies with auth and search missing leading question mark", () => {
      const result = stringifyParsedURL({
        protocol: "https:",
        auth: "user:pass",
        host: "example.com",
        pathname: "/path",
        search: "foo=bar",
        hash: "#hash",
      });
      expect(result).toBe("https://user:pass@example.com/path?foo=bar#hash");
    });
  });

  describe("parseFilename", () => {
    it("returns the filename from a URL", () => {
      const result = parseFilename("http://example.com/path/to/filename.ext");
      expect(result).toBe("filename.ext");
    });

    it("returns filename with strict option when extension exists", () => {
      const result = parseFilename("http://example.com/path/to/filename.ext", { strict: true });
      expect(result).toBe("filename.ext");
    });

    it("returns undefined with strict option when no extension", () => {
      const result = parseFilename("http://example.com/path/to/.hidden-file", { strict: true });
      expect(result).toBeUndefined();
    });

    it("handles empty input", () => {
      const result = parseFilename();
      expect(result).toBeUndefined();
    });
  });
});
```
