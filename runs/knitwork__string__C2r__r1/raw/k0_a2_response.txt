```ts
import { describe, it, expect } from "vitest";
import { genString, escapeString, genSafeVariableName } from "../../src/string";

describe("genString", () => {
  it("should generate a double quoted string by default", () => {
    const cases = [
      { input: "hello", expected: "\"hello\"" },
      { input: "hello \"world\"", expected: "\"hello \\\"world\\\"\"" },
      { input: "", expected: "\"\"" }
    ];

    for (const { input, expected } of cases) {
      expect(genString(input)).toBe(expected);
    }
  });

  it("should generate a single quoted string when singleQuotes option is true", () => {
    const cases = [
      { input: "hello", expected: "'hello'" },
      { input: "hello 'world'", expected: "'hello \\'world\\''" },
      { input: "hello \"world\"", expected: "'hello \\\"world\\\"'" },
      { input: "", expected: "''" }
    ];

    for (const { input, expected } of cases) {
      expect(genString(input, { singleQuotes: true })).toBe(expected);
    }
  });

  it("should handle newlines and special characters correctly", () => {
    const cases = [
      { input: "hello\nworld", expected: "\"hello\\nworld\"" },
      { input: "hello\nworld", options: { singleQuotes: true }, expected: "'hello\\nworld'" }
    ];

    for (const { input, options, expected } of cases) {
      expect(genString(input, options)).toBe(expected);
    }
  });
});

describe("escapeString", () => {
  it("should return the string as is if no escape needed", () => {
    const cases = [
      { input: "hello", expected: "hello" },
      { input: "foo_bar-123", expected: "foo_bar-123" },
      { input: "", expected: "" }
    ];

    for (const { input, expected } of cases) {
      expect(escapeString(input)).toBe(expected);
    }
  });

  it("should escape special characters like newlines, quotes, backslashes, and line separators", () => {
    const cases = [
      { input: "line1\nline2", expected: "line1\\nline2" },
      { input: "line1\rline2", expected: "line1\\rline2" },
      { input: "it's fine", expected: "it\\'s fine" },
      { input: "back\\slash", expected: "back\\\\slash" },
      { input: "sep\u2028Line", expected: "sep\\u2028Line" },
      { input: "sep\u2029Line", expected: "sep\\u2029Line" }
    ];

    for (const { input, expected } of cases) {
      expect(escapeString(input)).toBe(expected);
    }
  });
});

describe("genSafeVariableName", () => {
  it("should prefix reserved names with an underscore", () => {
    const cases = [
      { input: "class", expected: "_class" },
      { input: "return", expected: "_return" },
      { input: "undefined", expected: "_undefined" },
      { input: "null", expected: "_null" }
    ];

    for (const { input, expected } of cases) {
      expect(genSafeVariableName(input)).toBe(expected);
    }
  });

  it("should prefix names starting with a digit with an underscore", () => {
    const cases = [
      { input: "123var", expected: "_123var" },
      { input: "0", expected: "_0" }
    ];

    for (const { input, expected } of cases) {
      expect(genSafeVariableName(input)).toBe(expected);
    }
  });

  it("should replace non-word characters with their char code prefixed by an underscore", () => {
    const cases = [
      { input: "foo-bar", expected: "foo_45bar" },
      { input: "foo.bar", expected: "foo_46bar" },
      { input: "foo bar", expected: "foo_32bar" }
    ];

    for (const { input, expected } of cases) {
      expect(genSafeVariableName(input)).toBe(expected);
    }
  });

  it("should return valid variable names unchanged", () => {
    const cases = [
      { input: "fooBar", expected: "fooBar" },
      { input: "_private", expected: "_private" }
    ];

    for (const { input, expected } of cases) {
      expect(genSafeVariableName(input)).toBe(expected);
    }
  });
});
```
