```ts
import { describe, it, expect } from "vitest";
import { $URL, createURL } from "../../src/url";

describe("$URL", () => {
  it("should parse absolute URL correctly", () => {
    const url = new $URL("https://example.com/path?foo=bar#hash");
    expect(url.protocol).toBe("https:");
    expect(url.host).toBe("example.com");
    expect(url.pathname).toBe("/path");
    expect(url.query).toEqual({ foo: "bar" });
    expect(url.hash).toBe("#hash");
    expect(url.hostname).toBe("example.com");
    expect(url.port).toBe("");
    expect(url.hasProtocol).toBe(6);
    expect(url.isAbsolute).toBe(true);
    expect(url.search).toBe("?foo=bar");
    expect(url.origin).toBe("https://example.com");
    expect(url.href).toBe("https://example.com/path?foo=bar#hash");
    expect(url.toString()).toBe("https://example.com/path?foo=bar#hash");
    expect(url.toJSON()).toBe("https://example.com/path?foo=bar#hash");
  });

  it("should handle default empty input", () => {
    const url = new $URL();
    expect(url.protocol).toBe("");
    expect(url.host).toBe("");
    expect(url.pathname).toBe("");
    expect(url.query).toEqual({});
    expect(url.hash).toBe("");
    expect(url.href).toBe("");
  });

  it("should throw TypeError for non-string input", () => {
    expect(() => new $URL(123 as any)).toThrow(TypeError);
    expect(() => new $URL(123 as any)).toThrow("URL input should be string received number (123)");
  });

  it("should parse auth, port, username, password correctly", () => {
    const url = new $URL("http://user:pass@localhost:8080/path");
    expect(url.username).toBe("user");
    expect(url.password).toBe("pass");
    expect(url.port).toBe("8080");
    expect(url.hostname).toBe("localhost");
    expect(url.host).toBe("localhost:8080");
    expect(url.auth).toBe("user:pass");
    expect(url.encodedAuth).toBe("user:pass");
    expect(url.href).toBe("http://user:pass@localhost:8080/path");
  });

  it("should handle searchParams getter correctly", () => {
    const url = new $URL("https://example.com/?a=1&b=2");
    const params = url.searchParams;
    expect(params.get("a")).toBe("1");
    expect(params.get("b")).toBe("2");
  });

  it("should handle array query parameters in searchParams", () => {
    const url = new $URL("https://example.com/?tag=js&tag=ts");
    const params = url.searchParams;
    expect(params.getAll("tag")).toEqual(["js", "ts"]);
  });

  it("should append a URL without protocol correctly", () => {
    const base = new $URL("https://example.com/base/");
    const extra = new $URL("subpath?a=2#newhash");

    base.append(extra);

    expect(base.pathname).toBe("/base/subpath");
    expect(base.query).toEqual({ a: "2" });
    expect(base.hash).toBe("#newhash");
    expect(base.href).toBe("https://example.com/base/subpath?a=2#newhash");
  });

  it("should throw error when appending a URL with protocol", () => {
    const base = new $URL("https://example.com/base");
    const extra = new $URL("https://other.com/path");

    expect(() => base.append(extra)).toThrow("Cannot append a URL with protocol");
  });
});

describe("createURL", () => {
  it("should create a $URL instance", () => {
    const url = createURL("https://example.com/foo");
    expect(url).toBeInstanceOf($URL);
    expect(url.href).toBe("https://example.com/foo");
  });
});
```
