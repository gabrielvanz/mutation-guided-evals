```ts
import { describe, it, expect } from "vitest";
import { defu, createDefu, defuFn, defuArrayFn } from "../../src/defu";

describe("defu", () => {
  it("should merge basic objects with defaults", () => {
    const base = { a: 1, c: 3 };
    const defaults = { a: 2, b: 2 };
    const result = defu(base, defaults);

    expect(result).toEqual({ a: 1, b: 2, c: 3 });
  });

  it("should handle multiple arguments in correct priority order", () => {
    const result = defu({ a: 1 }, { a: 2, b: 2 }, { b: 3, c: 3 });

    expect(result).toEqual({ a: 1, b: 2, c: 3 });
  });

  it("should ignore null and undefined values in base object", () => {
    const base = { a: null, b: undefined, c: 1 };
    const defaults = { a: 10, b: 20, c: 30 };
    const result = defu(base, defaults);

    expect(result).toEqual({ a: 10, b: 20, c: 1 });
  });

  it("should recursively merge plain objects", () => {
    const base = {
      nested: { a: 1, c: 3 }
    };
    const defaults = {
      nested: { a: 2, b: 2 }
    };
    const result = defu(base, defaults);

    expect(result).toEqual({
      nested: { a: 1, b: 2, c: 3 }
    });
  });

  it("should concatenate arrays", () => {
    const base = { arr: [3, 4] };
    const defaults = { arr: [1, 2] };
    const result = defu(base, defaults);

    expect(result).toEqual({ arr: [3, 4, 1, 2] });
  });

  it("should protect against prototype pollution", () => {
    const base = JSON.parse('{"__proto__": {"polluted": true}, "constructor": {"prototype": {"polluted": true}}, "a": 1}');
    const defaults = { a: 2 };
    const result = defu(base, defaults);

    expect(result).toEqual({ a: 1 });
    expect(({} as any).polluted).toBeUndefined();
  });

  it("should fallback to empty object if defaults is not a plain object", () => {
    const base = { a: 1 };
    const result = defu(base, null as any);

    expect(result).toEqual({ a: 1 });
  });
});

describe("createDefu", () => {
  it("should allow custom mergers", () => {
    const custom = createDefu((object, key, value) => {
      if (key === "special") {
        object[key] = `custom_${value}`;
        return true;
      }
    });

    const result = custom({ special: "value" }, { special: "default" });

    expect(result).toEqual({ special: "custom_value" });
  });
});

describe("defuFn", () => {
  it("should call functions when merging existing keys", () => {
    const base = {
      callback: (val: number) => val + 1
    };
    const defaults = {
      callback: 10
    };
    const result = defuFn(base, defaults);

    expect(result).toEqual({
      callback: 11
    });
  });

  it("should not call functions if key does not exist in object", () => {
    const base = {};
    const defaults = {
      callback: (val: number) => val + 1
    };
    const result = defuFn(base, defaults);

    expect(typeof result.callback).toBe("function");
  });
});

describe("defuArrayFn", () => {
  it("should call functions only when merging defined arrays", () => {
    const base = {
      arr: (val: number[]) => [...val, 3]
    };
    const defaults = {
      arr: [1, 2]
    };
    const result = defuArrayFn(base, defaults);

    expect(result).toEqual({
      arr: [1, 2, 3]
    });
  });

  it("should not call functions if target is not an array", () => {
    const base = {
      arr: (val: string) => `${val}_modified`
    };
    const defaults = {
      arr: "not_an_array"
    };
    const result = defuArrayFn(base, defaults);

    expect(typeof result.arr).toBe("function");
  });
});
```
