```ts
import { describe, it, expect } from "vitest";
import {
  parseURL,
  parsePath,
  parseAuth,
  parseHost,
  stringifyParsedURL,
  parseFilename
} from "../../src/parse";

describe("parseURL", () => {
  it("should parse absolute URLs with protocol, host, search, and hash", () => {
    const result = parseURL("http://foo.com/foo?test=123#token");
    expect(result.protocol).toBe("http:");
    expect(result.auth).toBe("");
    expect(result.host).toBe("foo.com");
    expect(result.pathname).toBe("/foo");
    expect(result.search).toBe("?test=123");
    expect(result.hash).toBe("#token");
  });

  it("should parse relative URLs without protocol", () => {
    const result = parseURL("foo.com/foo?test=123#token");
    expect(result).toEqual({
      pathname: "foo.com/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should parse relative URLs with a default protocol", () => {
    const result = parseURL("foo.com/foo?test=123#token", "https://");
    expect(result.protocol).toBe("https:");
    expect(result.auth).toBe("");
    expect(result.host).toBe("foo.com");
    expect(result.pathname).toBe("/foo");
    expect(result.search).toBe("?test=123");
    expect(result.hash).toBe("#token");
  });

  it("should handle special protocols like blob, data, javascript, and vbscript", () => {
    const result = parseURL("data:text/plain;base64,SGVsbG8=");
    expect(result).toEqual({
      protocol: "data:",
      pathname: "text/plain;base64,SGVsbG8=",
      href: "data:text/plain;base64,SGVsbG8=",
      auth: "",
      host: "",
      search: "",
      hash: ""
    });
  });

  it("should handle empty input", () => {
    const result = parseURL("");
    expect(result).toEqual({
      pathname: "",
      search: "",
      hash: ""
    });
  });

  it("should parse file protocol correctly", () => {
    const result = parseURL("file:///C:/path/to/file");
    expect(result.protocol).toBe("file:");
    expect(result.pathname).toBe("C:/path/to/file");
  });

  it("should parse URLs with authentication", () => {
    const result = parseURL("https://user:pass@example.com/path");
    expect(result.protocol).toBe("https:");
    expect(result.auth).toBe("user:pass");
    expect(result.host).toBe("example.com");
    expect(result.pathname).toBe("/path");
    expect(result.search).toBe("");
    expect(result.hash).toBe("");
  });
});

describe("parsePath", () => {
  it("should split path, search, and hash", () => {
    const result = parsePath("http://foo.com/foo?test=123#token");
    expect(result).toEqual({
      pathname: "http://foo.com/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should handle empty input", () => {
    const result = parsePath("");
    expect(result).toEqual({
      pathname: "",
      search: "",
      hash: ""
    });
  });
});

describe("parseAuth", () => {
  it("should parse username and password", () => {
    const result = parseAuth("admin:secret");
    expect(result).toEqual({
      username: "admin",
      password: "secret"
    });
  });

  it("should handle empty input", () => {
    const result = parseAuth("");
    expect(result).toEqual({
      username: "",
      password: ""
    });
  });

  it("should decode percent-encoded auth strings", () => {
    const result = parseAuth("foo%40bar:baz%2Fqux");
    expect(result).toEqual({
      username: "foo@bar",
      password: "baz/qux"
    });
  });
});

describe("parseHost", () => {
  it("should parse hostname and port", () => {
    const result = parseHost("foo.com:8080");
    expect(result).toEqual({
      hostname: "foo.com",
      port: "8080"
    });
  });

  it("should handle host without port", () => {
    const result = parseHost("foo.com");
    expect(result).toEqual({
      hostname: "foo.com",
      port: undefined
    });
  });

  it("should handle empty input", () => {
    const result = parseHost("");
    expect(result).toEqual({
      hostname: "",
      port: undefined
    });
  });
});

describe("stringifyParsedURL", () => {
  it("should stringify a parsed URL object back to a string", () => {
    const parsed = {
      protocol: "http:",
      host: "foo.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token",
      auth: "user:pass"
    };
    const result = stringifyParsedURL(parsed);
    expect(result).toBe("http://user:pass@foo.com/foo?test=123#token");
  });

  it("should handle minimal parsed objects", () => {
    const parsed = {
      pathname: "/foo"
    };
    const result = stringifyParsedURL(parsed);
    expect(result).toBe("/foo");
  });

  it("should add search prefix if missing", () => {
    const parsed = {
      pathname: "/foo",
      search: "test=123"
    };
    const result = stringifyParsedURL(parsed);
    expect(result).toBe("/foo?test=123");
  });
});

describe("parseFilename", () => {
  it("should return the filename from a URL", () => {
    const result = parseFilename("http://example.com/path/to/filename.ext");
    expect(result).toBe("filename.ext");
  });

  it("should return undefined in strict mode for files without extension", () => {
    const result = parseFilename("/path/to/.hidden-file", { strict: true });
    expect(result).toBeUndefined();
  });

  it("should return filename in strict mode when extension exists", () => {
    const result = parseFilename("/path/to/filename.ext", { strict: true });
    expect(result).toBe("filename.ext");
  });

  it("should handle empty input", () => {
    const result = parseFilename("");
    expect(result).toBeUndefined();
  });
});
```
