```ts
import { describe, it, expect } from "vitest";
import {
  parseURL,
  parsePath,
  parseAuth,
  parseHost,
  stringifyParsedURL,
  parseFilename
} from "../../src/parse";

describe("parseURL", () => {
  it("should parse standard absolute URLs", () => {
    const result = parseURL("http://foo.com/foo?test=123#token");
    expect(result).toMatchObject({
      protocol: "http:",
      auth: "",
      host: "foo.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should parse relative URLs as paths when no default protocol is provided", () => {
    const result = parseURL("foo.com/foo?test=123#token");
    expect(result).toEqual({
      pathname: "foo.com/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should parse relative URLs using default protocol", () => {
    const result = parseURL("foo.com/foo?test=123#token", "https://");
    expect(result).toMatchObject({
      protocol: "https:",
      auth: "",
      host: "foo.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should handle empty input", () => {
    const result = parseURL("");
    expect(result).toEqual({
      pathname: "",
      search: "",
      hash: ""
    });
  });

  it("should parse special protocols like blob, data, javascript, vbscript", () => {
    const result = parseURL("data:text/plain;base64,SGVsbG8=");
    expect(result).toEqual({
      protocol: "data:",
      pathname: "text/plain;base64,SGVsbG8=",
      href: "data:text/plain;base64,SGVsbG8=",
      auth: "",
      host: "",
      search: "",
      hash: ""
    });
  });

  it("should parse URLs with basic authentication", () => {
    const result = parseURL("http://user:pass@foo.com/path");
    expect(result).toMatchObject({
      protocol: "http:",
      auth: "user:pass",
      host: "foo.com",
      pathname: "/path",
      search: "",
      hash: ""
    });
  });

  it("should handle file protocol paths correctly", () => {
    const result = parseURL("file:///C:/path/to/file");
    expect(result).toMatchObject({
      protocol: "file:",
      auth: "",
      host: "",
      pathname: "C:/path/to/file",
      search: "",
      hash: ""
    });
  });
});

describe("parsePath", () => {
  it("should split a full URL path string into pathname, search, and hash", () => {
    const result = parsePath("http://foo.com/foo?test=123#token");
    expect(result).toEqual({
      pathname: "http://foo.com/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should handle empty path input", () => {
    const result = parsePath("");
    expect(result).toEqual({
      pathname: "",
      search: "",
      hash: ""
    });
  });
});

describe("parseAuth", () => {
  it("should parse username and password from auth string", () => {
    const result = parseAuth("user:pass");
    expect(result).toEqual({
      username: "user",
      password: "pass"
    });
  });

  it("should handle empty auth input", () => {
    const result = parseAuth("");
    expect(result).toEqual({
      username: "",
      password: ""
    });
  });
});

describe("parseHost", () => {
  it("should parse hostname and port from host string", () => {
    const result = parseHost("foo.com:8080");
    expect(result).toEqual({
      hostname: "foo.com",
      port: "8080"
    });
  });

  it("should handle host without port", () => {
    const result = parseHost("foo.com");
    expect(result).toEqual({
      hostname: "foo.com",
      port: undefined
    });
  });

  it("should handle empty host input", () => {
    const result = parseHost("");
    expect(result).toEqual({
      hostname: "",
      port: undefined
    });
  });
});

describe("stringifyParsedURL", () => {
  it("should stringify a parsed URL object back to a string", () => {
    const parsed = parseURL("http://foo.com/foo?test=123#token");
    const stringified = stringifyParsedURL(parsed);
    expect(stringified).toBe("http://foo.com/foo?test=123#token");
  });

  it("should stringify partial parsed URL objects", () => {
    const stringified = stringifyParsedURL({
      host: "bar.com",
      pathname: "/foo",
      protocol: "http:"
    });
    expect(stringified).toBe("http://bar.com/foo");
  });

  it("should handle empty parsed object", () => {
    const stringified = stringifyParsedURL({});
    expect(stringified).toBe("");
  });
});

describe("parseFilename", () => {
  it("should return the last segment of a path as filename", () => {
    const filename = parseFilename("http://example.com/path/to/filename.ext");
    expect(filename).toBe("filename.ext");
  });

  it("should return undefined in strict mode for files without extension", () => {
    const filename = parseFilename("/path/to/.hidden-file", { strict: true });
    expect(filename).toBeUndefined();
  });

  it("should return filename in strict mode when extension exists", () => {
    const filename = parseFilename("/path/to/file.txt", { strict: true });
    expect(filename).toBe("file.txt");
  });

  it("should handle empty input for filename parsing", () => {
    const filename = parseFilename("");
    expect(filename).toBeUndefined();
  });
});
```
