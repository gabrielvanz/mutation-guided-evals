```ts
import { describe, it, expect } from "vitest";
import {
  parseURL,
  parsePath,
  parseAuth,
  parseHost,
  stringifyParsedURL,
  parseFilename
} from "../../src/parse";

describe("parseURL", () => {
  it("should parse absolute URLs with protocol, host, pathname, search, and hash", () => {
    const result = parseURL("http://foo.com/foo?test=123#token");
    expect(result).toEqual({
      protocol: "http:",
      auth: "",
      host: "foo.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should parse relative URLs without protocol", () => {
    const result = parseURL("foo.com/foo?test=123#token");
    expect(result).toMatchObject({
      pathname: "foo.com/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should apply default protocol when input lacks one", () => {
    const result = parseURL("foo.com/foo?test=123#token", "https://");
    expect(result).toEqual({
      protocol: "https:",
      auth: "",
      host: "foo.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should handle empty string input", () => {
    const result = parseURL("");
    expect(result).toEqual({
      pathname: "",
      search: "",
      hash: ""
    });
  });

  it("should handle special protocols like blob:", () => {
    const result = parseURL("blob:https://example.com/uuid");
    expect(result).toEqual({
      protocol: "blob:",
      pathname: "https://example.com/uuid",
      href: "blob:https://example.com/uuid",
      auth: "",
      host: "",
      search: "",
      hash: ""
    });
  });

  it("should parse URLs with authentication credentials", () => {
    const result = parseURL("http://user:pass@foo.com/foo");
    expect(result).toEqual({
      protocol: "http:",
      auth: "user:pass",
      host: "foo.com",
      pathname: "/foo",
      search: "",
      hash: ""
    });
  });

  it("should handle file protocol paths correctly", () => {
    const result = parseURL("file:///C:/path/to/file");
    expect(result).toEqual({
      protocol: "file:",
      auth: "",
      host: "",
      pathname: "/C:/path/to/file",
      search: "",
      hash: ""
    });
  });
});

describe("parsePath", () => {
  it("should split path into pathname, search, and hash", () => {
    const result = parsePath("http://foo.com/foo?test=123#token");
    expect(result).toEqual({
      pathname: "http://foo.com/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should handle empty path input", () => {
    const result = parsePath("");
    expect(result).toEqual({
      pathname: "",
      search: "",
      hash: ""
    });
  });
});

describe("parseAuth", () => {
  it("should parse username and password and decode them", () => {
    const result = parseAuth("user:pass");
    expect(result).toEqual({
      username: "user",
      password: "pass"
    });
  });

  it("should handle empty auth input", () => {
    const result = parseAuth("");
    expect(result).toEqual({
      username: "",
      password: ""
    });
  });
});

describe("parseHost", () => {
  it("should parse hostname and port from host string", () => {
    const result = parseHost("foo.com:8080");
    expect(result).toEqual({
      hostname: "foo.com",
      port: "8080"
    });
  });

  it("should handle host string without port", () => {
    const result = parseHost("foo.com");
    expect(result).toEqual({
      hostname: "foo.com",
      port: undefined
    });
  });

  it("should handle empty host input", () => {
    const result = parseHost("");
    expect(result).toEqual({
      hostname: "",
      port: undefined
    });
  });
});

describe("stringifyParsedURL", () => {
  it("should stringify a parsed URL object back to a string", () => {
    const parsed = parseURL("http://foo.com/foo?test=123#token");
    const result = stringifyParsedURL(parsed);
    expect(result).toBe("http://foo.com/foo?test=123#token");
  });

  it("should handle partial parsed objects without protocol or host", () => {
    const result = stringifyParsedURL({
      pathname: "/foo",
      search: "test=123",
      hash: "#token"
    });
    expect(result).toBe("/foo?test=123#token");
  });

  it("should include auth when present", () => {
    const result = stringifyParsedURL({
      protocol: "http:",
      auth: "user:pass",
      host: "foo.com",
      pathname: "/foo",
      search: "",
      hash: ""
    });
    expect(result).toBe("http://user:pass@foo.com/foo");
  });
});

describe("parseFilename", () => {
  it("should return the last segment of the path as filename", () => {
    const filename = parseFilename("http://example.com/path/to/filename.ext");
    expect(filename).toBe("filename.ext");
  });

  it("should return undefined in strict mode when filename has no extension", () => {
    const filename = parseFilename("/path/to/.hidden-file", { strict: true });
    expect(filename).toBeUndefined();
  });

  it("should return filename in strict mode when it has an extension", () => {
    const filename = parseFilename("/path/to/file.txt", { strict: true });
    expect(filename).toBe("file.txt");
  });

  it("should return undefined for empty input", () => {
    const filename = parseFilename("");
    expect(filename).toBeUndefined();
  });
});
```
