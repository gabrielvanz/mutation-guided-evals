```ts
import { describe, it, expect } from "vitest";
import defuDefault, { defu, createDefu, defuFn, defuArrayFn } from "../../src/defu";

describe("defu", () => {
  it("exports defu as both named and default export", () => {
    expect(defu).toBe(defuDefault);
  });

  it("merges basic objects with priority to earlier arguments", () => {
    const result = defu({ a: 1, b: 2 }, { b: 3, c: 4 });
    expect(result).toEqual({ a: 1, b: 2, c: 4 });
  });

  it("supports multiple arguments", () => {
    const result = defu({ a: 1 }, { a: 2, b: 2 }, { b: 3, c: 3 });
    expect(result).toEqual({ a: 1, b: 2, c: 3 });
  });

  it("deeply merges nested plain objects", () => {
    const result = defu(
      { nested: { a: 1, deep: { x: 10 } } },
      { nested: { a: 2, b: 3, deep: { x: 20, y: 30 } } },
    );
    expect(result).toEqual({
      nested: { a: 1, b: 3, deep: { x: 10, y: 30 } },
    });
  });

  it("concatenates arrays with base values first", () => {
    const result = defu({ items: [1, 2] }, { items: [3, 4] });
    expect(result).toEqual({ items: [1, 2, 3, 4] });
  });

  it("ignores null and undefined values in base object", () => {
    const result = defu(
      { a: null, b: undefined, c: 0, d: false, e: "" },
      { a: "defaultA", b: "defaultB", c: 1, d: true, e: "defaultE" },
    );
    expect(result).toEqual({
      a: "defaultA",
      b: "defaultB",
      c: 0,
      d: false,
      e: "",
    });
  });

  it("handles non-plain-object defaults gracefully", () => {
    expect(defu({ a: 1 }, null as any)).toEqual({ a: 1 });
    expect(defu({ a: 1 }, undefined as any)).toEqual({ a: 1 });
    expect(defu({ a: 1 }, "string" as any)).toEqual({ a: 1 });
    expect(defu({ a: 1 }, [1, 2] as any)).toEqual({ a: 1 });
  });

  it("ignores __proto__ and constructor keys", () => {
    const malicious = JSON.parse('{"__proto__": {"polluted": true}, "a": 1}');
    const result = defu(malicious, { a: 2, b: 3 });
    expect(result).toEqual({ a: 1, b: 3 });
    expect(({} as any).polluted).toBeUndefined();

    const withConstructor = { constructor: { prototype: { polluted: true } }, a: 1 };
    const result2 = defu(withConstructor, { a: 2 });
    expect(result2.a).toBe(1);
    expect(result2.constructor).toBe(Object);
  });

  it("overrides non-matching types instead of merging", () => {
    expect(defu({ a: [1, 2] }, { a: { b: 1 } })).toEqual({ a: [1, 2] });
    expect(defu({ a: { b: 1 } }, { a: [1, 2] })).toEqual({ a: { b: 1 } });
    expect(defu({ a: "hello" }, { a: { b: 1 } })).toEqual({ a: "hello" });
  });

  it("does not mutate input objects", () => {
    const base = { nested: { a: 1 }, list: [1] };
    const defaults = { nested: { a: 2, b: 3 }, list: [2] };
    const result = defu(base, defaults);

    expect(result).toEqual({ nested: { a: 1, b: 3 }, list: [1, 2] });
    expect(base).toEqual({ nested: { a: 1 }, list: [1] });
    expect(defaults).toEqual({ nested: { a: 2, b: 3 }, list: [2] });
  });
});

describe("createDefu", () => {
  it("accepts a custom merger that can override merge behavior", () => {
    const customDefu = createDefu((obj, key, value) => {
      if (typeof obj[key] === "number" && typeof value === "number") {
        obj[key] = obj[key] + value;
        return true;
      }
    });

    const result = customDefu({ count: 5, name: "a" }, { count: 10, name: "b" });
    expect(result).toEqual({ count: 15, name: "a" });
  });

  it("passes the nested namespace to the custom merger", () => {
    const namespaces: string[] = [];
    const customDefu = createDefu((_obj, key, _value, namespace) => {
      namespaces.push(`${namespace}:${key}`);
    });

    customDefu(
      { a: { b: { c: 1 } } },
      { a: { b: { c: 2 } } },
    );

    expect(namespaces).toContain(":a");
    expect(namespaces).toContain("a:b");
    expect(namespaces).toContain("a.b:c");
  });
});

describe("defuFn", () => {
  it("calls function values with the default value when default is defined", () => {
    const result = defuFn(
      {
        count: (val: number) => val * 2,
        items: (arr: number[]) => [...arr, 3],
      },
      {
        count: 5,
        items: [1, 2],
      },
    );

    expect(result).toEqual({
      count: 10,
      items: [1, 2, 3],
    });
  });

  it("assigns the function directly when default value is undefined", () => {
    const fn = (x: number) => x + 1;
    const result = defuFn({ fn }, {});
    expect(result.fn).toBe(fn);
  });
});

describe("defuArrayFn", () => {
  it("calls function values only when the default value is an array", () => {
    const nonArrayFn = () => "not-called";
    const result = defuArrayFn(
      {
        items: (arr: string[]) => arr.map((s) => s.toUpperCase()),
        other: nonArrayFn,
      },
      {
        items: ["a", "b"],
        other: "default",
      },
    );

    expect(result.items).toEqual(["A", "B"]);
    expect(result.other).toBe(nonArrayFn);
  });
});
```
