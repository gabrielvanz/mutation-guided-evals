```ts
import { describe, it, expect } from "vitest";
import { parseQuery, encodeQueryItem, stringifyQuery } from "../../src/query";

describe("parseQuery", () => {
  it("should parse simple query strings", () => {
    const cases = [
      { input: "foo=bar&baz=qux", expected: { foo: "bar", baz: "qux" } },
      { input: "?foo=bar&baz=qux", expected: { foo: "bar", baz: "qux" } },
      { input: "", expected: {} },
      { input: "?", expected: {} },
    ];

    for (const { input, expected } of cases) {
      expect(parseQuery(input)).toEqual(expected);
    }
  });

  it("should handle arrays and repeated keys", () => {
    const input = "tags=javascript&tags=web&tags=dev";
    const expected = { tags: ["javascript", "web", "dev"] };
    expect(parseQuery(input)).toEqual(expected);
  });

  it("should ignore prototype pollution keys", () => {
    const input = "__proto__=bar&constructor=baz&safe=value";
    const expected = { safe: "value" };
    expect(parseQuery(input)).toEqual(expected);
  });

  it("should handle empty values and missing equal signs", () => {
    const cases = [
      { input: "foo=&bar", expected: { foo: "", bar: "" } },
      { input: "foo", expected: { foo: "" } },
    ];

    for (const { input, expected } of cases) {
      expect(parseQuery(input)).toEqual(expected);
    }
  });
});

describe("encodeQueryItem", () => {
  it("should encode single and multiple query items", () => {
    const cases = [
      { key: "message", value: "Hello World", expected: "message=Hello+World" },
      { key: "tags", value: ["javascript", "web"], expected: "tags=javascript&tags=web" },
      { key: "num", value: 123, expected: "num=123" },
      { key: "bool", value: true, expected: "bool=true" },
      { key: "empty", value: "", expected: "empty" },
      { key: "nullVal", value: null, expected: "nullVal" },
    ];

    for (const { key, value, expected } of cases) {
      expect(encodeQueryItem(key, value as any)).toBe(expected);
    }
  });
});

describe("stringifyQuery", () => {
  it("should stringify query objects correctly", () => {
    const cases = [
      { input: { foo: "bar", baz: "qux" }, expected: "foo=bar&baz=qux" },
      { input: { foo: "bar", baz: undefined }, expected: "foo=bar" },
      { input: { tags: ["javascript", "web"] }, expected: "tags=javascript&tags=web" },
      { input: {}, expected: "" },
    ];

    for (const { input, expected } of cases) {
      expect(stringifyQuery(input)).toBe(expected);
    }
  });
});
```
