```ts
import { describe, it, expect } from "vitest";
import { genString, escapeString, genSafeVariableName } from "../../src/string";

describe("genString", () => {
  it("should generate a double quoted string by default", () => {
    const input = "hello world";
    const result = genString(input);
    expect(result).toBe("\"hello world\"");
  });

  it("should generate a single quoted string when singleQuotes is true", () => {
    const input = "hello world";
    const result = genString(input, { singleQuotes: true });
    expect(result).toBe("'hello world'");
  });

  it("should handle empty strings correctly", () => {
    expect(genString("")).toBe("\"\"");
    expect(genString("", { singleQuotes: true })).toBe("''");
  });

  it("should escape special characters and quotes", () => {
    const input = "line1\nline2 'quoted' \\backslash";
    expect(genString(input)).toBe("\"line1\\nline2 'quoted' \\\\backslash\"");
    expect(genString(input, { singleQuotes: true })).toBe("'line1\\nline2 \\'quoted\\' \\\\backslash'");
  });
});

describe("escapeString", () => {
  const cases = [
    { input: "plain", expected: "plain" },
    { input: "with\nnewline", expected: "with\\nnewline" },
    { input: "with\rreturn", expected: "with\\rreturn" },
    { input: "with'quote", expected: "with\\'quote" },
    { input: "with\\backslash", expected: "with\\\\backslash" },
    { input: "line2028\u2028separator", expected: "line2028\\u2028separator" },
    { input: "line2029\u2029separator", expected: "line2029\\u2029separator" },
  ];

  for (const { input, expected } of cases) {
    it(`should escape string: ${JSON.stringify(input)}`, () => {
      expect(escapeString(input)).toBe(expected);
    });
  }
});

describe("genSafeVariableName", () => {
  it("should return the same name if it is a safe identifier", () => {
    expect(genSafeVariableName("fooBar123")).toBe("fooBar123");
  });

  it("should prefix with underscore if name is a reserved keyword", () => {
    expect(genSafeVariableName("class")).toBe("_class");
    expect(genSafeVariableName("return")).toBe("_return");
  });

  it("should prefix with underscore if name starts with a digit", () => {
    expect(genSafeVariableName("123abc")).toBe("_123abc");
  });

  it("should replace non-word characters with their char code", () => {
    expect(genSafeVariableName("foo-bar")).toBe("foo_45bar");
    expect(genSafeVariableName("a.b")).toBe("a_46b");
  });
});
```
