import { describe, it, expect } from "vitest";
import { defu, createDefu, defuFn, defuArrayFn } from "../../src/defu";

describe("defu", () => {
  it("should merge basic objects", () => {
    const result = defu({ a: 1 }, { b: 2 });
    expect(result).toEqual({ a: 1, b: 2 });
  });

  it("should prioritize base object over defaults", () => {
    const result = defu({ a: 1 }, { a: 2, b: 3 });
    expect(result).toEqual({ a: 1, b: 3 });
  });

  it("should ignore null and undefined values in base object", () => {
    const result = defu({ a: null, b: undefined, c: 3 }, { a: 1, b: 2, c: 1 });
    expect(result).toEqual({ a: 1, b: 2, c: 3 });
  });

  it("should recursively merge nested plain objects with correct default namespace tracking", () => {
    const customMerger = createDefu((object, key, value, namespace) => {
      if (key === "target") {
        object[key] = `${namespace}:${value}`;
        return true;
      }
    });

    const result = customMerger(
      { a: { target: "val" } },
      { a: { target: "default" } }
    );
    expect(result).toEqual({ a: { target: "a:val" } });
  });

  it("should use dot as default namespace when omitted", () => {
    let capturedNamespace = "";
    const customMerger = createDefu((_object, _key, _value, namespace) => {
      capturedNamespace = namespace;
      return true;
    });

    customMerger({ a: 1 }, { a: 2 });
    expect(capturedNamespace).toBe(".");
  });

  it("should concatenate arrays", () => {
    const result = defu({ a: [1, 2] }, { a: [3, 4] });
    expect(result).toEqual({ a: [1, 2, 3, 4] });
  });

  it("should handle multiple arguments", () => {
    const result = defu({ a: 1 }, { b: 2 }, { c: 3 }, { a: 10 });
    expect(result).toEqual({ a: 1, b: 2, c: 3 });
  });

  it("should ignore __proto__ and constructor keys", () => {
    const result = defu(
      JSON.parse('{"__proto__": {"polluted": true}, "constructor": {"polluted": true}, "a": 1}'),
      { a: 2, b: 3 }
    );
    expect(result).toEqual({ a: 1, b: 3 });
    expect(({} as any).polluted).toBeUndefined();
  });

  it("should not allow prototype pollution via __proto__ key", () => {
    const malicious = JSON.parse('{"__proto__": {"polluted": true}}');
    const result = defu(malicious, { a: 1 });
    expect(result).toEqual({ a: 1 });
    expect(({} as any).polluted).toBeUndefined();
  });

  it("should handle non-plain object defaults gracefully by falling back to empty object", () => {
    const result = defu({ a: 1 }, "not-an-object" as any);
    expect(result).toEqual({ a: 1 });
  });
});

describe("createDefu", () => {
  it("should create a custom defu instance with a custom merger", () => {
    const customDefu = createDefu((object, key, value) => {
      if (key === "special") {
        object[key] = `custom_${value}`;
        return true;
      }
    });

    const result = customDefu({ special: "value", normal: 1 }, { special: "default", normal: 2 });
    expect(result).toEqual({ special: "custom_value", normal: 1 });
  });
});

describe("defuFn", () => {
  it("should execute function values passing current value when object[key] is defined", () => {
    const result = defuFn(
      {
        arr: (val: number[]) => [...val, 3],
      },
      {
        arr: [1, 2],
      }
    );
    expect(result).toEqual({ arr: [1, 2, 3] });
  });

  it("should NOT execute function values when object[key] is undefined", () => {
    const result = defuFn(
      {
        arr: (val: number[]) => [...val, 3],
      },
      {
        other: 1,
      }
    );
    expect(result).toEqual({
      arr: expect.any(Function),
      other: 1,
    });
  });

  it("should treat non-function values normally even if object[key] is defined", () => {
    const result = defuFn(
      {
        val: "override",
      },
      {
        val: "original",
      }
    );
    expect(result).toEqual({ val: "override" });
  });
});

describe("defuArrayFn", () => {
  it("should execute function values only when target is an array", () => {
    const result = defuArrayFn(
      {
        arr: (val: number[]) => [...val, 3],
        notArr: (val: string) => `${val}_modified`,
      },
      {
        arr: [1, 2],
        notArr: "original",
      }
    );
    expect(result).toEqual({
      arr: [1, 2, 3],
      notArr: expect.any(Function),
    });
  });

  it("should NOT execute non-function values when target is an array", () => {
    const result = defuArrayFn(
      {
        arr: [5, 6],
      },
      {
        arr: [1, 2],
      }
    );
    expect(result).toEqual({
      arr: [5, 6, 1, 2],
    });
  });
});
