```ts
import { describe, it, expect } from "vitest";
import { $URL, createURL } from "../../src/url";

describe("$URL", () => {
  it("should parse absolute URL with protocol, host, pathname, query and hash", () => {
    const url = new $URL("https://example.com:8080/path/to/page?foo=bar#hash");
    expect(url.protocol).toBe("https:");
    expect(url.host).toBe("example.com:8080");
    expect(url.hostname).toBe("example.com");
    expect(url.port).toBe("8080");
    expect(url.pathname).toBe("/path/to/page");
    expect(url.query).toEqual({ foo: "bar" });
    expect(url.hash).toBe("#hash");
    expect(url.search).toBe("?foo=bar");
    expect(url.href).toBe("https://example.com:8080/path/to/page?foo=bar#hash");
  });

  it("should handle empty input with default values", () => {
    const url = new $URL();
    expect(url.protocol).toBe("");
    expect(url.host).toBe("");
    expect(url.hostname).toBe("");
    expect(url.port).toBe("");
    expect(url.pathname).toBe("");
    expect(url.query).toEqual({});
    expect(url.hash).toBe("");
    expect(url.href).toBe("");
    expect(url.toString()).toBe("");
    expect(url.toJSON()).toBe("");
  });

  it("should throw TypeError for non-string input", () => {
    // @ts-expect-error testing invalid input type
    expect(() => new $URL(123)).toThrow(TypeError);
    // @ts-expect-error testing invalid input type
    expect(() => new $URL(null)).toThrow(
      "URL input should be string received object (null)"
    );
  });

  it("should parse auth, username, and password correctly", () => {
    const url = new $URL("https://user:pass@example.com/");
    expect(url.auth).toBe("user:pass");
    expect(url.username).toBe("user");
    expect(url.password).toBe("pass");
    expect(url.encodedAuth).toBe("user:pass");
    expect(url.href).toBe("https://user:pass@example.com/");
  });

  it("should handle auth without password", () => {
    const url = new $URL("https://user@example.com/");
    expect(url.username).toBe("user");
    expect(url.password).toBe("");
    expect(url.encodedAuth).toBe("user");
  });

  it("should compute hasProtocol and isAbsolute correctly", () => {
    const absUrl = new $URL("https://example.com");
    expect(absUrl.hasProtocol).toBeGreaterThan(0);
    expect(absUrl.isAbsolute).toBe(true);

    const relUrl = new $URL("/foo/bar");
    expect(relUrl.hasProtocol).toBe(0);
    expect(relUrl.isAbsolute).toBe(true);

    const relativePath = new $URL("foo/bar");
    expect(relativePath.hasProtocol).toBe(0);
    expect(relativePath.isAbsolute).toBe(false);
  });

  it("should provide searchParams as URLSearchParams", () => {
    const url = new $URL("https://example.com/?a=1&b=2");
    const params = url.searchParams;
    expect(params.get("a")).toBe("1");
    expect(params.get("b")).toBe("2");
  });

  it("should handle array values in query and searchParams", () => {
    const url = new $URL("https://example.com/?tag=a&tag=b");
    expect(url.query).toEqual({ tag: ["a", "b"] });
    const params = url.searchParams;
    expect(params.getAll("tag")).toEqual(["a", "b"]);
  });

  it("should append a relative URL without protocol", () => {
    const base = new $URL("https://example.com/base/");
    const toAppend = new $URL("sub?x=1#hash");
    base.append(toAppend);

    expect(base.pathname).toBe("/base/sub");
    expect(base.query).toEqual({ x: "1" });
    expect(base.hash).toBe("#hash");
  });

  it("should throw error when appending a URL with protocol", () => {
    const base = new $URL("https://example.com/");
    const other = new $URL("https://other.com/");
    expect(() => base.append(other)).toThrow("Cannot append a URL with protocol");
  });
});

describe("createURL", () => {
  it("should create and return an instance of $URL", () => {
    const url = createURL("https://example.com/test");
    expect(url).toBeInstanceOf($URL);
    expect(url.href).toBe("https://example.com/test");
  });
});
```
