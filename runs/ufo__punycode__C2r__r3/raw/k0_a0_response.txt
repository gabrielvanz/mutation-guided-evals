```ts
import { describe, it, expect } from "vitest";
import { toASCII } from "../../src/punycode";

describe("toASCII", () => {
  it("should return ASCII strings unchanged", () => {
    const cases = [
      { input: "example.com", expected: "example.com" },
      { input: "test", expected: "test" },
      { input: "123.456", expected: "123.456" }
    ];

    for (const { input, expected } of cases) {
      expect(toASCII(input)).toBe(expected);
    }
  });

  it("should convert non-ASCII domain names to punycode", () => {
    const cases = [
      { input: "münchen.de", expected: "xn--mnchen-3ya.de" },
      { input: "ñandu.com", expected: "xn--andu-1wa.com" },
      { input: "شبكة.مصر", expected: "xn--ngbc0gah.xn--wgbh1c" }
    ];

    for (const { input, expected } of cases) {
      expect(toASCII(input)).toBe(expected);
    }
  });

  it("should handle email addresses with non-ASCII domain parts", () => {
    expect(toASCII("user@münchen.de")).toBe("user@xn--mnchen-3ya.de");
  });

  it("should handle various dot separators like ideographic full stop", () => {
    expect(toASCII("example．com")).toBe("example.com");
  });

  it("should throw RangeError on overflow or invalid input", () => {
    expect(() => toASCII("\ud800")).toThrow(RangeError);
  });
});
```
