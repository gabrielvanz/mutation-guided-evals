```ts
import { describe, it, expect } from "vitest";
import {
  sep,
  normalize,
  join,
  resolve,
  normalizeString,
  isAbsolute,
  toNamespacedPath,
  extname,
  relative,
  dirname,
  format,
  basename,
  parse,
  matchesGlob
} from "../../src/_path";

describe("sep", () => {
  it("should be a forward slash", () => {
    expect(sep).toBe("/");
  });
});

describe("normalize", () => {
  it("should normalize empty path to current directory", () => {
    expect(normalize("")).toBe(".");
  });

  it("should normalize empty path with trailing slash if applicable", () => {
    // Tests path.length === 0 branch and trailing slash handling when result is empty
    expect(normalize("")).toBe(".");
  });

  it("should normalize simple paths", () => {
    expect(normalize("foo/bar//baz/..")).toBe("foo/bar");
  });

  it("should handle Windows paths and convert backslashes", () => {
    expect(normalize("foo\\bar\\baz")).toBe("foo/bar/baz");
    expect(normalize("C:\\foo\\bar")).toBe("C:/foo/bar");
  });

  it("should preserve trailing slash", () => {
    expect(normalize("foo/bar/")).toBe("foo/bar/");
    expect(normalize("/foo/")).toBe("/foo/");
  });

  it("should handle UNC paths", () => {
    expect(normalize("\\\\server\\share\\dir")).toBe("//server/share/dir");
    expect(normalize("\\\\foo")).toBe("//foo");
    expect(normalize("\\\\.\\foo")).toBe("//./foo");
  });

  it("should handle drive letter trailing slash additions", () => {
    expect(normalize("C:")).toBe("C:/");
    expect(normalize("c:")).toBe("c:/");
  });
});

describe("join", () => {
  it("should join path segments", () => {
    expect(join("foo", "bar", "baz")).toBe("foo/bar/baz");
  });

  it("should ignore empty segments", () => {
    expect(join("foo", "", "bar")).toBe("foo/bar");
  });

  it("should handle slashes correctly", () => {
    expect(join("/foo", "bar")).toBe("/foo/bar");
    expect(join("foo/", "/bar")).toBe("foo/bar");
    expect(join("foo/", "bar/")).toBe("foo/bar/");
  });

  it("should return dot for empty join", () => {
    expect(join()).toBe(".");
  });

  it("should handle trailing slash in join correctly", () => {
    expect(join("foo/", "bar")).toBe("foo/bar");
  });

  it("should slice leading slash on segment when both path and seg have slashes", () => {
    expect(join("foo/", "/bar")).toBe("foo/bar");
  });
});

describe("resolve", () => {
  it("should resolve absolute paths", () => {
    expect(resolve("/foo", "bar")).toBe("/foo/bar");
  });

  it("should resolve relative segments", () => {
    expect(resolve("foo", "bar", "../baz")).toContain("baz");
  });

  it("should handle empty arguments by using cwd", () => {
    expect(resolve("")).toBeDefined();
    expect(resolve("")).not.toBe("");
  });

  it("should handle UNC paths in resolve", () => {
    expect(resolve("\\\\server\\share")).toBe("//server/share");
  });

  it("should handle multiple segments including empty ones", () => {
    expect(resolve("/a", "", "b")).toBe("/a/b");
  });

  it("should resolve with negative index bounds correctly", () => {
    expect(resolve("", "")).toBeDefined();
  });

  it("should preserve resolvedUNC correctly", () => {
    expect(resolve("\\\\server\\share", "dir")).toBe("//server/share/dir");
  });
});

describe("normalizeString", () => {
  it("should normalize dot segments without allowing above root", () => {
    expect(normalizeString("foo/../bar", false)).toBe("bar");
  });

  it("should allow above root when specified", () => {
    expect(normalizeString("foo/../../bar", true)).toBe("../bar");
  });

  it("should handle empty paths", () => {
    expect(normalizeString("", false)).toBe("");
  });

  it("should handle complex dotdot logic correctly", () => {
    expect(normalizeString("a/b/../c", false)).toBe("a/c");
    expect(normalizeString("..", true)).toBe("..");
    expect(normalizeString("a/b/..", false)).toBe("a");
    expect(normalizeString("a/b/../..", false)).toBe("");
  });

  it("should handle triple dot or complex dot combinations without going above root unnecessarily", () => {
    expect(normalizeString(".../foo", false)).toBe(".../foo");
    expect(normalizeString("a/./b", false)).toBe("a/b");
    expect(normalizeString("a/..b", false)).toBe("a/..b");
  });
});

describe("isAbsolute", () => {
  it("should identify absolute POSIX paths", () => {
    expect(isAbsolute("/foo/bar")).toBe(true);
    expect(isAbsolute("foo/bar")).toBe(false);
  });

  it("should identify absolute Windows paths", () => {
    expect(isAbsolute("C:\\foo")).toBe(true);
    expect(isAbsolute("c:\\foo")).toBe(true);
    expect(isAbsolute("1:\\foo")).toBe(false);
    expect(isAbsolute("\\\\server\\share")).toBe(true);
  });

  it("should validate root folder expressions", () => {
    expect(isAbsolute("/C:")).toBe(true);
    expect(isAbsolute("/")).toBe(true);
    expect(isAbsolute("//server")).toBe(true);
  });
});

describe("toNamespacedPath", () => {
  it("should normalize windows paths for namespacing", () => {
    expect(toNamespacedPath("C:\\foo\\bar")).toBe("C:/foo/bar");
  });
});

describe("extname", () => {
  it("should return the file extension", () => {
    expect(extname("index.html")).toBe(".html");
    expect(extname("file.spec.ts")).toBe(".ts");
  });

  it("should return empty string when no extension", () => {
    expect(extname("README")).toBe("");
    expect(extname(".gitignore")).toBe("");
    expect(extname("..")).toBe("");
  });

  it("should handle dot-only file names", () => {
    expect(extname(".")).toBe("");
    expect(extname("..")).toBe("");
  });
});

describe("relative", () => {
  it("should compute relative path between two paths", () => {
    expect(relative("/data/orandea/test/aaa", "/data/orandea/impl/bbb")).toBe("../../impl/bbb");
  });

  it("should handle identical paths", () => {
    expect(relative("/foo/bar", "/foo/bar")).toBe("");
  });

  it("should handle different drive letters on Windows", () => {
    expect(relative("C:/foo/bar", "D:/foo/bar")).toBe("D:/foo/bar");
  });

  it("should handle relative paths with drive letters and distinct roots correctly", () => {
    expect(relative("C:/foo", "D:/bar")).toBe("D:/bar");
    expect(relative("C:/foo", "C:/bar")).toBe("../bar");
  });
});

describe("dirname", () => {
  it("should return the directory name of a path", () => {
    expect(dirname("/foo/bar/baz.js")).toBe("/foo/bar");
    expect(dirname("/foo/bar///")).toBe("/foo");
    expect(dirname("/foo/bar/")).toBe("/foo");
  });

  it("should return dot or root for root-level paths", () => {
    expect(dirname("/foo")).toBe("/");
    expect(dirname("foo")).toBe(".");
  });

  it("should handle drive letters in dirname", () => {
    expect(dirname("C:foo")).toBe(".");
    expect(dirname("C:/foo/bar")).toBe("C:/foo");
    expect(dirname("C:")).toBe("C:/");
  });
});

describe("format", () => {
  it("should format a path object into a string", () => {
    expect(
      format({
        root: "/",
        dir: "/path/to",
        base: "file.txt"
      })
    ).toBe("/path/to/file.txt");
  });

  it("should format using name and ext if base is missing", () => {
    expect(
      format({
        dir: "/path/to",
        name: "file",
        ext: ".txt"
      })
    ).toBe("/path/to/file.txt");
  });

  it("should prepend dot to ext if missing", () => {
    expect(
      format({
        dir: "/path/to",
        name: "file",
        ext: "txt"
      })
    ).toBe("/path/to/file.txt");
  });

  it("should handle empty or missing ext correctly without fallback artifacts", () => {
    expect(
      format({
        dir: "/path/to",
        name: "file",
        ext: ""
      })
    ).toBe("/path/to/file");
  });

  it("should handle root paths in format", () => {
    expect(
      format({
        root: "C:\\",
        base: "file.txt"
      })
    ).toBe("C:/file.txt");
  });
});

describe("basename", () => {
  it("should return the last portion of a path", () => {
    expect(basename("/foo/bar/baz.html")).toBe("baz.html");
    expect(basename("/foo/bar///")).toBe("bar");
  });

  it("should strip extension if provided", () => {
    expect(basename("/foo/bar/baz.html", ".html")).toBe("baz");
  });

  it("should handle trailing slashes in basename", () => {
    expect(basename("/foo/bar/")).toBe("bar");
  });
});

describe("parse", () => {
  it("should parse a path into an object", () => {
    expect(parse("/home/user/dir/file.txt")).toEqual({
      root: "/",
      dir: "/home/user/dir",
      base: "file.txt",
      ext: ".txt",
      name: "file"
    });
  });

  it("should parse paths without directories", () => {
    expect(parse("file.txt")).toEqual({
      root: "",
      dir: "",
      base: "file.txt",
      ext: ".txt",
      name: "file"
    });
  });

  it("should parse Windows roots correctly", () => {
    expect(parse("C:\\foo\\bar.txt").root).toBe("C:/");
  });

  it("should parse trailing slashes correctly in parse with multiple trailing slashes", () => {
    expect(parse("/foo/bar///")).toEqual({
      root: "/",
      dir: "/foo",
      base: "bar",
      ext: "",
      name: "bar"
    });
  });
});

describe("matchesGlob", () => {
  it("should match path against glob pattern", () => {
    expect(matchesGlob("/foo/bar.js", "*.js")).toBe(false);
    expect(matchesGlob("/foo/bar.js", "**/bar.js")).toBe(true);
    expect(matchesGlob("/foo/bar.js", ["*.txt", "**/*.js"])).toBe(true);
  });
});
```
