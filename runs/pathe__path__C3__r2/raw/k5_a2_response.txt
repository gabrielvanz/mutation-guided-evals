```ts
import { describe, it, expect } from "vitest";
import {
  sep,
  normalize,
  join,
  resolve,
  normalizeString,
  isAbsolute,
  toNamespacedPath,
  extname,
  relative,
  dirname,
  format,
  basename,
  parse,
  matchesGlob
} from "../../src/_path";

describe("sep", () => {
  it("should be a forward slash", () => {
    expect(sep).toBe("/");
  });
});

describe("normalize", () => {
  it("should normalize empty path to current directory", () => {
    expect(normalize("")).toBe(".");
  });

  it("should normalize simple paths", () => {
    expect(normalize("foo/bar//baz/..")).toBe("foo/bar");
  });

  it("should handle Windows paths and convert backslashes", () => {
    expect(normalize("foo\\bar\\baz")).toBe("foo/bar/baz");
    expect(normalize("C:\\foo\\bar")).toBe("C:/foo/bar");
    expect(normalize("C:")).toBe("C:/");
  });

  it("should preserve trailing slash", () => {
    expect(normalize("foo/bar/")).toBe("foo/bar/");
    expect(normalize("/foo/")).toBe("/foo/");
  });

  it("should handle UNC paths", () => {
    expect(normalize("\\\\server\\share\\dir")).toBe("//server/share/dir");
    expect(normalize("\\\\server\\share")).toBe("//server/share");
  });

  it("should normalize absolute paths correctly when result is root", () => {
    expect(normalize("/foo/..")).toBe("/");
  });
});

describe("join", () => {
  it("should join path segments", () => {
    expect(join("foo", "bar", "baz")).toBe("foo/bar/baz");
  });

  it("should ignore empty segments", () => {
    expect(join("foo", "", "bar")).toBe("foo/bar");
  });

  it("should handle slashes correctly", () => {
    expect(join("/foo", "bar")).toBe("/foo/bar");
    expect(join("foo/", "/bar")).toBe("foo/bar");
  });

  it("should return dot for empty join", () => {
    expect(join()).toBe(".");
  });
});

describe("resolve", () => {
  it("should resolve absolute paths", () => {
    expect(resolve("/foo", "bar")).toBe("/foo/bar");
  });

  it("should resolve relative segments", () => {
    expect(resolve("foo", "bar", "../baz")).toContain("baz");
  });

  it("should handle empty arguments by using cwd", () => {
    expect(resolve("")).toBeDefined();
  });

  it("should handle multiple arguments including root and cwd fallback", () => {
    expect(resolve("foo", "/bar")).toBe("/bar");
  });
});

describe("normalizeString", () => {
  it("should normalize dot segments without allowing above root", () => {
    expect(normalizeString("foo/../bar", false)).toBe("bar");
  });

  it("should allow above root when specified", () => {
    expect(normalizeString("../foo", true)).toBe("../foo");
  });

  it("should handle empty string", () => {
    expect(normalizeString("", false)).toBe("");
  });

  it("should handle single dot and double dot combinations precisely", () => {
    expect(normalizeString("a/./b/../c", false)).toBe("a/c");
    expect(normalizeString("..", false)).toBe("");
    expect(normalizeString(".", false)).toBe("");
    expect(normalizeString("a/..", false)).toBe("");
    expect(normalizeString("a/b/../../c", false)).toBe("c");
  });
});

describe("isAbsolute", () => {
  it.each([
    { input: "/foo/bar", expected: true },
    { input: "C:/foo/bar", expected: true },
    { input: "C:\\foo\\bar", expected: true },
    { input: "//server/share", expected: true },
    { input: "foo/bar", expected: false },
    { input: "", expected: false }
  ])("should return $expected for $input", ({ input, expected }) => {
    expect(isAbsolute(input)).toBe(expected);
  });
});

describe("toNamespacedPath", () => {
  it("should normalize windows path for namespaced path", () => {
    expect(toNamespacedPath("C:\\foo\\bar")).toBe("C:/foo/bar");
  });
});

describe("extname", () => {
  it.each([
    { input: "index.html", expected: ".html" },
    { input: "index.coffee.md", expected: ".md" },
    { input: "index.", expected: "." },
    { input: "index", expected: "" },
    { input: ".gitignore", expected: "" },
    { input: "..", expected: "" },
    { input: "", expected: "" }
  ])("should return extension for $input", ({ input, expected }) => {
    expect(extname(input)).toBe(expected);
  });
});

describe("relative", () => {
  it("should compute relative path between two paths", () => {
    expect(relative("/data/orandea/test/aaa", "/data/orandea/impl/bbb")).toBe("../../impl/bbb");
  });

  it("should handle same path", () => {
    expect(relative("/foo/bar", "/foo/bar")).toBe("");
  });

  it("should return absolute to target if drive letters differ", () => {
    expect(relative("C:/foo/bar", "D:/baz/qux")).toBe("D:/baz/qux");
  });
});

describe("dirname", () => {
  it.each([
    { input: "/foo/bar/baz", expected: "/foo/bar" },
    { input: "/foo/bar/baz/", expected: "/foo/bar" },
    { input: "/foo", expected: "/" },
    { input: "foo", expected: "." },
    { input: "C:/foo/bar", expected: "C:/foo" },
    { input: "C:/", expected: "/" }
  ])("should return dirname for $input", ({ input, expected }) => {
    expect(dirname(input)).toBe(expected);
  });
});

describe("format", () => {
  it("should format path object into string", () => {
    expect(
      format({
        root: "/",
        dir: "/home/user/dir",
        base: "file.txt",
        ext: ".txt",
        name: "file"
      })
    ).toBe("/home/user/dir/file.txt");
  });

  it("should format with extension (dot check)", () => {
    expect(
      format({
        root: "",
        dir: "",
        base: "",
        ext: ".txt",
        name: "file"
      })
    ).toBe(".txt");
    expect(
      format({
        root: "",
        dir: "",
        base: "file.txt",
        ext: ".txt",
        name: "file"
      })
    ).toBe("file.txt");
  });
});

describe("basename", () => {
  it.each([
    { input: "/foo/bar/baz.html", extension: undefined, expected: "baz.html" },
    { input: "/foo/bar/baz.html", extension: ".html", expected: "baz" },
    { input: "/foo/bar/baz.html", extension: "html", expected: "baz." },
    { input: "", extension: undefined, expected: "" },
    { input: "C:\\foo\\bar\\", extension: undefined, expected: "bar" }
  ])("should return basename for $input with ext $extension", ({ input, extension, expected }) => {
    expect(basename(input, extension)).toBe(expected);
  });
});

describe("parse", () => {
  it("should parse path into components", () => {
    const result = parse("/home/user/dir/file.txt");
    expect(result).toEqual({
      root: "/",
      dir: "/home/user/dir",
      base: "file.txt",
      ext: ".txt",
      name: "file"
    });
  });

  it("should parse path without directory", () => {
    const result = parse("file.txt");
    expect(result).toEqual({
      root: "",
      dir: "",
      base: "file.txt",
      ext: ".txt",
      name: "file"
    });
  });
});

describe("matchesGlob", () => {
  it("should match path against glob pattern", () => {
    expect(matchesGlob("foo/bar.js", "*.js")).toBe(false);
    expect(matchesGlob("foo/bar.js", "**/bar.js")).toBe(true);
  });
});
```
