```ts
import { describe, it, expect } from "vitest";
import { $URL, createURL } from "../../src/url";

describe("$URL", () => {
  it("should parse absolute URL with protocol, host, pathname, query and hash", () => {
    const url = new $URL("https://example.com:8080/path/to/page?foo=bar#hash");

    expect(url.protocol).toBe("https:");
    expect(url.host).toBe("example.com:8080");
    expect(url.hostname).toBe("example.com");
    expect(url.port).toBe("8080");
    expect(url.pathname).toBe("/path/to/page");
    expect(url.query).toEqual({ foo: "bar" });
    expect(url.hash).toBe("#hash");
    expect(url.hasProtocol).toBe(6);
    expect(url.isAbsolute).toBe(true);
    expect(url.search).toBe("?foo=bar");
    expect(url.origin).toBe("https://example.com:8080");
    expect(url.fullpath).toBe("/path/to/page?foo=bar#hash");
    expect(url.href).toBe("https://example.com:8080/path/to/page?foo=bar#hash");
    expect(url.toString()).toBe("https://example.com:8080/path/to/page?foo=bar#hash");
    expect(url.toJSON()).toBe("https://example.com:8080/path/to/page?foo=bar#hash");
  });

  it("should handle empty input with default values", () => {
    const url = new $URL();

    expect(url.protocol).toBe("");
    expect(url.host).toBe("");
    expect(url.hostname).toBe("");
    expect(url.port).toBe("");
    expect(url.pathname).toBe("");
    expect(url.query).toEqual({});
    expect(url.hash).toBe("");
    expect(url.hasProtocol).toBe(0);
    expect(url.isAbsolute).toBe(false);
    expect(url.search).toBe("");
    expect(url.origin).toBe("");
    expect(url.fullpath).toBe("");
    expect(url.href).toBe("");
  });

  it("should throw TypeError for non-string input", () => {
    expect(() => new $URL(123 as any)).toThrow(TypeError);
    expect(() => new $URL(123 as any)).toThrow("URL input should be string received number (123)");
  });

  it("should parse auth with username and password", () => {
    const url = new $URL("https://user:pass@example.com/");

    expect(url.auth).toBe("user:pass");
    expect(url.username).toBe("user");
    expect(url.password).toBe("pass");
    expect(url.encodedAuth).toBe("user:pass");
    expect(url.href).toBe("https://user:pass@example.com/");
  });

  it("should parse auth with only username", () => {
    const url = new $URL("https://user@example.com/");

    expect(url.auth).toBe("user");
    expect(url.username).toBe("user");
    expect(url.password).toBe("");
    expect(url.encodedAuth).toBe("user");
  });

  it("should return empty string for encodedAuth when auth is missing", () => {
    const url = new $URL("https://example.com/");

    expect(url.auth).toBe("");
    expect(url.encodedAuth).toBe("");
  });

  it("should generate searchParams correctly including arrays and values", () => {
    const url = new $URL("/test?a=1&a=2&b=hello");
    const params = url.searchParams;

    expect(params.getAll("a")).toEqual(["1", "2"]);
    expect(params.get("b")).toBe("hello");
  });

  it("should append a relative URL correctly", () => {
    const base = new $URL("https://example.com/base/");
    const toAppend = new $URL("sub/path?extra=1#newhash");

    base.append(toAppend);

    expect(base.pathname).toBe("/base/sub/path");
    expect(base.query).toEqual({ extra: "1" });
    expect(base.hash).toBe("#newhash");
  });

  it("should throw error when appending a URL with a protocol", () => {
    const base = new $URL("https://example.com/");
    const toAppend = new $URL("https://other.com/");

    expect(() => base.append(toAppend)).toThrow("Cannot append a URL with protocol");
  });
});

describe("createURL", () => {
  it("should create an instance of $URL", () => {
    const url = createURL("https://example.com/");

    expect(url).toBeInstanceOf($URL);
    expect(url.href).toBe("https://example.com/");
  });
});
```
