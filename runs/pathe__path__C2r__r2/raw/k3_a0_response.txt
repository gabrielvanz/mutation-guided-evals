```ts
import { describe, it, expect } from "vitest";
import {
  sep,
  normalize,
  join,
  resolve,
  normalizeString,
  isAbsolute,
  toNamespacedPath,
  extname,
  relative,
  dirname,
  format,
  basename,
  parse,
  matchesGlob
} from "../../src/_path";

describe("sep", () => {
  it("should be a forward slash", () => {
    expect(sep).toBe("/");
  });
});

describe("normalize", () => {
  it("should normalize paths correctly", () => {
    const cases = [
      { input: "", expected: "." },
      { input: "foo/bar//baz", expected: "foo/bar/baz" },
      { input: "foo/./bar/../baz", expected: "foo/baz" },
      { input: "C:\\foo\\bar", expected: "C:/foo/bar" },
      { input: "/foo/bar/..", expected: "/foo" },
      { input: "/", expected: "/" },
      { input: "C:", expected: "C:/" },
      { input: "//server/share/file", expected: "//server/share/file" },
      { input: "foo/", expected: "foo/" }
    ];

    for (const { input, expected } of cases) {
      expect(normalize(input)).toBe(expected);
    }
  });
});

describe("join", () => {
  it("should join path segments", () => {
    const cases = [
      { input: ["foo", "bar", "baz"], expected: "foo/bar/baz" },
      { input: ["/foo", "bar"], expected: "/foo/bar" },
      { input: ["foo/", "/bar"], expected: "foo/bar" },
      { input: ["", "foo"], expected: "foo" },
      { input: [], expected: "." }
    ];

    for (const { input, expected } of cases) {
      expect(join(...input)).toBe(expected);
    }
  });
});

describe("resolve", () => {
  it("should resolve absolute paths", () => {
    expect(resolve("/foo/bar", "./baz")).toBe("/foo/bar/baz");
    expect(resolve("/foo/bar", "/baz")).toBe("/baz");
    expect(resolve("")).not.toBe("");
  });
});

describe("normalizeString", () => {
  it("should normalize string paths with allowAboveRoot options", () => {
    expect(normalizeString("foo/../bar", false)).toBe("bar");
    expect(normalizeString("../foo", true)).toBe("../foo");
    expect(normalizeString("../foo", false)).toBe("foo");
  });
});

describe("isAbsolute", () => {
  it("should correctly identify absolute paths", () => {
    const cases = [
      { input: "/foo/bar", expected: true },
      { input: "C:/foo/bar", expected: true },
      { input: "C:\\foo\\bar", expected: true },
      { input: "//server/share", expected: true },
      { input: "foo/bar", expected: false },
      { input: "", expected: false }
    ];

    for (const { input, expected } of cases) {
      expect(isAbsolute(input)).toBe(expected);
    }
  });
});

describe("toNamespacedPath", () => {
  it("should normalize windows paths for namespacing", () => {
    expect(toNamespacedPath("C:\\foo\\bar")).toBe("C:/foo/bar");
    expect(toNamespacedPath("/foo/bar")).toBe("/foo/bar");
  });
});

describe("extname", () => {
  it("should return the file extension", () => {
    const cases = [
      { input: "index.html", expected: ".html" },
      { input: "index.coffee.md", expected: ".md" },
      { input: "index.", expected: "." },
      { input: "index", expected: "" },
      { input: ".gitignore", expected: "" },
      { input: "..", expected: "" }
    ];

    for (const { input, expected } of cases) {
      expect(extname(input)).toBe(expected);
    }
  });
});

describe("relative", () => {
  it("should compute relative path between two paths", () => {
    expect(relative("/data/orandea/test/aaa", "/data/orandea/impl/bbb")).toBe("../../impl/bbb");
    expect(relative("C:/orandea/test/aaa", "D:/orandea/impl/bbb")).toBe("D:/orandea/impl/bbb");
  });
});

describe("dirname", () => {
  it("should return the directory name of a path", () => {
    const cases = [
      { input: "/foo/bar/baz/asdf", expected: "/foo/bar/baz" },
      { input: "/foo/bar/", expected: "/foo" },
      { input: "foo", expected: "." },
      { input: "C:\\", expected: "C:/" }
    ];

    for (const { input, expected } of cases) {
      expect(dirname(input)).toBe(expected);
    }
  });
});

describe("format", () => {
  it("should format a path object into a path string", () => {
    expect(
      format({
        root: "/",
        dir: "/home/user/dir",
        base: "file.txt"
      })
    ).toBe("/home/user/dir/file.txt");

    expect(
      format({
        name: "file",
        ext: ".txt"
      })
    ).toBe("file.txt");
  });
});

describe("basename", () => {
  it("should return the last portion of a path", () => {
    expect(basename("/foo/bar/baz/asdf.html", ".html")).toBe("asdf");
    expect(basename("/foo/bar/baz/asdf.html")).toBe("asdf.html");
    expect(basename("")).toBe("");
  });
});

describe("parse", () => {
  it("should parse a path into an object", () => {
    const parsed = parse("/home/user/dir/file.txt");
    expect(parsed).toEqual({
      root: "/",
      dir: "/home/user/dir",
      base: "file.txt",
      ext: ".txt",
      name: "file"
    });
  });
});

describe("matchesGlob", () => {
  it("should match path against glob patterns", () => {
    expect(matchesGlob("/foo/bar.js", "*.js")).toBe(false);
    expect(matchesGlob("/foo/bar.js", "**/bar.js")).toBe(true);
    expect(matchesGlob("/foo/bar.js", ["*.txt", "**/bar.js"])).toBe(true);
  });
});
```
