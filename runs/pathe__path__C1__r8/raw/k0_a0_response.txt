```ts
import { describe, it, expect } from "vitest";
import {
  sep,
  normalize,
  join,
  resolve,
  normalizeString,
  isAbsolute,
  toNamespacedPath,
  extname,
  relative,
  dirname,
  format,
  basename,
  parse,
  matchesGlob,
} from "../../src/_path";

describe("_path module", () => {
  it("should have sep equal to '/'", () => {
    expect(sep).toBe("/");
  });

  describe("normalize", () => {
    it("should normalize empty string to '.'", () => {
      expect(normalize("")).toBe(".");
    });

    it("should normalize windows paths", () => {
      expect(normalize("foo\\bar\\baz")).toBe("foo/bar/baz");
    });

    it("should handle absolute paths and trailing separators", () => {
      expect(normalize("/foo/bar/")).toBe("/foo/bar/");
      expect(normalize("/")).toBe("/");
    });

    it("should handle UNC paths", () => {
      expect(normalize("\\\\server\\share")).toBe("//server/share");
      expect(normalize("\\\\server")).toBe("//.///server");
    });

    it("should handle drive letters", () => {
      expect(normalize("C:")).toBe("C:/");
      expect(normalize("C:\\foo")).toBe("C:/foo");
    });
  });

  describe("join", () => {
    it("should join path segments", () => {
      expect(join("/foo", "bar", "baz/asdf")).toBe("/foo/bar/baz/asdf");
      expect(join("foo", "", "bar")).toBe("foo/bar");
    });

    it("should handle empty arguments", () => {
      expect(join()).toBe(".");
    });
  });

  describe("resolve", () => {
    it("should resolve absolute paths", () => {
      expect(resolve("/foo/bar", "./baz")).toBe("/foo/bar/baz");
    });

    it("should resolve with multiple arguments", () => {
      expect(resolve("/foo", "/bar", "baz")).toBe("/bar/baz");
    });
  });

  describe("normalizeString", () => {
    it("should normalize path strings with allowAboveRoot", () => {
      expect(normalizeString("/foo/../bar", false)).toBe("/bar");
      expect(normalizeString("foo/../../bar", true)).toBe("../bar");
    });
  });

  describe("isAbsolute", () => {
    it("should correctly identify absolute paths", () => {
      expect(isAbsolute("/foo/bar")).toBe(true);
      expect(isAbsolute("C:/foo")).toBe(true);
      expect(isAbsolute("foo/bar")).toBe(false);
    });
  });

  describe("toNamespacedPath", () => {
    it("should normalize windows path for namespaced path", () => {
      expect(toNamespacedPath("C:\\foo\\bar")).toBe("C:/foo/bar");
    });
  });

  describe("extname", () => {
    it("should return extension of file", () => {
      expect(extname("index.html")).toBe(".html");
      expect(extname("index.coffee.md")).toBe(".md");
      expect(extname("index")).toBe("");
      expect(extname("..")).toBe("");
    });
  });

  describe("relative", () => {
    it("should calculate relative path between from and to", () => {
      expect(relative("/data/orandea/test/aaa", "/data/orandea/impl/bbb")).toBe("../../impl/bbb");
    });

    it("should handle different drive letters", () => {
      expect(relative("C:/foo", "D:/bar")).toBe("D:/bar");
    });
  });

  describe("dirname", () => {
    it("should return the directory name of a path", () => {
      expect(dirname("/foo/bar/baz/asdf")).toBe("/foo/bar/baz");
      expect(dirname("/foo")).toBe("/");
      expect(dirname("foo")).toBe(".");
    });
  });

  describe("format", () => {
    it("should format path object", () => {
      expect(
        format({
          root: "/",
          dir: "/home/user/dir",
          base: "file.txt",
        })
      ).toBe("/home/user/dir/file.txt");
    });
  });

  describe("basename", () => {
    it("should return the last portion of a path", () => {
      expect(basename("/foo/bar/baz/asdf.html")).toBe("asdf.html");
      expect(basename("/foo/bar/baz/asdf.html", ".html")).toBe("asdf");
    });
  });

  describe("parse", () => {
    it("should parse path into an object", () => {
      const parsed = parse("/home/user/dir/file.txt");
      expect(parsed).toEqual({
        root: "/",
        dir: "/home/user/dir",
        base: "file.txt",
        ext: ".txt",
        name: "file",
      });
    });
  });

  describe("matchesGlob", () => {
    it("should match path against glob pattern", () => {
      expect(matchesGlob("/foo/bar.js", "*.js")).toBe(false);
      expect(matchesGlob("/foo/bar.js", "/**/*.js")).toBe(true);
    });
  });
});
```
