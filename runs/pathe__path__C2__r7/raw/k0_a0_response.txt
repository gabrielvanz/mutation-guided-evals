```ts
import { describe, it, expect } from "vitest";
import {
  sep,
  normalize,
  join,
  resolve,
  normalizeString,
  isAbsolute,
  toNamespacedPath,
  extname,
  relative,
  dirname,
  format,
  basename,
  parse,
  matchesGlob
} from "../../src/_path";

describe("sep", () => {
  it("should be a forward slash", () => {
    expect(sep).toBe("/");
  });
});

describe("normalize", () => {
  it("should normalize empty path to dot", () => {
    expect(normalize("")).toBe(".");
  });

  it("should normalize duplicate slashes and backslashes", () => {
    expect(normalize("foo//bar\\baz")).toBe("foo/bar/baz");
  });

  it("should handle absolute paths and dots", () => {
    expect(normalize("/foo/../bar")).toBe("/bar");
  });

  it("should preserve trailing slash", () => {
    expect(normalize("/foo/bar/")).toBe("/foo/bar/");
  });

  it("should handle windows drive letters", () => {
    expect(normalize("C:\\foo\\bar")).toBe("C:/foo/bar");
    expect(normalize("C:")).toBe("C:/");
  });
});

describe("join", () => {
  it("should join multiple segments", () => {
    expect(join("foo", "bar", "baz")).toBe("foo/bar/baz");
  });

  it("should ignore empty segments", () => {
    expect(join("foo", "", "bar")).toBe("foo/bar");
  });

  it("should handle slashes between segments", () => {
    expect(join("/foo/", "/bar")).toBe("/foo/bar");
  });
});

describe("resolve", () => {
  it("should resolve relative paths against cwd", () => {
    expect(resolve("foo/bar")).toContain("/foo/bar");
  });

  it("should resolve absolute segments", () => {
    expect(resolve("/foo", "/bar", "baz")).toBe("/bar/baz");
  });

  it("should return root when arguments are empty", () => {
    expect(resolve()).not.toBe("");
  });
});

describe("normalizeString", () => {
  it("should normalize relative dot segments", () => {
    expect(normalizeString("foo/./bar/../baz", false)).toBe("foo/baz");
  });

  it("should allow above root when specified", () => {
    expect(normalizeString("../../foo", true)).toBe("../../foo");
  });

  it("should not allow above root when false", () => {
    expect(normalizeString("../../foo", false)).toBe("foo");
  });
});

describe("isAbsolute", () => {
  it.each([
    { input: "/foo/bar", expected: true },
    { input: "C:/foo/bar", expected: true },
    { input: "C:\\foo\\bar", expected: true },
    { input: "foo/bar", expected: false },
    { input: "", expected: false }
  ])("should return $expected for $input", ({ input, expected }) => {
    expect(isAbsolute(input)).toBe(expected);
  });
});

describe("toNamespacedPath", () => {
  it("should normalize windows paths", () => {
    expect(toNamespacedPath("C:\\foo\\bar")).toBe("C:/foo/bar");
  });
});

describe("extname", () => {
  it.each([
    { input: "index.ts", expected: ".ts" },
    { input: "file.tar.gz", expected: ".gz" },
    { input: "README", expected: "" },
    { input: ".gitignore", expected: "" },
    { input: "..", expected: "" }
  ])("should return $expected for $input", ({ input, expected }) => {
    expect(extname(input)).toBe(expected);
  });
});

describe("relative", () => {
  it("should compute relative path between two absolute paths", () => {
    expect(relative("/data/orandea/test/aaa", "/data/orandea/impl/bbb")).toBe("../../impl/bbb");
  });

  it("should return absolute destination if drive letters differ", () => {
    expect(relative("C:/foo/bar", "D:/foo/bar")).toBe("D:/foo/bar");
  });
});

describe("dirname", () => {
  it.each([
    { input: "/foo/bar/baz.js", expected: "/foo/bar" },
    { input: "/foo", expected: "/" },
    { input: "foo/bar", expected: "foo" },
    { input: "", expected: "." }
  ])("should return $expected for $input", ({ input, expected }) => {
    expect(dirname(input)).toBe(expected);
  });
});

describe("format", () => {
  it("should format path object into string", () => {
    expect(
      format({
        root: "/",
        dir: "/foo/bar",
        base: "baz.js",
        ext: ".js",
        name: "baz"
      })
    ).toBe("/foo/bar/baz.js");
  });

  it("should format path object with name and ext when base is missing", () => {
    expect(
      format({
        root: "/",
        dir: "/foo/bar",
        name: "baz",
        ext: "js"
      })
    ).toBe("/foo/bar/baz.js");
  });
});

describe("basename", () => {
  it("should return last segment", () => {
    expect(basename("/foo/bar/baz.js")).toBe("baz.js");
  });

  it("should strip extension if provided", () => {
    expect(basename("/foo/bar/baz.js", ".js")).toBe("baz");
  });
});

describe("parse", () => {
  it("should parse path into components", () => {
    expect(parse("/home/user/dir/file.txt")).toEqual({
      root: "/",
      dir: "/home/user/dir",
      base: "file.txt",
      ext: ".txt",
      name: "file"
    });
  });
});

describe("matchesGlob", () => {
  it("should match path against glob pattern", () => {
    expect(matchesGlob("/foo/bar/baz.ts", "**/*.ts")).toBe(true);
    expect(matchesGlob("/foo/bar/baz.js", "**/*.ts")).toBe(false);
  });
});
```
