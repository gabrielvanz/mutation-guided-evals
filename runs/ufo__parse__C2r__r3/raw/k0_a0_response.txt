```ts
import { describe, it, expect } from "vitest";
import {
  parseURL,
  parsePath,
  parseAuth,
  parseHost,
  stringifyParsedURL,
  parseFilename
} from "../../src/parse";

describe("parseURL", () => {
  it("should parse absolute URLs with protocol, host, pathname, search and hash", () => {
    const result = parseURL("http://foo.com/foo?test=123#token");
    expect(result).toEqual({
      protocol: "http:",
      auth: "",
      host: "foo.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should parse relative URLs without protocol", () => {
    const result = parseURL("foo.com/foo?test=123#token");
    expect(result).toEqual({
      pathname: "foo.com/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should parse relative URLs with default protocol", () => {
    const result = parseURL("foo.com/foo?test=123#token", "https://");
    expect(result).toEqual({
      protocol: "https:",
      auth: "",
      host: "foo.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should handle empty input", () => {
    const result = parseURL("");
    expect(result).toEqual({
      pathname: "",
      search: "",
      hash: ""
    });
  });

  it("should handle special protocols like blob, data, javascript, vbscript", () => {
    const result = parseURL("blob:http://example.com/123-456");
    expect(result).toEqual({
      protocol: "blob:",
      pathname: "http://example.com/123-456",
      href: "blob:http://example.com/123-456",
      auth: "",
      host: "",
      search: "",
      hash: ""
    });
  });

  it("should parse URLs with authentication", () => {
    const result = parseURL("https://user:pass@example.com/path");
    expect(result).toEqual({
      protocol: "https:",
      auth: "user:pass",
      host: "example.com",
      pathname: "/path",
      search: "",
      hash: ""
    });
  });

  it("should handle file protocol paths correctly", () => {
    const result = parseURL("file:///C:/path/to/file");
    expect(result).toEqual({
      protocol: "file:",
      auth: "",
      host: "",
      pathname: "C:/path/to/file",
      search: "",
      hash: ""
    });
  });

  it("should replace backslashes with forward slashes", () => {
    const result = parseURL("http://foo.com\\bar\\baz");
    expect(result.pathname).toBe("/bar/baz");
  });
});

describe("parsePath", () => {
  it("should split path into pathname, search, and hash", () => {
    const result = parsePath("http://foo.com/foo?test=123#token");
    expect(result).toEqual({
      pathname: "http://foo.com/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should handle empty input", () => {
    const result = parsePath("");
    expect(result).toEqual({
      pathname: "",
      search: "",
      hash: ""
    });
  });

  it("should handle path with only search", () => {
    const result = parsePath("/path?query=1");
    expect(result).toEqual({
      pathname: "/path",
      search: "?query=1",
      hash: ""
    });
  });

  it("should handle path with only hash", () => {
    const result = parsePath("/path#hash");
    expect(result).toEqual({
      pathname: "/path",
      search: "",
      hash: "#hash"
    });
  });
});

describe("parseAuth", () => {
  it("should parse username and password", () => {
    const result = parseAuth("admin:secret");
    expect(result).toEqual({
      username: "admin",
      password: "secret"
    });
  });

  it("should handle empty input", () => {
    const result = parseAuth("");
    expect(result).toEqual({
      username: "",
      password: ""
    });
  });

  it("should decode percent-encoded auth strings", () => {
    const result = parseAuth("foo%40bar:baz%2Fqux");
    expect(result).toEqual({
      username: "foo@bar",
      password: "baz/qux"
    });
  });
});

describe("parseHost", () => {
  it("should parse hostname and port", () => {
    const result = parseHost("foo.com:8080");
    expect(result).toEqual({
      hostname: "foo.com",
      port: "8080"
    });
  });

  it("should handle host without port", () => {
    const result = parseHost("foo.com");
    expect(result).toEqual({
      hostname: "foo.com",
      port: undefined
    });
  });

  it("should handle empty input", () => {
    const result = parseHost("");
    expect(result).toEqual({
      hostname: "",
      port: undefined
    });
  });
});

describe("stringifyParsedURL", () => {
  it("should stringify a parsed URL object back to string", () => {
    const parsed = {
      protocol: "http:",
      host: "bar.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token"
    };
    const result = stringifyParsedURL(parsed);
    expect(result).toBe("http://bar.com/foo?test=123#token");
  });

  it("should include auth when present", () => {
    const parsed = {
      protocol: "https:",
      auth: "user:pass",
      host: "example.com",
      pathname: "/"
    };
    const result = stringifyParsedURL(parsed);
    expect(result).toBe("https://user:pass@example.com/");
  });

  it("should format search query without leading question mark if missing", () => {
    const parsed = {
      pathname: "/foo",
      search: "test=123"
    };
    const result = stringifyParsedURL(parsed);
    expect(result).toBe("/foo?test=123");
  });

  it("should handle empty parsed object", () => {
    const result = stringifyParsedURL({});
    expect(result).toBe("");
  });
});

describe("parseFilename", () => {
  it("should extract filename from URL", () => {
    const result = parseFilename("http://example.com/path/to/filename.ext");
    expect(result).toBe("filename.ext");
  });

  it("should extract filename without strict mode", () => {
    const result = parseFilename("/path/to/.hidden-file");
    expect(result).toBe(".hidden-file");
  });

  it("should return undefined in strict mode for files without extension", () => {
    const result = parseFilename("/path/to/.hidden-file", { strict: true });
    expect(result).toBeUndefined();
  });

  it("should return filename in strict mode when extension exists", () => {
    const result = parseFilename("/path/to/filename.ext", { strict: true });
    expect(result).toBe("filename.ext");
  });

  it("should handle empty input", () => {
    const result = parseFilename("");
    expect(result).toBeUndefined();
  });
});
```
