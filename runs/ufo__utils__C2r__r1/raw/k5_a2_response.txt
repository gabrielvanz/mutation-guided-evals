import { describe, it, expect } from "vitest";
import {
  isRelative,
  hasProtocol,
  isScriptProtocol,
  hasTrailingSlash,
  withoutTrailingSlash,
  withTrailingSlash,
  hasLeadingSlash,
  withoutLeadingSlash,
  withLeadingSlash,
  cleanDoubleSlashes,
  withBase,
  withoutBase,
  withQuery,
  filterQuery,
  getQuery,
  isEmptyURL,
  isNonEmptyURL,
  joinURL,
  joinRelativeURL,
  withHttp,
  withHttps,
  withoutProtocol,
  withProtocol,
  normalizeURL,
  resolveURL,
  isSamePath,
  isEqual,
  withFragment,
  withoutFragment,
  withoutHost,
} from "../../src/utils";

describe("isRelative", () => {
  const cases = [
    { input: "./foo", expected: true },
    { input: "../foo", expected: true },
    { input: "/foo", expected: false },
    { input: "foo", expected: false },
    { input: "", expected: false },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${expected} for "${input}"`, () => {
      expect(isRelative(input)).toBe(expected);
    });
  }
});

describe("hasProtocol", () => {
  const cases = [
    { input: "https://example.com", opts: undefined, expected: true },
    { input: "//example.com", opts: undefined, expected: false },
    { input: "//example.com", opts: { acceptRelative: true }, expected: true },
    { input: "//example.com", opts: true, expected: true },
    { input: "ftp://example.com", opts: undefined, expected: true },
    { input: "data:text/plain", opts: undefined, expected: true },
    { input: "data:text/plain", opts: { strict: true }, expected: false },
    { input: "http://", opts: undefined, expected: true },
    { input: "/foo/bar", opts: undefined, expected: false },
  ];

  for (const { input, opts, expected } of cases) {
    it(`should return ${expected} for "${input}" with opts ${JSON.stringify(opts)}`, () => {
      expect(hasProtocol(input, opts as any)).toBe(expected);
    });
  }
});

describe("isScriptProtocol", () => {
  const cases = [
    { input: "javascript:", expected: true },
    { input: "data:", expected: true },
    { input: "blob:", expected: true },
    { input: "vbscript:", expected: true },
    { input: "https://example.com", expected: false },
    { input: undefined, expected: false },
    { input: "", expected: false },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${expected} for "${input}"`, () => {
      expect(isScriptProtocol(input)).toBe(expected);
    });
  }
});

describe("hasTrailingSlash", () => {
  const cases = [
    { input: "/foo/", respectQueryAndFragment: false, expected: true },
    { input: "/foo", respectQueryAndFragment: false, expected: false },
    { input: "/foo?query=true", respectQueryAndFragment: true, expected: false },
    { input: "/foo/?query=true", respectQueryAndFragment: true, expected: true },
    { input: "", respectQueryAndFragment: false, expected: false },
  ];

  for (const { input, respectQueryAndFragment, expected } of cases) {
    it(`should return ${expected} for "${input}" (respectQueryAndFragment: ${respectQueryAndFragment})`, () => {
      expect(hasTrailingSlash(input, respectQueryAndFragment)).toBe(expected);
    });
  }
});

describe("withoutTrailingSlash", () => {
  const cases = [
    { input: "/foo/", respectQueryAndFragment: false, expected: "/foo" },
    { input: "/foo", respectQueryAndFragment: false, expected: "/foo" },
    { input: "/path/?query=true", respectQueryAndFragment: true, expected: "/path?query=true" },
    { input: "/", respectQueryAndFragment: false, expected: "/" },
    { input: "", respectQueryAndFragment: false, expected: "/" },
  ];

  for (const { input, respectQueryAndFragment, expected } of cases) {
    it(`should return "${expected}" for "${input}"`, () => {
      expect(withoutTrailingSlash(input, respectQueryAndFragment)).toBe(expected);
    });
  }
});

describe("withTrailingSlash", () => {
  const cases = [
    { input: "/foo", respectQueryAndFragment: false, expected: "/foo/" },
    { input: "/foo/", respectQueryAndFragment: false, expected: "/foo/" },
    { input: "/path?query=true", respectQueryAndFragment: true, expected: "/path/?query=true" },
    { input: "", respectQueryAndFragment: false, expected: "/" },
    { input: "#hash", respectQueryAndFragment: true, expected: "#hash" },
  ];

  for (const { input, respectQueryAndFragment, expected } of cases) {
    it(`should return "${expected}" for "${input}"`, () => {
      expect(withTrailingSlash(input, respectQueryAndFragment)).toBe(expected);
    });
  }
});

describe("hasLeadingSlash", () => {
  const cases = [
    { input: "/foo", expected: true },
    { input: "foo", expected: false },
    { input: "", expected: false },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${expected} for "${input}"`, () => {
      expect(hasLeadingSlash(input)).toBe(expected);
    });
  }
});

describe("withoutLeadingSlash", () => {
  const cases = [
    { input: "/foo", expected: "foo" },
    { input: "foo", expected: "foo" },
    { input: "/", expected: "/" },
    { input: "", expected: "/" },
  ];

  for (const { input, expected } of cases) {
    it(`should return "${expected}" for "${input}"`, () => {
      expect(withoutLeadingSlash(input)).toBe(expected);
    });
  }
});

describe("withLeadingSlash", () => {
  const cases = [
    { input: "foo", expected: "/foo" },
    { input: "/foo", expected: "/foo" },
    { input: "", expected: "/" },
  ];

  for (const { input, expected } of cases) {
    it(`should return "${expected}" for "${input}"`, () => {
      expect(withLeadingSlash(input)).toBe(expected);
    });
  }
});

describe("cleanDoubleSlashes", () => {
  const cases = [
    { input: "//foo//bar//", expected: "/foo/bar/" },
    { input: "http://example.com/analyze//http://localhost:3000//", expected: "http://example.com/analyze/http://localhost:3000/" },
    { input: "", expected: "" },
  ];

  for (const { input, expected } of cases) {
    it(`should return "${expected}" for "${input}"`, () => {
      expect(cleanDoubleSlashes(input)).toBe(expected);
    });
  }
});

describe("withBase", () => {
  const cases = [
    { input: "/foo/bar", base: "/foo", expected: "/foo/bar" },
    { input: "/foo/bar", base: "/baz", expected: "/baz/foo/bar" },
    { input: "/admin-dashboard", base: "/admin/", expected: "/admin/admin-dashboard" },
    { input: "https://example.com", base: "/foo", expected: "https://example.com" },
    { input: "/foo", base: "", expected: "/foo" },
  ];

  for (const { input, base, expected } of cases) {
    it(`should return "${expected}" for withBase("${input}", "${base}")`, () => {
      expect(withBase(input, base)).toBe(expected);
    });
  }
});

describe("withoutBase", () => {
  const cases = [
    { input: "/foo/bar", base: "/foo", expected: "/bar" },
    { input: "/foo/bar", base: "/baz", expected: "/foo/bar" },
    { input: "/admin-dashboard", base: "/admin/", expected: "/admin-dashboard" },
    { input: "/legacy//evil.com", base: "/legacy", expected: "/evil.com" },
    { input: "/foo", base: "", expected: "/foo" },
  ];

  for (const { input, base, expected } of cases) {
    it(`should return "${expected}" for withoutBase("${input}", "${base}")`, () => {
      expect(withoutBase(input, base)).toBe(expected);
    });
  }
});

describe("withQuery", () => {
  it("should add or replace the query section of the URL", () => {
    expect(withQuery("/foo?page=a", { token: "secret" })).toBe("/foo?page=a&token=secret");
    expect(withQuery("/foo?page=a", { page: "b" })).toBe("/foo?page=b");
  });
});

describe("filterQuery", () => {
  it("should filter the query section using predicate", () => {
    expect(filterQuery("/foo?bar=1&baz=2", (key) => key !== "bar")).toBe("/foo?baz=2");
    expect(filterQuery("/foo", (key) => key !== "bar")).toBe("/foo");
  });
});

describe("getQuery", () => {
  it("should parse and decode the query object of an input URL", () => {
    expect(getQuery("http://foo.com/foo?test=123&unicode=%E5%A5%BD")).toEqual({
      test: "123",
      unicode: "好",
    });
  });
});

describe("isEmptyURL & isNonEmptyURL", () => {
  it("isEmptyURL should correctly identify empty or '/' URLs", () => {
    expect(isEmptyURL("")).toBe(true);
    expect(isEmptyURL("/")).toBe(true);
    expect(isEmptyURL("/foo")).toBe(false);
  });

  it("isNonEmptyURL should correctly identify non-empty, non-'/' URLs", () => {
    expect(isNonEmptyURL("/foo")).toBe(true);
    expect(isNonEmptyURL("")).toBeFalsy();
    expect(isNonEmptyURL("/")).toBe(false);
  });
});

describe("joinURL", () => {
  const cases = [
    { base: "a", input: ["/b", "/c"], expected: "a/b/c" },
    { base: "/a/", input: ["/b/"], expected: "/a/b/" },
    { base: "", input: ["/a", "b"], expected: "/a/b" },
  ];

  for (const { base, input, expected } of cases) {
    it(`should join segments into "${expected}"`, () => {
      expect(joinURL(base, ...input)).toBe(expected);
    });
  }
});

describe("joinRelativeURL", () => {
  const cases = [
    { input: ["/a", "../b", "./c"], expected: "/b/c" },
    { input: ["a", "b", "c"], expected: "a/b/c" },
    { input: ["/a/", "../"], expected: "/" },
  ];

  for (const { input, expected } of cases) {
    it(`should join relative URL to "${expected}"`, () => {
      expect(joinRelativeURL(...input)).toBe(expected);
    });
  }
});

describe("withHttp, withHttps, withoutProtocol, withProtocol", () => {
  it("withHttp should set protocol to http://", () => {
    expect(withHttp("https://example.com")).toBe("http://example.com");
  });

  it("withHttps should set protocol to https://", () => {
    expect(withHttps("http://example.com")).toBe("https://example.com");
  });

  it("withoutProtocol should remove protocol", () => {
    expect(withoutProtocol("http://example.com")).toBe("example.com");
  });

  it("withProtocol should add or replace protocol", () => {
    expect(withProtocol("http://example.com", "ftp://")).toBe("ftp://example.com");
    expect(withProtocol("example.com", "https://")).toBe("https://example.com");
  });
});

describe("normalizeURL", () => {
  it("should normalize the input URL", () => {
    expect(normalizeURL("test?query=123 123#hash, test")).toBe("test?query=123+123#hash,%20test");
    expect(normalizeURL("http://localhost:3000")).toBe("http://localhost:3000");
  });
});

describe("resolveURL", () => {
  it("should resolve multiple URL segments", () => {
    expect(resolveURL("http://foo.com/foo?test=123#token", "bar", "baz")).toBe(
      "http://foo.com/foo/bar/baz?test=123#token",
    );
  });

  it("should throw TypeError when base is not a string", () => {
    expect(() => resolveURL(123 as any, "bar")).toThrow(TypeError);
  });

  it("should return base when filteredInputs is empty", () => {
    expect(resolveURL("http://foo.com")).toBe("http://foo.com");
  });
});

describe("isSamePath", () => {
  const cases = [
    { p1: "/foo", p2: "/foo/", expected: true },
    { p1: "/foo%20bar", p2: "/foo bar", expected: true },
    { p1: "/foo", p2: "/bar", expected: false },
  ];

  for (const { p1, p2, expected } of cases) {
    it(`should return ${expected} for isSamePath("${p1}", "${p2}")`, () => {
      expect(isSamePath(p1, p2)).toBe(expected);
    });
  }
});

describe("isEqual", () => {
  const cases = [
    { a: "/foo", b: "foo", options: undefined, expected: true },
    { a: "foo/", b: "foo", options: undefined, expected: true },
    { a: "/foo bar", b: "/foo%20bar", options: undefined, expected: true },
    { a: "/foo", b: "foo", options: { leadingSlash: true }, expected: false },
    { a: "foo/", b: "foo", options: { trailingSlash: true }, expected: false },
    { a: "/foo bar", b: "/foo%20bar", options: { encoding: true }, expected: false },
  ];

  for (const { a, b, options, expected } of cases) {
    it(`should return ${expected} for isEqual("${a}", "${b}", ${JSON.stringify(options)})`, () => {
      expect(isEqual(a, b, options)).toBe(expected);
    });
  }
});

describe("withFragment", () => {
  it("should add or replace the fragment section of the URL", () => {
    expect(withFragment("/foo", "bar")).toBe("/foo#bar");
    expect(withFragment("/foo#bar", "baz")).toBe("/foo#baz");
    expect(withFragment("/foo#bar", "")).toBe("/foo#bar");
    expect(withFragment("/foo", "#")).toBe("/foo");
  });
});

describe("withoutFragment", () => {
  it("should remove the fragment section from the URL", () => {
    expect(withoutFragment("http://example.com/foo?q=123#bar")).toBe("http://example.com/foo?q=123");
  });
});

describe("withoutHost", () => {
  it("should remove the host from the URL", () => {
    expect(withoutHost("http://example.com/foo?q=123#bar")).toBe("/foo?q=123#bar");
  });
});
