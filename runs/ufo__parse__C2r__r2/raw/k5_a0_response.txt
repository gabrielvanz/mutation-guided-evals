```ts
import { describe, it, expect } from "vitest";
import {
  parseURL,
  parsePath,
  parseAuth,
  parseHost,
  stringifyParsedURL,
  parseFilename
} from "../../src/parse";

describe("parseURL", () => {
  it("should parse absolute URLs with protocol, host, pathname, search, and hash", () => {
    const result = parseURL("http://foo.com/foo?test=123#token");
    expect(result).toEqual({
      protocol: "http:",
      auth: "",
      host: "foo.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should parse relative URLs without protocol", () => {
    const result = parseURL("foo.com/foo?test=123#token");
    expect(result).toMatchObject({
      pathname: "foo.com/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should apply default protocol when input lacks one", () => {
    const result = parseURL("foo.com/foo?test=123#token", "https://");
    expect(result).toEqual({
      protocol: "https:",
      auth: "",
      host: "foo.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should handle empty string input", () => {
    const result = parseURL("");
    expect(result).toEqual({
      pathname: "",
      search: "",
      hash: ""
    });
  });

  it("should parse special protocols like blob, data, javascript, vbscript", () => {
    const result = parseURL("data:text/plain;base64,SGVsbG8=");
    expect(result).toEqual({
      protocol: "data:",
      pathname: "text/plain;base64,SGVsbG8=",
      href: "data:text/plain;base64,SGVsbG8=",
      auth: "",
      host: "",
      search: "",
      hash: ""
    });
  });

  it("should parse file URLs correctly", () => {
    const result = parseURL("file:///C:/path/to/file");
    expect(result).toEqual({
      protocol: "file:",
      auth: "",
      host: "",
      pathname: "/C:/path/to/file",
      search: "",
      hash: ""
    });
  });

  it("should parse URLs with auth credentials", () => {
    const result = parseURL("http://user:pass@foo.com/path");
    expect(result).toEqual({
      protocol: "http:",
      auth: "user:pass",
      host: "foo.com",
      pathname: "/path",
      search: "",
      hash: ""
    });
  });

  it("should normalize backslashes to forward slashes", () => {
    const result = parseURL("http:\\\\foo.com\\bar\\baz");
    expect(result).toEqual({
      protocol: "http:",
      auth: "",
      host: "foo.com",
      pathname: "/bar/baz",
      search: "",
      hash: ""
    });
  });
});

describe("parsePath", () => {
  it("should split path into pathname, search, and hash", () => {
    const result = parsePath("http://foo.com/foo?test=123#token");
    expect(result).toEqual({
      pathname: "http://foo.com/foo",
      search: "?test=123",
      hash: "#token"
    });
  });

  it("should handle empty path input", () => {
    const result = parsePath("");
    expect(result).toEqual({
      pathname: "",
      search: "",
      hash: ""
    });
  });
});

describe("parseAuth", () => {
  it("should parse username and password", () => {
    const result = parseAuth("user:password");
    expect(result).toEqual({
      username: "user",
      password: "password"
    });
  });

  it("should handle empty auth input", () => {
    const result = parseAuth("");
    expect(result).toEqual({
      username: "",
      password: ""
    });
  });

  it("should decode percent-encoded auth strings", () => {
    const result = parseAuth("john%20doe:secret%231");
    expect(result).toEqual({
      username: "john doe",
      password: "secret#1"
    });
  });
});

describe("parseHost", () => {
  it("should parse hostname and port", () => {
    const result = parseHost("foo.com:8080");
    expect(result).toEqual({
      hostname: "foo.com",
      port: "8080"
    });
  });

  it("should parse hostname without port", () => {
    const result = parseHost("foo.com");
    expect(result).toEqual({
      hostname: "foo.com",
      port: undefined
    });
  });

  it("should handle empty host input", () => {
    const result = parseHost("");
    expect(result).toEqual({
      hostname: "",
      port: undefined
    });
  });
});

describe("stringifyParsedURL", () => {
  it("should stringify a parsed URL object back to string", () => {
    const parsed = {
      protocol: "http:",
      host: "foo.com",
      pathname: "/foo",
      search: "?test=123",
      hash: "#token",
      auth: "user:pass"
    };
    const result = stringifyParsedURL(parsed);
    expect(result).toBe("http://user:pass@foo.com/foo?test=123#token");
  });

  it("should stringify partial parsed URL without protocol or auth", () => {
    const parsed = {
      pathname: "/foo",
      search: "test=123",
      hash: "#token"
    };
    const result = stringifyParsedURL(parsed);
    expect(result).toBe("/foo?test=123#token");
  });

  it("should handle empty parsed object", () => {
    const result = stringifyParsedURL({});
    expect(result).toBe("");
  });
});

describe("parseFilename", () => {
  it("should return the filename from a URL path", () => {
    const result = parseFilename("http://example.com/path/to/filename.ext");
    expect(result).toBe("filename.ext");
  });

  it("should return undefined when no filename is present", () => {
    const result = parseFilename("http://example.com/path/to/");
    expect(result).toBe(undefined);
  });

  it("should return filename without extension in non-strict mode", () => {
    const result = parseFilename("/path/to/.hidden-file");
    expect(result).toBe(".hidden-file");
  });

  it("should return undefined in strict mode for files without valid extension or leading dot files", () => {
    const result = parseFilename("/path/to/.hidden-file", { strict: true });
    expect(result).toBe(undefined);
  });

  it("should return filename in strict mode when extension exists", () => {
    const result = parseFilename("/path/to/script.ts", { strict: true });
    expect(result).toBe("script.ts");
  });

  it("should handle empty input for parseFilename", () => {
    const result = parseFilename("");
    expect(result).toBe(undefined);
  });
});
```
