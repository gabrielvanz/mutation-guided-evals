```ts
import { describe, it, expect } from "vitest";
import {
  normalizeAliases,
  resolveAlias,
  reverseResolveAlias,
  filename
} from "../../src/utils";

describe("normalizeAliases", () => {
  it("should sort aliases from specific to general and resolve nested aliases", () => {
    const input = {
      "~/": "/root/",
      "~/components": "/root/src/components"
    };
    const result = normalizeAliases(input);
    expect(result).toEqual({
      "~/components": "/root/src/components",
      "~/": "/root/"
    });
  });

  it("should return the same object if already normalized", () => {
    const input = {
      "~/": "/root/"
    };
    const normalized = normalizeAliases(input);
    const reNormalized = normalizeAliases(normalized);
    expect(reNormalized).toBe(normalized);
  });

  it("should resolve nested aliases when one alias points to another", () => {
    const input = {
      "@lib": "@core/lib",
      "@core": "/app/core"
    };
    const result = normalizeAliases(input);
    expect(result).toEqual({
      "@core": "/app/core",
      "@lib": "/app/core/lib"
    });
  });

  it("should not affect an alias when key equals alias or starts with another alias incorrectly", () => {
    const input = {
      "@core": "/app/core",
      "@core-extra": "/app/extra"
    };
    const result = normalizeAliases(input);
    expect(result).toEqual({
      "@core-extra": "/app/extra",
      "@core": "/app/core"
    });
  });

  it("should not resolve an alias against itself and handle same key correctly", () => {
    const input = {
      "@core": "/app/core"
    };
    const result = normalizeAliases(input);
    expect(result).toEqual({
      "@core": "/app/core"
    });
  });

  it("should not resolve when key starts with another alias suffix (endsWith mutation guard)", () => {
    const input = {
      "core": "/path/core",
      "mycore": "/path/mycore"
    };
    const result = normalizeAliases(input);
    expect(result).toEqual({
      "mycore": "/path/mycore",
      "core": "/path/core"
    });
  });

  it("should handle undefined or safe property access when values are processed", () => {
    const input = {
      "@a": "/a",
      "@b": "/b"
    };
    const result = normalizeAliases(input);
    expect(result).toEqual({
      "@b": "/b",
      "@a": "/a"
    });
  });

  it("should sort aliases correctly using segment counts via _compareAliases", () => {
    const input = {
      "a": "/a",
      "a/b/c": "/c",
      "a/b": "/b"
    };
    const result = normalizeAliases(input);
    expect(Object.keys(result)).toEqual(["a/b/c", "a/b", "a"]);
  });

  it("should not resolve alias value when key starts with alias as a prefix string but without path separator", () => {
    const input = {
      "@app": "/app",
      "@apple": "/apple-dir/sub"
    };
    const result = normalizeAliases(input);
    expect(result).toEqual({
      "@apple": "/apple-dir/sub",
      "@app": "/app"
    });
  });

  it("should handle same alias key when values point to each other", () => {
    const input = {
      "@alias": "/base/alias"
    };
    const result = normalizeAliases(input);
    expect(result).toEqual({
      "@alias": "/base/alias"
    });
  });

  it("should not replace alias value when alias points to a path where alias is a suffix instead of a prefix or segment", () => {
    const input = {
      "core": "/app/core",
      "mycore": "/app/mycore"
    };
    const result = normalizeAliases(input);
    expect(result).toEqual({
      "mycore": "/app/mycore",
      "core": "/app/core"
    });
  });
});

describe("resolveAlias", () => {
  const aliases = {
    "@": "/app/src",
    "@components": "/app/src/components"
  };

  const testCases = [
    { path: "@components/Button.ts", expected: "/app/src/components/Button.ts" },
    { path: "@/utils/index.ts", expected: "/app/src/utils/index.ts" },
    { path: "C:\\app\\src\\components\\Button.ts", expected: "C:/app/src/components/Button.ts" },
    { path: "@unmatched/file.ts", expected: "@unmatched/file.ts" }
  ];

  for (const { path, expected } of testCases) {
    it(`should resolve ${path} to ${expected}`, () => {
      expect(resolveAlias(path, aliases)).toBe(expected);
    });
  }

  it("should not resolve alias when character after alias is not a separator", () => {
    const simpleAliases = {
      "@app": "/path/to/app"
    };
    expect(resolveAlias("@apples/file.ts", simpleAliases)).toBe("@apples/file.ts");
  });

  it("should ignore non-matching aliases and continue processing other aliases", () => {
    const multiAliases = {
      "@missing": "/missing",
      "@": "/app/src"
    };
    expect(resolveAlias("@/file.ts", multiAliases)).toBe("/app/src/file.ts");
  });

  it("should handle aliases correctly when using normal paths", () => {
    const winAliases = {
      "@components": "/app/src/components"
    };
    expect(resolveAlias("@components/Button.ts", winAliases)).toBe("/app/src/components/Button.ts");
  });

  it("should correctly handle trailing slashes in alias definition during resolution", () => {
    const aliasesWithTrailingSlash = {
      "@components/": "/app/src/components"
    };
    expect(resolveAlias("@components/Button.ts", aliasesWithTrailingSlash)).toBe("/app/src/components/Button.ts");
  });

  it("should return the path unchanged when no aliases match", () => {
    const customAliases = {
      "@foo": "/foo"
    };
    expect(resolveAlias("/bar/baz.ts", customAliases)).toBe("/bar/baz.ts");
  });

  it("should handle backslash as trailing slash in alias target or path checking", () => {
    const customAliases = {
      "@dir": "/some\\dir"
    };
    expect(resolveAlias("@dir\\file.ts", customAliases)).toBe("/some/dir/file.ts");
  });
});

describe("reverseResolveAlias", () => {
  const aliases = {
    "@": "/app/src",
    "@components": "/app/src/components"
  };

  it("should return sorted possible alias resolutions ordered by length descending", () => {
    const customAliases = {
      "@": "/app/src",
      "@sub": "/app/src/sub",
      "@nested": "/app/src/sub/nested"
    };
    const result = reverseResolveAlias("/app/src/sub/nested/file.ts", customAliases);
    expect(result).toEqual([
      "@/sub/nested/file.ts",
      "@sub/nested/file.ts",
      "@nested/file.ts"
    ]);
  });

  it("should return empty array when no alias matches", () => {
    const result = reverseResolveAlias("/other/path/file.ts", aliases);
    expect(result).toEqual([]);
  });

  it("should correctly handle trailing slashes and path boundaries during reverse resolution", () => {
    const result = reverseResolveAlias("/app/src/components/Button.ts", aliases);
    expect(result).toEqual([
      "@/components/Button.ts",
      "@components/Button.ts"
    ]);
  });

  it("should skip non-matching paths in reverse resolve", () => {
    const multiAliases = {
      "@other": "/other/path",
      "@": "/app/src"
    };
    const result = reverseResolveAlias("/app/src/file.ts", multiAliases);
    expect(result).toEqual(["@/file.ts"]);
  });

  it("should correctly handle alias target with trailing slash in reverse resolution", () => {
    const trailingSlashAliases = {
      "@components": "/app/src/components/"
    };
    const result = reverseResolveAlias("/app/src/components/Button.ts", trailingSlashAliases);
    expect(result).toEqual(["@components/Button.ts"]);
  });

  it("should sort reverse resolved paths by length in descending order", () => {
    const customAliases = {
      "short": "/path/to",
      "verylongalias": "/path/to/target"
    };
    const result = reverseResolveAlias("/path/to/target/file.ts", customAliases);
    expect(result[0].length).toBeGreaterThan(result[result.length - 1].length);
  });
});

describe("filename", () => {
  const testCases = [
    { input: "/path/to/file.ts", expected: "file" },
    { input: "C:\\path\\to\\file.d.ts", expected: "file.d" },
    { input: "/path/to/.gitignore", expected: ".gitignore" },
    { input: "file-only.js", expected: "file-only" },
    { input: "/path/to/", expected: undefined },
    { input: "", expected: undefined }
  ];

  for (const { input, expected } of testCases) {
    it(`should extract filename for ${JSON.stringify(input)}`, () => {
      expect(filename(input)).toBe(expected);
    });
  }
});
```
