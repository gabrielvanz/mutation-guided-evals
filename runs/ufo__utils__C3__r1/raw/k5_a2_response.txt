describe("isRelative", () => {
  const cases = [
    { input: "./foo", expected: true },
    { input: "../foo", expected: true },
    { input: "/foo", expected: false },
    { input: "foo", expected: false },
    { input: "", expected: false },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${expected} for "${input}"`, () => {
      expect(isRelative(input)).toBe(expected);
    });
  }
});

describe("hasProtocol", () => {
  const cases = [
    { input: "https://example.com", expected: true },
    { input: "ftp://example.com", expected: true },
    { input: "data:text/plain", expected: true },
    { input: "a:b", expected: false },
    { input: "//example.com", expected: false },
    { input: "/foo/bar", expected: false },
    { input: "", expected: false },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${expected} for "${input}"`, () => {
      expect(hasProtocol(input)).toBe(expected);
    });
  }

  it("should accept relative URLs when acceptRelative option is true", () => {
    expect(hasProtocol("//example.com", { acceptRelative: true })).toBe(true);
    expect(hasProtocol("//example.com", true)).toBe(true);
  });

  it("should respect strict option for protocols", () => {
    expect(hasProtocol("data:text/plain", { strict: true })).toBe(false);
    expect(hasProtocol("https://example.com", { strict: true })).toBe(true);
    expect(hasProtocol("http://example.com", { strict: true })).toBe(true);
  });

  it("should correctly handle strict protocol anchors", () => {
    expect(hasProtocol("abc https://example.com", { strict: true })).toBe(true);
  });
});

describe("isScriptProtocol", () => {
  const cases = [
    { input: "javascript:", expected: true },
    { input: "data:", expected: true },
    { input: "blob:", expected: true },
    { input: "vbscript:", expected: true },
    { input: "HTTPS:", expected: false },
    { input: "https://example.com", expected: false },
    { input: "", expected: false },
    { input: undefined, expected: false },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${expected} for protocol "${input}"`, () => {
      expect(isScriptProtocol(input)).toBe(expected);
    });
  }

  it("should be case-insensitive and strict about whitespace", () => {
    expect(isScriptProtocol("JAVASCRIPT:")).toBe(true);
    expect(isScriptProtocol("\tjavascript:")).toBe(true);
    expect(isScriptProtocol(" SCRIPT:javascript:")).toBe(false);
  });
});

describe("hasTrailingSlash", () => {
  const cases = [
    { input: "/foo/", expected: true },
    { input: "/foo", expected: false },
    { input: "", expected: false },
  ];

  for (const { input, expected } of cases) {
    it(`should return ${expected} for "${input}" without respecting query`, () => {
      expect(hasTrailingSlash(input)).toBe(expected);
    });
  }

  it("should respect query and fragment when specified", () => {
    expect(hasTrailingSlash("/foo?query=true", true)).toBe(false);
    expect(hasTrailingSlash("/foo/?query=true", true)).toBe(true);
    expect(hasTrailingSlash("/foo/#hash", true)).toBe(true);
    expect(hasTrailingSlash("/foo/#", true)).toBe(true);
  });
});

describe("withoutTrailingSlash", () => {
  const cases = [
    { input: "/foo/", expected: "/foo" },
    { input: "/foo", expected: "/foo" },
    { input: "", expected: "/" },
  ];

  for (const { input, expected } of cases) {
    it(`should return "${expected}" for "${input}"`, () => {
      expect(withoutTrailingSlash(input)).toBe(expected);
    });
  }

  it("should respect query and fragment when specified", () => {
    expect(withoutTrailingSlash("/path/?query=true", true)).toBe("/path?query=true");
    expect(withoutTrailingSlash("/path/#hash", true)).toBe("/path#hash");
    expect(withoutTrailingSlash("/path/", true)).toBe("/path");
    expect(withoutTrailingSlash("", true)).toBe("/");
    expect(withoutTrailingSlash("/path?q=1/", true)).toBe("/path?q=1/");
  });

  it("should bypass respectQueryAndFragment when false or omitted", () => {
    expect(withoutTrailingSlash("/foo/?bar=1", false)).toBe("/foo/?bar=1");
  });
});

describe("withTrailingSlash", () => {
  const cases = [
    { input: "/foo", expected: "/foo/" },
    { input: "/foo/", expected: "/foo/" },
    { input: "", expected: "/" },
  ];

  for (const { input, expected } of cases) {
    it(`should return "${expected}" for "${input}"`, () => {
      expect(withTrailingSlash(input)).toBe(expected);
    });
  }

  it("should respect query and fragment when specified", () => {
    expect(withTrailingSlash("/path?query=true", true)).toBe("/path/?query=true");
    expect(withTrailingSlash("/path#hash", true)).toBe("/path/#hash");
    expect(withTrailingSlash("#hash", true)).toBe("#hash");
    expect(withTrailingSlash("", true)).toBe("/");
    expect(withTrailingSlash("/path/", true)).toBe("/path/");
  });

  it("should return false correctly if already has trailing slash with respectQueryAndFragment", () => {
    expect(withTrailingSlash("/path/?q=1", true)).toBe("/path/?q=1");
  });
});

describe("withBase", () => {
  const cases = [
    { input: "/foo/bar", base: "/foo", expected: "/foo/bar" },
    { input: "/foo/bar", base: "/baz", expected: "/baz/foo/bar" },
    { input: "https://example.com", base: "/foo", expected: "https://example.com" },
    { input: "/admin-dashboard", base: "/admin/", expected: "/admin/admin-dashboard" },
  ];

  for (const { input, base, expected } of cases) {
    it(`should return "${expected}" for input "${input}" with base "${base}"`, () => {
      expect(withBase(input, base)).toBe(expected);
    });
  }
});

describe("withoutBase", () => {
  const cases = [
    { input: "/foo/bar", base: "/foo", expected: "/bar" },
    { input: "/baz/foo", base: "/foo", expected: "/baz/foo" },
    { input: "/admin-dashboard", base: "/admin/", expected: "/admin-dashboard" },
    { input: "/legacy//evil.com", base: "/legacy", expected: "/evil.com" },
    { input: "/foo", base: "", expected: "/foo" },
  ];

  for (const { input, base, expected } of cases) {
    it(`should return "${expected}" for input "${input}" without base "${base}"`, () => {
      expect(withoutBase(input, base)).toBe(expected);
    });
  }

  it("should not strip global slashes if base is empty string or empty URL", () => {
    expect(withoutBase("/foo", "")).toBe("/foo");
    expect(withoutBase("/foo", "/")).toBe("/foo");
  });

  it("should strip multiple consecutive leading slashes correctly", () => {
    expect(withoutBase("/legacy///evil.com", "/legacy")).toBe("/evil.com");
  });
});

describe("withQuery", () => {
  it("should add or replace the query section of the URL", () => {
    expect(withQuery("/foo?page=a", { token: "secret" })).toBe("/foo?page=a&token=secret");
    expect(withQuery("/foo?page=a", { page: "b" })).toBe("/foo?page=b");
  });
});

describe("filterQuery", () => {
  it("should filter query keys using predicate", () => {
    expect(filterQuery("/foo?bar=1&baz=2", (key) => key !== "bar")).toBe("/foo?baz=2");
    expect(filterQuery("/foo", (key) => key !== "bar")).toBe("/foo");
  });
});

describe("isNonEmptyURL", () => {
  it("should return false for empty or slash URL", () => {
    expect(isNonEmptyURL("")).toBeFalsy();
    expect(isNonEmptyURL("/")).toBeFalsy();
    expect(isNonEmptyURL("/foo")).toBe(true);

describe("joinURL", () => {
  it("should join multiple URL segments into a single URL", () => {
    expect(joinURL("a", "/b", "/c")).toBe("a/b/c");
    expect(joinURL("http://example.com", "/foo", "bar")).toBe("http://example.com/foo/bar");
  });
});

describe("joinRelativeURL", () => {
  it("should join relative URL segments", () => {
    expect(joinRelativeURL("/a", "../b", "./c")).toBe("/b/c");
    expect(joinRelativeURL("/", "a")).toBe("/a");
    expect(joinRelativeURL("/a/b", "..")).toBe("/a");
    expect(joinRelativeURL("http://example.com", "a", "..", "b")).toBe("http://example.com/b");
    expect(joinRelativeURL("./a", "../b")).toBe("./b");
    expect(joinRelativeURL("a/", "b/")).toBe("a/b/");
  });

  it("should handle single slash segment inputs correctly", () => {
    expect(joinRelativeURL("/a", "/", "b")).toBe("/a/b");
    expect(joinRelativeURL("/a/b/c", "../..")).toBe("/a");
    expect(joinRelativeURL("./a", "./b")).toBe("./a/b");
    expect(joinRelativeURL("../a", "b")).toBe("../a/b");
  });
});

describe("withHttp", () => {

describe("withoutProtocol", () => {
  it("should remove protocol from the input", () => {
    expect(withoutProtocol("http://example.com")).toBe("example.com");
    expect(withoutProtocol("example.com")).toBe("example.com");
  });

describe("withProtocol", () => {
  it("should add or replace protocol of the input URL", () => {
    expect(withProtocol("http://example.com", "ftp://")).toBe("ftp://example.com");
    expect(withProtocol("example.com", "https://")).toBe("https://example.com");
  });

  it("should handle protocol-relative paths or missing matches correctly", () => {
    expect(withProtocol("//example.com", "https://")).toBe("https://example.com");
    expect(withProtocol("foo/bar", "http://")).toBe("http://foo/bar");
  });
});

describe("normalizeURL", () => {
  it("should normalize the input URL", () => {
    expect(normalizeURL("test?query=123%20123#hash,%20test")).toBe("test?query=123+123#hash,%20test");
    expect(normalizeURL("http://localhost:3000")).toBe("http://localhost:3000");
  });
});

describe("resolveURL", () => {
  it("should resolve multiple URL segments", () => {
    expect(resolveURL("http://foo.com/foo?test=123#token", "bar", "baz")).toBe("http://foo.com/foo/bar/baz?test=123#token");
    expect(resolveURL("http://foo.com/foo?a=1", "?b=2")).toBe("http://foo.com/foo?a=1&b=2");
    expect(resolveURL("http://foo.com/foo", "bar?a=1")).toBe("http://foo.com/foo/bar?a=1");
  });

  it("should throw TypeError if base is not a string", () => {
    expect(() => resolveURL(123 as any, "bar")).toThrow(TypeError);
  });

  it("should return base when no valid input segments are provided", () => {
    expect(resolveURL("http://foo.com", "", "/")).toBe("http://foo.com");
  });

  it("should correctly handle merging search queries with short-circuit and non-empty checks", () => {
    expect(resolveURL("http://foo.com?a=1", "bar?b=2")).toBe("http://foo.com/bar?a=1&b=2");
    expect(resolveURL("http://foo.com", "?b=2")).toBe("http://foo.com?b=2");
  });
});

describe("isSamePath", () => {
  it("should check if two paths are equal", () => {
    expect(isSamePath("/foo", "/foo/")).toBe(true);
    expect(isSamePath("/foo%20bar", "/foo bar/")).toBe(true);
    expect(isSamePath("/foo", "/bar")).toBe(false);
  });
});

describe("isEqual", () => {
  it("should check if two paths are equal with default options", () => {
    expect(isEqual("/foo", "foo")).toBe(true);
    expect(isEqual("foo/", "foo")).toBe(true);
    expect(isEqual("/foo bar", "/foo%20bar")).toBe(true);
  });

  it("should support strict compare options", () => {
    expect(isEqual("/foo", "foo", { leadingSlash: true })).toBe(false);
    expect(isEqual("foo/", "foo", { trailingSlash: true })).toBe(false);
    expect(isEqual("/foo bar", "/foo%20bar", { encoding: true })).toBe(false);
  });
});

describe("withFragment", () => {
  it("should add or replace the fragment section of the URL", () => {
    expect(withFragment("/foo", "bar")).toBe("/foo#bar");
    expect(withFragment("/foo#bar", "baz")).toBe("/foo#baz");
    expect(withFragment("/foo#bar", "")).toBe("/foo#bar");
    expect(withFragment("/foo", "#")).toBe("/foo");
  });
});

describe("withoutFragment", () => {
  it("should remove the fragment section from the URL", () => {
    expect(withoutFragment("http://example.com/foo?q=123#bar")).toBe("http://example.com/foo?q=123");
  });
});

describe("withoutHost", () => {
  it("should remove the host from the URL", () => {
    expect(withoutHost("http://example.com/foo?q=123#bar")).toBe("/foo?q=123#bar");
  });
});
