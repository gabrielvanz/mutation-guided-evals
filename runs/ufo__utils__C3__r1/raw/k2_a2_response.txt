```ts
import { describe, it, expect } from "vitest";
import {
  isRelative,
  hasProtocol,
  isScriptProtocol,
  hasTrailingSlash,
  withoutTrailingSlash,
  withTrailingSlash,
  hasLeadingSlash,
  withoutLeadingSlash,
  withLeadingSlash,
  cleanDoubleSlashes,
  withBase,
  withoutBase,
  withQuery,
  filterQuery,
  getQuery,
  isEmptyURL,
  isNonEmptyURL,
  joinURL,
  joinRelativeURL,
  withHttp,
  withHttps,
  withoutProtocol,
  withProtocol,
  normalizeURL,
  resolveURL,
  isSamePath,
  isEqual,
  withFragment,
  withoutFragment,
  withoutHost,
} from "../../src/utils";

describe("isRelative", () => {
  it.each([
    { input: "./foo", expected: true },
    { input: "../foo", expected: true },
    { input: "/foo", expected: false },
    { input: "foo", expected: false },
    { input: "", expected: false },
  ])("should return $expected for $input", ({ input, expected }) => {
    expect(isRelative(input)).toBe(expected);
  });
});

describe("hasProtocol", () => {
  it.each([
    { input: "https://example.com", expected: true },
    { input: "ftp://example.com", expected: true },
    { input: "data:text/plain", expected: true },
    { input: "//example.com", expected: false },
    { input: "//example.com", opts: { acceptRelative: true }, expected: true },
    { input: "//example.com", opts: true, expected: true },
    { input: "data:text/plain", opts: { strict: true }, expected: false },
    { input: "https://example.com", opts: { strict: true }, expected: true },
    { input: "/foo/bar", expected: false },
  ])("should correctly determine protocol for $input", ({ input, opts, expected }) => {
    expect(hasProtocol(input, opts as any)).toBe(expected);
  });
});

describe("isScriptProtocol", () => {
  it.each([
    { input: "javascript:", expected: true },
    { input: "data:", expected: true },
    { input: "blob:", expected: true },
    { input: "vbscript:", expected: true },
    { input: "https://example.com", expected: false },
    { input: undefined, expected: false },
  ])("should correctly identify script protocol for $input", ({ input, expected }) => {
    expect(isScriptProtocol(input)).toBe(expected);
  });
});

describe("hasTrailingSlash", () => {
  it.each([
    { input: "/foo/", expected: true },
    { input: "/foo", expected: false },
    { input: "/foo?query=true", respect: true, expected: false },
    { input: "/foo/?query=true", respect: true, expected: true },
    { input: "", respect: false, expected: false },
  ])("should return $expected for hasTrailingSlash($input, $respect)", ({ input, respect, expected }) => {
    expect(hasTrailingSlash(input, respect)).toBe(expected);
  });
});

describe("withoutTrailingSlash", () => {
  it.each([
    { input: "/foo/", respect: false, expected: "/foo" },
    { input: "/foo", respect: false, expected: "/foo" },
    { input: "", respect: false, expected: "/" },
    { input: "/path/?query=true", respect: true, expected: "/path?query=true" },
    { input: "/path/#hash", respect: true, expected: "/path#hash" },
    { input: "/", respect: false, expected: "/" },
  ])("should remove trailing slash correctly for $input", ({ input, respect, expected }) => {
    expect(withoutTrailingSlash(input, respect)).toBe(expected);
  });
});

describe("withTrailingSlash", () => {
  it.each([
    { input: "/foo", respect: false, expected: "/foo/" },
    { input: "/foo/", respect: false, expected: "/foo/" },
    { input: "", respect: false, expected: "/" },
    { input: "/path?query=true", respect: true, expected: "/path/?query=true" },
    { input: "#hash", respect: true, expected: "#hash" },
  ])("should ensure trailing slash correctly for $input", ({ input, respect, expected }) => {
    expect(withTrailingSlash(input, respect)).toBe(expected);
  });
});

describe("hasLeadingSlash", () => {
  it.each([
    { input: "/foo", expected: true },
    { input: "foo", expected: false },
    { input: "", expected: false },
  ])("should return $expected for hasLeadingSlash($input)", ({ input, expected }) => {
    expect(hasLeadingSlash(input)).toBe(expected);
  });
});

describe("withoutLeadingSlash", () => {
  it.each([
    { input: "/foo", expected: "foo" },
    { input: "foo", expected: "foo" },
    { input: "", expected: "/" },
  ])("should remove leading slash for $input", ({ input, expected }) => {
    expect(withoutLeadingSlash(input)).toBe(expected);
  });
});

describe("withLeadingSlash", () => {
  it.each([
    { input: "foo", expected: "/foo" },
    { input: "/foo", expected: "/foo" },
    { input: "", expected: "/" },
  ])("should ensure leading slash for $input", ({ input, expected }) => {
    expect(withLeadingSlash(input)).toBe(expected);
  });
});

describe("cleanDoubleSlashes", () => {
  it.each([
    { input: "//foo//bar//", expected: "/foo/bar/" },
    { input: "http://example.com/analyze//http://localhost:3000//", expected: "http://example.com/analyze/http://localhost:3000/" },
    { input: "", expected: "" },
  ])("should clean double slashes for $input", ({ input, expected }) => {
    expect(cleanDoubleSlashes(input)).toBe(expected);
  });
});

describe("withBase", () => {
  it.each([
    { input: "/foo/bar", base: "/foo", expected: "/foo/bar" },
    { input: "/foo/bar", base: "/baz", expected: "/baz/foo/bar" },
    { input: "/admin-dashboard", base: "/admin/", expected: "/admin/admin-dashboard" },
    { input: "https://example.com", base: "/foo", expected: "https://example.com" },
    { input: "/foo/bar", base: "", expected: "/foo/bar" },
  ])("should prepend base $base to $input correctly", ({ input, base, expected }) => {
    expect(withBase(input, base)).toBe(expected);
  });
});

describe("withoutBase", () => {
  it.each([
    { input: "/foo/bar", base: "/foo", expected: "/bar" },
    { input: "/admin-dashboard", base: "/admin/", expected: "/admin-dashboard" },
    { input: "/foo/bar", base: "/baz", expected: "/foo/bar" },
    { input: "/foo/bar", base: "", expected: "/foo/bar" },
    { input: "/legacy//evil.com", base: "/legacy", expected: "/evil.com" },
  ])("should remove base $base from $input correctly", ({ input, base, expected }) => {
    expect(withoutBase(input, base)).toBe(expected);
  });
});

describe("withQuery", () => {
  it("should add or replace query parameters", () => {
    expect(withQuery("/foo?page=a", { token: "secret" })).toBe("/foo?page=a&token=secret");
  });
});

describe("filterQuery", () => {
  it("should filter query parameters based on predicate", () => {
    expect(filterQuery("/foo?bar=1&baz=2", (key) => key !== "bar")).toBe("/foo?baz=2");
  });

  it("should return input as-is if no query exists", () => {
    expect(filterQuery("/foo", () => true)).toBe("/foo");
  });
});

describe("getQuery", () => {
  it("should parse and decode query object", () => {
    expect(getQuery("http://foo.com/foo?test=123&unicode=%E5%A5%BD")).toEqual({
      test: "123",
      unicode: "好",
    });
  });
});

describe("isEmptyURL / isNonEmptyURL", () => {
  it.each([
    { url: "", empty: true, nonEmpty: false },
    { url: "/", empty: true, nonEmpty: false },
    { url: "/foo", empty: false, nonEmpty: true },
  ])("should correctly check empty/non-empty status for $url", ({ url, empty, nonEmpty }) => {
    expect(isEmptyURL(url)).toBe(empty);
    expect(isNonEmptyURL(url)).toBe(nonEmpty);
  });
});

describe("joinURL", () => {
  it.each([
    { base: "a", inputs: ["/b", "/c"], expected: "a/b/c" },
    { base: "", inputs: ["/b", "c"], expected: "/b/c" },
  ])("should join URL segments correctly", ({ base, inputs, expected }) => {
    expect(joinURL(base, ...inputs)).toBe(expected);
  });
});

describe("joinRelativeURL", () => {
  it.each([
    { inputs: ["/a", "../b", "./c"], expected: "/b/c" },
    { inputs: ["./a", "./b"], expected: "./a/b" },
  ])("should join relative URL segments correctly", ({ inputs, expected }) => {
    expect(joinRelativeURL(...inputs)).toBe(expected);
  });
});

describe("withHttp / withHttps / withoutProtocol / withProtocol", () => {
  it("should add http protocol", () => {
    expect(withHttp("https://example.com")).toBe("http://example.com");
  });

  it("should add https protocol", () => {
    expect(withHttps("http://example.com")).toBe("https://example.com");
  });

  it("should remove protocol", () => {
    expect(withoutProtocol("http://example.com")).toBe("example.com");
  });

  it("should replace protocol with custom one", () => {
    expect(withProtocol("http://example.com", "ftp://")).toBe("ftp://example.com");
    expect(withProtocol("example.com", "http://")).toBe("http://example.com");
  });
});

describe("normalizeURL", () => {
  it("should normalize URL correctly", () => {
    expect(normalizeURL("test?query=123 123#hash, test")).toBe("test?query=123+123#hash,%20test");
    expect(normalizeURL("http://localhost:3000")).toBe("http://localhost:3000");
  });
});

describe("resolveURL", () => {
  it("should resolve multiple URL segments", () => {
    expect(resolveURL("http://foo.com/foo?test=123#token", "bar", "baz")).toBe("http://foo.com/foo/bar/baz?test=123#token");
  });

  it("should return base when no inputs are provided", () => {
    expect(resolveURL("http://foo.com")).toBe("http://foo.com");
  });

  it("should throw TypeError when base is not a string", () => {
    expect(() => resolveURL(123 as any)).toThrow(TypeError);
  });
});

describe("isSamePath", () => {
  it.each([
    { p1: "/foo", p2: "/foo/", expected: true },
    { p1: "/foo%20bar", p2: "/foo bar", expected: true },
    { p1: "/foo", p2: "/bar", expected: false },
  ])("should compare paths $p1 and $p2 as $expected", ({ p1, p2, expected }) => {
    expect(isSamePath(p1, p2)).toBe(expected);
  });
});

describe("isEqual", () => {
  it.each([
    { a: "/foo", b: "foo", opts: undefined, expected: true },
    { a: "foo/", b: "foo", opts: undefined, expected: true },
    { a: "/foo bar", b: "/foo%20bar", opts: undefined, expected: true },
    { a: "/foo", b: "foo", opts: { leadingSlash: true }, expected: false },
    { a: "foo/", b: "foo", opts: { trailingSlash: true }, expected: false },
    { a: "/foo bar", b: "/foo%20bar", opts: { encoding: true }, expected: false },
  ])("should evaluate isEqual for $a and $b", ({ a, b, opts, expected }) => {
    expect(isEqual(a, b, opts)).toBe(expected);
  });
});

describe("withFragment", () => {
  it.each([
    { input: "/foo", hash: "bar", expected: "/foo#bar" },
    { input: "/foo#bar", hash: "baz", expected: "/foo#baz" },
    { input: "/foo#bar", hash: "", expected: "/foo#bar" },
    { input: "/foo", hash: "#", expected: "/foo" },
  ])("should handle fragments for $input with hash $hash", ({ input, hash, expected }) => {
    expect(withFragment(input, hash)).toBe(expected);
  });
});

describe("withoutFragment", () => {
  it("should remove fragment from URL", () => {
    expect(withoutFragment("http://example.com/foo?q=123#bar")).toBe("http://example.com/foo?q=123");
  });
});

describe("withoutHost", () => {
  it("should remove host from URL", () => {
    expect(withoutHost("http://example.com/foo?q=123#bar")).toBe("/foo?q=123#bar");
  });
});
```
