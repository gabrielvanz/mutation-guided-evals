```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on protected route", async () => {
      const response = await request(app).get("/carrinhos");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token format is invalid", async () => {
      const response = await request(app)
        .get("/carrinhos")
        .set("authorization", "InvalidFormat token123");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is malformed Bearer token", async () => {
      const response = await request(app)
        .get("/carrinhos")
        .set("authorization", "Bearer");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is completely invalid", async () => {
      const response = await request(app)
        .get("/carrinhos")
        .set("authorization", "Bearer invalid.jwt.token");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 403 when a non-admin user tries to access admin-only route", async () => {
      const uniqueSuffix = Math.random().toString(36).substring(2, 9);
      const email = `user_${uniqueSuffix}@test.com`;
      const password = "password123";

      await request(app).post("/usuarios").send({
        nome: "Test User",
        email,
        password,
        administrador: "false"
      });

      const loginRes = await request(app).post("/login").send({
        email,
        password
      });

      const token = loginRes.body.authorization;

      const response = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Teste",
          preco: 10,
          descricao: "Desc",
          quantidade: 10
        });

      expect(response.status).toBe(403);
      expect(response.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 when sending invalid JSON payload", async () => {
      const response = await request(app)
        .post("/usuarios")
        .set("Content-Type", "application/json")
        .send('{"nome": "bad json');

      expect(response.status).toBe(400);
      expect(response.body).toHaveProperty("message");
    });
  });
});
```
