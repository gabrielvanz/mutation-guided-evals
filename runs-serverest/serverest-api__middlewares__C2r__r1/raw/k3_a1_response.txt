```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on protected routes", async () => {
      const response = await request(app).get("/carrinhos");
      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token has bad format", async () => {
      const response = await request(app)
        .get("/carrinhos")
        .set("authorization", "InvalidFormat 12345");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is invalid or malformed", async () => {
      const response = await request(app)
        .get("/carrinhos")
        .set("authorization", "Bearer invalid.token.string");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 403 when a non-admin user tries to access admin-only routes", async () => {
      const email = "regular_user_test@test.com";
      await request(app)
        .post("/usuarios")
        .send({
          nome: "Regular User",
          email,
          password: "testpassword",
          administrador: "false"
        });

      const loginResponse = await request(app)
        .post("/login")
        .send({
          email,
          password: "testpassword"
        });

      const token = loginResponse.body.authorization;

      const response = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Teste",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(response.status).toBe(403);
      expect(response.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should handle malformed JSON body errors", async () => {
      const response = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send("{ invalid json ");

      expect(response.status).toBe(400);
      expect(response.body.message).toContain("Adicione aspas em todos os valores");
    });

    it("should handle URI decode error on malformed URL parameters", async () => {
      const response = await request(app).get("/usuarios/%E0%A2%A");

      expect(response.status).toBe(400);
      expect(response.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });
  });
});
```
