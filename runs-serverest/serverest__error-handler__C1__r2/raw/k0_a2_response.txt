import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler middleware", () => {
  it("should handle headers already sent when response is not writable ended", () => {
    const error = new Error("Test error");
    const req = { method: "GET", originalUrl: "/test", timedout: false, socket: { destroy: () => {} } };
    const res = { headersSent: true, writableEnded: false };
    const next = () => {};

    let socketDestroyed = false;
    req.socket.destroy = () => { socketDestroyed = true; };

    errorHandler(error, req, res, next);

    expect(socketDestroyed).toBe(true);
  });

  it("should handle ValidationError by returning status 400", () => {
    const error = new Error("Validation failed") as any;
    error.name = "ValidationError";
    error.inner = [];
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      status: (code: number) => {
        statusCode = code;
        return {
          json: (data: any) => {
            jsonBody = data;
            return data;
          }
        };
      }
    };
    const next = () => {};

    errorHandler(error, req, res as any, next);
    expect(statusCode).toBe(400);
    expect(jsonBody).toBeDefined();
  });

  it("should handle URIError by returning status 400", () => {
    const error = new Error("URI malformed");
