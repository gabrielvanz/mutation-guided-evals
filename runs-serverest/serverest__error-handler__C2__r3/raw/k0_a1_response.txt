```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent when request timed out", () => {
    const error = new Error("Timeout occurred");
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: {
        destroy: () => {}
      }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});
    expect(res.headersSent).toBe(true);
  });

  it("should handle headersSent when request did not timeout and response is not writable ended", () => {
    const error = new Error("Some error");
    let socketDestroyed = false;
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/test",
      socket: {
        destroy: () => {
          socketDestroyed = true;
        }
      }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});
    expect(socketDestroyed).toBe(true);
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: []
    };
    const req = {};
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return res;
      },
      json: (body: any) => {
        jsonBody = body;
        return res;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonBody).toBeDefined();
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = {};
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return res;
      },
      json: (body: any) => {
        jsonBody = body;
        return res;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = {
      type: "entity.parse.failed"
    };
    const req = {};
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return res;
      },
      json: (body: any) => {
        jsonBody = body;
        return res;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const error = {
      type: "entity.too.large"
    };
    const req = {};
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return res;
      },
      json: (body: any) => {
        jsonBody = body;
        return res;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = {
      type: "charset.unsupported",
      charset: "utf-16"
    };
    const req = {};
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return res;
      },
      json: (body: any) => {
        jsonBody = body;
        return res;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = {
      type: "encoding.unsupported",
      encoding: "deflate"
    };
    const req = {};
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return res;
      },
      json: (body: any) => {
        jsonBody = body;
        return res;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle client error with Z_ prefix code", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "Data error"
    };
    const req = {};
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return res;
      },
      json: (body: any) => {
        jsonBody = body;
        return res;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT code", () => {
    const error = {
      code: "ETIMEDOUT"
    };
    const req = {};
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return res;
      },
      json: (body: any) => {
        jsonBody = body;
        return res;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(408);
  });

  it("should handle generic client error", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "some.client.error"
    };
    const req = {};
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return res;
      },
      json: (body: any) => {
        jsonBody = body;
        return res;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.client.error"
    });
  });

  it("should handle internal error with type", () => {
    const error = {
      type: "database.error",
      stack: "at stack"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return res;
      },
      json: (body: any) => {
        jsonBody = body;
        return res;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(500);
    expect(jsonBody.error).toBe("database.error");
  });
});
```
