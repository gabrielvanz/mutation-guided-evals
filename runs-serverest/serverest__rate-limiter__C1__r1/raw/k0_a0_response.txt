```ts
import { describe, it, expect, vi } from "vitest";
import mod from "../../src/middlewares/rate-limiter";

describe("rate-limiter middleware", () => {
  it("exports a function as default", () => {
    expect(typeof mod).toBe("function");
  });

  it("exports helper factories and constants", () => {
    expect(typeof mod.criarLimitador).toBe("function");
    expect(typeof mod.criarOrcamentoDeBytes).toBe("function");
    expect(typeof mod.criarRateLimiter).toBe("function");
    expect(typeof mod.BYTES_POR_HORA).toBe("number");
  });

  it("bypasses rate limiting in test environment", async () => {
    const req = {};
    const res = {
      locals: {
        etapas: {
          ignorar: vi.fn(),
        },
      },
    };
    const next = vi.fn();

    await mod(req, res, next);

    expect(res.locals.etapas.ignorar).toHaveBeenCalled();
    expect(next).toHaveBeenCalled();
  });
});
```
