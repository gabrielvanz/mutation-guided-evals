import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent when request timed out", () => {
    const error = new Error("Timeout after response");
    const socketDestroy = { destroy: () => {} };
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/users",
      socket: socketDestroy
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };
    const next = () => {};

    const result = errorHandler(error, req, res, next);
    expect(result).toBeUndefined();
  });

  it("should handle headersSent when request did not timeout", () => {
    const error = new Error("Late error");
    const socketDestroy = { destroy: () => {} };
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/login",
      socket: socketDestroy
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };
    const next = () => {};

    const result = errorHandler(error, req, res, next);
    expect(result).toBeUndefined();
  });

  it("should handle ValidationError", () => {
    const error = { name: "ValidationError", details: [] };
    const req = { body: {} };
    const jsonMock = { message: "Erro de schema" };
    const res = {
      headersSent: false,
      status: (code: number) => {
        expect(code).toBe(400);
        return {
          json: (body: unknown) => {
            expect(body).toBeDefined();
            return body;
          }
        };
      }
    };
    const next = () => {};

    const result = errorHandler(error, req, res, next);
    expect(result).toBeDefined();
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = { body: {} };
    const res = {
      headersSent: false,
      status: (code: number) => {
        expect(code).toBe(400);
        return {
          json: (body: unknown) => {
            expect(body).toEqual({
              message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
            });
            return body;
          }
        };
      }
    };
    const next = () => {};

    const result = errorHandler(error, req, res, next);
    expect(result).toBeDefined();
  });

  it("should handle entity.parse.failed error type", () => {

            });
            return body;
          }
        };
      }
    };
    const next = () => {};

    const result = errorHandler(error, req, res, next);
    expect(result).toBeDefined();
  });

  it("should handle entity.too.large error type", () => {
    const error = { type: "entity.too.large" };
    const req = { body: {} };
    const res = {
      headersSent: false,
      status: (code: number) => {
        expect(code).toBe(413);
        return {
          json: (body: unknown) => {
            expect(body).toEqual({ message: "Payload too large" });
            return body;
          }
        };
      }
    };
    const next = () => {};

    const result = errorHandler(error, req, res, next);
    expect(result).toBeDefined();
  });

  it("should handle charset.unsupported error type", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = { body: {} };
    const res = {
      headersSent: false,

        return {
          json: (body: unknown) => {
            expect(body).toEqual({ message: "Unsupported charset", charset: "utf-16" });

      }

        return {

    };

  it("should handle internal error with error type present", () => {
    const error = { type: "database.error", message: "DB down" };
    const req = { body: { test: 1 } };
    const res = {
      headersSent: false,

        expect(code).toBe(500);
        return {

            expect(body).toHaveProperty("message");
            expect(body).toHaveProperty("error", "database.error");
            expect(body).toHaveProperty("version");

          }
        };
      }
    };
    const next = () => {};
