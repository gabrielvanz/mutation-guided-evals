```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent by destroying socket when not timedout", () => {
    const error = new Error("Test error");
    error.stack = "stack trace";
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: false,
      socket: {
        destroy: () => {}
      }
    };
    let destroyed = false;
    req.socket.destroy = () => {
      destroyed = true;
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});
    expect(destroyed).toBe(true);
  });

  it("should handle headersSent when response is already writableEnded", () => {
    const error = new Error("Test error");
    error.stack = "stack trace";
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: true,
      socket: {
        destroy: () => {}
      }
    };
    let destroyed = false;
    req.socket.destroy = () => {
      destroyed = true;
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    errorHandler(error, req, res, () => {});
    expect(destroyed).toBe(false);
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: [{ message: "field is required" }]
    };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonBody = body;
        return this;
      }
    };
    const req = { body: {} };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({ message: "field is required" });
  });

  it("should handle URIError", () => {
    const error = new URIError("malformed URI");
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonBody = body;
        return this;
      }
    };
    const req = { body: {} };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = {
      type: "entity.parse.failed"
    };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonBody = body;
        return this;
      }
    };
    const req = { body: {} };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it.each([
    "entity.too.large",
    "parameters.too.many"
  ])("should handle payload too large for type %s", (type) => {
    const error = { type };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonBody = body;
        return this;
      }
    };
    const req = { body: {} };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = {
      type: "charset.unsupported",
      charset: "utf-16"
    };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonBody = body;
        return this;
      }
    };
    const req = { body: {} };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = {
      type: "encoding.unsupported",
      encoding: "deflate"
    };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonBody = body;
        return this;
      }
    };
    const req = { body: {} };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle client error with Z_ prefix code", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "unexpected end of file"
    };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonBody = body;
        return this;
      }
    };
    const req = { body: {} };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = {
      code: "ETIMEDOUT"
    };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonBody = body;
        return this;
      }
    };
    const req = { body: {} };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(408);
    expect(jsonBody).toHaveProperty("message");
  });

  it("should handle generic client error", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "some.client.error"
    };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonBody = body;
        return this;
      }
    };
    const req = { body: {} };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.client.error"
    });
  });

  it("should handle internal error with error type", () => {
    const error = {
      type: "database.error",
      stack: "error stack"
    };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonBody = body;
        return this;
      }
    };
    const req = { body: {} };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(500);
    expect(jsonBody.error).toBe("database.error");
    expect(jsonBody).toHaveProperty("message");
    expect(jsonBody).toHaveProperty("version");
  });
});
```
