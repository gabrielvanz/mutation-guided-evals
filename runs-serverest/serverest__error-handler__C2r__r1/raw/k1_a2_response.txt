```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headers already sent with timedout request", () => {
    const error = new Error("Test error");
    error.stack = "Error: Test error";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const result = errorHandler(error, req, res, () => {});
    expect(result).toBeUndefined();
  });

  it("should handle headers already sent without timeout and writable ended", () => {
    const error = new Error("Test error");
    error.stack = "Error: Test error";
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/test",
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    const result = errorHandler(error, req, res, () => {});
    expect(result).toBeUndefined();
  });

  it("should return 400 for ValidationError", () => {
    const error = { name: "ValidationError", details: [] };
    const req = {};
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toBeDefined();
  });

  it("should return 400 for URIError", () => {
    const error = new URIError("Malformed URI");
    const req = {};
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should return 400 for entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = {};
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should return 413 for entity.too.large", () => {
    const error = { type: "entity.too.large" };
    const req = {};
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should return 413 for parameters.too.many", () => {
    const error = { type: "parameters.too.many" };
    const req = {};
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should return 415 for charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = {};
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should return 415 for encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate-x" };
    const req = {};
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate-x" });
  });

  it("should return 400 for client error with Z_ prefix code", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "invalid block type"
    };
    const req = {};
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should return 408 for ETIMEDOUT code", () => {
    const error = { code: "ETIMEDOUT" };
    const req = {};
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(408);
    expect(jsonBody).toHaveProperty("message");
  });

  it("should return general client error when expose is true and statusCode is 4xx", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "custom.error"
    };
    const req = {};
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.error"
    });
  });

  it("should return 500 with error type when error has type and is not handled specifically", () => {
    const error = {
      type: "database.error",
      stack: "Error stack"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(500);
    expect(jsonBody).toHaveProperty("message");
    expect(jsonBody.error).toBe("database.error");
    expect(jsonBody).toHaveProperty("version");
  });
});
```
