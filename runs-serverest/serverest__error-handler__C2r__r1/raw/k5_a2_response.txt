```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headers already sent with timedout request", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: {
        destroy: () => {}
      }
    };
    const res = {
      headersSent: true,
      writableEnded: false,
      status: (code: number) => ({ json: (body: any) => ({ code, body }), send: () => {} })
    };

    const result = errorHandler(error, req as any, res as any, () => {});
    expect(result).toBeUndefined();
  });

  it("should handle headers already sent and response writable ended", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/test",
      socket: {
        destroy: () => {}
      }
    };
    const res = {
      headersSent: true,
      writableEnded: true,
      status: (code: number) => ({ json: (body: any) => ({ code, body }), send: () => {} })
    };

    const result = errorHandler(error, req as any, res as any, () => {});
    expect(result).toBeUndefined();
  });

  it("should handle ValidationError", () => {
    const error: any = new Error("Validation error");
    error.name = "ValidationError";
    error.details = [{ message: "Field required", path: ["field"] }];
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req as any, res as any, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toHaveProperty("message");
  });

  it("should handle URIError", () => {
    const error: any = new Error("URI malformed");
    error.name = "URIError";
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req as any, res as any, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error: any = new Error("Parse failed");
    error.type = "entity.parse.failed";
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req as any, res as any, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it.each([
    "entity.too.large",
    "parameters.too.many"
  ])("should handle payload too large errors: %s", (errorType) => {
    const error: any = new Error("Too large");
    error.type = errorType;
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req as any, res as any, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error: any = new Error("Unsupported charset");
    error.type = "charset.unsupported";
    error.charset = "iso-8859-1";
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req as any, res as any, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "iso-8859-1" });
  });

  it("should handle encoding.unsupported", () => {
    const error: any = new Error("Unsupported encoding");
    error.type = "encoding.unsupported";
    error.encoding = "deflate";
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req as any, res as any, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle Zlib compression error from client", () => {
    const error: any = new Error("Z_DATA_ERROR");
    error.code = "Z_DATA_ERROR";
    error.expose = true;
    error.statusCode = 400;
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req as any, res as any, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error: any = new Error("Timeout");
    error.code = "ETIMEDOUT";
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req as any, res as any, () => {});
    expect(statusCode).toBe(408);
    expect(jsonBody).toHaveProperty("message");
  });

  it("should handle general client error", () => {
    const error: any = new Error("Bad request");
    error.expose = true;
    error.statusCode = 422;
    error.type = "custom.error.type";
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req as any, res as any, () => {});
    expect(statusCode).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.error.type"
    });
  });

  it("should handle internal error with error type", () => {
    const error: any = new Error("Internal failure");
    error.type = "database.error";
    const req = { body: { test: true } };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req as any, res as any, () => {});
    expect(statusCode).toBe(500);
    expect(jsonBody).toHaveProperty("message");
    expect(jsonBody).toHaveProperty("error", "database.error");
    expect(jsonBody).toHaveProperty("version");
  });
});
```
