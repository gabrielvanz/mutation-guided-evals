```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

async function getAdminToken() {
  const email = `admin_${Math.floor(Math.random() * 1000000)}@qa.com`;
  await request(app)
    .post("/usuarios")
    .send({
      nome: "Admin User",
      email,
      password: "password",
      administrador: "true"
    });

  const loginRes = await request(app)
    .post("/login")
    .send({
      email,
      password: "password"
    });

  return loginRes.body.authorization;
}

describe("produtos API", () => {
  describe("GET /produtos", () => {
    it("should return all products", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto Teste ${Math.floor(Math.random() * 1000000)}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Descricao teste",
          quantidade: 10
        });

      const res = await request(app).get("/produtos");
      expect(res.status).toBe(200);
      expect(res.body).toHaveProperty("quantidade");
      expect(res.body).toHaveProperty("produtos");
      expect(Array.isArray(res.body.produtos)).toBe(true);
      expect(res.body.quantidade).toBeGreaterThanOrEqual(1);
    });

    it("should filter products by query parameters", async () => {
      const token = await getAdminToken();
      const uniqueName = `FilterProd_${Math.floor(Math.random() * 1000000)}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 250,
          descricao: "Filtro",
          quantidade: 5
        });

      const res = await request(app).get(`/produtos?nome=${uniqueName}`);
      expect(res.status).toBe(200);
      expect(res.body.quantidade).toBe(1);
      expect(res.body.produtos[0].nome).toBe(uniqueName);
    });
  });

  describe("GET /produtos/:id", () => {
    it("should return a single product by id", async () => {
      const token = await getAdminToken();
      const uniqueName = `GetOne_${Math.floor(Math.random() * 1000000)}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 50,
          descricao: "Get one desc",
          quantidade: 20
        });

      const productId = postRes.body._id;

      const res = await request(app).get(`/produtos/${productId}`);
      expect(res.status).toBe(200);
      expect(res.body._id).toBe(productId);
      expect(res.body.nome).toBe(uniqueName);
    });

    it("should return 400 when product id does not exist", async () => {
      const res = await request(app).get("/produtos/1234567890abcdef");
      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Produto não encontrado" });
    });

    it("should return 400 when product id format is invalid", async () => {
      const res = await request(app).get("/produtos/invalid-id");
      expect(res.status).toBe(400);
    });
  });

  describe("POST /produtos", () => {
    it("should create a product when admin and valid payload", async () => {
      const token = await getAdminToken();
      const uniqueName = `New Product ${Math.floor(Math.random() * 1000000)}`;
      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 450,
          descricao: "Mouse Gamer",
          quantidade: 12,
          imagem: "mouse.jpg"
        });

      expect(res.status).toBe(201);
      expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
      expect(res.body).toHaveProperty("_id");
    });

    it("should return 400 when product name is already used", async () => {
      const token = await getAdminToken();
      const uniqueName = `Duplicate_${Math.floor(Math.random() * 1000000)}`;
      const payload = {
        nome: uniqueName,
        preco: 100,
        descricao: "Desc",
        quantidade: 10
      };

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send(payload);

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send(payload);

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });

    it("should return 401 when token is missing", async () => {
      const res = await request(app)
        .post("/produtos")
        .send({
          nome: "Sem Token",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(401);
      expect(res.body).toHaveProperty("message", "Token de acesso ausente.");
    });
  });

  describe("DELETE /produtos/:id", () => {
    it("should delete a product when admin and product exists", async () => {
      const token = await getAdminToken();
      const uniqueName = `DeleteProd_${Math.floor(Math.random() * 1000000)}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Del desc",
          quantidade: 10
        });

      const productId = postRes.body._id;

      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro excluído com sucesso" });
    });

    it("should return success/none when deleting a non-existent product id", async () => {
      const token = await getAdminToken();
      const res = await request(app)
        .delete("/produtos/9999999999999999")
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Nenhum registro excluído" });
    });
  });

  describe("PUT /produtos/:id", () => {
    it("should update an existing product or create if not exists", async () => {
      const token = await getAdminToken();
      const uniqueName = `PutProd_${Math.floor(Math.random() * 1000000)}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Put desc",
          quantidade: 10
        });

      const productId = postRes.body._id;
      const updatedName = `Updated_${uniqueName}`;

      const res = await request(app)
        .put(`/produtos/${productId}`)
        .set("authorization", token)
        .send({
          nome: updatedName,
          preco: 150,
          descricao: "Updated desc",
          quantidade: 5
        });

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro alterado com sucesso" });
    });

    it("should return 400 when updating with a name already used by another product", async () => {
      const token = await getAdminToken();
      const name1 = `Prod1_${Math.floor(Math.random() * 1000000)}`;
      const name2 = `Prod2_${Math.floor(Math.random() * 1000000)}`;

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({ nome: name1, preco: 100, descricao: "Desc", quantidade: 10 });

      const postRes2 = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({ nome: name2, preco: 100, descricao: "Desc", quantidade: 10 });

      const productId2 = postRes2.body._id;

      const res = await request(app)
        .put(`/produtos/${productId2}`)
        .set("authorization", token)
        .send({ nome: name1, preco: 200, descricao: "Desc", quantidade: 5 });

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });
  });
});
```
