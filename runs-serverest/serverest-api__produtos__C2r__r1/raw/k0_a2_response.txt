```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Produtos API", () => {
  let adminToken: string;
  let counter = 0;

  const getAdminToken = async () => {
    if (adminToken) return adminToken;
    counter++;
    const email = `admin_${counter}@qa.com`;
    await request(app)
      .post("/usuarios")
      .send({
        nome: "Admin Teste",
        email,
        password: "teste",
        administrador: "true"
      });
    const loginRes = await request(app)
      .post("/login")
      .send({
        email,
        password: "teste"
      });
    adminToken = loginRes.body.authorization;
    return adminToken;
  };

  describe("GET /produtos", () => {
    it("should return all products", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Produto ${counter}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Descricao",
          quantidade: 10
        });

      const res = await request(app).get("/produtos");
      expect(res.status).toBe(200);
      expect(res.body).toHaveProperty("quantidade");
      expect(res.body).toHaveProperty("produtos");
      expect(Array.isArray(res.body.produtos)).toBe(true);
      expect(res.body.quantidade).toBeGreaterThan(0);
    });

    it("should filter products by query parameters", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Filtravel ${counter}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 250,
          descricao: "Descricao filtro",
          quantidade: 5
        });

      const res = await request(app).get(`/produtos?nome=${uniqueName}`);
      expect(res.status).toBe(200);
      expect(res.body.quantidade).toBe(1);
      expect(res.body.produtos[0].nome).toBe(uniqueName);
    });
  });

  describe("GET /produtos/:id", () => {
    it("should return a product by valid id", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Produto GetOne ${counter}`;
      const createRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 150,
          descricao: "Descricao",
          quantidade: 12
        });

      const productId = createRes.body._id;

      const res = await request(app).get(`/produtos/${productId}`);
      expect(res.status).toBe(200);
      expect(res.body._id).toBe(productId);
      expect(res.body.nome).toBe(uniqueName);
    });

    it("should return 400 when product id does not exist", async () => {
      const res = await request(app).get("/produtos/abcdef1234567890");
      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Produto não encontrado" });
    });

    it("should return validation error for invalid product id format", async () => {
      const res = await request(app).get("/produtos/invalid-id");
      expect(res.status).toBe(400);
    });
  });

  describe("POST /produtos", () => {
    it("should create a new product when authenticated as admin", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Novo Produto ${counter}`;
      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 499,
          descricao: "Smartphone",
          quantidade: 20
        });

      expect(res.status).toBe(201);
      expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
      expect(res.body).toHaveProperty("_id");
    });

    it("should return 400 when product name is already used", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Duplicado ${counter}`;
      const produtoData = {
        nome: uniqueName,
        preco: 100,
        descricao: "Teste",
        quantidade: 10
      };

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send(produtoData);

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send(produtoData);

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });

    it("should return 401 when token is missing", async () => {
      const res = await request(app)
        .post("/produtos")
        .send({
          nome: "Sem Token",
          preco: 100,
          descricao: "Teste",
          quantidade: 10
        });

      expect(res.status).toBe(401);
    });
  });

  describe("PUT /produtos/:id", () => {
    it("should update an existing product or create if not found", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Produto Put ${counter}`;
      const createRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Original",
          quantidade: 5
        });

      const productId = createRes.body._id;

      const res = await request(app)
        .put(`/produtos/${productId}`)
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 120,
          descricao: "Atualizado",
          quantidade: 8
        });

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro alterado com sucesso" });
    });

    it("should return 400 when updating with an already used product name", async () => {
      const token = await getAdminToken();
      counter++;
      const name1 = `Prod 1 ${counter}`;
      counter++;
      const name2 = `Prod 2 ${counter}`;

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({ nome: name1, preco: 10, descricao: "D1", quantidade: 1 });

      const createRes2 = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({ nome: name2, preco: 20, descricao: "D2", quantidade: 2 });

      const productId2 = createRes2.body._id;

      const res = await request(app)
        .put(`/produtos/${productId2}`)
        .set("authorization", token)
        .send({
          nome: name1,
          preco: 20,
          descricao: "D2",
          quantidade: 2
        });

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });
  });

  describe("DELETE /produtos/:id", () => {
    it("should delete a product successfully when it is not in any cart", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Produto Delete ${counter}`;
      const createRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 50,
          descricao: "Deletar",
          quantidade: 5
        });

      const productId = createRes.body._id;

      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro excluído com sucesso" });
    });

    it("should return DELETE_NONE message when deleting non-existent product", async () => {
      const token = await getAdminToken();
      const res = await request(app)
        .delete("/produtos/abcdef1234567890")
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Nenhum registro excluído" });
    });
  });
});
```
