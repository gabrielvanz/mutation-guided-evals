```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Produtos API", () => {
  let adminToken: string;
  let counter = 0;

  async function getAdminToken() {
    if (adminToken) return adminToken;
    counter++;
    const email = `admin_${counter}@qa.com`;
    await request(app)
      .post("/usuarios")
      .send({
        nome: "Admin User",
        email,
        password: "password",
        administrador: "true"
      });
    const loginRes = await request(app)
      .post("/login")
      .send({
        email,
        password: "password"
      });
    adminToken = loginRes.body.authorization;
    return adminToken;
  }

  describe("GET /produtos", () => {
    it("should list all products", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Produto Teste ${counter}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Descricao",
          quantidade: 10
        });

      const res = await request(app).get("/produtos");
      expect(res.status).toBe(200);
      expect(res.body).toHaveProperty("quantidade");
      expect(res.body).toHaveProperty("produtos");
      expect(Array.isArray(res.body.produtos)).toBe(true);
      expect(res.body.quantidade).toBeGreaterThanOrEqual(1);
    });

    it("should filter products by query parameters", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Filtravel ${counter}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 250,
          descricao: "Teste filtro",
          quantidade: 5
        });

      const res = await request(app).get(`/produtos?nome=${uniqueName}`);
      expect(res.status).toBe(200);
      expect(res.body.quantidade).toBe(1);
      expect(res.body.produtos[0].nome).toBe(uniqueName);
    });
  });

  describe("GET /produtos/:id", () => {
    it("should get a product by id", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `GetOne ${counter}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 50,
          descricao: "Desc",
          quantidade: 2
        });

      const productId = postRes.body._id;
      const res = await request(app).get(`/produtos/${productId}`);
      expect(res.status).toBe(200);
      expect(res.body._id).toBe(productId);
      expect(res.body.nome).toBe(uniqueName);
    });

    it("should return 400 when product id does not exist", async () => {
      const res = await request(app).get("/produtos/abcdef1234567890");
      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Produto não encontrado" });
    });

    it("should validate id format", async () => {
      const res = await request(app).get("/produtos/invalid-id");
      expect(res.status).toBe(400);
    });
  });

  describe("POST /produtos", () => {
    it("should create a product successfully when user is admin", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Novo Produto ${counter}`;
      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 499,
          descricao: "Produto legal",
          quantidade: 20
        });

      expect(res.status).toBe(201);
      expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
      expect(res.body).toHaveProperty("_id");
    });

    it("should return 400 when product name is already used", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Duplicado ${counter}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 10,
          descricao: "Desc",
          quantidade: 5
        });

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 20,
          descricao: "Outra desc",
          quantidade: 5
        });

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });

    it("should reject creation without token", async () => {
      const res = await request(app)
        .post("/produtos")
        .send({
          nome: "Sem Token",
          preco: 10,
          descricao: "Desc",
          quantidade: 5
        });

      expect(res.status).toBe(401);
      expect(res.body).toHaveProperty("message");
    });

    it("should reject creation when user is not admin", async () => {
      counter++;
      const email = `user_${counter}@qa.com`;
      await request(app)
        .post("/usuarios")
        .send({
          nome: "Normal User",
          email,
          password: "password",
          administrador: "false"
        });
      const loginRes = await request(app)
        .post("/login")
        .send({
          email,
          password: "password"
        });
      const userToken = loginRes.body.authorization;

      const res = await request(app)
        .post("/produtos")
        .set("authorization", userToken)
        .send({
          nome: "Produto Comum",
          preco: 10,
          descricao: "Desc",
          quantidade: 5
        });

      expect(res.status).toBe(403);
      expect(res.body).toHaveProperty("message", "Rota exclusiva para administradores");
    });

    it("should validate required fields in schema", async () => {
      const token = await getAdminToken();
      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          preco: 10
        });

      expect(res.status).toBe(400);
    });
  });

  describe("PUT /produtos/:id", () => {
    it("should update an existing product or create if not found", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Put Produto ${counter}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Antiga",
          quantidade: 10
        });

      const productId = postRes.body._id;
      counter++;
      const res = await request(app)
        .put(`/produtos/${productId}`)
        .set("authorization", token)
        .send({
          nome: `Put Produto ${counter}`,
          preco: 150,
          descricao: "Nova",
          quantidade: 15
        });

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro alterado com sucesso" });
    });

    it("should return 400 when updating with a name already used by another product", async () => {
      const token = await getAdminToken();
      counter++;
      const name1 = `Prod A ${counter}`;
      counter++;
      const name2 = `Prod B ${counter}`;

      const res1 = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({ nome: name1, preco: 10, descricao: "D", quantidade: 1 });

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({ nome: name2, preco: 10, descricao: "D", quantidade: 1 });

      const id1 = res1.body._id;

      const res = await request(app)
        .put(`/produtos/${id1}`)
        .set("authorization", token)
        .send({ nome: name2, preco: 20, descricao: "D", quantidade: 2 });

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });

    it("should reject update without authorization", async () => {
      const res = await request(app)
        .put("/produtos/abcdef1234567890")
        .send({
          nome: "Teste",
          preco: 10,
          descricao: "Desc",
          quantidade: 1
        });

      expect(res.status).toBe(401);
    });
  });

  describe("DELETE /produtos/:id", () => {
    it("should delete a product successfully", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Delete Produto ${counter}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Deletar",
          quantidade: 5
        });

      const productId = postRes.body._id;
      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro excluído com sucesso" });
    });

    it("should return DELETE_NONE message when product does not exist", async () => {
      const token = await getAdminToken();
      const res = await request(app)
        .delete("/produtos/abcdef1234567890")
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Nenhum registro excluído" });
    });

    it("should prevent deletion of product associated with a cart", async () => {
      const token = await getAdminToken();
      counter++;
      const uniqueName = `Produto Carrinho ${counter}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Carrinho",
          quantidade: 10
        });

      const productId = postRes.body._id;

      counter++;
      const email = `client_${counter}@qa.com`;
      await request(app)
        .post("/usuarios")
        .send({ nome: "Client", email, password: "pwd", administrador: "false" });
      const loginRes = await request(app)
        .post("/login")
        .send({ email, password: "pwd" });
      const clientToken = loginRes.body.authorization;

      await request(app)
        .post("/carrinhos")
        .set("authorization", clientToken)
        .send({
          produtos: [{ idProduto: productId, quantidade: 2 }]
        });

      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(400);
      expect(res.body).toHaveProperty("message", "Não é permitido excluir produto que faz parte de carrinho");
      expect(res.body).toHaveProperty("idCarrinhos");
      expect(Array.isArray(res.body.idCarrinhos)).toBe(true);
    });

    it("should reject delete without authorization", async () => {
      const res = await request(app).delete("/produtos/abcdef1234567890");
      expect(res.status).toBe(401);
    });
  });
});
```
