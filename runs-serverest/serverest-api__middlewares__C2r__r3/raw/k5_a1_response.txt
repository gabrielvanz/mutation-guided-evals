```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("middlewares API", () => {
  describe("authentication-middleware & error-handler", () => {
    it("should reject request to protected route without authorization header", async () => {
      const res = await request(app).get("/usuarios");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({ message: "Token de acesso ausente, inválido ou expirado" });
    });

    it("should reject request with malformed authorization header", async () => {
      const res = await request(app)
        .get("/usuarios")
        .set("authorization", "InvalidTokenFormat");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({ message: "Token de acesso ausente, inválido ou expirado" });
    });

    it("should reject request with invalid bearer token", async () => {
      const res = await request(app)
        .get("/usuarios")
        .set("authorization", "Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.invalid.signature");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({ message: "Token de acesso ausente, inválido ou expirado" });
    });

    it("should reject non-admin user trying to access admin-only routes", async () => {
      const uniqueSuffix = Math.random().toString(36).substring(7);
      const email = `user_${uniqueSuffix}@test.com`;
      await request(app).post("/usuarios").send({
        nome: "Comum",
        email,
        password: "teste",
        administrador: "false"
      });

      const loginRes = await request(app).post("/login").send({
        email,
        password: "teste"
      });

      const token = loginRes.body.authorization;

      const res = await request(app).get("/usuarios").set("authorization", token);

      expect(res.status).toBe(403);
      expect(res.body).toEqual({ message: "Rota exclusiva para administradores" });
    });

    it("should handle malformed JSON body via error handler with 400 status", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send("{ invalid json ");

      expect(res.status).toBe(400);
      expect(res.body.message).toBeDefined();
    });

    it("should handle URIError for invalid URL encoding parameters", async () => {
      const res = await request(app).get("/usuarios/%C0%C0");

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });
  });
});
```
