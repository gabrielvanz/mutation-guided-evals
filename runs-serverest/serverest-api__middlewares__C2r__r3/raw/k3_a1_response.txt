```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on a protected route", async () => {
      const res = await request(app).get("/carrinhos");
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token has bad format", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "InvalidFormat 123456");
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is invalid", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "Bearer invalid.token.string");
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 403 when a non-admin user tries to access an admin-only route", async () => {
      const email = "regular_user_auth_test@qa.com";
      await request(app)
        .post("/usuarios")
        .send({
          nome: "Regular User",
          email,
          password: "teste",
          administrador: "false"
        });

      const loginRes = await request(app)
        .post("/login")
        .send({ email, password: "teste" });

      const token = loginRes.body.authorization;

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Admin Check",
          preco: 10,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(403);
      expect(res.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 with schema validation message on invalid body payload for usuarios", async () => {
      const res = await request(app)
        .post("/usuarios")
        .send({
          nome: "",
          email: "not-an-email",
          password: "",
          administrador: "true"
        });

      expect(res.status).toBe(400);
      expect(res.body).toHaveProperty("email");
    });

    it("should handle URIError and return 400 for invalid URI parameters", async () => {
      const res = await request(app).get("/usuarios/%E0%A4%A");
      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });

    it("should handle malformed JSON and return 400 with parse error message", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send("{\"nome\": malformed}");

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
      });
    });

    it("should return 415 when charset is unsupported", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json; charset=utf-7")
        .send({ nome: "Teste", email: "teste_charset@qa.com", password: "teste", administrador: "true" });

      expect(res.status).toBe(415);
      expect(res.body).toEqual({
        message: "Unsupported charset",
        charset: "utf-7"
      });
    });
  });
});
```
