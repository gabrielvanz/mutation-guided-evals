```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on protected route", async () => {
      const res = await request(app).get("/carrinhos");
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token format is bad", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "InvalidFormat token123");
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is invalid", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "Bearer invalid.jwt.token");
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 403 when user is not admin on admin-only route", async () => {
      const email = `nonadmin_${Date.now()}@qa.com`;
      await request(app).post("/usuarios").send({
        nome: "Non Admin",
        email,
        password: "teste",
        administrador: "false"
      });

      const loginRes = await request(app).post("/login").send({
        email,
        password: "teste"
      });

      const token = loginRes.body.authorization;

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Teste",
          preco: 10,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(403);
      expect(res.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should handle URIError and return 400", async () => {
      const res = await request(app).get("/usuarios/%E0%A4%A");
      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });

    it("should handle invalid JSON body parse error and return 400", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send("{\"nome\": invalid json}");

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
      });
    });
  });
});
```
