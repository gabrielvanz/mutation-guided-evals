```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on a protected route", async () => {
      const response = await request(app).get("/carrinhos");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token format is invalid", async () => {
      const response = await request(app)
        .get("/carrinhos")
        .set("authorization", "InvalidFormat token123");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is completely malformed or invalid", async () => {
      const response = await request(app)
        .get("/carrinhos")
        .set("authorization", "Bearer not.a.valid.jwt");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 403 when a non-admin user tries to access an admin-only route", async () => {
      const email = `nonadmin_user@test.com`;
      const createResponse = await request(app)
        .post("/usuarios")
        .send({
          nome: "Non Admin",
          email,
          password: "teste",
          administrador: "false"
        });

      expect(createResponse.status).toBe(201);

      const loginResponse = await request(app)
        .post("/login")
        .send({
          email,
          password: "teste"
        });

      expect(loginResponse.status).toBe(200);
      const token = loginResponse.body.authorization;

      const response = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Teste",
          preco: 10,
          descricao: "Desc",
          quantidade: 10
        });

      expect(response.status).toBe(403);
      expect(response.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 with schema validation message when sending invalid user data", async () => {
      const response = await request(app)
        .post("/usuarios")
        .send({
          nome: "",
          email: "invalid-email",
          password: "",
          administrador: "true"
        });

      expect(response.status).toBe(400);
      expect(response.body).toHaveProperty("email");
    });

    it("should return 400 when sending invalid JSON syntax", async () => {
      const response = await request(app)
        .post("/usuarios")
        .set("Content-Type", "application/json")
        .send("{ invalid json }");

      expect(response.status).toBe(400);
      expect(response.body).toEqual({
        message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
      });
    });

    it("should return 400 when URI decode error occurs with invalid characters in URL parameters", async () => {
      const response = await request(app).get("/usuarios/%C0%C0");

      expect(response.status).toBe(400);
      expect(response.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });
  });
});
```
