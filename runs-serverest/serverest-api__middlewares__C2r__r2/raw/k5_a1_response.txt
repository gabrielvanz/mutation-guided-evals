```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on a protected route", async () => {
      const res = await request(app).get("/carrinhos");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({ message: "Token de acesso ausente, inválido ou expirado" });
    });

    it("should return 401 when token has bad format", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "InvalidFormat token123");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({ message: "Token de acesso ausente, inválido ou expirado" });
    });

    it("should return 403 when user is not an administrator on an admin-only route", async () => {
      const email = "non_admin_user@test.com";
      const userRes = await request(app).post("/usuarios").send({
        nome: "Non Admin",
        email,
        password: "teste",
        administrador: "false"
      });

      expect(userRes.status).toBe(201);

      const loginRes = await request(app).post("/login").send({
        email,
        password: "teste"
      });

      expect(loginRes.status).toBe(200);
      const token = loginRes.body.authorization;

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Exclusivo",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(403);
      expect(res.body).toEqual({ message: "Rota exclusiva para administradores" });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 when malformed JSON body is sent", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send("{ invalid json }");

      expect(res.status).toBe(400);
      expect(res.body.message).toBeDefined();
    });

    it("should handle URI decode errors gracefully", async () => {
      const res = await request(app).get("/usuarios/%E0%A4%A");

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });
  });
});
```
