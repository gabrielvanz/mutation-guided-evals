import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosService from "../../src/services/carrinhos-service";
import produtosService from "../../src/services/produtos-service";
import authService from "../../src/services/auth-service";
import usuariosService from "../../src/services/usuarios-service";

vi.mock("../../src/services/produtos-service", () => ({
  default: {

    updateQuantidadePorLote: vi.fn(),

vi.mock("../../src/services/auth-service", () => ({
  default: {
    verifyToken: vi.fn()
  }
}));

vi.mock("../../src/services/usuarios-service", () => ({

describe("extrairProdutosDuplicados", () => {
  it("should return an empty array when there are no duplicate products", () => {
    const produtos = [
      { idProduto: "1", quantidade: 2 },
      { idProduto: "2", quantidade: 1 }
    ];
    const result = carrinhosService.extrairProdutosDuplicados(produtos);
    expect(result).toEqual([]);
  });

  it("should extract duplicate product IDs when products appear multiple times", () => {
    const produtos = [
      { idProduto: "1", quantidade: 2 },
      { idProduto: "2", quantidade: 1 },
      { idProduto: "1", quantidade: 3 }
    ];
    const result = carrinhosService.extrairProdutosDuplicados(produtos);
    expect(result).toEqual(["1"]);
  });
});

describe("precoTotal", () => {
  it("should calculate total price correctly for a list of products", () => {
    const produtos = [
      { precoUnitario: 10, quantidade: 2 },
      { precoUnitario: 5, quantidade: 4 }
    ];
    const result = carrinhosService.precoTotal(produtos);
    expect(result).toBe(40);
  });

  it("should return zero when product list is empty", () => {
    const result = carrinhosService.precoTotal([]);
    expect(result).toBe(0);
  });
});

describe("quantidadeTotal", () => {

      { quantidade: 2 },

describe("totaisSaoSeguros", () => {
  it("should return true when both totals are safe integers", () => {
    const result = carrinhosService.totaisSaoSeguros(100, 50);
    expect(result).toBe(true);
  });

  it("should return false when price total is not a safe integer", () => {
    const result = carrinhosService.totaisSaoSeguros(Number.MAX_SAFE_INTEGER + 1, 10);
    expect(result).toBe(false);
  });

  it("should return false when quantity total is not a safe integer", () => {
    const result = carrinhosService.totaisSaoSeguros(100, Number.MAX_SAFE_INTEGER + 1);
    expect(result).toBe(false);
  });
});

describe("getProdutosComPrecoUnitarioOuErro", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  it("should return produtosComPrecoUnitario when batch lookup succeeds", async () => {
    const produtos = [{ idProduto: "prod1", quantidade: 2 }];
    const produtosComPreco = [{ idProduto: "prod1", precoUnitario: 15, quantidade: 2 }];
    vi.mocked(produtosService.getPrecosUnitariosPorLote).mockResolvedValueOnce({ produtosComPreco, error: undefined });

    const result = await carrinhosService.getProdutosComPrecoUnitarioOuErro(produtos);
    expect(result).toEqual({ produtosComPrecoUnitario: produtosComPreco });
  });

  it("should return formatted error when batch lookup fails", async () => {
    const produtos = [{ idProduto: "prod1", quantidade: 2 }];
    const errorObj = { message: "Produto não encontrado", item: { idProduto: "prod1" }, index: 0, statusCode: 400 };
    vi.mocked(produtosService.getPrecosUnitariosPorLote).mockResolvedValueOnce({ error: errorObj });

    const result = await carrinhosService.getProdutosComPrecoUnitarioOuErro(produtos);
    expect(result).toEqual({
      error: {
        message: "Produto não encontrado",
        item: { idProduto: "prod1", index: 0 },
        index: 0,
        statusCode: 400
      }
    });
  });
});

describe("removeCarrinho", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  it("should restore stock and delete the cart by id", async () => {
    const carrinho = {
      _id: "cart123",
      produtos: [{ idProduto: "prod1", quantidade: 3 }]
    };

    const restoreSpy = vi.mocked(produtosService.restoreQuantidadePorLote).mockResolvedValueOnce(undefined as never);
    const deleteSpy = vi.spyOn(carrinhosService, "deleteById").mockResolvedValueOnce(undefined as never);

    await carrinhosService.removeCarrinho(carrinho);

    expect(restoreSpy).toHaveBeenCalledWith(carrinho.produtos);
    expect(deleteSpy).toHaveBeenCalledWith("cart123");
  });
});

describe("getCarrinhoDoUsuario", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  it("should retrieve cart using resolved user id from authorization token", async () => {
    vi.mocked(authService.verifyToken).mockReturnValueOnce({ email: "a@b.com", password: "123" } as never);
    vi.mocked(usuariosService.getDadosDoUsuario).mockResolvedValueOnce({ _id: "user999" } as never);

    const getAllSpy = vi.spyOn(carrinhosService, "getAll").mockResolvedValueOnce([{ _id: "cart123" }] as never);

    const result = await carrinhosService.getCarrinhoDoUsuario("Bearer token");

    expect(authService.verifyToken).toHaveBeenCalledWith("Bearer token");
    expect(usuariosService.getDadosDoUsuario).toHaveBeenCalledWith({ email: "a@b.com", password: "123" });
    expect(getAllSpy).toHaveBeenCalledWith({ idUsuario: "user999" });
    expect(result).toEqual([{ _id: "cart123" }]);
  });
});

describe("usuarioJaPossuiCarrinho", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  it("should return user id and existence check result", async () => {
    vi.mocked(authService.verifyToken).mockReturnValueOnce({ email: "a@b.com", password: "123" } as never);
    vi.mocked(usuariosService.getDadosDoUsuario).mockResolvedValueOnce({ _id: "user999" } as never);

    const cartObj = { _id: "cart123", idUsuario: "user999" };
    const existeSpy = vi.spyOn(carrinhosService, "existeCarrinho").mockResolvedValueOnce(cartObj as never);

    const result = await carrinhosService.usuarioJaPossuiCarrinho("Bearer token");

    expect(existeSpy).toHaveBeenCalledWith({ idUsuario: "user999" });
    expect(result).toEqual({
      idUsuario: "user999",
      possuiCarrinho: cartObj
    });
  });
});
