describe("extrairProdutosDuplicados", () => {
  it("should return an empty array when there are no duplicate products", () => {
    const produtos = [
      { idProduto: "1", quantidade: 2 },
      { idProduto: "2", quantidade: 1 }
    ];
    const result = carrinhosService.extrairProdutosDuplicados(produtos);
    expect(result).toEqual([]);
  });

  it("should extract duplicate product IDs when products appear multiple times", () => {
    const produtos = [
      { idProduto: "1", quantidade: 2 },
      { idProduto: "2", quantidade: 1 },
      { idProduto: "1", quantidade: 3 }
    ];
    const result = carrinhosService.extrairProdutosDuplicados(produtos);
    expect(result).toEqual(["1"]);
  });
});

describe("precoTotal", () => {
  it("should calculate total price correctly for a list of products", () => {
    const produtos = [
      { precoUnitario: 10, quantidade: 2 },
      { precoUnitario: 5, quantidade: 4 }
    ];
    const result = carrinhosService.precoTotal(produtos);
    expect(result).toBe(40);

  it("should return zero when product list is empty", () => {
    const result = carrinhosService.precoTotal([]);
    expect(result).toBe(0);
  });
});

describe("quantidadeTotal", () => {

describe("totaisSaoSeguros", () => {
  it("should return true when both totals are safe integers", () => {
    const result = carrinhosService.totaisSaoSeguros(100, 50);
    expect(result).toBe(true);
  });

  it("should return false when price total is not a safe integer", () => {
    const result = carrinhosService.totaisSaoSeguros(Number.MAX_SAFE_INTEGER + 1, 10);
    expect(result).toBe(false);
  });

  it("should return false when quantity total is not a safe integer", () => {
    const result = carrinhosService.totaisSaoSeguros(100, Number.MAX_SAFE_INTEGER + 1);
    expect(result).toBe(false);
  });
});

describe("getProdutosComPrecoUnitarioOuErro", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  it("should return produtosComPrecoUnitario when batch lookup succeeds", async () => {
    const produtos = [{ idProduto: "prod-1", quantidade: 2 }];
    const produtosComPreco = [{ idProduto: "prod-1", precoUnitario: 15, quantidade: 2 }];
    vi.mocked(produtosService.getPrecosUnitariosPorLote).mockResolvedValueOnce({ produtosComPreco } as never);

    const result = await carrinhosService.getProdutosComPrecoUnitarioOuErro(produtos);
    expect(result).toEqual({ produtosComPrecoUnitario: produtosComPreco });
  });

  it("should return formatted error when batch lookup fails", async () => {
    const produtos = [{ idProduto: "prod-1", quantidade: 2 }];
    const errorObj = { message: "Produto não encontrado", item: { idProduto: "prod-1" }, index: 0, statusCode: 400 };
    vi.mocked(produtosService.getPrecosUnitariosPorLote).mockResolvedValueOnce({ error: errorObj } as never);

    const result = await carrinhosService.getProdutosComPrecoUnitarioOuErro(produtos);
    expect(result).toEqual({
      error: {
        message: "Produto não encontrado",
        item: { idProduto: "prod-1", index: 0 },
        index: 0,
        statusCode: 400
      }
    });
  });
});

describe("removeCarrinho", () => {
  it("should restore stock and delete carrinho by id", async () => {
    const carrinho = { _id: "cart-1", produtos: [{ idProduto: "p1", quantidade: 2 }] };
    const restoreSpy = vi.spyOn(produtosService, "restoreQuantidadePorLote").mockResolvedValueOnce(undefined as never);
    const deleteSpy = vi.spyOn(carrinhosService, "deleteById").mockResolvedValueOnce(1 as never);

    await carrinhosService.removeCarrinho(carrinho);

    expect(restoreSpy).toHaveBeenCalledWith(carrinho.produtos);
    expect(deleteSpy).toHaveBeenCalledWith("cart-1");
  });
});

describe("getCarrinhoDoUsuario", () => {
  it("should resolve token, get user id, and query getAll with correct user id", async () => {
    vi.mocked(authService.verifyToken).mockReturnValueOnce({ email: "test@test.com", password: "123" } as never);
    vi.mocked(usuariosService.getDadosDoUsuario).mockResolvedValueOnce({ _id: "user-99" } as never);
    const getAllSpy = vi.spyOn(carrinhosService, "getAll").mockResolvedValueOnce([] as never);

    const result = await carrinhosService.getCarrinhoDoUsuario("Bearer token123");

    expect(authService.verifyToken).toHaveBeenCalledWith("Bearer token123");
    expect(usuariosService.getDadosDoUsuario).toHaveBeenCalledWith({ email: "test@test.com", password: "123" });
    expect(getAllSpy).toHaveBeenCalledWith({ idUsuario: "user-99" });
    expect(result).toEqual([]);
  });
});

describe("usuarioJaPossuiCarrinho", () => {
  it("should verify token, get user id, and check cart existence with user id", async () => {
    vi.mocked(authService.verifyToken).mockReturnValueOnce({ email: "test@test.com", password: "123" } as never);
    vi.mocked(usuariosService.getDadosDoUsuario).mockResolvedValueOnce({ _id: "user-99" } as never);
    const existeCarrinhoSpy = vi.spyOn(carrinhosService, "existeCarrinho").mockResolvedValueOnce(null as never);

    const result = await carrinhosService.usuarioJaPossuiCarrinho("Bearer token123");

    expect(authService.verifyToken).toHaveBeenCalledWith("Bearer token123");
    expect(usuariosService.getDadosDoUsuario).toHaveBeenCalledWith({ email: "test@test.com", password: "123" });

    expect(result).toEqual({
      idUsuario: "user-99",
      possuiCarrinho: null
    });
  });
});
