```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Carrinhos API", () => {
  it("should get an empty list of carts", async () => {
    const res = await request(app).get("/carrinhos");
    expect(res.status).toBe(200);
    expect(res.body).toEqual({ quantidade: 0, carrinhos: [] });
  });

  it("should return cart not found for a non-existent id", async () => {
    const res = await request(app).get("/carrinhos/1234567890abcdef");
    expect(res.status).toBe(400);
    expect(res.body).toEqual({ message: "Carrinho não encontrado" });
  });

  it("should fail to create a cart without authentication", async () => {
    const res = await request(app)
      .post("/carrinhos")
      .send({
        produtos: [{ idProduto: "123", quantidade: 1 }]
      });
    expect(res.status).toBe(401);
    expect(res.body).toEqual({ message: "Token de acesso ausente" });
  });

  it("should cancel purchase when user has no cart", async () => {
    const email = "user_nocart@email.com";
    const password = "password123";

    await request(app).post("/usuarios").send({
      nome: "Test User No Cart",
      email,
      password,
      administrador: "false"
    });

    const loginRes = await request(app).post("/login").send({ email, password });
    const token = loginRes.body.authorization;

    const res = await request(app)
      .delete("/carrinhos/cancelar-compra")
      .set("authorization", token);

    expect(res.status).toBe(200);
    expect(res.body).toEqual({ message: "Não foi encontrado nenhum carrinho" });
  });

  it("should conclude purchase when user has no cart", async () => {
    const email = "user_nocart_concluir@email.com";
    const password = "password123";

    await request(app).post("/usuarios").send({
      nome: "Test User No Cart Concluir",
      email,
      password,
      administrador: "false"
    });

    const loginRes = await request(app).post("/login").send({ email, password });
    const token = loginRes.body.authorization;

    const res = await request(app)
      .delete("/carrinhos/concluir-compra")
      .set("authorization", token);

    expect(res.status).toBe(200);
    expect(res.body).toEqual({ message: "Não foi encontrado nenhum carrinho" });
  });

  it("should fail schema validation for post when body is missing produtos", async () => {
    const email = "user_schema@email.com";
    const password = "password123";

    await request(app).post("/usuarios").send({
      nome: "Test User Schema",
      email,
      password,
      administrador: "false"
    });

    const loginRes = await request(app).post("/login").send({ email, password });
    const token = loginRes.body.authorization;

    const res = await request(app)
      .post("/carrinhos")
      .set("authorization", token)
      .send({});

    expect(res.status).toBe(400);
    expect(res.body).toHaveProperty("message");
  });
});
```
