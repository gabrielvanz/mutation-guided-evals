```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import service from "../../src/services/carrinhos-service";
import constant from "../../src/utils/constants";

vi.mock("../../src/services/carrinhos-service", () => ({
  getAll: vi.fn(),
  getOne: vi.fn(),
  usuarioJaPossuiCarrinho: vi.fn(),
  extrairProdutosDuplicados: vi.fn(),
  getProdutosComPrecoUnitarioOuErro: vi.fn(),
  precoTotal: vi.fn(),
  quantidadeTotal: vi.fn(),
  totaisSaoSeguros: vi.fn(),
  baixarEstoque: vi.fn(),
  criarCarrinho: vi.fn(),
  getCarrinhoDoUsuario: vi.fn(),
  removeCarrinho: vi.fn(),
  concluiCompra: vi.fn()
}));

const createMockRes = () => {
  const res: any = {};
  res.status = vi.fn().mockReturnValue(res);
  res.send = vi.fn().mockReturnValue(res);
  res.json = vi.fn().mockReturnValue(res);
  return res;
};

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  describe("get", () => {
    it("should return all carts with quantity", async () => {
      const mockCarrinhos = [{ _id: "1" }, { _id: "2" }];
      vi.mocked(service.getAll).mockResolvedValue(mockCarrinhos as any);

      const req = { query: {} };
      const res = createMockRes();

      await carrinhosController.get(req, res);

      expect(service.getAll).toHaveBeenCalledWith(req.query);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: 2,
        carrinhos: mockCarrinhos
      });
    });
  });

  describe("getOne", () => {
    it("should return 400 when cart is not found", async () => {
      vi.mocked(service.getOne).mockResolvedValue(null);

      const req = { params: { id: "invalid-id" } };
      const res = createMockRes();

      await carrinhosController.getOne(req, res);

      expect(service.getOne).toHaveBeenCalledWith("invalid-id");
      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });

    it("should return 200 with cart when found", async () => {
      const mockCarrinho = { _id: "1", produtos: [] };
      vi.mocked(service.getOne).mockResolvedValue(mockCarrinho as any);

      const req = { params: { id: "1" } };
      const res = createMockRes();

      await carrinhosController.getOne(req, res);

      expect(service.getOne).toHaveBeenCalledWith("1");
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });
  });

  describe("post", () => {
    it("should return 400 when user already has a cart", async () => {
      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: true
      });

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [] }
      };
      const res = createMockRes();
      res.locals = { etapas: { regras: vi.fn(), regra: vi.fn() } };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return 400 when duplicate products are present", async () => {
      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValue(["prod-1"]);

      const req = {
        headers: { authorization: "Bearer token" },
        body: {
          produtos: [
            produtoMock("prod-1", 1),
            produtoMock("prod-1", 2)
          ]
        }
      };
      const res = createMockRes();
      res.locals = { etapas: { regras: vi.fn(), regra: vi.fn() } };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["prod-1"]
      });
    });

    it("should return error when service returns getProdutosComPrecoUnitarioOuErro error", async () => {
      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValue([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValue({
        produtosComPrecoUnitario: [],
        error: { statusCode: 400, message: constant.PRODUCT_NOT_FOUND, index: 0, item: 0 }
      } as any);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [produtoMock("prod-1", 1)] }
      };
      const res = createMockRes();
      res.locals = { etapas: { regras: vi.fn(), regra: vi.fn() } };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.PRODUCT_NOT_FOUND,
        item: 0
      });
    });

    it("should return 400 when cart totals are not safe", async () => {
      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValue([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValue({
        produtosComPrecoUnitario: [produtoMock("prod-1", 1)],
        error: null
      } as any);
      vi.mocked(service.precoTotal).mockReturnValue(9999999);
      vi.mocked(service.quantidadeTotal).mockReturnValue(1);
      vi.mocked(service.totaisSaoSeguros).mockReturnValue(false);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [produtoMock("prod-1", 1)] }
      };
      const res = createMockRes();
      res.locals = { etapas: { regras: vi.fn(), regra: vi.fn() } };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });

    it("should successfully create cart when all validations pass", async () => {
      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValue([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValue({
        produtosComPrecoUnitario: [produtoMock("prod-1", 2)],
        error: null
      } as any);
      vi.mocked(service.precoTotal).mockReturnValue(50);
      vi.mocked(service.quantidadeTotal).mockReturnValue(2);
      vi.mocked(service.totaisSaoSeguros).mockReturnValue(true);
      vi.mocked(service.criarCarrinho).mockResolvedValue({ _id: "cart-1" } as any);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod-1", quantidade: "2" }] }
      };
      const res = createMockRes();
      const etapas = { regras: vi.fn(), regra: vi.fn(), abrir: vi.fn() };
      res.locals = { etapas };

      await carrinhosController.post(req, res);

      expect(etapas.abrir).toHaveBeenCalledWith("write");
      expect(service.baixarEstoque).toHaveBeenCalled();
      expect(service.criarCarrinho).toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.POST_SUCCESS,
        _id: "cart-1"
      });
    });
  });

  describe("cancelarCompra", () => {
    it("should remove cart and replenish stock when user has a cart", async () => {
      const mockCart = { _id: "cart-1" };
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValue([mockCart] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockRes();

      await carrinhosController.cancelarCompra(req, res);

      expect(service.removeCarrinho).toHaveBeenCalledWith(mockCart);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}`
      });
    });

    it("should return NO_CART message when user has no cart", async () => {
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValue([] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockRes();

      await carrinhosController.cancelarCompra(req, res);

      expect(service.removeCarrinho).not.toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase when user has a cart", async () => {
      const mockCart = { _id: "cart-1" };
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValue([mockCart] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockRes();

      await carrinhosController.concluirCompra(req, res);

      expect(service.concluiCompra).toHaveBeenCalledWith(mockCart);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("should return NO_CART message when user has no cart", async () => {
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValue([] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockRes();

      await carrinhosController.concluirCompra(req, res);

      expect(service.concluiCompra).not.toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});

function produtoMock(idProduto: string, quantidade: number) {
  return { idProduto, quantidade };
}
```
