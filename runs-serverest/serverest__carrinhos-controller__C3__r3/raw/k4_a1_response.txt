```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import constant from "../../src/utils/constants";
import service from "../../src/services/carrinhos-service";

vi.mock("../../src/services/carrinhos-service");

describe("carrinhos-controller", () => {
  let req: any;
  let res: any;

  beforeEach(() => {
    vi.clearAllMocks();
    req = {
      query: {},
      params: {},
      headers: {},
      body: {}
    };
    res = {
      status: vi.fn().mockReturnThis(),
      send: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
      locals: {}
    };
  });

  describe("get", () => {
    it("should return all carts with quantity", async () => {
      const carrinhosMock = [{ _id: "1" }, { _id: "2" }];
      vi.spyOn(service, "getAll").mockResolvedValue(carrinhosMock as any);

      await carrinhosController.get(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: 2,
        carrinhos: carrinhosMock
      });
    });
  });

  describe("getOne", () => {
    it("should return cart when found", async () => {
      req.params.id = "123";
      const carrinhoMock = { _id: "123", produtos: [] };
      vi.spyOn(service, "getOne").mockResolvedValue(carrinhoMock as any);

      await carrinhosController.getOne(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(carrinhoMock);
    });

    it("should return 400 when cart is not found", async () => {
      req.params.id = "999";
      vi.spyOn(service, "getOne").mockResolvedValue(null);

      await carrinhosController.getOne(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });
  });

  describe("post", () => {
    it("should create a cart successfully", async () => {
      res.locals.etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      req.headers.authorization = "Bearer token";
      req.body = {
        produtos: [{ idProduto: "p1", quantidade: "2" }]
      };

      vi.spyOn(service, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: false
      });
      vi.spyOn(service, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(service, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "p1", quantidade: 2, precoUnitario: 10 }],
        error: null
      });
      vi.spyOn(service, "precoTotal").mockReturnValue(20);
      vi.spyOn(service, "quantidadeTotal").mockReturnValue(2);
      vi.spyOn(service, "totaisSaoSeguros").mockReturnValue(true);
      vi.spyOn(service, "baixarEstoque").mockResolvedValue(undefined);
      vi.spyOn(service, "criarCarrinho").mockResolvedValue({ _id: "c1" } as any);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.POST_SUCCESS,
        _id: "c1"
      });
    });

    it("should return 400 if user already has a cart", async () => {
      res.locals.etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      req.headers.authorization = "Bearer token";
      req.body = { produtos: [] };

      vi.spyOn(service, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: true
      });

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return 400 if there are duplicate products", async () => {
      res.locals.etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      req.headers.authorization = "Bearer token";
      req.body = {
        produtos: [
          { idProduto: "p1", quantidade: 1 },
          { idProduto: "p1", quantidade: 1 }
        ]
      };

      vi.spyOn(service, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: false
      });
      vi.spyOn(service, "extrairProdutosDuplicados").mockReturnValue(["p1"]);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["p1"]
      });
    });

    it("should return error from service when product error occurs", async () => {
      res.locals.etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      req.headers.authorization = "Bearer token";
      req.body = {
        produtos: [{ idProduto: "p1", quantidade: 1 }]
      };

      vi.spyOn(service, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: false
      });
      vi.spyOn(service, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(service, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: [],
        error: {
          statusCode: 400,
          message: constant.PRODUCT_NOT_FOUND,
          index: 0,
          item: "p1"
        }
      });

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.PRODUCT_NOT_FOUND,
        item: "p1"
      });
    });

    it("should return 400 if cart totals are too big", async () => {
      res.locals.etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      req.headers.authorization = "Bearer token";
      req.body = {
        produtos: [{ idProduto: "p1", quantidade: 1 }]
      };

      vi.spyOn(service, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: false
      });
      vi.spyOn(service, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(service, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "p1", quantidade: 1, precoUnitario: 10 }],
        error: null
      });
      vi.spyOn(service, "precoTotal").mockReturnValue(Number.MAX_SAFE_INTEGER);
      vi.spyOn(service, "quantidadeTotal").mockReturnValue(1);
      vi.spyOn(service, "totaisSaoSeguros").mockReturnValue(false);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });
  });

  describe("cancelarCompra", () => {
    it("should remove cart and replenish stock if cart exists", async () => {
      req.headers.authorization = "Bearer token";
      const carrinhoMock = { _id: "c1" };
      vi.spyOn(service, "getCarrinhoDoUsuario").mockResolvedValue([carrinhoMock] as any);
      vi.spyOn(service, "removeCarrinho").mockResolvedValue(undefined);

      await carrinhosController.cancelarCompra(req, res);

      expect(service.removeCarrinho).toHaveBeenCalledWith(carrinhoMock);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}`
      });
    });

    it("should return NO_CART if user has no cart", async () => {
      req.headers.authorization = "Bearer token";
      vi.spyOn(service, "getCarrinhoDoUsuario").mockResolvedValue([]);

      await carrinhosController.cancelarCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase if cart exists", async () => {
      req.headers.authorization = "Bearer token";
      const carrinhoMock = { _id: "c1" };
      vi.spyOn(service, "getCarrinhoDoUsuario").mockResolvedValue([carrinhoMock] as any);
      vi.spyOn(service, "concluiCompra").mockResolvedValue(undefined);

      await carrinhosController.concluirCompra(req, res);

      expect(service.concluiCompra).toHaveBeenCalledWith(carrinhoMock);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("should return NO_CART if user has no cart", async () => {
      req.headers.authorization = "Bearer token";
      vi.spyOn(service, "getCarrinhoDoUsuario").mockResolvedValue([]);

      await carrinhosController.concluirCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});
```
