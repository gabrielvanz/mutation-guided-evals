```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

async function getAdminToken(suffix: string) {
  const email = `admin_${suffix}@qa.com`;
  await request(app)
    .post("/usuarios")
    .send({
      nome: "Admin User",
      email,
      password: "password",
      administrador: "true"
    });

  const loginRes = await request(app)
    .post("/login")
    .send({
      email,
      password: "password"
    });

  return loginRes.body.authorization;
}

describe("produtos API", () => {
  describe("GET /produtos", () => {
    it("should return empty list when no products exist", async () => {
      const res = await request(app).get("/produtos");
      expect(res.status).toBe(200);
      expect(res.body).toEqual({ quantidade: 0, produtos: [] });
    });

    it("should return list of products when products exist", async () => {
      const token = await getAdminToken("1");
      const uniqueName = "Produto A";

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Descricao",
          quantidade: 10
        });

      const res = await request(app).get("/produtos");
      expect(res.status).toBe(200);
      expect(res.body.quantidade).toBeGreaterThan(0);
      expect(res.body.produtos.some((p: any) => p.nome === uniqueName)).toBe(true);
    });

    it("should filter products by name query parameter", async () => {
      const token = await getAdminToken("2");
      const uniqueName = "FiltravelB";

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 50,
          descricao: "Desc",
          quantidade: 5
        });

      const res = await request(app).get(`/produtos?nome=${uniqueName}`);
      expect(res.status).toBe(200);
      expect(res.body.quantidade).toBe(1);
      expect(res.body.produtos[0].nome).toBe(uniqueName);
    });
  });

  describe("GET /produtos/:id", () => {
    it("should return a product by valid id", async () => {
      const token = await getAdminToken("3");
      const uniqueName = "Produto ID C";

      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 200,
          descricao: "Desc",
          quantidade: 15
        });

      const productId = postRes.body._id;

      const res = await request(app).get(`/produtos/${productId}`);
      expect(res.status).toBe(200);
      expect(res.body._id).toBe(productId);
      expect(res.body.nome).toBe(uniqueName);
    });

    it("should return 400 when product id does not exist", async () => {
      const res = await request(app).get("/produtos/1234567890abcdef");
      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Produto não encontrado" });
    });

    it("should return validation error for invalid id format", async () => {
      const res = await request(app).get("/produtos/invalid-id");
      expect(res.status).toBe(400);
    });
  });

  describe("POST /produtos", () => {
    it("should create a product successfully when user is admin and data is valid", async () => {
      const token = await getAdminToken("4");
      const uniqueName = "Novo Produto D";

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 450,
          descricao: "Mouse Gamer",
          quantidade: 100,
          imagem: "http://imagem.com/mouse.jpg"
        });

      expect(res.status).toBe(201);
      expect(res.body.message).toBe("Cadastro com sucesso");
      expect(res.body._id).toBeDefined();
    });

    it("should return 400 when product name is already used", async () => {
      const token = await getAdminToken("5");
      const uniqueName = "RepetidoE";

      const productData = {
        nome: uniqueName,
        preco: 100,
        descricao: "Desc",
        quantidade: 10
      };

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send(productData);

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send(productData);

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });

    it("should return 401 when token is missing", async () => {
      const res = await request(app)
        .post("/produtos")
        .send({
          nome: "Sem Token",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(401);
    });

    it("should return 403 when user is not admin", async () => {
      const email = "user_nonadmin@qa.com";
      await request(app)
        .post("/usuarios")
        .send({
          nome: "Regular User",
          email,
          password: "password",
          administrador: "false"
        });

      const loginRes = await request(app)
        .post("/login")
        .send({
          email,
          password: "password"
        });

      const token = loginRes.body.authorization;

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Comum",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(403);
    });

    it("should return 400 when required fields are missing", async () => {
      const token = await getAdminToken("6");

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          preco: 100
        });

      expect(res.status).toBe(400);
    });
  });

  describe("PUT /produtos/:id", () => {
    it("should update a product successfully", async () => {
      const token = await getAdminToken("7");
      const uniqueName = "Para Atualizar F";
      const updatedName = "Atualizado G";

      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const productId = postRes.body._id;

      const res = await request(app)
        .put(`/produtos/${productId}`)
        .set("authorization", token)
        .send({
          nome: updatedName,
          preco: 150,
          descricao: "Desc Atualizada",
          quantidade: 20
        });

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro alterado com sucesso" });
    });

    it("should return 400 when updating to a name already used by another product", async () => {
      const token = await getAdminToken("8");
      const name1 = "Produto Um H";
      const name2 = "Produto Dois I";

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: name1,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: name2,
          preco: 200,
          descricao: "Desc",
          quantidade: 20
        });

      const productId2 = postRes.body._id;

      const res = await request(app)
        .put(`/produtos/${productId2}`)
        .set("authorization", token)
        .send({
          nome: name1,
          preco: 200,
          descricao: "Desc",
          quantidade: 20
        });

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });

    it("should return 401 when token is missing on put", async () => {
      const res = await request(app)
        .put("/produtos/1234567890abcdef")
        .send({
          nome: "Teste",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(401);
    });
  });

  describe("DELETE /produtos/:id", () => {
    it("should delete a product successfully when it is not in a cart", async () => {
      const token = await getAdminToken("9");
      const uniqueName = "Para Deletar J";

      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const productId = postRes.body._id;

      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro excluído com sucesso" });
    });

    it("should return delete none message when product does not exist", async () => {
      const token = await getAdminToken("10");

      const res = await request(app)
        .delete("/produtos/1234567890abcdef")
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Nenhum registro foi excluído" });
    });

    it("should return 401 when token is missing on delete", async () => {
      const res = await request(app).delete("/produtos/1234567890abcdef");
      expect(res.status).toBe(401);
    });
  });
});
```
