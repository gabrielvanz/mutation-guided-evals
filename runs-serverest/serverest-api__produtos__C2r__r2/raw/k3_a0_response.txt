```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Produtos API", () => {
  let adminToken: string;

  const getAdminToken = async () => {
    if (adminToken) return adminToken;
    const email = `admin_${Date.now()}_${Math.random().toString(36).substring(2, 7)}@qa.com`;
    await request(app)
      .post("/usuarios")
      .send({
        nome: "Admin Teste",
        email,
        password: "teste",
        administrador: "true"
      });
    const loginRes = await request(app)
      .post("/login")
      .send({ email, password: "teste" });
    adminToken = loginRes.body.authorization;
    return adminToken;
  };

  describe("GET /produtos", () => {
    it("should return all products and quantity", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto List ${Date.now()}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Descricao",
          quantidade: 10
        });

      const res = await request(app).get("/produtos");
      expect(res.status).toBe(200);
      expect(res.body).toHaveProperty("quantidade");
      expect(res.body).toHaveProperty("produtos");
      expect(Array.isArray(res.body.produtos)).toBe(true);
      expect(res.body.quantidade).toBeGreaterThanOrEqual(1);
    });

    it("should filter products by query parameters", async () => {
      const token = await getAdminToken();
      const uniqueName = `FiltroExato_${Date.now()}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 250,
          descricao: "Filtro desc",
          quantidade: 5
        });

      const res = await request(app).get(`/produtos?nome=${uniqueName}`);
      expect(res.status).toBe(200);
      expect(res.body.quantidade).toBe(1);
      expect(res.body.produtos[0].nome).toBe(uniqueName);
    });
  });

  describe("GET /produtos/:id", () => {
    it("should return a single product by id", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto One ${Date.now()}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 150,
          descricao: "Desc",
          quantidade: 5
        });

      const productId = postRes.body._id;

      const res = await request(app).get(`/produtos/${productId}`);
      expect(res.status).toBe(200);
      expect(res.body._id).toBe(productId);
      expect(res.body.nome).toBe(uniqueName);
    });

    it("should return 400 when product does not exist", async () => {
      const fakeId = "1234567890abcdef";
      const res = await request(app).get(`/produtos/${fakeId}`);
      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Produto não encontrado." });
    });
  });

  describe("POST /produtos", () => {
    it("should create a product successfully as admin", async () => {
      const token = await getAdminToken();
      const uniqueName = `Novo Produto ${Date.now()}`;
      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 499,
          descricao: "Smartphone moderno",
          quantidade: 20,
          imagem: "http://imagem.com/img.png"
        });

      expect(res.status).toBe(201);
      expect(res.body.message).toBe("Cadastro realizado com sucesso.");
      expect(res.body).toHaveProperty("_id");
    });

    it("should fail when creating a product with an existing name", async () => {
      const token = await getAdminToken();
      const uniqueName = `Duplicado ${Date.now()}`;
      const productData = {
        nome: uniqueName,
        preco: 100,
        descricao: "Desc",
        quantidade: 10
      };

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send(productData);

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send(productData);

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome." });
    });

    it("should fail when unauthenticated", async () => {
      const res = await request(app)
        .post("/produtos")
        .send({
          nome: "Sem Token",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(401);
      expect(res.body).toHaveProperty("message");
    });

    it("should fail when user is not admin", async () => {
      const email = `common_${Date.now()}@qa.com`;
      await request(app)
        .post("/usuarios")
        .send({
          nome: "Common User",
          email,
          password: "teste",
          administrador: "false"
        });
      const loginRes = await request(app)
        .post("/login")
        .send({ email, password: "teste" });
      const commonToken = loginRes.body.authorization;

      const res = await request(app)
        .post("/produtos")
        .set("authorization", commonToken)
        .send({
          nome: "Produto Comum",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(403);
      expect(res.body).toHaveProperty("message");
    });
  });

  describe("PUT /produtos/:id", () => {
    it("should update an existing product successfully", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto Put ${Date.now()}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const productId = postRes.body._id;
      const updatedName = `Produto Atualizado ${Date.now()}`;

      const res = await request(app)
        .put(`/produtos/${productId}`)
        .set("authorization", token)
        .send({
          nome: updatedName,
          preco: 200,
          descricao: "Nova desc",
          quantidade: 15
        });

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro alterado com sucesso." });
    });

    it("should create product via put if id does not exist", async () => {
      const token = await getAdminToken();
      const fakeId = "9999999999abcdef";
      const uniqueName = `Upsert Produto ${Date.now()}`;

      const res = await request(app)
        .put(`/produtos/${fakeId}`)
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 300,
          descricao: "Upsert desc",
          quantidade: 5
        });

      expect(res.status).toBe(201);
      expect(res.body.message).toBe("Cadastro realizado com sucesso.");
      expect(res.body).toHaveProperty("_id");
    });
  });

  describe("DELETE /produtos/:id", () => {
    it("should delete an existing product successfully", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto Delete ${Date.now()}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const productId = postRes.body._id;

      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro excluído com sucesso." });
    });

    it("should return delete none when deleting non-existent product", async () => {
      const token = await getAdminToken();
      const fakeId = "1234567890abcdef";

      const res = await request(app)
        .delete(`/produtos/${fakeId}`)
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Nenhum registro excluído." });
    });

    it("should not allow deleting a product included in a cart", async () => {
      const token = await getAdminToken();
      const userEmail = `cart_user_${Date.now()}@qa.com`;
      await request(app)
        .post("/usuarios")
        .send({
          nome: "Cart User",
          email: userEmail,
          password: "teste",
          administrador: "false"
        });
      const userLogin = await request(app)
        .post("/login")
        .send({ email: userEmail, password: "teste" });
      const userToken = userLogin.body.authorization;

      const uniqueName = `Produto No Cart ${Date.now()}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 50,
          descricao: "Desc",
          quantidade: 10
        });
      const productId = postRes.body._id;

      await request(app)
        .post("/carrinhos")
        .set("authorization", userToken)
        .send({
          produtos: [{ idProduto: productId, quantidade: 2 }]
        });

      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(400);
      expect(res.body.message).toBe("Não é permitido excluir produto que faz parte de carrinho");
      expect(Array.isArray(res.body.idCarrinhos)).toBe(true);
    });
  });
});
```
