```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on protected routes", async () => {
      const response = await request(app).get("/carrinhos");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token format is invalid", async () => {
      const response = await request(app)
        .get("/carrinhos")
        .set("authorization", "InvalidFormat token123");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is malformed or invalid", async () => {
      const response = await request(app)
        .get("/carrinhos")
        .set("authorization", "Bearer invalidtokenstring");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 403 when a non-admin user attempts to access admin-only routes", async () => {
      const regularUser = {
        nome: "Regular User",
        email: "regular_user_auth@qa.com",
        password: "teste",
        administrador: "false"
      };

      await request(app).post("/usuarios").send(regularUser);

      const loginRes = await request(app).post("/login").send({
        email: regularUser.email,
        password: regularUser.password
      });
      expect(loginRes.status).toBe(200);
      const token = loginRes.body.authorization;

      const response = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Admin Check",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(response.status).toBe(403);
      expect(response.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 when sending invalid JSON payload (body-parser entity parse failed)", async () => {
      const response = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send('{"nome": "bad json');

      expect(response.status).toBe(400);
      expect(response.body.message).toBeDefined();
    });

    it("should return 400 when URL contains invalid URI encoding", async () => {
      const response = await request(app).get("/usuarios/%E0%A4%A");

      expect(response.status).toBe(400);
      expect(response.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });
  });
});
```
