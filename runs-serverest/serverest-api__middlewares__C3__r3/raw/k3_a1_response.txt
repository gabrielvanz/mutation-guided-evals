```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on a protected route", async () => {
      const res = await request(app).get("/carrinhos");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token format is invalid", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "InvalidFormat token123");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is malformed", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "Bearer not.a.jwt.token");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 403 when a non-admin user tries to access admin-only routes", async () => {
      const email = "user_non_admin@test.com";
      await request(app).post("/usuarios").send({
        nome: "Test User",
        email,
        password: "password",
        administrador: "false"
      });

      const loginRes = await request(app).post("/login").send({
        email,
        password: "password"
      });

      const token = loginRes.body.authorization;

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Teste",
          preco: 10,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(403);
      expect(res.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 with schema validation error messages when required fields are missing", async () => {
      const res = await request(app).post("/usuarios").send({
        email: "incomplete@test.com"
      });

      expect(res.status).toBe(400);
      expect(res.body).toHaveProperty("nome");
    });

    it("should return 400 when URI decode error occurs", async () => {
      const res = await request(app).get("/usuarios/%E0%A0%A");

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });

    it("should handle entity parse failed (bad JSON)", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send("{ invalid json");

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
      });
    });

    it("should return 415 when unsupported charset is provided", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json; charset=utf-7")
        .send({});

      expect(res.status).toBe(415);
      expect(res.body).toEqual({
        message: "Unsupported charset",
        charset: "utf-7"
      });
    });

    it("should return 415 when unsupported content encoding is provided", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("content-encoding", "unsupported-encoding")
        .send({});

      expect(res.status).toBe(415);
      expect(res.body).toEqual({
        message: "Unsupported content encoding",
        encoding: "unsupported-encoding"
      });
    });
  });
});
```
