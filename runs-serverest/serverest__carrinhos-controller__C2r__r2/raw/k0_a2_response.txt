```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import service from "../../src/services/carrinhos-service";
import constant from "../../src/utils/constants";

vi.mock("../../src/services/carrinhos-service", () => ({
  getAll: vi.fn(),
  getOne: vi.fn(),
  usuarioJaPossuiCarrinho: vi.fn(),
  extrairProdutosDuplicados: vi.fn(),
  getProdutosComPrecoUnitarioOuErro: vi.fn(),
  precoTotal: vi.fn(),
  quantidadeTotal: vi.fn(),
  totaisSaoSeguros: vi.fn(),
  baixarEstoque: vi.fn(),
  criarCarrinho: vi.fn(),
  getCarrinhoDoUsuario: vi.fn(),
  removeCarrinho: vi.fn(),
  concluiCompra: vi.fn()
}));

const createMockRes = () => {
  const res: any = {};
  res.status = vi.fn().mockReturnValue(res);
  res.send = vi.fn().mockReturnValue(res);
  res.json = vi.fn().mockReturnValue(res);
  return res;
};

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  describe("get", () => {
    it("should return all carts with quantidade", async () => {
      const req = { query: {} };
      const res = createMockRes();
      const mockCarrinhos = [{ _id: "1" }, { _id: "2" }];

      vi.mocked(service.getAll).mockResolvedValueOnce(mockCarrinhos as any);

      await carrinhosController.get(req, res);

      expect(service.getAll).toHaveBeenCalledWith(req.query);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ quantidade: 2, carrinhos: mockCarrinhos });
    });
  });

  describe("getOne", () => {
    it("should return cart when found", async () => {
      const req = { params: { id: "123" } };
      const res = createMockRes();
      const mockCarrinho = { _id: "123", produtos: [] };

      vi.mocked(service.getOne).mockResolvedValueOnce(mockCarrinho as any);

      await carrinhosController.getOne(req, res);

      expect(service.getOne).toHaveBeenCalledWith("123");
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });

    it("should return 400 when cart is not found", async () => {
      const req = { params: { id: "999" } };
      const res = createMockRes();

      vi.mocked(service.getOne).mockResolvedValueOnce(null as any);

      await carrinhosController.getOne(req, res);

      expect(service.getOne).toHaveBeenCalledWith("999");
      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });
  });

  describe("post", () => {
    it("should create cart successfully", async () => {
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: "2" }] }
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const res = createMockRes();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({ idUsuario: "u1", possuiCarrinho: false });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValueOnce([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValueOnce({
        produtosComPrecoUnitario: [{ idProduto: "p1", quantidade: 2, precoUnitario: 10 }]
      } as any);
      vi.mocked(service.precoTotal).mockReturnValueOnce(20);
      vi.mocked(service.quantidadeTotal).mockReturnValueOnce(2);
      vi.mocked(service.totaisSaoSeguros).mockReturnValueOnce(true);
      vi.mocked(service.baixarEstoque).mockResolvedValueOnce(undefined as any);
      vi.mocked(service.criarCarrinho).mockResolvedValueOnce({ _id: "c1" } as any);

      await carrinhosController.post(req, res);

      expect(etapas.regras).toHaveBeenCalledWith(
        "single_cart",
        "no_duplicate_product",
        "product_exists",
        "enough_stock",
        "safe_totals"
      );
      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({ message: constant.POST_SUCCESS, _id: "c1" });
    });

    it("should return 400 if user already has a cart", async () => {
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [] }
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const res = createMockRes();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({ idUsuario: "u1", possuiCarrinho: true });

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return 400 if duplicate products exist", async () => {
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1" }, { idProduto: "p1" }] }
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const res = createMockRes();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({ idUsuario: "u1", possuiCarrinho: false });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValueOnce(["p1"]);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_WITH_DUPLICATE_PRODUCT, idProdutosDuplicados: ["p1"] });
    });

    it("should return error status if product not found or out of stock", async () => {
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }] }
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const res = createMockRes();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({ idUsuario: "u1", possuiCarrinho: false });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValueOnce([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValueOnce({
        error: { message: constant.PRODUCT_NOT_FOUND, statusCode: 400, item: 0 }
      } as any);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.PRODUCT_NOT_FOUND, item: 0 });
    });

    it("should return 400 if totals are not safe", async () => {
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }] }
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const res = createMockRes();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({ idUsuario: "u1", possuiCarrinho: false });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValueOnce([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValueOnce({
        produtosComPrecoUnitario: []
      } as any);
      vi.mocked(service.precoTotal).mockReturnValueOnce(999999999);
      vi.mocked(service.quantidadeTotal).mockReturnValueOnce(999999999);
      vi.mocked(service.totaisSaoSeguros).mockReturnValueOnce(false);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });
  });

  describe("cancelarCompra", () => {
    it("should remove cart and replenish stock when cart exists", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockRes();
      const mockCarrinho = { _id: "c1" };

      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValueOnce([mockCarrinho] as any);
      vi.mocked(service.removeCarrinho).mockResolvedValueOnce(undefined as any);

      await carrinhosController.cancelarCompra(req, res);

      expect(service.removeCarrinho).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}` });
    });

    it("should return NO_CART message when user has no cart", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockRes();

      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValueOnce([] as any);

      await carrinhosController.cancelarCompra(req, res);

      expect(service.removeCarrinho).not.toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase when cart exists", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockRes();
      const mockCarrinho = { _id: "c1" };

      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValueOnce([mockCarrinho] as any);
      vi.mocked(service.concluiCompra).mockResolvedValueOnce(undefined as any);

      await carrinhosController.concluirCompra(req, res);

      expect(service.concluiCompra).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("should return NO_CART message when user has no cart", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockRes();

      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValueOnce([] as any);

      await carrinhosController.concluirCompra(req, res);

      expect(service.concluiCompra).not.toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});
```
