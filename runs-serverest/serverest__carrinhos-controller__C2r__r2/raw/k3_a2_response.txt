```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import service from "../../src/services/carrinhos-service";
import constant from "../../src/utils/constants";

vi.mock("../../src/services/carrinhos-service", () => ({
  getAll: vi.fn(),
  getOne: vi.fn(),
  usuarioJaPossuiCarrinho: vi.fn(),
  extrairProdutosDuplicados: vi.fn(),
  getProdutosComPrecoUnitarioOuErro: vi.fn(),
  precoTotal: vi.fn(),
  quantidadeTotal: vi.fn(),
  totaisSaoSeguros: vi.fn(),
  baixarEstoque: vi.fn(),
  criarCarrinho: vi.fn(),
  getCarrinhoDoUsuario: vi.fn(),
  removeCarrinho: vi.fn(),
  concluiCompra: vi.fn()
}));

const mockResponse = () => {
  const res: any = {};
  res.status = vi.fn().mockReturnValue(res);
  res.send = vi.fn().mockReturnValue(res);
  res.json = vi.fn().mockReturnValue(res);
  return res;
};

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  describe("get", () => {
    it("should return all carts with quantity", async () => {
      const req = { query: {} };
      const res = mockResponse();
      const mockCarrinhos = [{ _id: "1" }, { _id: "2" }];
      vi.mocked(service.getAll).mockResolvedValueOnce(mockCarrinhos as any);

      await carrinhosController.get(req as any, res as any);

      expect(service.getAll).toHaveBeenCalledWith({});
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: 2,
        carrinhos: mockCarrinhos
      });
    });
  });

  describe("getOne", () => {
    it("should return cart when found", async () => {
      const req = { params: { id: "123" } };
      const res = mockResponse();
      const mockCarrinho = { _id: "123", produtos: [] };
      vi.mocked(service.getOne).mockResolvedValueOnce(mockCarrinho as any);

      await carrinhosController.getOne(req as any, res as any);

      expect(service.getOne).toHaveBeenCalledWith("123");
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });

    it("should return 400 when cart is not found", async () => {
      const req = { params: { id: "invalid" } };
      const res = mockResponse();
      vi.mocked(service.getOne).mockResolvedValueOnce(null);

      await carrinhosController.getOne(req as any, res as any);

      expect(service.getOne).toHaveBeenCalledWith("invalid");
      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });
  });

  describe("post", () => {
    it("should create cart successfully when valid data is provided", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: "2" }] }
      };
      const res = mockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({ idUsuario: "u1", possuiCarrinho: false });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValueOnce([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValueOnce({
        produtosComPrecoUnitario: [{ idProduto: "p1", quantidade: 2, precoUnitario: 10 }],
        error: null
      });
      vi.mocked(service.precoTotal).mockReturnValueOnce(20);
      vi.mocked(service.quantidadeTotal).mockReturnValueOnce(2);
      vi.mocked(service.totaisSaoSeguros).mockReturnValueOnce(true);
      vi.mocked(service.criarCarrinho).mockResolvedValueOnce({ _id: "c1" } as any);

      await carrinhosController.post(req as any, res as any);

      expect(etapas.regras).toHaveBeenCalledWith("single_cart", "no_duplicate_product", "product_exists", "enough_stock", "safe_totals");
      expect(etapas.regra).toHaveBeenCalledWith("single_cart", true);
      expect(etapas.abrir).toHaveBeenCalledWith("write");
      expect(service.baixarEstoque).toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.POST_SUCCESS,
        _id: "c1"
      });
    });

    it("should return 400 when user already has a cart", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [] }
      };
      const res = mockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({ idUsuario: "u1", possuiCarrinho: true });

      await carrinhosController.post(req as any, res as any);

      expect(etapas.regra).toHaveBeenCalledWith("single_cart", false);
      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return 400 when duplicate products exist", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }, { idProduto: "p1", quantidade: 2 }] }
      };
      const res = mockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({ idUsuario: "u1", possuiCarrinho: false });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValueOnce(["p1"]);

      await carrinhosController.post(req as any, res as any);

      expect(etapas.regra).toHaveBeenCalledWith("no_duplicate_product", false);
      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["p1"]
      });
    });

    it("should return error status when product is not found or out of stock", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }] }
      };
      const res = mockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({ idUsuario: "u1", possuiCarrinho: false });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValueOnce([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValueOnce({
        produtosComPrecoUnitario: [],
        error: { message: constant.PRODUCT_NOT_FOUND, statusCode: 400, index: 0, item: 0 } as any
      });

      await carrinhosController.post(req as any, res as any);

      expect(etapas.regra).toHaveBeenCalledWith("product_exists", false, { item: 0 });
      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.PRODUCT_NOT_FOUND,
        item: 0
      });
    });

    it("should return 400 when totals are too big", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }] }
      };
      const res = mockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({ idUsuario: "u1", possuiCarrinho: false });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValueOnce([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValueOnce({
        produtosComPrecoUnitario: [{ idProduto: "p1", quantidade: 1, precoUnitario: 10 }],
        error: null
      });
      vi.mocked(service.precoTotal).mockReturnValueOnce(999999999);
      vi.mocked(service.quantidadeTotal).mockReturnValueOnce(999999999);
      vi.mocked(service.totaisSaoSeguros).mockReturnValueOnce(false);

      await carrinhosController.post(req as any, res as any);

      expect(etapas.regra).toHaveBeenCalledWith("safe_totals", false);
      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });
  });

  describe("cancelarCompra", () => {
    it("should remove cart and replenish stock when user has a cart", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = mockResponse();
      const cart = { _id: "c1" };
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValueOnce([cart] as any);

      await carrinhosController.cancelarCompra(req as any, res as any);

      expect(service.removeCarrinho).toHaveBeenCalledWith(cart);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}`
      });
    });

    it("should return NO_CART when user does not have a cart", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = mockResponse();
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValueOnce([]);

      await carrinhosController.cancelarCompra(req as any, res as any);

      expect(service.removeCarrinho).not.toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase when user has a cart", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = mockResponse();
      const cart = { _id: "c1" };
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValueOnce([cart] as any);

      await carrinhosController.concluirCompra(req as any, res as any);

      expect(service.concluiCompra).toHaveBeenCalledWith(cart);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("should return NO_CART when user does not have a cart", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = mockResponse();
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValueOnce([]);

      await carrinhosController.concluirCompra(req as any, res as any);

      expect(service.concluiCompra).not.toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});
```
