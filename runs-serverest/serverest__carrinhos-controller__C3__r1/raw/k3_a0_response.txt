```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import service from "../../src/services/carrinhos-service";
import constant from "../../src/utils/constants";

vi.mock("../../src/services/carrinhos-service");
vi.mock("../../src/utils/constants", () => ({
  CART_NOT_FOUND: "Não foi encontrado carrinho para este id",
  LIMIT_JUST_ONE_CART: "Não é permitido possuir mais de 1 carrinho",
  CART_WITH_DUPLICATE_PRODUCT: "Não é permitido possuir produtos duplicados",
  PRODUCT_NOT_FOUND: "Produto não encontrado",
  CART_TOTALS_TOO_BIG: "Totais do carrinho muito grandes",
  POST_SUCCESS: "Cadastro realizado com sucesso",
  DELETE_SUCCESS: "Registro excluído com sucesso",
  REPLENISHED_STOCK: "Estoque dos produtos reabastecido",
  NO_CART: "Não foi encontrado carrinho para este usuário"
}));

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  describe("get", () => {
    it("deve retornar todos os carrinhos com status 200", async () => {
      const carrinhosMock = [{ _id: "1", produtos: [] }];
      vi.mocked(service.getAll).mockResolvedValueOnce(carrinhosMock as any);

      const req = { query: {} };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.get(req as any, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: 1,
        carrinhos: carrinhosMock
      });
    });
  });

  describe("getOne", () => {
    it("deve retornar o carrinho quando encontrado com status 200", async () => {
      const carrinhoMock = { _id: "1", produtos: [] };
      vi.mocked(service.getOne).mockResolvedValueOnce(carrinhoMock as any);

      const req = { params: { id: "1" } };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        json: vi.fn()
      };

      await carrinhosController.getOne(req as any, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(carrinhoMock);
    });

    it("deve retornar 400 se o carrinho não for encontrado", async () => {
      vi.mocked(service.getOne).mockResolvedValueOnce(null);

      const req = { params: { id: "invalid" } };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        json: vi.fn()
      };

      await carrinhosController.getOne(req as any, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });
  });

  describe("post", () => {
    it("deve retornar 400 se o usuário já possui carrinho", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = { headers: { authorization: "Bearer token" }, body: { produtos: [] } };
      const res: any = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({
        idUsuario: "user1",
        possuiCarrinho: true
      });

      await carrinhosController.post(req as any, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("deve retornar 400 se houver produtos duplicados", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: "2" }] }
      };
      const res: any = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({
        idUsuario: "user1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValueOnce(["p1"]);

      await carrinhosController.post(req as any, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["p1"]
      });
    });

    it("deve retornar erro do serviço se houver falha ao buscar preço ou estoque", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: "2" }] }
      };
      const res: any = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({
        idUsuario: "user1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValueOnce([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValueOnce({
        produtosComPrecoUnitario: [],
        error: { statusCode: 400, message: constant.PRODUCT_NOT_FOUND, index: 0, item: 0 } as any
      });

      await carrinhosController.post(req as any, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.PRODUCT_NOT_FOUND,
        item: 0
      });
    });

    it("deve retornar 400 se os totais do carrinho forem muito grandes", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: "2" }] }
      };
      const res: any = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({
        idUsuario: "user1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValueOnce([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValueOnce({
        produtosComPrecoUnitario: [{ idProduto: "p1", precoUnitario: 10, quantidade: 2 }] as any,
        error: null
      });
      vi.mocked(service.precoTotal).mockReturnValueOnce(1000);
      vi.mocked(service.quantidadeTotal).mockReturnValueOnce(2);
      vi.mocked(service.totaisSaoSeguros).mockReturnValueOnce(false);

      await carrinhosController.post(req as any, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });

    it("deve criar o carrinho com sucesso e retornar 201", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: "2" }] }
      };
      const res: any = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValueOnce({
        idUsuario: "user1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValueOnce([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValueOnce({
        produtosComPrecoUnitario: [{ idProduto: "p1", precoUnitario: 10, quantidade: 2 }] as any,
        error: null
      });
      vi.mocked(service.precoTotal).mockReturnValueOnce(20);
      vi.mocked(service.quantidadeTotal).mockReturnValueOnce(2);
      vi.mocked(service.totaisSaoSeguros).mockReturnValueOnce(true);
      vi.mocked(service.criarCarrinho).mockResolvedValueOnce({ _id: "cart123" } as any);

      await carrinhosController.post(req as any, res);

      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.POST_SUCCESS,
        _id: "cart123"
      });
    });
  });

  describe("cancelarCompra", () => {
    it("deve remover o carrinho e repor o estoque se o carrinho existir", async () => {
      const carrinhoObj = { _id: "cart1" };
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValueOnce([carrinhoObj] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req as any, res);

      expect(service.removeCarrinho).toHaveBeenCalledWith(carrinhoObj);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}`
      });
    });

    it("deve retornar mensagem de nenhum carrinho se o usuário não possuir carrinho", async () => {
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValueOnce([]);

      const req = { headers: { authorization: "Bearer token" } };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req as any, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("deve concluir a compra e retornar sucesso se o carrinho existir", async () => {
      const carrinhoObj = { _id: "cart1" };
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValueOnce([carrinhoObj] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req as any, res);

      expect(service.concluiCompra).toHaveBeenCalledWith(carrinhoObj);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("deve retornar mensagem de nenhum carrinho se o usuário não possuir carrinho ao concluir", async () => {
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValueOnce([]);

      const req = { headers: { authorization: "Bearer token" } };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req as any, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});
```
