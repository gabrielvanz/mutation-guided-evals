```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import service from "../../src/services/carrinhos-service";
import constant from "../../src/utils/constants";

vi.mock("../../src/services/carrinhos-service", () => ({
  getAll: vi.fn(),
  getOne: vi.fn(),
  usuarioJaPossuiCarrinho: vi.fn(),
  extrairProdutosDuplicados: vi.fn(),
  getProdutosComPrecoUnitarioOuErro: vi.fn(),
  precoTotal: vi.fn(),
  quantidadeTotal: vi.fn(),
  totaisSaoSeguros: vi.fn(),
  baixarEstoque: vi.fn(),
  criarCarrinho: vi.fn(),
  getCarrinhoDoUsuario: vi.fn(),
  removeCarrinho: vi.fn(),
  concluiCompra: vi.fn()
}));

const createMockResponse = () => {
  const res: any = {};
  res.status = vi.fn().mockReturnValue(res);
  res.send = vi.fn().mockReturnValue(res);
  res.json = vi.fn().mockReturnValue(res);
  return res;
};

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  describe("get", () => {
    it("should return all carrinhos with status 200", async () => {
      const req = { query: {} };
      const res = createMockResponse();
      const mockCarrinhos = [{ _id: "1", produtos: [] }];

      vi.mocked(service.getAll).mockResolvedValue(mockCarrinhos as any);

      await carrinhosController.get(req, res);

      expect(service.getAll).toHaveBeenCalledWith(req.query);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: 1,
        carrinhos: mockCarrinhos
      });
    });
  });

  describe("getOne", () => {
    it("should return cart when found with status 200", async () => {
      const req = { params: { id: "123" } };
      const res = createMockResponse();
      const mockCarrinho = { _id: "123", produtos: [] };

      vi.mocked(service.getOne).mockResolvedValue(mockCarrinho as any);

      await carrinhosController.getOne(req, res);

      expect(service.getOne).toHaveBeenCalledWith("123");
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });

    it("should return CART_NOT_FOUND with status 400 when cart does not exist", async () => {
      const req = { params: { id: "invalid" } };
      const res = createMockResponse();

      vi.mocked(service.getOne).mockResolvedValue(null);

      await carrinhosController.getOne(req, res);

      expect(service.getOne).toHaveBeenCalledWith("invalid");
      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });
  });

  describe("post", () => {
    it("should create a cart successfully", async () => {
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: "2" }] }
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const res = createMockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({ idUsuario: "u1", possuiCarrinho: false });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValue([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "p1", quantidade: 2, precoUnitario: 10 }],
        error: null
      });
      vi.mocked(service.precoTotal).mockReturnValue(20);
      vi.mocked(service.quantidadeTotal).mockReturnValue(2);
      vi.mocked(service.totaisSaoSeguros).mockReturnValue(true);
      vi.mocked(service.baixarEstoque).mockResolvedValue(undefined);
      vi.mocked(service.criarCarrinho).mockResolvedValue({ _id: "c1" } as any);

      await carrinhosController.post(req, res);

      expect(etapas.regras).toHaveBeenCalledWith("single_cart", "no_duplicate_product", "product_exists", "enough_stock", "safe_totals");
      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({ message: constant.POST_SUCCESS, _id: "c1" });
    });

    it("should return LIMIT_JUST_ONE_CART when user already has a cart", async () => {
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [] }
      };
      const etapas = { regras: vi.fn(), regra: vi.fn(), abrir: vi.fn() };
      const res = createMockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({ idUsuario: "u1", possuiCarrinho: true });

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return CART_WITH_DUPLICATE_PRODUCT when duplicate products are found", async () => {
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1" }, { idProduto: "p1" }] }
      };
      const etapas = { regras: vi.fn(), regra: vi.fn(), abrir: vi.fn() };
      const res = createMockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({ idUsuario: "u1", possuiCarrinho: false });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValue(["p1"]);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_WITH_DUPLICATE_PRODUCT, idProdutosDuplicados: ["p1"] });
    });

    it("should return error from service when product not found or insufficient stock", async () => {
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }] }
      };
      const etapas = { regras: vi.fn(), regra: vi.fn(), abrir: vi.fn() };
      const res = createMockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({ idUsuario: "u1", possuiCarrinho: false });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValue([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValue({
        produtosComPrecoUnitario: [],
        error: { message: constant.PRODUCT_NOT_FOUND, statusCode: 400, item: 0 } as any
      });

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.PRODUCT_NOT_FOUND, item: 0 });
    });

    it("should return CART_TOTALS_TOO_BIG when totals are not safe", async () => {
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }] }
      };
      const etapas = { regras: vi.fn(), regra: vi.fn(), abrir: vi.fn() };
      const res = createMockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({ idUsuario: "u1", possuiCarrinho: false });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValue([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "p1", quantidade: 1, precoUnitario: 10 }],
        error: null
      });
      vi.mocked(service.precoTotal).mockReturnValue(10);
      vi.mocked(service.quantidadeTotal).mockReturnValue(1);
      vi.mocked(service.totaisSaoSeguros).mockReturnValue(false);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });
  });

  describe("cancelarCompra", () => {
    it("should remove cart and replenish stock when cart exists", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockResponse();
      const mockCarrinho = { _id: "c1" };

      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValue([mockCarrinho] as any);
      vi.mocked(service.removeCarrinho).mockResolvedValue(undefined as any);

      await carrinhosController.cancelarCompra(req, res);

      expect(service.removeCarrinho).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}` });
    });

    it("should return NO_CART when user has no cart", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockResponse();

      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValue([] as any);

      await carrinhosController.cancelarCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase when cart exists", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockResponse();
      const mockCarrinho = { _id: "c1" };

      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValue([mockCarrinho] as any);
      vi.mocked(service.concluiCompra).mockResolvedValue(undefined as any);

      await carrinhosController.concluirCompra(req, res);

      expect(service.concluiCompra).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("should return NO_CART when user has no cart to conclude", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockResponse();

      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValue([] as any);

      await carrinhosController.concluirCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});
```
