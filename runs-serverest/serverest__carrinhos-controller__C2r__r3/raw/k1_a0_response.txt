```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import service from "../../src/services/carrinhos-service";
import constant from "../../src/utils/constants";

vi.mock("../../src/services/carrinhos-service");
vi.mock("../../src/utils/constants", () => ({
  CART_NOT_FOUND: "Não foi encontrado carrinho para este id",
  LIMIT_JUST_ONE_CART: "Não é permitido possuir mais de 1 carrinho",
  CART_WITH_DUPLICATE_PRODUCT: "Não é permitido possuir produtos duplicados",
  PRODUCT_NOT_FOUND: "Produto não encontrado",
  CART_TOTALS_TOO_BIG: "Totais do carrinho muito grandes",
  POST_SUCCESS: "Cadastro realizado com sucesso",
  DELETE_SUCCESS: "Registro excluído com sucesso",
  REPLENISHED_STOCK: "Estoque reabastecido",
  NO_CART: "Não foi encontrado carrinho para este usuário"
}));

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  describe("get", () => {
    it("should return all carts with quantity", async () => {
      const mockCarrinhos = [{ _id: "1" }, { _id: "2" }];
      vi.mocked(service.getAll).mockResolvedValue(mockCarrinhos as any);

      const req = { query: {} };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.get(req as any, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: 2,
        carrinhos: mockCarrinhos
      });
    });
  });

  describe("getOne", () => {
    it("should return 400 when cart is not found", async () => {
      vi.mocked(service.getOne).mockResolvedValue(null);

      const req = { params: { id: "invalid" } };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        json: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.getOne(req as any, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });

    it("should return 200 with cart when found", async () => {
      const mockCarrinho = { _id: "1", produtos: [] };
      vi.mocked(service.getOne).mockResolvedValue(mockCarrinho as any);

      const req = { params: { id: "1" } };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.getOne(req as any, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });
  });

  describe("post", () => {
    it("should return 400 when user already has a cart", async () => {
      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: true
      });

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [] }
      };
      const res: any = {
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn()
          }
        },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return 400 when there are duplicate products", async () => {
      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValue(["prod-1"]);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod-1", quantidade: "2" }] }
      };
      const res: any = {
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn()
          }
        },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["prod-1"]
      });
    });

    it("should return error status when product service returns an error", async () => {
      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValue([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValue({
        produtosComPrecoUnitario: [],
        error: {
          statusCode: 400,
          message: constant.PRODUCT_NOT_FOUND,
          item: 0
        }
      });

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod-1", quantidade: "2" }] }
      };
      const res: any = {
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn()
          }
        },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.PRODUCT_NOT_FOUND,
        item: 0
      });
    });

    it("should return 400 when totals are too big", async () => {
      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValue([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "prod-1", quantidade: 2, precoUnitario: 10 }],
        error: null
      });
      vi.mocked(service.precoTotal).mockReturnValue(999999999);
      vi.mocked(service.quantidadeTotal).mockReturnValue(10);
      vi.mocked(service.totaisSaoSeguros).mockReturnValue(false);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod-1", quantidade: "2" }] }
      };
      const res: any = {
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn()
          }
        },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });

    it("should successfully create a cart", async () => {
      vi.mocked(service.usuarioJaPossuiCarrinho).mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados).mockReturnValue([]);
      const mockProdutos = [{ idProduto: "prod-1", quantidade: 2, precoUnitario: 10 }];
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro).mockResolvedValue({
        produtosComPrecoUnitario: mockProdutos,
        error: null
      });
      vi.mocked(service.precoTotal).mockReturnValue(20);
      vi.mocked(service.quantidadeTotal).mockReturnValue(2);
      vi.mocked(service.totaisSaoSeguros).mockReturnValue(true);
      vi.mocked(service.criarCarrinho).mockResolvedValue({ _id: "cart-1" } as any);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod-1", quantidade: "2" }] }
      };
      const res: any = {
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn(),
            abrir: vi.fn()
          }
        },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res);

      expect(service.baixarEstoque).toHaveBeenCalledWith(mockProdutos);
      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.POST_SUCCESS,
        _id: "cart-1"
      });
    });
  });

  describe("cancelarCompra", () => {
    it("should remove cart and replenish stock when cart exists", async () => {
      const mockCarrinho = { _id: "cart-1" };
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValue([mockCarrinho] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req as any, res);

      expect(service.removeCarrinho).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}`
      });
    });

    it("should return NO_CART message when cart does not exist", async () => {
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValue([] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req as any, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase when cart exists", async () => {
      const mockCarrinho = { _id: "cart-1" };
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValue([mockCarrinho] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req as any, res);

      expect(service.concluiCompra).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("should return NO_CART message when cart does not exist", async () => {
      vi.mocked(service.getCarrinhoDoUsuario).mockResolvedValue([] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res: any = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req as any, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});
```
