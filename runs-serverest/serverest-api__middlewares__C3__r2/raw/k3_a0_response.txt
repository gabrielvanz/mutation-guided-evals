```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on a protected route", async () => {
      const res = await request(app).get("/carrinhos");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token format is invalid (missing Bearer prefix)", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "invalid-token-format");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is malformed or invalid", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "Bearer invalid.jwt.token");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 403 when a non-admin user tries to access an admin-only route", async () => {
      const uniqueEmail = `user_${Date.now()}_${Math.random()}@qa.com`;
      const createUserRes = await request(app)
        .post("/usuarios")
        .send({
          nome: "User Comum",
          email: uniqueEmail,
          password: "teste",
          administrador: "false"
        });

      expect(createUserRes.status).toBe(201);

      const loginRes = await request(app)
        .post("/login")
        .send({
          email: uniqueEmail,
          password: "teste"
        });

      expect(loginRes.status).toBe(200);
      const token = loginRes.body.authorization;

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: `Produto_${Date.now()}`,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(403);
      expect(res.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 when sending invalid JSON payload", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("Content-Type", "application/json")
        .send("{\"nome\": \"Incompleto\"");

      expect(res.status).toBe(400);
      expect(res.body.message).toBeDefined();
    });

    it("should handle invalid URI encoding gracefully and return 400", async () => {
      const res = await request(app).get("/usuarios/%C0%C0");

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });
  });
});
```
