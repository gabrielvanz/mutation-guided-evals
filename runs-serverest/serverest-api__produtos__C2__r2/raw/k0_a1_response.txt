```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

async function getAdminToken() {
  const email = `adm_user@email.com`;
  await request(app)
    .post("/usuarios")
    .send({
      nome: "Admin User",
      email,
      password: "password123",
      administrador: "true"
    });

  const loginRes = await request(app)
    .post("/login")
    .send({
      email,
      password: "password123"
    });

  return loginRes.body.authorization;
}

describe("GET /produtos", () => {
  it("should return all products and quantity", async () => {
    const nome = `Produto Teste`;
    const token = await getAdminToken();

    await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send({
        nome,
        preco: 100,
        descricao: "Teste descricao",
        quantidade: 10
      });

    const res = await request(app).get("/produtos");
    expect(res.status).toBe(200);
    expect(res.body).toHaveProperty("quantidade");
    expect(res.body).toHaveProperty("produtos");
    expect(Array.isArray(res.body.produtos)).toBe(true);
    expect(res.body.quantidade).toBeGreaterThanOrEqual(1);
  });

  it("should filter products by query parameters", async () => {
    const uniqueName = `UniqueProdName`;
    const token = await getAdminToken();

    await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send({
        nome: uniqueName,
        preco: 250,
        descricao: "Desc",
        quantidade: 5
      });

    const res = await request(app).get(`/produtos?nome=${uniqueName}`);
    expect(res.status).toBe(200);
    expect(res.body.quantidade).toBe(1);
    expect(res.body.produtos[0].nome).toBe(uniqueName);
  });
});

describe("GET /produtos/:id", () => {
  it("should return a product by valid id", async () => {
    const token = await getAdminToken();
    const createRes = await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send({
        nome: `ProdGetOne`,
        preco: 50,
        descricao: "Desc",
        quantidade: 2
      });

    const id = createRes.body._id;
    const res = await request(app).get(`/produtos/${id}`);
    expect(res.status).toBe(200);
    expect(res.body._id).toBe(id);
    expect(res.body).toHaveProperty("nome");
  });

  it("should return 400 when product id does not exist", async () => {
    const res = await request(app).get("/produtos/abcdef1234567890");
    expect(res.status).toBe(400);
    expect(res.body).toEqual({ message: "Produto não encontrado" });
  });

  it("should return validation error for invalid id format", async () => {
    const res = await request(app).get("/produtos/invalid-id");
    expect(res.status).toBe(400);
  });
});

describe("POST /produtos", () => {
  it("should create a product when user is admin and payload is valid", async () => {
    const token = await getAdminToken();
    const res = await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send({
        nome: `Novo Produto`,
        preco: 200,
        descricao: "Produto legal",
        quantidade: 15,
        imagem: "http://imagem.com/img.jpg"
      });

    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
    expect(res.body).toHaveProperty("_id");
  });

  it("should return 400 when product name is already used", async () => {
    const token = await getAdminToken();
    const nome = `Duplicado Produto`;
    const payload = {
      nome,
      preco: 100,
      descricao: "Desc",
      quantidade: 10
    };

    await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send(payload);

    const res = await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send(payload);

    expect(res.status).toBe(400);
    expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
  });

  it("should reject product creation without authentication", async () => {
    const res = await request(app)
      .post("/produtos")
      .send({
        nome: "Sem Token",
        preco: 100,
        descricao: "Desc",
        quantidade: 10
      });

    expect(res.status).toBe(401);
  });
});

describe("PUT /produtos/:id", () => {
  it("should update an existing product successfully", async () => {
    const token = await getAdminToken();
    const createRes = await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send({
        nome: `Produto Update`,
        preco: 100,
        descricao: "Desc",
        quantidade: 10
      });

    const id = createRes.body._id;
    const res = await request(app)
      .put(`/produtos/${id}`)
      .set("authorization", token)
      .send({
        nome: `Produto Atualizado`,
        preco: 150,
        descricao: "Nova Desc",
        quantidade: 20
      });

    expect(res.status).toBe(200);
    expect(res.body).toEqual({ message: "Registro alterado com sucesso" });
  });

  it("should create product via put if id does not exist", async () => {
    const token = await getAdminToken();
    const randomId = "1234567890abcdef";
    const res = await request(app)
      .put(`/produtos/${randomId}`)
      .set("authorization", token)
      .send({
        nome: `Produto Upsert`,
        preco: 300,
        descricao: "Upsert desc",
        quantidade: 5
      });

    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
    expect(res.body).toHaveProperty("_id");
  });

  it("should reject put when name is already used by another product", async () => {
    const token = await getAdminToken();
    const name1 = `Prod A`;
    const name2 = `Prod B`;

    const p1 = await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send({ nome: name1, preco: 10, descricao: "D", quantidade: 1 });

    await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send({ nome: name2, preco: 10, descricao: "D", quantidade: 1 });

    const id1 = p1.body._id;

    const res = await request(app)
      .put(`/produtos/${id1}`)
      .set("authorization", token)
      .send({
        nome: name2,
        preco: 15,
        descricao: "D",
        quantidade: 2
      });

    expect(res.status).toBe(400);
    expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
  });
});

describe("DELETE /produtos/:id", () => {
  it("should delete a product successfully when it is not in a cart", async () => {
    const token = await getAdminToken();
    const createRes = await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send({
        nome: `Produto Delete`,
        preco: 50,
        descricao: "Desc",
        quantidade: 5
      });

    const id = createRes.body._id;
    const res = await request(app)
      .delete(`/produtos/${id}`)
      .set("authorization", token);

    expect(res.status).toBe(200);
    expect(res.body).toEqual({ message: "Registro excluído com sucesso" });
  });

  it("should return delete none message when product id does not exist", async () => {
    const token = await getAdminToken();
    const res = await request(app)
      .delete("/produtos/abcdef1234567890")
      .set("authorization", token);

    expect(res.status).toBe(200);
    expect(res.body).toEqual({ message: "Nenhum registro excluído" });
  });

  it("should prevent deleting a product that is present in a cart", async () => {
    const token = await getAdminToken();
    const prodRes = await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send({
        nome: `Produto Carrinho`,
        preco: 100,
        descricao: "Desc",
        quantidade: 10
      });

    const produtoId = prodRes.body._id;

    const userEmail = `user_cart@email.com`;
    await request(app)
      .post("/usuarios")
      .send({
        nome: "Cart User",
        email: userEmail,
        password: "password123",
        administrador: "false"
      });

    const loginRes = await request(app)
      .post("/login")
      .send({ email: userEmail, password: "password123" });

    const userToken = loginRes.body.authorization;

    await request(app)
      .post("/carrinhos")
      .set("authorization", userToken)
      .send({
        produtos: [{ idProduto: produtoId, quantidade: 2 }]
      });

    const res = await request(app)
      .delete(`/produtos/${produtoId}`)
      .set("authorization", token);

    expect(res.status).toBe(400);
    expect(res.body).toHaveProperty("message", "Não é permitido excluir produto que faz parte de carrinho");
    expect(res.body).toHaveProperty("idCarrinhos");
    expect(Array.isArray(res.body.idCarrinhos)).toBe(true);
  });
});
```
