```ts
import { describe, it, expect } from "vitest";
import produtosService from "../../src/services/produtos-service";

describe("getAll", () => {
  it("should return all products matching query", async () => {
    const produtos = await produtosService.getAll({});
    expect(Array.isArray(produtos)).toBe(true);
  });
});

describe("getOne", () => {
  it("should return null when product does not exist", async () => {
    const produto = await produtosService.getOne("inexistente123");
    expect(produto).toBeNull();
  });
});

describe("existeProduto", () => {
  it("should return null when product is not found by search criteria", async () => {
    const produto = await produtosService.existeProduto({ nome: "Produto Inexistente XYZ" });
    expect(produto).toBeNull();
  });
});

describe("criarProduto", () => {
  it("should create a product and format values correctly", async () => {
    const novoProduto = {
      nome: "  Produto Teste  ",
      preco: "199.99",
      quantidade: "10",
      descricao: "Teste",
      imagem: " http://imagem.com/img.png "
    };

    const criado = await produtosService.criarProduto(novoProduto);
    expect(criado).toHaveProperty("_id");
    expect(criado.nome).toBe("Produto Teste");
    expect(criado.preco).toBe(199.99);
    expect(criado.quantidade).toBe(10);
    expect(criado.imagem).toBe("http://imagem.com/img.png");
    expect(criado).toHaveProperty("alteradoEm");
  });
});

describe("getPrecosUnitariosPorLote", () => {
  it("should return error when product in batch does not exist", async () => {
    const resultado = await produtosService.getPrecosUnitariosPorLote([
      { idProduto: "inexistente999", quantidade: 1 }
    ]);
    expect(resultado).toHaveProperty("error");
    expect(resultado.error.statusCode).toBe(400);
    expect(resultado.error.message).toBe("Produto não encontrado");
  });
});

describe("deleteById", () => {
  it("should return 0 removed count when deleting non-existent product", async () => {
    const removidos = await produtosService.deleteById("inexistente123");
    expect(removidos).toBe(0);
  });
});

describe("createOrUpdateById", () => {
  it("should create or update a product by id with formatted values", async () => {
    const body = {
      nome: "  Produto Update  ",
      preco: "50",
      quantidade: "5"
    };

    const resultado = await produtosService.createOrUpdateById("id-teste-up", body);
    expect(resultado).toHaveProperty("_id", "id-teste-up");
    expect(resultado.nome).toBe("Produto Update");
    expect(resultado.preco).toBe(50);
    expect(resultado.quantidade).toBe(5);
    expect(resultado).toHaveProperty("alteradoEm");
  });
});

describe("updateById", () => {
  it("should update a product directly by id", async () => {
    const resultado = await produtosService.updateById("id-direto", {
      $set: { nome: "Atualizado Direto", preco: 10, quantidade: 2 }
    });
    expect(resultado).toHaveProperty("_id", "id-direto");
    expect(resultado.nome).toBe("Atualizado Direto");
  });
});
```
