```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Usuarios API", () => {
  it("should get empty or list of users", async () => {
    const res = await request(app).get("/usuarios");
    expect(res.status).toBe(200);
    expect(res.body).toHaveProperty("quantidade");
    expect(res.body).toHaveProperty("usuarios");
    expect(Array.isArray(res.body.usuarios)).toBe(true);
  });

  it("should create a new user successfully", async () => {
    const uniqueEmail = "user_test_success@test.com";
    const res = await request(app)
      .post("/usuarios")
      .send({
        nome: "Test User",
        email: uniqueEmail,
        password: "password123",
        administrador: "true"
      });

    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
    expect(res.body).toHaveProperty("_id");
  });

  it("should reject user creation with duplicate email", async () => {
    const uniqueEmail = "dup_user_test@test.com";
    const payload = {
      nome: "First User",
      email: uniqueEmail,
      password: "password123",
      administrador: "false"
    };

    const firstRes = await request(app).post("/usuarios").send(payload);
    expect(firstRes.status).toBe(201);

    const secondRes = await request(app).post("/usuarios").send({
      ...payload,
      nome: "Second User"
    });

    expect(secondRes.status).toBe(400);
    expect(secondRes.body).toEqual({ message: "Este email já está sendo usado" });
  });

  it("should prevent updating user email to one already used by another user", async () => {
    const email1 = "user1_put_dup@test.com";
    const email2 = "user2_put_dup@test.com";

    const res1 = await request(app).post("/usuarios").send({
      nome: "User One",
      email: email1,
      password: "123",
      administrador: "false"
    });
    const id1 = res1.body._id;

    const res2 = await request(app).post("/usuarios").send({
      nome: "User Two",
      email: email2,
      password: "123",
      administrador: "false"
    });
    const id2 = res2.body._id;

    // Try to update User Two's email to User One's email
    const putRes = await request(app).put(`/usuarios/${id2}`).send({
      nome: "User Two Updated",
      email: email1,
      password: "123",
      administrador: "false"
    });

    expect(putRes.status).toBe(400);
    expect(putRes.body).toEqual({ message: "Este email já está sendo usado" });
  });

  it("should return user not found for non-existent id in getOne", async () => {
    const fakeId = "1234567890abcdef";
    const res = await request(app).get(`/usuarios/${fakeId}`);
    expect(res.status).toBe(400);
    expect(res.body).toEqual({ message: "Usuário não encontrado" });
  });

  it("should get a specific user by id", async () => {
    const uniqueEmail = "getone_user_test@test.com";
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Get One User",
        email: uniqueEmail,
        password: "password123",
        administrador: "true"
      });

    const userId = createRes.body._id;

    const res = await request(app).get(`/usuarios/${userId}`);
    expect(res.status).toBe(200);
    expect(res.body).toHaveProperty("_id", userId);
    expect(res.body).toHaveProperty("email", uniqueEmail);
  });

  it("should delete an existing user", async () => {
    const uniqueEmail = "del_user_test@test.com";
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "To Delete",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    const userId = createRes.body._id;

    const res = await request(app).delete(`/usuarios/${userId}`);
    expect(res.status).toBe(200);
    expect(res.body).toEqual({ message: "Registro excluído com sucesso" });
  });

  it("should prevent deleting a user who has a cart", async () => {
    const uniqueEmail = "del_cart_user@test.com";
    const userRes = await request(app).post("/usuarios").send({
      nome: "User With Cart",
      email: uniqueEmail,
      password: "123",
      administrador: "false"
    });
    const userId = userRes.body._id;

    const prodRes = await request(app).post("/produtos").send({
      nome: "Produto Teste Carrinho",
      preco: 10,
      descricao: "Desc",
      quantidade: 5
    });
    const produtoId = prodRes.body._id;

    // Login as the user to get a token for creating a cart
    const loginRes = await request(app).post("/login").send({
      email: uniqueEmail,
      password: "123"
    });
    const token = loginRes.body.authorization;

    const cartRes = await request(app)
      .post("/carrinhos")
      .set("authorization", token)
      .send({
        produtos: [{ idProduto: produtoId, quantidade: 1 }]
      });
    expect(cartRes.status).toBe(201);
    const cartId = cartRes.body._id;

    // Attempt to delete user
    const delRes = await request(app).delete(`/usuarios/${userId}`);
    expect(delRes.status).toBe(400);
    expect(delRes.body).toEqual({
      message: "Não é permitido excluir usuário com carrinho cadastrado",
      idCarrinho: cartId
    });
  });

  it("should return delete none when user does not exist", async () => {
    const nonExistentId = "nonexistentid123";
    const res = await request(app).delete(`/usuarios/${nonExistentId}`);
    expect(res.status).toBe(200);
    expect(res.body).toEqual({ message: "Nenhum registro excluído" });
  });

  it("should update an existing user with put", async () => {
    const uniqueEmail = "put_user_test@test.com";
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Before Update",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    const userId = createRes.body._id;

    const res = await request(app)
      .put(`/usuarios/${userId}`)
      .send({
        nome: "After Update",
        email: uniqueEmail,
        password: "newpassword123",
        administrador: "true"
      });

    expect(res.status).toBe(200);
    expect(res.body).toEqual({ message: "Registro alterado com sucesso" });
  });

  it("should upsert a user with put if id does not exist", async () => {
    const newId = "fedcba0987654321";
    const uniqueEmail = "upsert_user_test@test.com";
    const res = await request(app)
      .put(`/usuarios/${newId}`)
      .send({
        nome: "Upserted User",
        email: uniqueEmail,
        password: "password123",
        administrador: "true"
      });

    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
    expect(res.body).toHaveProperty("_id");
  });
});
```
