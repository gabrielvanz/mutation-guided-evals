```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent when request timed out", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const result = errorHandler(error, req, res, () => {});
    expect(result).toBeUndefined();
  });

  it("should handle headersSent when request did not time out and response is not writableEnded", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    let socketDestroyCalled = false;
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/users",
      socket: {
        destroy: () => {
          socketDestroyCalled = true;
        }
      }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});
    expect(socketDestroyCalled).toBe(true);
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: [{ message: "field is required" }]
    };
    const req = {};
    const jsonMock = (data: any) => data;
    const statusMock = (code: number) => ({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const response = errorHandler(error, req, res as any, () => {});
    expect(response).toEqual({ message: "field is required" });
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = {};
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res as any, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed error type", () => {
    const error = {
      type: "entity.parse.failed",
      message: "JSON parse error"
    };
    const req = {};
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res as any, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large error type", () => {
    const error = { type: "entity.too.large" };
    const req = {};
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res as any, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle parameters.too.many error type", () => {
    const error = { type: "parameters.too.many" };
    const req = {};
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res as any, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported error type", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = {};
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res as any, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported error type", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate-x" };
    const req = {};
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res as any, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate-x" });
  });

  it("should handle client error with Z_ prefix in code", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "Bad compression"
    };
    const req = {};
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res as any, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = { code: "ETIMEDOUT" };
    const req = {};
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res as any, () => {});
    expect(statusCode).toBe(408);
    expect(jsonBody).toEqual({ message: "Tempo limite esgotado" });
  });

  it("should handle general client error where expose is true and status is 4xx", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "some.client.error",
      message: "Unprocessable"
    };
    const req = {};
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res as any, () => {});
    expect(statusCode).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.client.error"
    });
  });

  it("should handle unhandled error with error type returning 500", () => {
    const error = {
      type: "database.error",
      stack: "stack trace",
      message: "db down"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res as any, () => {});
    expect(statusCode).toBe(500);
    expect(jsonBody).toHaveProperty("message");
    expect(jsonBody).toHaveProperty("error", "database.error");
    expect(jsonBody).toHaveProperty("version");
  });
});
```
