```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headers already sent with writableEnded false", () => {
    const error = new Error("Test error");
    const socket = { destroy: () => {} };
    let socketDestroyCalled = false;
    socket.destroy = () => {
      socketDestroyCalled = true;
    };
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: false,
      socket
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});

    expect(socketDestroyCalled).toBe(true);
  });

  it("should handle headers already sent with writableEnded true", () => {
    const error = new Error("Test error");
    const socket = { destroy: () => {} };
    let socketDestroyCalled = false;
    socket.destroy = () => {
      socketDestroyCalled = true;
    };
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: true,
      socket
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    errorHandler(error, req, res, () => {});

    expect(socketDestroyCalled).toBe(false);
  });

  it("should handle ValidationError", () => {
    const error = { name: "ValidationError", details: [{ message: "error" }] };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonCalledWith).toBeDefined();
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonCalledWith).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonCalledWith).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large" };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCalled).toBe(413);
    expect(jsonCalledWith).toEqual({ message: "Payload too large" });
  });

  it("should handle parameters.too.many", () => {
    const error = { type: "parameters.too.many" };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCalled).toBe(413);
    expect(jsonCalledWith).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCalled).toBe(415);
    expect(jsonCalledWith).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCalled).toBe(415);
    expect(jsonCalledWith).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle client error with Z_ prefix in code", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "compression error"
    };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonCalledWith).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = { code: "ETIMEDOUT" };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCalled).toBe(408);
    expect(jsonCalledWith).toHaveProperty("message");
  });

  it("should handle generic client error", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "some.client.error",
      message: "invalid request"
    };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCalled).toBe(422);
    expect(jsonCalledWith).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.client.error"
    });
  });

  it("should handle server error with error type", () => {
    const error = {
      type: "database.error",
      message: "db down",
      stack: "at stack..."
    };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, { body: {} }, res, () => {});

    expect(statusCalled).toBe(500);
    expect(jsonCalledWith).toHaveProperty("message");
    expect(jsonCalledWith).toHaveProperty("error", "database.error");
    expect(jsonCalledWith).toHaveProperty("version");
  });
});
```
