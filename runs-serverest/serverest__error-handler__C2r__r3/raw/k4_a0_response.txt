```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headers already sent with timedout request", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: { destroy: () => {} }
    };
    let destroyed = false;
    req.socket.destroy = () => {
      destroyed = true;
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});
    expect(destroyed).toBe(true);
  });

  it("should handle headers already sent when response is already writableEnded", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/test",
      socket: { destroy: () => {} }
    };
    let destroyed = false;
    req.socket.destroy = () => {
      destroyed = true;
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    errorHandler(error, req, res, () => {});
    expect(destroyed).toBe(false);
  });

  it("should handle ValidationError", () => {
    const error = { name: "ValidationError", details: [] };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, {}, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toBeDefined();
  });

  it("should handle URIError", () => {
    const error = new URIError("malformed URI");
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, {}, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, {}, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large" };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, {}, res, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle parameters.too.many", () => {
    const error = { type: "parameters.too.many" };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, {}, res, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, {}, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, {}, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle client error with Z_ prefix code", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "data error"
    };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, {}, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = { code: "ETIMEDOUT" };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, {}, res, () => {});
    expect(statusCode).toBe(408);
    expect(jsonBody).toEqual({ message: "Limite de tempo excedido. Tente novamente mais tarde." });
  });

  it("should handle general client error", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "invalid.param"
    };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, { body: {} }, res, () => {});
    expect(statusCode).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "invalid.param"
    });
  });

  it("should handle internal error with type", () => {
    const error = { type: "some.internal.error", stack: "stack" };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, { body: {} }, res, () => {});
    expect(statusCode).toBe(500);
    expect(jsonBody).toEqual(
      expect.objectContaining({
        message: "Erro interno no Servidor",
        error: "some.internal.error"
      })
    );
  });
});
```
