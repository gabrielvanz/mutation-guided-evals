import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent by destroying socket when not timedout", () => {
    const error = new Error("Test error");
    error.stack = "stack";
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: false,
      socket: {
        destroy: () => {}
      }
    };
    let destroyed = false;
    req.socket.destroy = () => {
      destroyed = true;
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});

    expect(destroyed).toBe(true);
  });

  it("should not destroy socket when headersSent and writableEnded is true", () => {
    const error = new Error("Test error");
    error.stack = "stack";
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: true,
      socket: {
        destroy: () => {
          throw new Error("Should not be called");
        }
      }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    expect(() => errorHandler(error, req, res, () => {})).not.toThrow();
  });

  it("should handle ValidationError by returning status 400 with schema message", () => {
    const error = {
      name: "ValidationError",
      details: [{ message: "Field required", path: ["field"] }]
    };
    let statusCode = 0;
    let jsonBody: unknown = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: unknown) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toBeDefined();
  });

  it("should handle URIError by returning status 400", () => {
    const error = new URIError("Malformed URI");
    let statusCode = 0;
    let jsonBody: unknown = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: unknown) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed by returning status 400", () => {
    const error = {
      type: "entity.parse.failed"
    };
    let statusCode = 0;
    let jsonBody: unknown = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: unknown) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large by returning status 413", () => {
    const error = {
      type: "entity.too.large"
    };
    let statusCode = 0;
    let jsonBody: unknown = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: unknown) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle parameters.too.many by returning status 413", () => {
    const error = {
      type: "parameters.too.many"
    };
    let statusCode = 0;
    let jsonBody: unknown = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: unknown) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported by returning status 415", () => {
    const error = {
      type: "charset.unsupported",
      charset: "utf-16"
    };
    let statusCode = 0;
    let jsonBody: unknown = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: unknown) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported by returning status 415", () => {
    const error = {
      type: "encoding.unsupported",
      encoding: "deflate"
    };
    let statusCode = 0;
    let jsonBody: unknown = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: unknown) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle Z_ compressed body errors as client error with status 400", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "Unexpected end of file"
    };
    let statusCode = 0;
    let jsonBody: unknown = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: unknown) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT code with status 408", () => {
    const error = {
      code: "ETIMEDOUT"
    };
    let statusCode = 0;
    let jsonBody: unknown = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: unknown) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCode).toBe(408);
    expect(jsonBody).toBeDefined();
  });

  it("should handle general client error (expose true, 4xx) correctly", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "custom.error.type"
    };
    let statusCode = 0;
    let jsonBody: unknown = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: unknown) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {}, res, () => {});

    expect(statusCode).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.error.type"
    });
  });

  it("should handle unhandled errors with error type returning 500", () => {
    const error = {
      type: "some.server.error",
      stack: "err-stack"
    };
    let statusCode = 0;
    let jsonBody: unknown = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: unknown) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, { body: {} }, res, () => {});

    expect(statusCode).toBe(500);
    expect(jsonBody).toHaveProperty("message");
    expect(jsonBody).toHaveProperty("error", "some.server.error");
    expect(jsonBody).toHaveProperty("version");
  });
});
