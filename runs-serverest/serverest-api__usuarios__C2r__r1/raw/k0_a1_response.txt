```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Usuarios API", () => {
  it("should get all usuarios", async () => {
    const res = await request(app).get("/usuarios");
    expect(res.status).toBe(200);
    expect(res.body).toHaveProperty("quantidade");
    expect(res.body).toHaveProperty("usuarios");
    expect(Array.isArray(res.body.usuarios)).toBe(true);
  });

  it("should create a new usuario successfully", async () => {
    const res = await request(app)
      .post("/usuarios")
      .send({
        nome: "Test User",
        email: "test_unique_user_1@example.com",
        password: "password123",
        administrador: "true"
      });

    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
    expect(res.body).toHaveProperty("_id");
  });

  it("should return an error when creating a usuario with an existing email", async () => {
    const userData = {
      nome: "User One",
      email: "test_duplicate_email@example.com",
      password: "password123",
      administrador: "false"
    };

    const firstRes = await request(app).post("/usuarios").send(userData);
    expect(firstRes.status).toBe(201);

    const secondRes = await request(app).post("/usuarios").send({
      ...userData,
      nome: "User Two"
    });

    expect(secondRes.status).toBe(400);
    expect(secondRes.body).toEqual({ message: "Este email já está sendo usado" });
  });

  it("should return validation error when missing required fields on post", async () => {
    const res = await request(app)
      .post("/usuarios")
      .send({
        nome: "Incomplete User"
      });

    expect(res.status).toBe(400);
  });

  it("should get a single usuario by id", async () => {
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Get One User",
        email: "getone_unique_user@example.com",
        password: "password123",
        administrador: "true"
      });

    expect(createRes.status).toBe(201);
    const userId = createRes.body._id;

    const getRes = await request(app).get(`/usuarios/${userId}`);
    expect(getRes.status).toBe(200);
    expect(getRes.body).toHaveProperty("_id", userId);
    expect(getRes.body).toHaveProperty("email", "getone_unique_user@example.com");
  });

  it("should return 400 when getting a non-existent usuario with valid id format", async () => {
    const fakeId = "1234567890abcdef";
    const res = await request(app).get(`/usuarios/${fakeId}`);
    expect(res.status).toBe(400);
    expect(res.body).toEqual({ message: "Usuário não encontrado" });
  });

  it("should delete an existing usuario", async () => {
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Delete User",
        email: "delete_unique_user@example.com",
        password: "password123",
        administrador: "false"
      });

    expect(createRes.status).toBe(201);
    const userId = createRes.body._id;

    const deleteRes = await request(app).delete(`/usuarios/${userId}`);
    expect(deleteRes.status).toBe(200);
    expect(deleteRes.body).toEqual({ message: "Registro excluído com sucesso" });
  });

  it("should handle delete when usuario does not exist", async () => {
    const fakeId = "nonexistentid123";
    const deleteRes = await request(app).delete(`/usuarios/${fakeId}`);
    expect(deleteRes.status).toBe(200);
    expect(deleteRes.body).toEqual({ message: "Nenhum registro excluído" });
  });

  it("should update an existing usuario via put", async () => {
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Original Name",
        email: "put_unique_user@example.com",
        password: "password123",
        administrador: "false"
      });

    expect(createRes.status).toBe(201);
    const userId = createRes.body._id;

    const putRes = await request(app)
      .put(`/usuarios/${userId}`)
      .send({
        nome: "Updated Name",
        email: "put_unique_user@example.com",
        password: "newpassword123",
        administrador: "true"
      });

    expect(putRes.status).toBe(200);
    expect(putRes.body).toEqual({ message: "Registro alterado com sucesso" });
  });

  it("should create via put if usuario does not exist", async () => {
    const fakeId = "abcdef1234567890";

    const putRes = await request(app)
      .put(`/usuarios/${fakeId}`)
      .send({
        nome: "Upserted User",
        email: "putcreate_unique_user@example.com",
        password: "password123",
        administrador: "false"
      });

    expect(putRes.status).toBe(201);
    expect(putRes.body).toHaveProperty("message", "Cadastro realizado com sucesso");
    expect(putRes.body).toHaveProperty("_id");
  });
});
```
