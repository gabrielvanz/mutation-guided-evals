```ts
import { describe, it, expect, vi } from "vitest";
import mod from "../../src/middlewares/rate-limiter";

describe("rate-limiter", () => {
  it("should ignore rate limiting in test or local environment", async () => {
    const req = {};
    const res = {
      locals: {
        etapas: {
          ignorar: vi.fn()
        }
      }
    };
    const next = vi.fn();

    await mod(req, res, next);

    expect(res.locals.etapas.ignorar).toHaveBeenCalled();
    expect(next).toHaveBeenCalled();
  });

  it("should expose criarLimitador, criarOrcamentoDeBytes, criarRateLimiter, and BYTES_POR_HORA", () => {
    expect(typeof mod.criarLimitador).toBe("function");
    expect(typeof mod.criarOrcamentoDeBytes).toBe("function");
    expect(typeof mod.criarRateLimiter).toBe("function");
    expect(typeof mod.BYTES_POR_HORA).toBe("number");
  });
});

describe("criarOrcamentoDeBytes", () => {
  it("should initialize with default duration and allow checking msAteLiberar", async () => {
    const orcamento = mod.criarOrcamentoDeBytes({ bytes: 100, segundos: 10 });
    expect(orcamento.duracaoMs).toBe(10000);

    const ms = await orcamento.msAteLiberar("127.0.0.1");
    expect(ms).toBe(0);
  });

  it("should not penalize when size is zero or negative", async () => {
    const orcamento = mod.criarOrcamentoDeBytes({ bytes: 100, segundos: 10 });
    orcamento.registrar("127.0.0.1", 0);
    orcamento.registrar("127.0.0.1", -10);

    const ms = await orcamento.msAteLiberar("127.0.0.1");
    expect(ms).toBe(0);
  });
});

describe("criarLimitador", () => {
  it("should allow requests that do not exceed limits", async () => {
    const mockGlobal = {
      consume: vi.fn().mockResolvedValue(true),
      duration: 30
    };
    const mockPorIp = {
      consume: vi.fn().mockResolvedValue(true),
      duration: 30
    };
    const mockOrcamento = {
      msAteLiberar: vi.fn().mockResolvedValue(0),
      duracaoMs: 3600000
    };

    const limitador = mod.criarLimitador({
      global: mockGlobal,
      porIp: mockPorIp,
      orcamento: mockOrcamento
    });

    const req = { method: "GET", path: "/usuarios", ip: "127.0.0.1", on: vi.fn() };
    const res = { getHeader: vi.fn().mockReturnValue("10") };

    const resultado = await limitador(req, res);
    expect(resultado).toBeNull();
    expect(mockOrcamento.msAteLiberar).toHaveBeenCalledWith("127.0.0.1");
    expect(mockGlobal.consume).toHaveBeenCalledWith("global");
    expect(mockPorIp.consume).toHaveBeenCalledWith("127.0.0.1");
  });

  it("should block requests when byte budget is exceeded", async () => {
    const mockOrcamento = {
      msAteLiberar: vi.fn().mockResolvedValue(5000),
      duracaoMs: 3600000
    };

    const limitador = mod.criarLimitador({
      orcamento: mockOrcamento
    });

    const req = { method: "GET", path: "/usuarios", ip: "127.0.0.1" };
    const res = {};

    const resultado = await limitador(req, res);
    expect(resultado).toEqual({ tipo: "orcamento-de-bytes", ms: 5000 });
  });

  it("should handle error in byte budget check gracefully", async () => {
    const mockOrcamento = {
      msAteLiberar: vi.fn().mockRejectedValue(new Error("Storage error")),
      duracaoMs: 3600000
    };

    const limitador = mod.criarLimitador({
      orcamento: mockOrcamento
    });

    const req = { method: "GET", path: "/usuarios", ip: "127.0.0.1" };
    const res = {};

    const resultado = await limitador(req, res);
    expect(resultado?.tipo).toBe("orcamento-de-bytes");
    expect(resultado?.ms).toBe(3600000);
    expect(resultado?.erro).toBeDefined();
  });
});

describe("criarRateLimiter", () => {
  it("should call next when limitador returns null", async () => {
    const mockLimitador = vi.fn().mockResolvedValue(null);
    const rateLimiter = mod.criarRateLimiter({ limitador: mockLimitador });

    const req = {};
    const res = {};
    const next = vi.fn();

    await rateLimiter(req, res, next);
    expect(next).toHaveBeenCalled();
  });

  it("should send 429 and set Retry-After header when blocked", async () => {
    const mockLimitador = vi.fn().mockResolvedValue({ tipo: "requisicoes-por-ip", ms: 15000 });
    const rateLimiter = mod.criarRateLimiter({ limitador: mockLimitador });

    const req = { ip: "127.0.0.1" };
    const res = {
      set: vi.fn(),
      status: vi.fn().mockReturnThis(),
      send: vi.fn()
    };
    const next = vi.fn();

    await rateLimiter(req, res, next);

    expect(res.set).toHaveBeenCalledWith("Retry-After", "15");
    expect(res.status).toHaveBeenCalledWith(429);
    expect(res.send).toHaveBeenCalled();
    expect(next).not.toHaveBeenCalled();
  });
});
```
