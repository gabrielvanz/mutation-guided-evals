```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent and destroy socket when not timedout", () => {
    const error = new Error("Test error");
    const req = { method: "GET", originalUrl: "/test", timedout: false, socket: { destroy: () => {} } };
    let socketDestroyed = false;
    req.socket.destroy = () => {
      socketDestroyed = true;
    };
    const res = { headersSent: true, writableEnded: false };

    errorHandler(error, req, res, () => {});

    expect(socketDestroyed).toBe(true);
  });

  it("should handle headersSent without destroying socket when writableEnded is true", () => {
    const error = new Error("Test error");
    const req = { method: "GET", originalUrl: "/test", timedout: true, socket: { destroy: () => {} } };
    let socketDestroyed = false;
    req.socket.destroy = () => {
      socketDestroyed = true;
    };
    const res = { headersSent: true, writableEnded: true };

    errorHandler(error, req, res, () => {});

    expect(socketDestroyed).toBe(false);
  });

  it("should return 400 for ValidationError", () => {
    const error = { name: "ValidationError", details: [] };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toBeDefined();
  });

  it("should return 400 for URIError", () => {
    const error = new URIError("Malformed URI");
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should return 400 for entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should return 413 for entity.too.large", () => {
    const error = { type: "entity.too.large" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should return 415 for charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should return 415 for encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should return 400 for client error with Z_ prefix code", () => {
    const error = { expose: true, statusCode: 400, code: "Z_DATA_ERROR", message: "invalid data" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should return 408 for ETIMEDOUT error code", () => {
    const error = { code: "ETIMEDOUT" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(408);
    expect(jsonBody).toHaveProperty("message");
  });

  it("should return client error for generic client error (expose true, 4xx)", () => {
    const error = { expose: true, statusCode: 422, type: "custom.type" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.type"
    });
  });

  it("should return 500 with error type when error has type", () => {
    const error = { type: "some.internal.type", stack: "stack-trace" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(500);
    expect(jsonBody).toHaveProperty("message");
    expect(jsonBody).toHaveProperty("error", "some.internal.type");
    expect(jsonBody).toHaveProperty("version");
  });
});
```
