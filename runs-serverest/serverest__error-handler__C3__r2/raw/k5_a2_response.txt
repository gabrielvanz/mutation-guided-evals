```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headers already sent with timedout request", () => {
    const error = new Error("Test error");
    error.stack = "Error: Test error";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };
    const next = () => {};

    const result = errorHandler(error, req, res, next);
    expect(result).toBeUndefined();
  });

  it("should handle headers already sent without timedout request", () => {
    const error = new Error("Test error");
    error.stack = "Error: Test error";
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/test",
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };
    const next = () => {};

    const result = errorHandler(error, req, res, next);
    expect(result).toBeUndefined();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: []
    };
    const req = {};
    let statusCalled = 0;
    let jsonArg: any;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCalled = code;
        return {
          json(data: any) {
            jsonArg = data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonArg).toBeDefined();
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = {};
    let statusCalled = 0;
    let jsonArg: any;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCalled = code;
        return {
          json(data: any) {
            jsonArg = data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonArg).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed error type", () => {
    const error = {
      type: "entity.parse.failed",
      message: "JSON parse error"
    };
    const req = {};
    let statusCalled = 0;
    let jsonArg: any;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCalled = code;
        return {
          json(data: any) {
            jsonArg = data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonArg).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it.each([
    ["entity.too.large"],
    ["parameters.too.many"]
  ])("should handle payload too large error types: %s", (errorType) => {
    const error = { type: errorType };
    const req = {};
    let statusCalled = 0;
    let jsonArg: any;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCalled = code;
        return {
          json(data: any) {
            jsonArg = data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(413);
    expect(jsonArg).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported error type", () => {
    const error = {
      type: "charset.unsupported",
      charset: "utf-16"
    };
    const req = {};
    let statusCalled = 0;
    let jsonArg: any;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCalled = code;
        return {
          json(data: any) {
            jsonArg = data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(415);
    expect(jsonArg).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported error type", () => {
    const error = {
      type: "encoding.unsupported",
      encoding: "deflate"
    };
    const req = {};
    let statusCalled = 0;
    let jsonArg: any;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCalled = code;
        return {
          json(data: any) {
            jsonArg = data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(415);
    expect(jsonArg).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle client error with Z_ prefix code", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "Invalid compression"
    };
    const req = {};
    let statusCalled = 0;
    let jsonArg: any;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCalled = code;
        return {
          json(data: any) {
            jsonArg = data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonArg).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = {
      code: "ETIMEDOUT"
    };
    const req = {};
    let statusCalled = 0;
    let jsonArg: any;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCalled = code;
        return {
          json(data: any) {
            jsonArg = data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(408);
    expect(jsonArg).toHaveProperty("message");
  });

  it("should handle generic client error", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "some.client.error",
      message: "Bad request details"
    };
    const req = {};
    let statusCalled = 0;
    let jsonArg: any;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCalled = code;
        return {
          json(data: any) {
            jsonArg = data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(422);
    expect(jsonArg).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.client.error"
    });
  });

  it("should handle internal error with error type", () => {
    const error = {
      type: "database.error",
      message: "DB down",
      stack: "Error stack"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonArg: any;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCalled = code;
        return {
          json(data: any) {
            jsonArg = data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(500);
    expect(jsonArg).toEqual(expect.objectContaining({
      error: "database.error"
    }));
  });
});
```
