```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headers already sent with timedout request", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };
    const next = () => {};

    const result = errorHandler(error, req, res, next);
    expect(result).toBeUndefined();
  });

  it("should handle headers already sent without timeout", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/test",
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };
    const next = () => {};

    const result = errorHandler(error, req, res, next);
    expect(result).toBeUndefined();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: [{ message: "Field is required" }]
    };
    const req = {};
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status(s) {
        statusCalled = s;
        return this;
      },
      json(data) {
        jsonCalledWith = data;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonCalledWith).toBeDefined();
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = {};
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status(s) {
        statusCalled = s;
        return this;
      },
      json(data) {
        jsonCalledWith = data;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonCalledWith).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = {};
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status(s) {
        statusCalled = s;
        return this;
      },
      json(data) {
        jsonCalledWith = data;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonCalledWith).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large" };
    const req = {};
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status(s) {
        statusCalled = s;
        return this;
      },
      json(data) {
        jsonCalledWith = data;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(413);
    expect(jsonCalledWith).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-8" };
    const req = {};
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status(s) {
        statusCalled = s;
        return this;
      },
      json(data) {
        jsonCalledWith = data;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(415);
    expect(jsonCalledWith).toEqual({ message: "Unsupported charset", charset: "utf-8" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = {};
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status(s) {
        statusCalled = s;
        return this;
      },
      json(data) {
        jsonCalledWith = data;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(415);
    expect(jsonCalledWith).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle Zlib compression errors as client error", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "invalid block type"
    };
    const req = {};
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status(s) {
        statusCalled = s;
        return this;
      },
      json(data) {
        jsonCalledWith = data;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonCalledWith).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error", () => {
    const error = { code: "ETIMEDOUT" };
    const req = {};
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status(s) {
        statusCalled = s;
        return this;
      },
      json(data) {
        jsonCalledWith = data;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(408);
    expect(jsonCalledWith).toHaveProperty("message");
  });

  it("should handle general client errors", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "invalid.param"
    };
    const req = {};
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status(s) {
        statusCalled = s;
        return this;
      },
      json(data) {
        jsonCalledWith = data;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(422);
    expect(jsonCalledWith).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "invalid.param"
    });
  });

  it("should handle internal error with error type", () => {
    const error = {
      type: "some.internal.error",
      stack: "stacktrace"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status(s) {
        statusCalled = s;
        return this;
      },
      json(data) {
        jsonCalledWith = data;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(500);
    expect(jsonCalledWith).toHaveProperty("error", "some.internal.error");
  });
});
```
