```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle error when headers are already sent and request timed out", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const result = errorHandler(error, req, res, () => {});
    expect(result).toBeUndefined();
  });

  it("should handle schema validation error", () => {
    const error = {
      name: "ValidationError",
      details: [{ message: "field is required" }]
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (data) => {
            jsonResponse = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonResponse).toBeDefined();
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    error.stack = "URIError stack";
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (data) => {
            jsonResponse = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonResponse).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity parse failed error", () => {
    const error = {
      type: "entity.parse.failed",
      stack: "parse error stack"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (data) => {
            jsonResponse = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonResponse).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity too large error", () => {
    const error = {
      type: "entity.too.large",
      stack: "too large stack"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (data) => {
            jsonResponse = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(413);
    expect(jsonResponse).toEqual({ message: "Payload too large" });
  });

  it("should handle parameters too many error", () => {
    const error = {
      type: "parameters.too.many",
      stack: "too many stack"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (data) => {
            jsonResponse = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(413);
    expect(jsonResponse).toEqual({ message: "Payload too large" });
  });

  it("should handle unsupported charset error", () => {
    const error = {
      type: "charset.unsupported",
      charset: "utf-16",
      stack: "charset stack"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (data) => {
            jsonResponse = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonResponse).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle unsupported content encoding error", () => {
    const error = {
      type: "encoding.unsupported",
      encoding: "br",
      stack: "encoding stack"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (data) => {
            jsonResponse = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonResponse).toEqual({ message: "Unsupported content encoding", encoding: "br" });
  });

  it("should handle client error with Z_ prefix code", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "Data error",
      stack: "zlib stack"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (data) => {
            jsonResponse = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonResponse).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = {
      code: "ETIMEDOUT",
      stack: "timeout stack"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (data) => {
            jsonResponse = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(408);
    expect(jsonResponse).toEqual({ message: "O servidor demorou para responder. Tente novamente mais tarde." });
  });

  it("should handle general client error", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "invalid.param",
      stack: "client error stack"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (data) => {
            jsonResponse = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(422);
    expect(jsonResponse).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "invalid.param"
    });
  });

  it("should handle internal server error with type", () => {
    const error = {
      type: "database.error",
      stack: "server error stack"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (data) => {
            jsonResponse = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(500);
    expect(jsonResponse.message).toBe("Erro interno no servidor. Abra uma issue em https://github.com/ServeRest/ServeRest/issues/new/choose se o problema persistir.");
    expect(jsonResponse.error).toBe("database.error");
  });
});
```
