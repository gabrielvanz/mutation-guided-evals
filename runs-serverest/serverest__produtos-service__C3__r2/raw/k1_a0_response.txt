```ts
import { describe, it, expect } from "vitest";
import produtosService from "../../src/services/produtos-service";
import constant from "../../src/utils/constants";

describe("getAll", () => {
  it("should return all products matching query", async () => {
    const produtos = await produtosService.getAll({});
    expect(Array.isArray(produtos)).toBe(true);
  });
});

describe("getOne", () => {
  it("should return null when product does not exist", async () => {
    const produto = await produtosService.getOne("id-inexistente");
    expect(produto).toBeNull();
  });
});

describe("existeProduto", () => {
  it("should return null when product is not found", async () => {
    const produto = await produtosService.existeProduto({ nome: "Produto Inexistente Xpto" });
    expect(produto).toBeNull();
  });
});

describe("criarProduto", () => {
  it("should create a new product and format values", async () => {
    const novoProduto = {
      nome: "  Logitech MX Master 3  ",
      preco: "250",
      descricao: "Mouse ergônomico",
      quantidade: "15",
      imagem: "  imagem.jpg  "
    };

    const criado = await produtosService.criarProduto(novoProduto);
    expect(criado).toHaveProperty("_id");
    expect(criado.nome).toBe("Logitech MX Master 3");
    expect(criado.preco).toBe(250);
    expect(criado.quantidade).toBe(15);
    expect(criado.imagem).toBe("imagem.jpg");
    expect(criado).toHaveProperty("alteradoEm");
  });
});

describe("getPrecosUnitariosPorLote", () => {
  it("should return product not found error for missing product in batch", async () => {
    const resultado = await produtosService.getPrecosUnitariosPorLote([
      { idProduto: "id-inexistente-lote", quantidade: 1 }
    ]);
    expect(resultado).toHaveProperty("error");
    expect(resultado.error.statusCode).toBe(400);
    expect(resultado.error.message).toBe(constant.PRODUCT_NOT_FOUND);
    expect(resultado.error.item.idProduto).toBe("id-inexistente-lote");
  });

  it("should return insufficient stock error when quantity exceeds stock in batch", async () => {
    const criado = await produtosService.criarProduto({
      nome: "Teclado Mecânico",
      preco: "300",
      quantidade: "5"
    });

    const resultado = await produtosService.getPrecosUnitariosPorLote([
      { idProduto: criado._id, quantidade: 10 }
    ]);

    expect(resultado).toHaveProperty("error");
    expect(resultado.error.statusCode).toBe(400);
    expect(resultado.error.message).toBe(constant.INSUFFICIENT_STOCK);
    expect(resultado.error.item.quantidadeEstoque).toBe(5);
    expect(resultado.error.item.quantidade).toBe(10);
  });

  it("should return products with unit prices when all items are valid", async () => {
    const criado = await produtosService.criarProduto({
      nome: "Monitor 4K",
      preco: "1500",
      quantidade: "10"
    });

    const resultado = await produtosService.getPrecosUnitariosPorLote([
      { idProduto: criado._id, quantidade: 2 }
    ]);

    expect(resultado).toHaveProperty("produtosComPreco");
    expect(resultado.produtosComPreco).toHaveLength(1);
    expect(resultado.produtosComPreco[0].precoUnitario).toBe(1500);
  });
});

describe("getPrecoUnitarioOuErro", () => {
  it("should return product not found error when product does not exist", async () => {
    const resultado = await produtosService.getPrecoUnitarioOuErro({
      idProduto: "inexistente-unitario",
      quantidade: 1
    });

    expect(resultado).toHaveProperty("error");
    expect(resultado.error.statusCode).toBe(400);
    expect(resultado.error.message).toBe(constant.PRODUCT_NOT_FOUND);
  });

  it("should return insufficient stock error when quantity exceeds stock", async () => {
    const criado = await produtosService.criarProduto({
      nome: "Webcam HD",
      preco: "200",
      quantidade: "3"
    });

    const resultado = await produtosService.getPrecoUnitarioOuErro({
      idProduto: criado._id,
      quantidade: 4
    });

    expect(resultado).toHaveProperty("error");
    expect(resultado.error.statusCode).toBe(400);
    expect(resultado.error.message).toBe(constant.INSUFFICIENT_STOCK);
    expect(resultado.error.item.quantidadeEstoque).toBe(3);
  });

  it("should return unit price when product exists and has sufficient stock", async () => {
    const criado = await produtosService.criarProduto({
      nome: "Headset Gamer",
      preco: "400",
      quantidade: "10"
    });

    const resultado = await produtosService.getPrecoUnitarioOuErro({
      idProduto: criado._id,
      quantidade: 2
    });

    expect(resultado).toEqual({ precoUnitario: 400 });
  });
});

describe("deleteById", () => {
  it("should delete product by id", async () => {
    const removidos = await produtosService.deleteById("id-inexistente-remover");
    expect(removidos).toBe(0);
  });
});

describe("updateById", () => {
  it("should update product by id with upsert returning updated docs", async () => {
    const atualizado = await produtosService.updateById("id-inexistente-atualizar", {
      $set: { nome: "Atualizado" }
    });
    expect(atualizado).toBeDefined();
    expect(atualizado.nome).toBe("Atualizado");
  });
});

describe("createOrUpdateById", () => {
  it("should format values and update product by id", async () => {
    const resultado = await produtosService.createOrUpdateById("id-put-test", {
      nome: "  Produto PUT  ",
      preco: "100",
      quantidade: "5"
    });
    expect(resultado).toBeDefined();
    expect(resultado.nome).toBe("Produto PUT");
    expect(resultado.preco).toBe(100);
    expect(resultado.quantidade).toBe(5);
    expect(resultado).toHaveProperty("alteradoEm");
  });
});
```
