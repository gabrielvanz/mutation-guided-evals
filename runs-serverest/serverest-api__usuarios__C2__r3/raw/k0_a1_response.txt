```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("usuarios API", () => {
  it("should get all usuarios", async () => {
    const uniqueEmail = "user_all@test.com";
    await request(app)
      .post("/usuarios")
      .send({
        nome: "Test User",
        email: uniqueEmail,
        password: "password123",
        administrador: "true"
      });

    const res = await request(app).get("/usuarios");
    expect(res.status).toBe(200);
    expect(res.body).toHaveProperty("quantidade");
    expect(res.body).toHaveProperty("usuarios");
    expect(Array.isArray(res.body.usuarios)).toBe(true);
    expect(res.body.quantidade).toBeGreaterThan(0);
  });

  it("should get one usuario by id", async () => {
    const uniqueEmail = "user_one@test.com";
    const postRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Single User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    const userId = postRes.body._id;

    const res = await request(app).get(`/usuarios/${userId}`);
    expect(res.status).toBe(200);
    expect(res.body).toHaveProperty("_id", userId);
    expect(res.body).toHaveProperty("email", uniqueEmail);
  });

  it("should return 400 when usuario id is not found for getOne", async () => {
    const res = await request(app).get("/usuarios/1234567890abcdef");
    expect(res.status).toBe(400);
    expect(res.body).toEqual({ message: "Usuário não encontrado" });
  });

  it("should create a new usuario", async () => {
    const uniqueEmail = "new_user@test.com";
    const res = await request(app)
      .post("/usuarios")
      .send({
        nome: "New User",
        email: uniqueEmail,
        password: "password123",
        administrador: "true"
      });

    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
    expect(res.body).toHaveProperty("_id");
  });

  it("should return 400 when creating a usuario with an existing email", async () => {
    const uniqueEmail = "duplicate@test.com";
    const userData = {
      nome: "User One",
      email: uniqueEmail,
      password: "password123",
      administrador: "true"
    };

    await request(app).post("/usuarios").send(userData);

    const res = await request(app).post("/usuarios").send({
      nome: "User Two",
      email: uniqueEmail,
      password: "password456",
      administrador: "false"
    });

    expect(res.status).toBe(400);
    expect(res.body).toEqual({ message: "Este email já está sendo usado" });
  });

  it("should delete an existing usuario", async () => {
    const uniqueEmail = "delete_user@test.com";
    const postRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Delete User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    const userId = postRes.body._id;

    const res = await request(app).delete(`/usuarios/${userId}`);
    expect(res.status).toBe(200);
    expect(res.body).toEqual({ message: "Registro excluído com sucesso" });
  });

  it("should return delete none when deleting a non-existent usuario", async () => {
    const res = await request(app).delete("/usuarios/nonexistentid12");
    expect(res.status).toBe(200);
    expect(res.body).toEqual({ message: "Nenhum registro excluído" });
  });

  it("should update an existing usuario using put", async () => {
    const uniqueEmail = "put_user@test.com";
    const postRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Put User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    const userId = postRes.body._id;
    const updatedEmail = "updated@test.com";

    const res = await request(app)
      .put(`/usuarios/${userId}`)
      .send({
        nome: "Updated Name",
        email: updatedEmail,
        password: "newpassword123",
        administrador: "true"
      });

    expect(res.status).toBe(200);
    expect(res.body).toEqual({ message: "Registro alterado com sucesso" });
  });

  it("should create a usuario via put if id does not exist (upsert)", async () => {
    const uniqueEmail = "upsert_user@test.com";
    const nonExistentId = "abcdef1234567891";

    const res = await request(app)
      .put(`/usuarios/${nonExistentId}`)
      .send({
        nome: "Upserted User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
    expect(res.body).toHaveProperty("_id", nonExistentId);
  });
});
```
