```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on a protected route", async () => {
      const res = await request(app).get("/carrinhos");
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token format is invalid", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "InvalidFormat token123");
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is invalid or malformed Bearer token", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "Bearer not.a.valid.token");
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 403 when a non-admin user tries to access admin-only routes", async () => {
      const email = "user_auth_test@email.com";
      const password = "password123";

      const createRes = await request(app)
        .post("/usuarios")
        .send({
          nome: "Normal User",
          email,
          password,
          administrador: "false"
        });
      expect(createRes.status).toBe(201);

      const loginRes = await request(app)
        .post("/login")
        .send({
          email,
          password
        });
      expect(loginRes.status).toBe(200);
      const token = loginRes.body.authorization;

      const adminRouteRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Teste",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });
      expect(adminRouteRes.status).toBe(403);
      expect(adminRouteRes.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 when malformed JSON body is sent", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("Content-Type", "application/json")
        .send('{"nome": "Invalid JSON');

      expect(res.status).toBe(400);
      expect(res.body).toHaveProperty("message");
    });

    it("should handle invalid URI encoding gracefully with a 400 response", async () => {
      const res = await request(app).get("/usuarios/%C0%C0");
      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });
  });
});
```
