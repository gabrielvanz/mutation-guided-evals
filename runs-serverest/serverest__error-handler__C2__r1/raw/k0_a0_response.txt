```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent by destroying socket when not timedout", () => {
    const error = new Error("Test error");
    error.stack = "stack";
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: false,
      socket: {
        destroy: () => {}
      }
    };
    let destroyed = false;
    req.socket.destroy = () => {
      destroyed = true;
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});

    expect(destroyed).toBe(true);
  });

  it("should handle headersSent without destroying socket when writableEnded is true", () => {
    const error = new Error("Test error");
    error.stack = "stack";
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: true,
      socket: {
        destroy: () => {}
      }
    };
    let destroyed = false;
    req.socket.destroy = () => {
      destroyed = true;
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    errorHandler(error, req, res, () => {});

    expect(destroyed).toBe(false);
  });

  it("should handle ValidationError", () => {
    const error = new Error("Validation error");
    error.name = "ValidationError";
    const req = {};
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (b) => {
            jsonBody = b;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonBody).toEqual({});
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = {};
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (b) => {
            jsonBody = b;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = new Error("Parse failed");
    error.type = "entity.parse.failed";
    const req = {};
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (b) => {
            jsonBody = b;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const error = new Error("Too large");
    error.type = "entity.too.large";
    const req = {};
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (b) => {
            jsonBody = b;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle parameters.too.many", () => {
    const error = new Error("Too many parameters");
    error.type = "parameters.too.many";
    const req = {};
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (b) => {
            jsonBody = b;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = new Error("Unsupported charset");
    error.type = "charset.unsupported";
    error.charset = "utf-16";
    const req = {};
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (b) => {
            jsonBody = b;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = new Error("Unsupported encoding");
    error.type = "encoding.unsupported";
    error.encoding = "deflate-x";
    const req = {};
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (b) => {
            jsonBody = b;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate-x" });
  });

  it("should handle client error with Z_ code", () => {
    const error = new Error("Zlib data error");
    error.code = "Z_DATA_ERROR";
    error.expose = true;
    error.statusCode = 400;
    const req = {};
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (b) => {
            jsonBody = b;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = new Error("Timeout");
    error.code = "ETIMEDOUT";
    const req = {};
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (b) => {
            jsonBody = b;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(408);
    expect(jsonBody).toHaveProperty("message");
  });

  it("should handle general client error", () => {
    const error = new Error("Bad request");
    error.expose = true;
    error.statusCode = 422;
    error.type = "custom.type";
    const req = {};
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (b) => {
            jsonBody = b;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.type"
    });
  });

  it("should handle fallback internal server error with error type", () => {
    const error = new Error("Server crash");
    error.type = "server.crash";
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s) => {
        statusCalled = s;
        return {
          json: (b) => {
            jsonBody = b;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(500);
    expect(jsonBody).toHaveProperty("message");
    expect(jsonBody).toHaveProperty("error", "server.crash");
    expect(jsonBody).toHaveProperty("version");
  });
});
```
