import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headers already sent with writableEnded false", () => {
    const error = new Error("Test error");
    const socket = { destroy: () => {} };
    let socketDestroyCalled = false;
    socket.destroy = () => {
      socketDestroyCalled = true;
    };
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: false,
      socket
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});

    expect(socketDestroyCalled).toBe(true);
  });

  it("should handle headers already sent with writableEnded true", () => {
    const error = new Error("Test error");
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: true,
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    let statusCalled = false;
    res.status = () => {
      statusCalled = true;
      return res;
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(false);
  });

  it("should handle ValidationError", () => {
    const error = { name: "ValidationError", details: [] };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toBeDefined();
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle parameters.too.many", () => {
    const error = { type: "parameters.too.many" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle compressed body error (Z_DATA_ERROR)", () => {
    const error = { expose: true, statusCode: 400, code: "Z_DATA_ERROR", message: "data error" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error", () => {
    const error = { code: "ETIMEDOUT" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(408);
    expect(jsonBody).toEqual({ message: "Requisição expirou (timeout)" });
  });

  it("should handle generic client error (expose true, statusCode 422)", () => {
    const error = { expose: true, statusCode: 422, type: "custom.error" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.error"
    });
  });

  it("should handle internal server error with type", () => {
    const error = { type: "server.error", stack: "stack-trace" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(500);
    expect(jsonBody.message).toBe("Erro interno no servidor");
    expect(jsonBody.error).toBe("server.error");
    expect(jsonBody.version).toBeDefined();
  });
});
