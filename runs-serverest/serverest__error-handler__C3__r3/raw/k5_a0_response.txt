import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent and destroy socket if not writableEnded", () => {
    const error = new Error("Test error");
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: false,
      socket: {
        destroy: () => {}
      }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const result = errorHandler(error, req, res, () => {});
    expect(result).toBeUndefined();
  });

  it("should handle headersSent and not destroy socket if writableEnded", () => {
    const error = new Error("Test error");
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: true,
      socket: {
        destroy: () => {
          throw new Error("Should not be called");
        }
      }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    const result = errorHandler(error, req, res, () => {});
    expect(result).toBeUndefined();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: [{ message: "field is required" }]
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonCalledWith).toEqual({ message: "field is required" });
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = { body: {} };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonCalledWith).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonCalledWith).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large" };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(413);
    expect(jsonCalledWith).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(415);
    expect(jsonCalledWith).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(415);
    expect(jsonCalledWith).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle Z_ compressed body error", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "invalid block type"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(400);
    expect(jsonCalledWith).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error", () => {
    const error = { code: "ETIMEDOUT" };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(408);
    expect(jsonCalledWith).toHaveProperty("message");
  });

  it("should handle generic client error", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "some.client.error",
      message: "Bad request"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(422);
    expect(jsonCalledWith).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.client.error"
    });
  });

  it("should handle internal error with type", () => {
    const error = {
      type: "database.error",
      message: "DB connection failed",
      stack: "stacktrace"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonCalledWith = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonCalledWith = data;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCalled).toBe(500);
    expect(jsonCalledWith).toHaveProperty("message");
    expect(jsonCalledWith).toHaveProperty("error", "database.error");
    expect(jsonCalledWith).toHaveProperty("version");
  });
});
