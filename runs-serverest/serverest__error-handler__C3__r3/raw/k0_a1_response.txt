import { describe, it, expect } from "vitest";

describe("errorHandler", () => {

    const error = new Error("Test error");
    error.stack = "Error: Test error";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };
    const next = () => {};

    const result = errorHandler(error, req, res, next);
    expect(result).toBeUndefined();
  });

  it("should handle headers already sent without timeout and writable ended", () => {
    const error = new Error("Test error");
    error.stack = "Error: Test error";
    const req = {
      timedout: false,

      originalUrl: "/test",
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,

      status: function (code) {
        this.statusCode = code;
        return this;
      },
      json: function (body) {
        this.body = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
