import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headers already sent with timedout request", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: {
        destroy: () => {}
      }
    };
    const res = {
      headersSent: true,
      writableEnded: false,
      status: (code: number) => ({ json: (body: any) => ({ code, body }) })
    };

    const result = errorHandler(error, req as any, res as any, () => {});
    expect(result).toBeUndefined();
  });

  it("should handle headers already sent without timedout request", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    let destroyed = false;
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/test",
      socket: {
        destroy: () => { destroyed = true; }
      }
    };
    const res = {
      headersSent: true,
      writableEnded: false,
      status: (code: number) => ({ json: (body: any) => ({ code, body }) })
    };

    errorHandler(error, req as any, res as any, () => {});
    expect(destroyed).toBe(true);
  });

  it("should handle ValidationError", () => {
    const error = { name: "ValidationError", details: [{ message: "error" }] };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {} as any, res as any, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toBeDefined();
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {} as any, res as any, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {} as any, res as any, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody.message).toContain("Adicione aspas em todos os valores");
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large" };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {} as any, res as any, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle parameters.too.many", () => {
    const error = { type: "parameters.too.many" };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {} as any, res as any, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {} as any, res as any, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate-x" };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {} as any, res as any, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate-x" });
  });

  it("should handle client error with Z_ prefix code", () => {
    const error = { expose: true, statusCode: 400, code: "Z_DATA_ERROR", message: "zlib error" };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {} as any, res as any, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody.message).toContain("Corpo comprimido inválido");
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = { code: "ETIMEDOUT" };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, {} as any, res as any, () => {});
    expect(statusCode).toBe(408);
    expect(jsonBody).toBeDefined();
  });

  it("should handle generic client error", () => {
    const error = { expose: true, statusCode: 422, type: "some.type" };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, { body: {} } as any, res as any, () => {});
    expect(statusCode).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.type"
    });
  });

  it("should handle server error with error type", () => {
    const error = { type: "server.error", stack: "stack" };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, { body: {} } as any, res as any, () => {});
    expect(statusCode).toBe(500);
    expect(jsonBody.error).toBe("server.error");
  });
});
