import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headers already sent with timedout request", () => {
    const error = new Error("Test error");
    const req = { timedout: true, method: "GET", originalUrl: "/test", socket: { destroy: () => {} } };
    const res = { headersSent: true, writableEnded: false };
    const next = () => {};

    const result = errorHandler(error, req, res, next);
    expect(result).toBeUndefined();
  });

  it("should handle headers already sent without timeout and destroy socket if not writable ended", () => {
    const error = new Error("Test error");
    let destroyed = false;
    const req = { timedout: false, method: "GET", originalUrl: "/test", socket: { destroy: () => { destroyed = true; } } };
    const res = { headersSent: true, writableEnded: false };
    const next = () => {};

    errorHandler(error, req, res, next);
    expect(destroyed).toBe(true);
  });

  it("should handle ValidationError", () => {
    const error = { name: "ValidationError", details: [] };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toBeDefined();
  });

  it("should handle URIError", () => {
    const error = new URIError("URI malformed");
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed error", () => {
    const error = { type: "entity.parse.failed" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large error", () => {
    const error = { type: "entity.too.large" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle parameters.too.many error", () => {
    const error = { type: "parameters.too.many" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported error", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported error", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle Z_ compressed body client error", () => {
    const error = { expose: true, statusCode: 400, code: "Z_DATA_ERROR", message: "data error" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error", () => {
    const error = { code: "ETIMEDOUT" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(408);
    expect(jsonBody).toBeDefined();
  });

  it("should handle general client error", () => {
    const error = { expose: true, statusCode: 422, type: "some.type" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.type"
    });
  });

  it("should handle server error with error type", () => {
    const error = { type: "server.error", stack: "stack-trace" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCode = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(500);
    expect(jsonBody.error).toBe("server.error");
    expect(jsonBody.version).toBeDefined();
  });
});
