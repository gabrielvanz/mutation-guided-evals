```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on a protected route", async () => {
      const res = await request(app).get("/carrinhos");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token has bad format", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "InvalidFormat token123");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is invalid or malformed", async () => {
      const res = await request(app)
        .get("/carrinhos")
        .set("authorization", "Bearer invalid.jwt.token");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 403 when user is not an administrator on admin-only route", async () => {
      const email = `regular_${Date.now()}@test.com`;
      const userRes = await request(app)
        .post("/usuarios")
        .send({
          nome: "Regular User",
          email,
          password: "testpassword",
          administrador: "false"
        });

      expect(userRes.status).toBe(201);

      const loginRes = await request(app)
        .post("/login")
        .send({
          email,
          password: "testpassword"
        });

      expect(loginRes.status).toBe(200);
      const token = loginRes.body.authorization;

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: `Produto_${Date.now()}`,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(403);
      expect(res.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 when malformed JSON is sent in request body", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("Content-Type", "application/json")
        .send('{"nome": "bad json"');

      expect(res.status).toBe(400);
      expect(res.body.message).toContain("Adicione aspas em todos os valores");
    });

    it("should return 400 when URI decode error occurs", async () => {
      const res = await request(app).get("/usuarios?_id=%E0%A3%A3");

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });

    it("should return 400 for schema validation errors on invalid user payload", async () => {
      const res = await request(app)
        .post("/usuarios")
        .send({
          email: "not-an-email",
          password: "123"
        });

      expect(res.status).toBe(400);
      expect(res.body).toHaveProperty("nome");
    });
  });
});
```
