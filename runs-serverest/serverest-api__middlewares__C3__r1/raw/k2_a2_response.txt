```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("middlewares API", () => {
  describe("authentication-middleware & error-handler", () => {
    it("should reject request to protected route without authorization header", async () => {
      const response = await request(app).get("/carrinhos");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should reject request with malformed authorization header", async () => {
      const response = await request(app)
        .get("/carrinhos")
        .set("authorization", "InvalidTokenFormat");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should reject request with invalid token string", async () => {
      const response = await request(app)
        .get("/carrinhos")
        .set("authorization", "Bearer invalid.jwt.token");

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should reject non-admin user trying to access admin-only routes", async () => {
      let counter = 1;
      const getUnique = () => `test_${counter++}@qa.com`;

      const regularUser = {
        nome: "Test User",
        email: getUnique(),
        password: "teste",
        administrador: "false"
      };

      const createRes = await request(app).post("/usuarios").send(regularUser);
      expect(createRes.status).toBe(201);

      const loginRes = await request(app).post("/login").send({
        email: regularUser.email,
        password: regularUser.password
      });
      expect(loginRes.status).toBe(200);
      const token = loginRes.body.authorization;

      const response = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Test Product",
          preco: 10,
          descricao: "Desc",
          quantidade: 10
        });

      expect(response.status).toBe(403);
      expect(response.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });

    it("should handle malformed JSON body via error-handler middleware", async () => {
      const response = await request(app)
        .post("/usuarios")
        .set("Content-Type", "application/json")
        .send("{ malformed json ");

      expect(response.status).toBe(400);
      expect(response.body).toHaveProperty("message");
      expect(response.body.message).toContain("Adicione aspas em todos os valores");
    });
  });
});
```
