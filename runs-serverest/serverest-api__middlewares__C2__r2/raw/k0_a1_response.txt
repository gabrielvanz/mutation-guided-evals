```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("middlewares API", () => {
  describe("authentication-middleware & error-handler", () => {
    it("should reject request to protected route without authorization header", async () => {
      const res = await request(app).get("/usuarios");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido, expirado ou usuário não encontrado. Refaça o login."
      });
    });

    it("should reject request with malformed authorization header", async () => {
      const res = await request(app)
        .get("/usuarios")
        .set("authorization", "InvalidTokenFormat");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido, expirado ou usuário não encontrado. Refaça o login."
      });
    });

    it("should reject request with invalid token", async () => {
      const res = await request(app)
        .get("/usuarios")
        .set("authorization", "Bearer invalid.jwt.token");

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido, expirado ou usuário não encontrado. Refaça o login."
      });
    });

    it("should reject non-admin user trying to access admin-only routes", async () => {
      const userPayload = {
        nome: "Comum Teste",
        email: "comum_test_middleware@email.com",
        password: "teste",
        administrador: "false"
      };

      const createRes = await request(app).post("/usuarios").send(userPayload);
      expect(createRes.status).toBe(201);

      const loginRes = await request(app).post("/login").send({
        email: userPayload.email,
        password: userPayload.password
      });

      expect(loginRes.status).toBe(200);
      const token = loginRes.body.authorization;

      const res = await request(app)
        .get("/usuarios")
        .set("authorization", token);

      expect(res.status).toBe(403);
      expect(res.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });

    it("should handle URIError (invalid URL encoding) with 400", async () => {
      const res = await request(app).get("/usuarios/%E0%A3%A");

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });

    it("should handle malformed JSON body (entity.parse.failed) with 400", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send("{invalid json");

      expect(res.status).toBe(400);
      expect(res.body).toHaveProperty("message");
      expect(res.body.message).toContain("Adicione aspas em todos os valores");
    });
  });
});
```
