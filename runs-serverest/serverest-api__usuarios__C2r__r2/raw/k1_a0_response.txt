```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Usuarios API", () => {
  it("should get all usuarios successfully", async () => {
    const res = await request(app).get("/usuarios");
    expect(res.status).toBe(200);
    expect(res.body).toHaveProperty("quantidade");
    expect(res.body).toHaveProperty("usuarios");
    expect(Array.isArray(res.body.usuarios)).toBe(true);
  });

  it("should filter usuarios by query parameters", async () => {
    const uniqueEmail = `user_${Date.now()}@test.com`;
    const uniqueName = `User ${Date.now()}`;

    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: uniqueName,
        email: uniqueEmail,
        password: "password123",
        administrador: "true"
      });

    expect(createRes.status).toBe(201);

    const res = await request(app).get(`/usuarios?email=${uniqueEmail}`);
    expect(res.status).toBe(200);
    expect(res.body.quantidade).toBe(1);
    expect(res.body.usuarios[0].email).toBe(uniqueEmail);
  });

  it("should return 400 when getting a usuario with invalid id format", async () => {
    const res = await request(app).get("/usuarios/invalid-id");
    expect(res.status).toBe(400);
  });

  it("should return 400 when usuario is not found", async () => {
    const res = await request(app).get("/usuarios/1234567890abcdef");
    expect(res.status).toBe(400);
    expect(res.body).toEqual({ message: "Usuário não encontrado" });
  });

  it("should create a new usuario successfully", async () => {
    const uniqueEmail = `create_${Date.now()}@test.com`;
    const res = await request(app)
      .post("/usuarios")
      .send({
        nome: "Test User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
    expect(res.body).toHaveProperty("_id");
  });

  it("should return 400 when creating a usuario with an existing email", async () => {
    const uniqueEmail = `duplicate_${Date.now()}@test.com`;
    const payload = {
      nome: "Test User",
      email: uniqueEmail,
      password: "password123",
      administrador: "false"
    };

    const firstRes = await request(app).post("/usuarios").send(payload);
    expect(firstRes.status).toBe(201);

    const secondRes = await request(app).post("/usuarios").send(payload);
    expect(secondRes.status).toBe(400);
    expect(secondRes.body).toEqual({ message: "Este email já está sendo usado" });
  });

  it("should return 400 when required fields are missing on post", async () => {
    const res = await request(app)
      .post("/usuarios")
      .send({
        nome: "Incomplete User"
      });

    expect(res.status).toBe(400);
  });

  it("should delete an existing usuario successfully", async () => {
    const uniqueEmail = `delete_${Date.now()}@test.com`;
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Delete User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    expect(createRes.status).toBe(201);
    const userId = createRes.body._id;

    const deleteRes = await request(app).delete(`/usuarios/${userId}`);
    expect(deleteRes.status).toBe(200);
    expect(deleteRes.body).toEqual({ message: "Registro excluído com sucesso" });
  });

  it("should return DELETE_NONE when deleting a non-existent usuario id", async () => {
    const deleteRes = await request(app).delete("/usuarios/nonexistentid123");
    expect(deleteRes.status).toBe(200);
    expect(deleteRes.body).toEqual({ message: "Nenhum registro excluído" });
  });

  it("should not allow deleting a usuario that has an active cart", async () => {
    const uniqueEmail = `cart_user_${Date.now()}@test.com`;
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Cart User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    expect(createRes.status).toBe(201);
    const userId = createRes.body._id;

    // Create a product first so we can add it to a cart
    const prodRes = await request(app)
      .post("/produtos")
      .set("authorization", "") // If needed, but let's test directly via carrinhos if endpoint allows or mock data
      .send({
        nome: `Product ${Date.now()}`,
        preco: 100,
        descricao: "Test product",
        quantidade: 10
      });

    // Login or obtain token if required for carrinhos, or just insert cart directly/via endpoint
    // Let's use the login endpoint to get a token if required, or test via POST /carrinhos
    const loginRes = await request(app).post("/login").send({
      email: uniqueEmail,
      password: "password123"
    });

    if (loginRes.status === 200 && loginRes.body.authorization) {
      const token = loginRes.body.authorization;
      if (prodRes.status === 201) {
        const prodId = prodRes.body._id;
        await request(app)
          .post("/carrinhos")
          .set("authorization", token)
          .send({
            produtos: [{ idProduto: prodId, quantidade: 1 }]
          });

        const deleteRes = await request(app).delete(`/usuarios/${userId}`);
        expect(deleteRes.status).toBe(400);
        expect(deleteRes.body).toHaveProperty("message", "Não é permitido excluir usuário com carrinho cadastrado");
      }
    }
  });

  it("should update an existing usuario successfully (PUT)", async () => {
    const uniqueEmail = `put_${Date.now()}@test.com`;
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Put User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    expect(createRes.status).toBe(201);
    const userId = createRes.body._id;

    const updateRes = await request(app)
      .put(`/usuarios/${userId}`)
      .send({
        nome: "Updated Put User",
        email: uniqueEmail,
        password: "newpassword",
        administrador: "true"
      });

    expect(updateRes.status).toBe(200);
    expect(updateRes.body).toEqual({ message: "Registro alterado com sucesso" });
  });

  it("should upsert/create via PUT if usuario does not exist", async () => {
    const uniqueEmail = `upsert_${Date.now()}@test.com`;
    const nonExistentId = "abcdef0123456789";

    const updateRes = await request(app)
      .put(`/usuarios/${nonExistentId}`)
      .send({
        nome: "Upserted User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    expect(updateRes.status).toBe(201);
    expect(updateRes.body).toHaveProperty("message", "Cadastro realizado com sucesso");
    expect(updateRes.body).toHaveProperty("_id", nonExistentId);
  });
});
```
