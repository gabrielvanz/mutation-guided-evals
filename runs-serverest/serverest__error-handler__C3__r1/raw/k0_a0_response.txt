```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent and destroy socket if not writableEnded", () => {
    const error = new Error("Test error");
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: false,
      socket: {
        destroy: () => {}
      }
    };
    let destroyed = false;
    req.socket.destroy = () => {
      destroyed = true;
    };

    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});

    expect(destroyed).toBe(true);
  });

  it("should handle headersSent without destroying socket if writableEnded is true", () => {
    const error = new Error("Test error");
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: true,
      socket: {
        destroy: () => {}
      }
    };
    let destroyed = false;
    req.socket.destroy = () => {
      destroyed = true;
    };

    const res = {
      headersSent: true,
      writableEnded: true
    };

    errorHandler(error, req, res, () => {});

    expect(destroyed).toBe(false);
  });

  it("should return 400 for ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: [{ message: "Campo obrigatório" }]
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonResult = null;

    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonResult = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonResult).toBeDefined();
  });

  it("should return 400 for URIError", () => {
    const error = new URIError("Malformed URI");
    const req = { body: {} };
    let statusCalled = 0;
    let jsonResult = null;

    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonResult = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonResult).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should return 400 for entity.parse.failed", () => {
    const error = {
      type: "entity.parse.failed"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonResult = null;

    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonResult = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonResult).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should return 413 for entity.too.large", () => {
    const error = {
      type: "entity.too.large"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonResult = null;

    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonResult = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(413);
    expect(jsonResult).toEqual({ message: "Payload too large" });
  });

  it("should return 413 for parameters.too.many", () => {
    const error = {
      type: "parameters.too.many"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonResult = null;

    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonResult = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(413);
    expect(jsonResult).toEqual({ message: "Payload too large" });
  });

  it("should return 415 for charset.unsupported", () => {
    const error = {
      type: "charset.unsupported",
      charset: "utf-16"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonResult = null;

    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonResult = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(415);
    expect(jsonResult).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should return 415 for encoding.unsupported", () => {
    const error = {
      type: "encoding.unsupported",
      encoding: "br"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonResult = null;

    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonResult = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(415);
    expect(jsonResult).toEqual({ message: "Unsupported content encoding", encoding: "br" });
  });

  it("should return 400 for client error with Z_ code", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "data error"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonResult = null;

    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonResult = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonResult).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should return 408 for ETIMEDOUT code", () => {
    const error = {
      code: "ETIMEDOUT"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonResult = null;

    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonResult = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(408);
    expect(jsonResult).toBeDefined();
  });

  it("should return general client error when expose is true and status 4xx", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "custom.error",
      message: "Invalid"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonResult = null;

    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonResult = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(422);
    expect(jsonResult).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.error"
    });
  });

  it("should return 500 with error type when error has type and no other rules match", () => {
    const error = {
      type: "database.error",
      message: "db down",
      stack: "stack"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonResult = null;

    const res = {
      headersSent: false,
      status: (code) => {
        statusCalled = code;
        return {
          json: (data) => {
            jsonResult = data;
            return data;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(500);
    expect(jsonResult).toHaveProperty("message");
    expect(jsonResult).toHaveProperty("error", "database.error");
    expect(jsonResult).toHaveProperty("version");
  });
});
```
