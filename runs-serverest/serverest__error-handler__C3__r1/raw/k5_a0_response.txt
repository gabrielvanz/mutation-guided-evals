```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent by destroying socket when not writableEnded and req.timedout is false", () => {
    const error = new Error("Test error");
    error.stack = "stack";
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: false,
      socket: {
        destroy: () => {}
      }
    };
    let destroyed = false;
    req.socket.destroy = () => {
      destroyed = true;
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});

    expect(destroyed).toBe(true);
  });

  it("should handle headersSent without destroying socket when res.writableEnded is true", () => {
    const error = new Error("Test error");
    error.stack = "stack";
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: true,
      socket: {
        destroy: () => {
          throw new Error("Should not be called");
        }
      }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    expect(() => errorHandler(error, req, res, () => {})).not.toThrow();
  });

  it("should return 400 with schema validation message when error name is ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: [{ message: "Field is required", path: ["field"] }]
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonBody).toBeDefined();
  });

  it("should return 400 when error name is URIError", () => {
    const error = new Error("Malformed URI");
    error.name = "URIError";
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should return 400 when error type is entity.parse.failed", () => {
    const error = {
      type: "entity.parse.failed"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should return 413 when error type is entity.too.large", () => {
    const error = {
      type: "entity.too.large"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should return 413 when error type is parameters.too.many", () => {
    const error = {
      type: "parameters.too.many"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should return 415 when error type is charset.unsupported", () => {
    const error = {
      type: "charset.unsupported",
      charset: "utf-16"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should return 415 when error type is encoding.unsupported", () => {
    const error = {
      type: "encoding.unsupported",
      encoding: "deflate"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should return 400 when client error with Z_ prefix code occurs", () => {
    const error = new Error("Zlib error");
    (error as any).expose = true;
    (error as any).statusCode = 400;
    (error as any).code = "Z_DATA_ERROR";
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should return 408 when error code is ETIMEDOUT", () => {
    const error = {
      code: "ETIMEDOUT"
    };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(408);
    expect(jsonBody).toBeDefined();
  });

  it("should return general client error when ehErroDoCliente returns true", () => {
    const error = new Error("Bad request");
    (error as any).expose = true;
    (error as any).statusCode = 422;
    (error as any).type = "some.type";
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.type"
    });
  });

  it("should return 500 with INTERNAL_ERROR when error has a type", () => {
    const error = new Error("Internal server issue");
    (error as any).type = "database.error";
    error.stack = "stack trace";
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody: any = null;
    const res = {
      headersSent: false,
      status: (code: number) => {
        statusCalled = code;
        return {
          json: (body: any) => {
            jsonBody = body;
            return res;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(500);
    expect(jsonBody.error).toBe("database.error");
  });
});
```
