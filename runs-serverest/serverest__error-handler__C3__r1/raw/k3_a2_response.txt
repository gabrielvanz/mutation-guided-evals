import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent when response is not writableEnded", () => {
    const error = { message: "Test error", stack: "stack", name: "Error" };
    const req = { method: "GET", originalUrl: "/test", timedout: false, socket: { destroy: () => {} } };
    let destroyed = false;
    req.socket.destroy = () => {
      destroyed = true;
    };
    const res = { headersSent: true, writableEnded: false };

    errorHandler(error, req, res, () => {});

    expect(destroyed).toBe(true);
  });

  it("should handle headersSent when response is writableEnded", () => {
    const error = { message: "Test error", stack: "stack", name: "Error" };
    const req = { method: "GET", originalUrl: "/test", timedout: true, socket: { destroy: () => {} } };
    let destroyed = false;
    req.socket.destroy = () => {
      destroyed = true;
    };
    const res = { headersSent: true, writableEnded: true };

    errorHandler(error, req, res, () => {});

    expect(destroyed).toBe(false);
  });

  it("should handle ValidationError", () => {
    const error = { name: "ValidationError", details: [] };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toBeDefined();
  });

  it("should handle URIError", () => {
    const error = { name: "URIError", message: "URI malformed" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle parameters.too.many", () => {
    const error = { type: "parameters.too.many" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,

        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle Zlib client error", () => {
    const error = { expose: true, statusCode: 400, code: "Z_DATA_ERROR", message: "invalid stored block lengths" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."

  it("should handle ETIMEDOUT error", () => {
    const error = { code: "ETIMEDOUT" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;

      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(408);
    expect(jsonBody).toBeDefined();
  });

  it("should handle general client error", () => {
    const error = { expose: true, statusCode: 422, type: "some.client.error" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.client.error"
    });
  });

  it("should handle server error with type", () => {
    const error = { type: "database.error", stack: "stack" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status(code) {
        statusCode = code;
        return this;
      },
      json(body) {
        jsonBody = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCode).toBe(500);
    expect(jsonBody).toMatchObject({
      error: "database.error"
    });
  });
});
