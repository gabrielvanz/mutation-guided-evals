```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Produtos API", () => {
  let adminToken: string;

  const getAdminToken = async () => {
    if (adminToken) return adminToken;
    const email = `admin_${Date.now()}_${Math.random().toString(36).substring(7)}@qa.com`;
    await request(app)
      .post("/usuarios")
      .send({
        nome: "Admin Test",
        email,
        password: "teste",
        administrador: "true"
      });
    const loginRes = await request(app)
      .post("/login")
      .send({
        email,
        password: "teste"
      });
    adminToken = loginRes.body.authorization;
    return adminToken;
  };

  describe("GET /produtos", () => {
    it("should return all products", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto List ${Date.now()}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const res = await request(app).get("/produtos");
      expect(res.status).toBe(200);
      expect(res.body).toHaveProperty("quantidade");
      expect(res.body).toHaveProperty("produtos");
      expect(Array.isArray(res.body.produtos)).toBe(true);
      expect(res.body.quantidade).toBeGreaterThan(0);
    });

    it("should filter products by name query", async () => {
      const token = await getAdminToken();
      const uniqueName = `UniqueTarget_${Date.now()}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 50,
          descricao: "Desc",
          quantidade: 5
        });

      const res = await request(app).get(`/produtos?nome=${uniqueName}`);
      expect(res.status).toBe(200);
      expect(res.body.quantidade).toBe(1);
      expect(res.body.produtos[0].nome).toBe(uniqueName);
    });
  });

  describe("GET /produtos/:id", () => {
    it("should return a product by valid id", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto One ${Date.now()}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 200,
          descricao: "Desc",
          quantidade: 15
        });

      const productId = postRes.body._id;

      const res = await request(app).get(`/produtos/${productId}`);
      expect(res.status).toBe(200);
      expect(res.body._id).toBe(productId);
      expect(res.body.nome).toBe(uniqueName);
    });

    it("should return error when product does not exist", async () => {
      const res = await request(app).get("/produtos/abcdef0123456789");
      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Produto não encontrado" });
    });

    it("should fail schema validation for invalid id format", async () => {
      const res = await request(app).get("/produtos/invalid-id");
      expect(res.status).toBe(400);
    });
  });

  describe("POST /produtos", () => {
    it("should create a product when user is admin and data is valid", async () => {
      const token = await getAdminToken();
      const uniqueName = `New Produto ${Date.now()}`;
      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 450,
          descricao: "Mouse gamer",
          quantidade: 100,
          imagem: "http://imagem.com/mouse.jpg"
        });

      expect(res.status).toBe(201);
      expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
      expect(res.body).toHaveProperty("_id");
    });

    it("should not create a product with duplicate name", async () => {
      const token = await getAdminToken();
      const uniqueName = `Duplicate Produto ${Date.now()}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 150,
          descricao: "Other Desc",
          quantidade: 5
        });

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });

    it("should fail when token is missing", async () => {
      const res = await request(app)
        .post("/produtos")
        .send({
          nome: "Sem Token",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(401);
    });

    it("should fail schema validation when required fields are missing", async () => {
      const token = await getAdminToken();
      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Incompleto"
        });

      expect(res.status).toBe(400);
    });
  });

  describe("PUT /produtos/:id", () => {
    it("should update an existing product successfully", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto To Update ${Date.now()}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const productId = postRes.body._id;
      const updatedName = `Updated Produto ${Date.now()}`;

      const res = await request(app)
        .put(`/produtos/${productId}`)
        .set("authorization", token)
        .send({
          nome: updatedName,
          preco: 120,
          descricao: "New Desc",
          quantidade: 20
        });

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro alterado com sucesso" });
    });

    it("should create product via put if id does not exist", async () => {
      const token = await getAdminToken();
      const uniqueName = `Put Upsert ${Date.now()}`;
      const randomId = "1234567890abcdef";

      const res = await request(app)
        .put(`/produtos/${randomId}`)
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 300,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(201);
      expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
      expect(res.body).toHaveProperty("_id");
    });

    it("should fail to update if name already used by another product", async () => {
      const token = await getAdminToken();
      const name1 = `Prod One ${Date.now()}`;
      const name2 = `Prod Two ${Date.now()}`;

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({ nome: name1, preco: 10, descricao: "D", quantidade: 1 });

      const postRes2 = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({ nome: name2, preco: 10, descricao: "D", quantidade: 1 });

      const productId2 = postRes2.body._id;

      const res = await request(app)
        .put(`/produtos/${productId2}`)
        .set("authorization", token)
        .send({
          nome: name1,
          preco: 20,
          descricao: "D",
          quantidade: 2
        });

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });
  });

  describe("DELETE /produtos/:id", () => {
    it("should delete an existing product", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto To Delete ${Date.now()}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const productId = postRes.body._id;

      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro excluído com sucesso" });
    });

    it("should return delete none message if product does not exist", async () => {
      const token = await getAdminToken();
      const nonExistentId = "fedcba9876543210";

      const res = await request(app)
        .delete(`/produtos/${nonExistentId}`)
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Nenhum registro excluído" });
    });

    it("should not delete product if it is part of a cart", async () => {
      const token = await getAdminToken();
      const uniqueName = `Product Cart ${Date.now()}`;
      const postProd = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 50,
          descricao: "Desc",
          quantidade: 10
        });

      const productId = postProd.body._id;

      const userEmail = `user_cart_${Date.now()}@qa.com`;
      const userRes = await request(app)
        .post("/usuarios")
        .send({
          nome: "User Cart",
          email: userEmail,
          password: "teste",
          administrador: "false"
        });

      const userLogin = await request(app)
        .post("/login")
        .send({ email: userEmail, password: "teste" });

      const userToken = userLogin.body.authorization;

      await request(app)
        .post("/carrinhos")
        .set("authorization", userToken)
        .send({
          produtos: [{ idProduto: productId, quantidade: 2 }]
        });

      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(400);
      expect(res.body).toHaveProperty("message", "Não é permitido excluir produto que faz parte de carrinho");
      expect(res.body).toHaveProperty("idCarrinhos");
      expect(Array.isArray(res.body.idCarrinhos)).toBe(true);
    });
  });
});
```
