# Convenções de teste (projetos UnJS)

- Framework: Vitest. Imports no topo: `import { describe, it, expect } from "vitest";`.
- Estilo do código: aspas duplas, ponto e vírgula, indentação de 2 espaços.

- Para funções puras, prefira testes orientados a tabela (array de casos `{ input, expected }` percorrido com `for ... of` ou `it.each`), como nos testes originais do projeto.

// ===== file: src/controllers/carrinhos-controller.js =====
'use strict'

const constant = require('../utils/constants')
const service = require('../services/carrinhos-service')

exports.get = async (req, res) => {
  const carrinhos = await service.getAll(req.query)
  res.status(200).send({ quantidade: carrinhos.length, carrinhos })
}

exports.getOne = async (req, res) => {
  const { id } = req.params
  const carrinho = await service.getOne(id)
  if (!carrinho) {
    return res.status(400).json({ message: constant.CART_NOT_FOUND })
  }
  return res.status(200).send(carrinho)
}

exports.post = async (req, res) => {
  const { etapas } = res.locals
  etapas.regras('single_cart', 'no_duplicate_product', 'product_exists', 'enough_stock', 'safe_totals')

  const { idUsuario, possuiCarrinho } = await service.usuarioJaPossuiCarrinho(req.headers.authorization)
  etapas.regra('single_cart', !possuiCarrinho)
  if (possuiCarrinho) {
    return res.status(400).send({ message: constant.LIMIT_JUST_ONE_CART })
  }

  // O express-validation não devolve ao req.body o número que o Joi converteu. Sem o Number,
  // quantidade "2" seria gravada como texto e quantidadeTotal viraria "02". O + 0 transforma -0 em 0.
  const produtos = req.body.produtos.map(produto => ({ ...produto, quantidade: Number(produto.quantidade) + 0 }))

  const idProdutosDuplicados = service.extrairProdutosDuplicados(produtos)
  etapas.regra('no_duplicate_product', !idProdutosDuplicados.length)
  if (idProdutosDuplicados.length) {
    return res.status(400).send({ message: constant.CART_WITH_DUPLICATE_PRODUCT, idProdutosDuplicados })
  }

  const { produtosComPrecoUnitario, error } = await service.getProdutosComPrecoUnitarioOuErro(produtos)
  if (error) {
    // As duas regras são avaliadas item a item e param no primeiro item que falha.
    if (error.message === constant.PRODUCT_NOT_FOUND) {
      etapas.regra('product_exists', false, { item: error.index })
    } else {
      etapas.regra('product_exists', true)
      etapas.regra('enough_stock', false, { item: error.index })
    }
    return res.status(error.statusCode).send({ message: error.message, item: error.item })
  }
  etapas.regra('product_exists', true)
  etapas.regra('enough_stock', true)

  const precoTotal = service.precoTotal(produtosComPrecoUnitario)
  const quantidadeTotal = service.quantidadeTotal(produtosComPrecoUnitario)
  const totaisSeguros = service.totaisSaoSeguros(precoTotal, quantidadeTotal)
  etapas.regra('safe_totals', totaisSeguros)
  if (!totaisSeguros) {
    return res.status(400).send({ message: constant.CART_TOTALS_TOO_BIG })
  }

  etapas.abrir('write')
  await service.baixarEstoque(produtosComPrecoUnitario)

  const carrinho = { produtos: produtosComPrecoUnitario, precoTotal, quantidadeTotal, idUsuario }

  const dadosCadastrados = await service.criarCarrinho(carrinho)
  res.status(201).send({ message: constant.POST_SUCCESS, _id: dadosCadastrados._id })
}

exports.cancelarCompra = async (req, res) => {
  const carrinhoDoUsuario = await service.getCarrinhoDoUsuario(req.headers.authorization)

  if (carrinhoDoUsuario.length) {
    await service.removeCarrinho(carrinhoDoUsuario[0])
    return res.status(200).send({ message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}` })
  }

  res.status(200).send({ message: constant.NO_CART })
}

exports.concluirCompra = async (req, res) => {
  const carrinhoDoUsuario = await service.getCarrinhoDoUsuario(req.headers.authorization)

  if (carrinhoDoUsuario.length) {
    await service.concluiCompra(carrinhoDoUsuario[0])
    return res.status(200).send({ message: constant.DELETE_SUCCESS })
  }
  res.status(200).send({ message: constant.NO_CART })
}

'use strict'

const Datastore = require('nedb-promises')
const { join } = require('path')

const alterarValoresParaRegex = require('../utils/alterarValoresParaRegex')
const { continuarDaMaiorMarca, doMaisNovoAoMaisAntigo, marcarAlteracao, semMarca } = require('../utils/ordemDeAlteracao')
const authService = require('../services/auth-service')
const usuariosService = require('../services/usuarios-service')
const produtosService = require('../services/produtos-service')

const datastore = Datastore.create({ filename: join(__dirname, '../data/carrinhos.db'), autoload: true })
continuarDaMaiorMarca(datastore)

// Index frequently searched fields for better query performance
/* istanbul ignore next */
datastore.ensureIndex({ fieldName: 'idUsuario' })
/* istanbul ignore next */
datastore.ensureIndex({ fieldName: 'precoTotal' })
/* istanbul ignore next */
datastore.ensureIndex({ fieldName: 'quantidadeTotal' })
/* istanbul ignore next */
datastore.ensureIndex({ fieldName: 'produtos.idProduto' })

exports.getAll = async queryString => {
  queryString = alterarValoresParaRegex(queryString)
  return doMaisNovoAoMaisAntigo(await datastore.find(queryString))
}

exports.getOne = async id => {
  return semMarca(await datastore.findOne({ _id: id }))
}

exports.existeCarrinho = pesquisa => {
  return datastore.findOne(pesquisa)
}

exports.existeCarrinhoComProduto = idProduto => {
  return datastore.findOne({ produtos: { $elemMatch: { idProduto } } })
}

exports.criarCarrinho = async body => {
  return datastore.insert({ ...body, alteradoEm: marcarAlteracao() })
}

exports.extrairProdutosDuplicados = arrayProdutos => {
  const vistos = new Set()
  const duplicados = new Set()
  for (const { idProduto } of arrayProdutos) {
    if (vistos.has(idProduto)) duplicados.add(idProduto)
    vistos.add(idProduto)
  }
  return [...duplicados]
}

exports.getProdutosComPrecoUnitarioOuErro = async arrayProdutos => {
  // Use batch lookup instead of per-item queries to reduce round-trips
  const { produtosComPreco, error } = await produtosService.getPrecosUnitariosPorLote(arrayProdutos)

  if (error) {
    const item = { ...error.item, index: error.index }
    return { error: { ...error, item } }
  }

  return { produtosComPrecoUnitario: produtosComPreco }
}

exports.deleteById = async id => {
  return datastore.remove({ _id: id }, {})
}

exports.concluiCompra = async carrinho => {
  await this.deleteById(carrinho._id)
}

exports.removeCarrinho = async carrinho => {
  const { produtos, _id } = carrinho
  await this.reabasteceEstoque(produtos)
  await this.deleteById(_id)
}

exports.getCarrinhoDoUsuario = async (authorization) => {
  const _id = await idUsuario(authorization)
  return this.getAll({ idUsuario: _id })
}

exports.usuarioJaPossuiCarrinho = async (authorization) => {
  const _id = await idUsuario(authorization)
  return {
    idUsuario: _id,
    possuiCarrinho: await this.existeCarrinho({ idUsuario: _id })
  }
}

exports.precoTotal = produtos => produtos.reduce((total, produto) => total + produto.precoUnitario * produto.quantidade, 0)

exports.quantidadeTotal = produtos => produtos.reduce((total, produto) => total + produto.quantidade, 0)

// Acima de Number.MAX_SAFE_INTEGER o total seria gravado já arredondado, e o GET /carrinhos passaria a devolver
// um número que nenhum contrato com Joi.number() aceita, reprovando a lista inteira e não só esse carrinho.
exports.totaisSaoSeguros = (precoTotal, quantidadeTotal) =>
  Number.isSafeInteger(precoTotal) && Number.isSafeInteger(quantidadeTotal)

exports.baixarEstoque = async produtos => {
  // Use batch update to reduce round-trips instead of updating per-item
  await produtosService.updateQuantidadePorLote(produtos)
}

exports.reabasteceEstoque = async produtos => {
  // Use batch restore to reduce round-trips instead of restoring per-item
  await produtosService.restoreQuantidadePorLote(produtos)
}

const idUsuario = async (authorization) => {
  const { email, password } = authService.verifyToken(authorization)
  const { _id } = await usuariosService.getDadosDoUsuario({ email, password })
  return _id
}

'use strict'

const { Joi } = require('express-validation')

const filtroDeTexto = require('../utils/filtroDeTexto')

exports.schemaGet = {
  query: Joi.object({
    precoTotal: Joi.number().positive().integer(),
    quantidadeTotal: Joi.number().positive().integer(),
    idUsuario: filtroDeTexto,
    _id: filtroDeTexto
  })
}

exports.schemaGetOne = {
  params: Joi.object({
    id: Joi.string().regex(/^[a-zA-Z0-9]{16}$/).required()
  })
}

exports.schemaPost = {
  body: Joi.object({
    produtos: Joi.array().items(
      Joi.object({
        idProduto: Joi.string().required(),
        quantidade: Joi.number().positive().integer().required()
      })
    ).min(1).required()
  })
}

describe("carrinhos API", () => {

  });

  it("should require authentication to post a carrinho", async () => {
    const res = await request(app)

        produtos: [{ idProduto: "123456", quantidade: 1 }]
      });
    expect(res.status).toBe(401);
  });

  it("should require authentication to cancel a purchase", async () => {
    const res = await request(app).delete("/carrinhos/cancelar-compra");
    expect(res.status).toBe(401);
  });

  it("should require authentication to conclude a purchase", async () => {
    const res = await request(app).delete("/carrinhos/concluir-compra");
    expect(res.status).toBe(401);
  });
});
