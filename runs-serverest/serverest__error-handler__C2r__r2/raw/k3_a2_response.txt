```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headers already sent with timedout request", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    const req = {
      method: "GET",
      originalUrl: "/test",
      timedout: true,
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const result = errorHandler(error, req, res, () => {});
    expect(result).toBeUndefined();
  });

  it("should handle headers already sent without timeout", () => {
    const error = new Error("Test error");
    error.stack = "Error stack";
    const req = {
      method: "POST",
      originalUrl: "/test",
      timedout: false,
      socket: { destroy: () => {} }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    const result = errorHandler(error, req, res, () => {});
    expect(result).toBeUndefined();
  });

  it("should handle ValidationError", () => {
    const error = { name: "ValidationError", details: [{ message: "error", path: ["field"] }] };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toBeDefined();
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    error.stack = "stack";
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed", stack: "stack" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large", stack: "stack" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16", stack: "stack" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate", stack: "stack" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonBody).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle Zlib client error", () => {
    const error = { expose: true, statusCode: 400, code: "Z_DATA_ERROR", message: "invalid block type", stack: "stack" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonBody).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error", () => {
    const error = { code: "ETIMEDOUT", stack: "stack" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(408);
    expect(jsonBody).toHaveProperty("message");
  });

  it("should handle general client error", () => {
    const error = { expose: true, statusCode: 422, type: "custom.error", stack: "stack" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(422);
    expect(jsonBody).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.error"
    });
  });

  it("should handle internal error with type", () => {
    const error = { type: "database.error", stack: "stack", message: "DB fail" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (code) => {
        statusCode = code;
        return {
          json: (body) => {
            jsonBody = body;
            return body;
          }
        };
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(500);
    expect(jsonBody).toHaveProperty("error", "database.error");
    expect(jsonBody).toHaveProperty("version");
  });
});
```
