import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent by delegating to registrarErroDepoisDaResposta", () => {
    const error = new Error("Test error");
    const req = { method: "GET", originalUrl: "/test", timedout: false, socket: { destroy: () => {} } };
    let socketDestroyed = false;
    req.socket.destroy = () => {
      socketDestroyed = true;
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});

    expect(socketDestroyed).toBe(true);
  });

  it("should handle ValidationError by returning status 400 with schema error message", () => {
    const error = { name: "ValidationError", details: [{ message: "field is required", path: ["field"] }] };
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s: number) => {
        statusCalled = s;
        return res;
      },
      json: (b: any) => {
        jsonBody = b;
        return res;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonBody).toEqual({ "field": "field is required" });
  });

  it("should handle URIError by returning status 400", () => {
    const error = new URIError("Malformed URI");
    const req = { body: {} };
    let statusCalled = 0;
    let jsonBody = null;
    const res = {
      headersSent: false,
      status: (s: number) => {
        statusCalled = s;
        return res;
      },
      json: (b: any) => {
        jsonBody = b;
        return res;
      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(400);
    expect(jsonBody).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed error type", () => {
    const error = { type: "entity.parse.failed" };
    const req = { body: {} };
    let statusCalled = 0;

      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"

        jsonBody = b;

      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(413);
    expect(jsonBody).toEqual({ message: "Payload too large" });
  });

  it("should handle parameters.too.many error type", () => {

        statusCalled = s;

    let statusCalled = 0;
    let jsonBody = null;

      headersSent: false,

      }
    };

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(422);
    expect(jsonBody).toEqual({

      error: "custom.error"

  it("should handle internal error with type (status 500)", () => {
    const error = { type: "database.error", stack: "stack trace" };
    const req = { body: {} };
    let statusCalled = 0;

        statusCalled = s;
        return res;
      },
      json: (b: any) => {
        jsonBody = b;

    errorHandler(error, req, res, () => {});

    expect(statusCalled).toBe(500);
    expect(jsonBody.error).toBe("database.error");
    expect(jsonBody).toHaveProperty("version");
  });
});
