```ts
import { describe, it, expect } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler", () => {
  it("should handle headersSent when request timed out", () => {
    const error = new Error("Timeout after response");
    const socketDestroy = { destroy: () => {} };
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/users",
      socket: socketDestroy
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const next = () => {};
    errorHandler(error, req, res, next);
    expect(res.headersSent).toBe(true);
  });

  it("should handle headersSent when request did not timeout", () => {
    const error = new Error("Error after response");
    const socketDestroy = { destroy: () => {} };
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/login",
      socket: socketDestroy
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    const next = () => {};
    errorHandler(error, req, res, next);
    expect(res.headersSent).toBe(true);
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: [{ message: "Field is required" }]
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonResponse = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonResponse).not.toBeNull();
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonResponse = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonResponse).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonResponse = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonResponse).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonResponse = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(413);
    expect(jsonResponse).toEqual({ message: "Payload too large" });
  });

  it("should handle parameters.too.many", () => {
    const error = { type: "parameters.too.many" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonResponse = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(413);
    expect(jsonResponse).toEqual({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonResponse = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonResponse).toEqual({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonResponse = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(415);
    expect(jsonResponse).toEqual({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle Zlib compression client error", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "Unexpected end of file"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonResponse = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(400);
    expect(jsonResponse).toEqual({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = { code: "ETIMEDOUT" };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonResponse = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(408);
    expect(jsonResponse).toHaveProperty("message");
  });

  it("should handle generic client error (expose = true)", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "some.client.error",
      message: "Invalid entity"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonResponse = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(422);
    expect(jsonResponse).toEqual({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.client.error"
    });
  });

  it("should handle server error with error type", () => {
    const error = {
      type: "database.error",
      stack: "Error stack trace"
    };
    const req = { body: {} };
    let statusCode = 0;
    let jsonResponse: any = null;
    const res = {
      headersSent: false,
      status(code: number) {
        statusCode = code;
        return this;
      },
      json(body: any) {
        jsonResponse = body;
        return this;
      }
    };

    errorHandler(error, req, res, () => {});
    expect(statusCode).toBe(500);
    expect(jsonResponse.error).toBe("database.error");
    expect(jsonResponse).toHaveProperty("message");
    expect(jsonResponse).toHaveProperty("version");
  });
});
```
