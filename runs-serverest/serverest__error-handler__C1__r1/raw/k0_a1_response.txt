```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";

describe("errorHandler middleware", () => {
  it("handles headers already sent when response is not writable ended", () => {
    const error = new Error("Late error");
    const req = { method: "GET", originalUrl: "/test", timedout: true, socket: { destroy: vi.fn() } };
    const res = { headersSent: true, writableEnded: false };
    const next = vi.fn();

    errorHandler(error, req, res, next);

    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("handles headers already sent when response is writable ended", () => {
    const error = new Error("Late error");
    const req = { method: "GET", originalUrl: "/test", timedout: false, socket: { destroy: vi.fn() } };
    const res = { headersSent: true, writableEnded: true };
    const next = vi.fn();

    errorHandler(error, req, res, next);

    expect(req.socket.destroy).not.toHaveBeenCalled();
  });

  it("handles ValidationError (schema error)", () => {
    const error = { name: "ValidationError", inner: [] };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalled();
  });

  it("handles URIError", () => {
    const error = new Error("Malformed URI");
    error.name = "URIError";
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: 'Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente.'
    });
  });

  it("handles entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: 'Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225'
    });
  });

  it("handles entity.too.large", () => {
    const error = { type: "entity.too.large" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(413);
    expect(jsonMock).toHaveBeenCalledWith({ message: 'Payload too large' });
  });

  it("handles parameters.too.many", () => {
    const error = { type: "parameters.too.many" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(413);
    expect(jsonMock).toHaveBeenCalledWith({ message: 'Payload too large' });
  });

  it("handles charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(415);
    expect(jsonMock).toHaveBeenCalledWith({ message: 'Unsupported charset', charset: 'utf-16' });
  });

  it("handles encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(415);
    expect(jsonMock).toHaveBeenCalledWith({ message: 'Unsupported content encoding', encoding: 'deflate' });
  });

  it("handles client error with Z_ compression error code", () => {
    const error = { expose: true, statusCode: 400, code: "Z_DATA_ERROR", message: "zlib error" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: 'Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header.'
    });
  });

  it("handles ETIMEDOUT code", () => {
    const error = { code: "ETIMEDOUT" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(408);
    expect(jsonMock).toHaveBeenCalledWith({ message: expect.any(String) });
  });

  it("handles generic client error", () => {
    const error = { expose: true, statusCode: 400, type: "bad.request", message: "Bad request" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: 'Não foi possível ler a requisição. Confira o corpo e os headers enviados.',
      error: 'bad.request'
    });
  });

  it("handles internal server error with type", () => {
    const error = { type: "some.internal.error", message: "fail" };
    const req = { body: {} };
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(500);
    expect(jsonMock).toHaveBeenCalledWith(expect.objectContaining({
      message: expect.any(String),
      error: 'some.internal.error'
    }));
  });
});
```
