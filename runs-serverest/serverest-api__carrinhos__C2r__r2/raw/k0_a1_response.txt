```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("carrinhos API", () => {
  let counter = 0;
  function getUniqueSuffix() {
    counter++;
    return `u_${Date.now()}_${counter}`;
  }

  async function createRegisteredUser(uniqueSuffix: string) {
    const user = {
      nome: `User ${uniqueSuffix}`,
      email: `user_${uniqueSuffix}@email.com`,
      password: "password123",
      administrador: "true"
    };
    await request(app).post("/usuarios").send(user);
    const loginRes = await request(app).post("/login").send({
      email: user.email,
      password: user.password
    });
    return loginRes.body.authorization as string;
  }

  async function createProduct(uniqueSuffix: string, quantidade = 10) {
    const product = {
      nome: `Product ${uniqueSuffix}`,
      preco: 100,
      descricao: "Description",
      quantidade
    };
    const token = await createRegisteredUser(`prod_creator_${uniqueSuffix}`);
    const res = await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send(product);
    return res.body._id as string;
  }

  describe("GET /carrinhos", () => {
    it("should return empty list when no carts exist", async () => {
      const res = await request(app).get("/carrinhos");
      expect(res.status).toBe(200);
      expect(res.body).toEqual({ quantidade: expect.any(Number), carrinhos: expect.any(Array) });
    });

    it("should filter carts by query parameters", async () => {
      const suffix = getUniqueSuffix();
      const token = await createRegisteredUser(suffix);
      const productId = await createProduct(suffix, 20);

      await request(app)
        .post("/carrinhos")
        .set("authorization", token)
        .send({
          produtos: [{ idProduto: productId, quantidade: 2 }]
        });

      const res = await request(app).get("/carrinhos?quantidadeTotal=2");
      expect(res.status).toBe(200);
      expect(res.body.quantidade).toBeGreaterThan(0);
      expect(res.body.carrinhos[0].quantidadeTotal).toBe(2);
    });

    it("should fail validation for invalid query parameters", async () => {
      const res = await request(app).get("/carrinhos?precoTotal=invalid");
      expect(res.status).toBe(400);
    });
  });

  describe("GET /carrinhos/:id", () => {
    it("should return a cart by valid id", async () => {
      const suffix = getUniqueSuffix();
      const token = await createRegisteredUser(suffix);
      const productId = await createProduct(suffix, 20);

      const postRes = await request(app)
        .post("/carrinhos")
        .set("authorization", token)
        .send({
          produtos: [{ idProduto: productId, quantidade: 1 }]
        });

      const cartId = postRes.body._id;
      const res = await request(app).get(`/carrinhos/${cartId}`);
      expect(res.status).toBe(200);
      expect(res.body._id).toBe(cartId);
    });

    it("should return 400 when cart id is not found", async () => {
      const res = await request(app).get("/carrinhos/1234567890abcdef");
      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Carrinho não encontrado" });
    });

    it("should fail validation for malformed cart id", async () => {
      const res = await request(app).get("/carrinhos/invalid-id");
      expect(res.status).toBe(400);
    });
  });

  describe("POST /carrinhos", () => {
    it("should create a cart successfully and decrease stock", async () => {
      const suffix = getUniqueSuffix();
      const token = await createRegisteredUser(suffix);
      const productId = await createProduct(suffix, 15);

      const res = await request(app)
        .post("/carrinhos")
        .set("authorization", token)
        .send({
          produtos: [{ idProduto: productId, quantidade: 3 }]
        });

      expect(res.status).toBe(201);
      expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
      expect(res.body).toHaveProperty("_id");

      const productRes = await request(app).get(`/produtos/${productId}`);
      expect(productRes.body.quantidade).toBe(12);
    });

    it("should fail when user is not authenticated", async () => {
      const res = await request(app)
        .post("/carrinhos")
        .send({
          produtos: [{ idProduto: "1234567890abcdef", quantidade: 1 }]
        });

      expect(res.status).toBe(401);
      expect(res.body).toEqual({ message: "Token de acesso ausente" });
    });

    it("should fail when user already possesses a cart", async () => {
      const suffix = getUniqueSuffix();
      const token = await createRegisteredUser(suffix);
      const prod1 = await createProduct(`${suffix}_1`, 10);
      const prod2 = await createProduct(`${suffix}_2`, 10);

      await request(app)
        .post("/carrinhos")
        .set("authorization", token)
        .send({
          produtos: [{ idProduto: prod1, quantidade: 1 }]
        });

      const res = await request(app)
        .post("/carrinhos")
        .set("authorization", token)
        .send({
          produtos: [{ idProduto: prod2, quantidade: 1 }]
        });

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Não é permitido ter mais de 1 carrinho" });
    });

    it("should fail when product is duplicated in the request body", async () => {
      const suffix = getUniqueSuffix();
      const token = await createRegisteredUser(suffix);
      const productId = await createProduct(suffix, 10);

      const res = await request(app)
        .post("/carrinhos")
        .set("authorization", token)
        .send({
          produtos: [
            { idProduto: productId, quantidade: 1 },
            { idProduto: productId, quantidade: 2 }
          ]
        });

      expect(res.status).toBe(400);
      expect(res.body.message).toBe("Não é permitido possuir produtos duplicados no carrinho");
      expect(res.body.idProdutosDuplicados).toContain(productId);
    });

    it("should fail when product does not exist", async () => {
      const suffix = getUniqueSuffix();
      const token = await createRegisteredUser(suffix);

      const res = await request(app)
        .post("/carrinhos")
        .set("authorization", token)
        .send({
          produtos: [{ idProduto: "nonexistent0001", quantidade: 1 }]
        });

      expect(res.status).toBe(400);
      expect(res.body.message).toBe("Produto não encontrado");
    });

    it("should fail when stock is not enough", async () => {
      const suffix = getUniqueSuffix();
      const token = await createRegisteredUser(suffix);
      const productId = await createProduct(suffix, 2);

      const res = await request(app)
        .post("/carrinhos")
        .set("authorization", token)
        .send({
          produtos: [{ idProduto: productId, quantidade: 5 }]
        });

      expect(res.status).toBe(400);
      expect(res.body.message).toBe("Produto não possui quantidade suficiente");
    });

    it("should fail validation when body is missing produtos", async () => {
      const suffix = getUniqueSuffix();
      const token = await createRegisteredUser(suffix);

      const res = await request(app)
        .post("/carrinhos")
        .set("authorization", token)
        .send({});

      expect(res.status).toBe(400);
    });
  });

  describe("DELETE /carrinhos/concluir-compra", () => {
    it("should conclude purchase successfully and delete cart", async () => {
      const suffix = getUniqueSuffix();
      const token = await createRegisteredUser(suffix);
      const productId = await createProduct(suffix, 10);

      await request(app)
        .post("/carrinhos")
        .set("authorization", token)
        .send({
          produtos: [{ idProduto: productId, quantidade: 2 }]
        });

      const res = await request(app)
        .delete("/carrinhos/concluir-compra")
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro excluído com sucesso" });
    });

    it("should return message when user has no cart to conclude", async () => {
      const suffix = getUniqueSuffix();
      const token = await createRegisteredUser(suffix);

      const res = await request(app)
        .delete("/carrinhos/concluir-compra")
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Não existe carrinho para este usuário" });
    });

    it("should fail when unauthorized", async () => {
      const res = await request(app).delete("/carrinhos/concluir-compra");
      expect(res.status).toBe(401);
    });
  });

  describe("DELETE /carrinhos/cancelar-compra", () => {
    it("should cancel purchase, replenish stock, and delete cart", async () => {
      const suffix = getUniqueSuffix();
      const token = await createRegisteredUser(suffix);
      const productId = await createProduct(suffix, 10);

      await request(app)
        .post("/carrinhos")
        .set("authorization", token)
        .send({
          produtos: [{ idProduto: productId, quantidade: 4 }]
        });

      const res = await request(app)
        .delete("/carrinhos/cancelar-compra")
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body.message).toContain("Registro excluído com sucesso");
      expect(res.body.message).toContain("Estoque reabastecido");

      const productRes = await request(app).get(`/produtos/${productId}`);
      expect(productRes.body.quantidade).toBe(10);
    });

    it("should return message when user has no cart to cancel", async () => {
      const suffix = getUniqueSuffix();
      const token = await createRegisteredUser(suffix);

      const res = await request(app)
        .delete("/carrinhos/cancelar-compra")
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Não existe carrinho para este usuário" });
    });

    it("should fail when unauthorized", async () => {
      const res = await request(app).delete("/carrinhos/cancelar-compra");
      expect(res.status).toBe(401);
    });
  });
});
```
