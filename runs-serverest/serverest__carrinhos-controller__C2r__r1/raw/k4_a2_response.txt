```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import service from "../../src/services/carrinhos-service";
import constant from "../../src/utils/constants";

vi.mock("../../src/services/carrinhos-service", async (importOriginal) => {
  const actual: any = await importOriginal();
  return {
    ...actual,
    getAll: vi.fn(),
    getOne: vi.fn(),
    usuarioJaPossuiCarrinho: vi.fn(),
    extrairProdutosDuplicados: vi.fn(),
    getProdutosComPrecoUnitarioOuErro: vi.fn(),
    precoTotal: vi.fn(),
    quantidadeTotal: vi.fn(),
    totaisSaoSeguros: vi.fn(),
    baixarEstoque: vi.fn(),
    criarCarrinho: vi.fn(),
    getCarrinhoDoUsuario: vi.fn(),
    removeCarrinho: vi.fn(),
    concluiCompra: vi.fn()
  };
});

const createMockResponse = () => {
  const res: any = {};
  res.status = vi.fn().mockReturnValue(res);
  res.send = vi.fn().mockReturnValue(res);
  res.json = vi.fn().mockReturnValue(res);
  return res;
};

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  describe("get", () => {
    it("should return all carts with quantity", async () => {
      const req = { query: {} };
      const res = createMockResponse();
      const mockCarrinhos = [{ _id: "1" }, { _id: "2" }];

      vi.mocked(service.getAll as any).mockResolvedValueOnce(mockCarrinhos);

      await carrinhosController.get(req, res);

      expect(service.getAll).toHaveBeenCalledWith(req.query);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: 2,
        carrinhos: mockCarrinhos
      });
    });
  });

  describe("getOne", () => {
    it("should return cart when found", async () => {
      const req = { params: { id: "123" } };
      const res = createMockResponse();
      const mockCarrinho = { _id: "123", produtos: [] };

      vi.mocked(service.getOne as any).mockResolvedValueOnce(mockCarrinho);

      await carrinhosController.getOne(req, res);

      expect(service.getOne).toHaveBeenCalledWith("123");
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });

    it("should return 400 when cart is not found", async () => {
      const req = { params: { id: "999" } };
      const res = createMockResponse();

      vi.mocked(service.getOne as any).mockResolvedValueOnce(null);

      await carrinhosController.getOne(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });
  });

  describe("post", () => {
    it("should create a cart successfully", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: "2" }] }
      };
      const res = createMockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho as any).mockResolvedValueOnce({
        idUsuario: "u1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados as any).mockReturnValueOnce([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro as any).mockResolvedValueOnce({
        produtosComPrecoUnitario: [{ idProduto: "p1", quantidade: 2, precoUnitario: 10 }],
        error: null
      });
      vi.mocked(service.precoTotal as any).mockReturnValueOnce(20);
      vi.mocked(service.quantidadeTotal as any).mockReturnValueOnce(2);
      vi.mocked(service.totaisSaoSeguros as any).mockReturnValueOnce(true);
      vi.mocked(service.criarCarrinho as any).mockResolvedValueOnce({ _id: "c1" });

      await carrinhosController.post(req, res);

      expect(etapas.regras).toHaveBeenCalledWith(
        "single_cart",
        "no_duplicate_product",
        "product_exists",
        "enough_stock",
        "safe_totals"
      );
      expect(service.baixarEstoque).toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.POST_SUCCESS,
        _id: "c1"
      });
    });

    it("should return 400 when user already has a cart", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [] }
      };
      const res = createMockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho as any).mockResolvedValueOnce({
        idUsuario: "u1",
        possuiCarrinho: true
      });

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return 400 when cart has duplicate products", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }, { idProduto: "p1", quantidade: 2 }] }
      };
      const res = createMockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho as any).mockResolvedValueOnce({
        idUsuario: "u1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados as any).mockReturnValueOnce(["p1"]);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["p1"]
      });
    });

    it("should return error when product is not found or out of stock", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "invalid", quantidade: 1 }] }
      };
      const res = createMockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho as any).mockResolvedValueOnce({
        idUsuario: "u1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados as any).mockReturnValueOnce([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro as any).mockResolvedValueOnce({
        produtosComPrecoUnitario: [],
        error: {
          statusCode: 400,
          message: constant.PRODUCT_NOT_FOUND,
          item: 0,
          index: 0
        }
      });

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.PRODUCT_NOT_FOUND,
        item: 0
      });
    });

    it("should return 400 when totals are not safe", async () => {
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }] }
      };
      const res = createMockResponse();
      res.locals = { etapas };

      vi.mocked(service.usuarioJaPossuiCarrinho as any).mockResolvedValueOnce({
        idUsuario: "u1",
        possuiCarrinho: false
      });
      vi.mocked(service.extrairProdutosDuplicados as any).mockReturnValueOnce([]);
      vi.mocked(service.getProdutosComPrecoUnitarioOuErro as any).mockResolvedValueOnce({
        produtosComPrecoUnitario: [{ idProduto: "p1", quantidade: 1, precoUnitario: 10 }],
        error: null
      });
      vi.mocked(service.precoTotal as any).mockReturnValueOnce(999999999);
      vi.mocked(service.quantidadeTotal as any).mockReturnValueOnce(1);
      vi.mocked(service.totaisSaoSeguros as any).mockReturnValueOnce(false);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });
  });

  describe("cancelarCompra", () => {
    it("should remove cart and replenish stock when cart exists", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockResponse();
      const mockCarrinho = { _id: "c1" };

      vi.mocked(service.getCarrinhoDoUsuario as any).mockResolvedValueOnce([mockCarrinho]);

      await carrinhosController.cancelarCompra(req, res);

      expect(service.removeCarrinho).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}`
      });
    });

    it("should return NO_CART when user has no cart", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockResponse();

      vi.mocked(service.getCarrinhoDoUsuario as any).mockResolvedValueOnce([]);

      await carrinhosController.cancelarCompra(req, res);

      expect(service.removeCarrinho).not.toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase when cart exists", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockResponse();
      const mockCarrinho = { _id: "c1" };

      vi.mocked(service.getCarrinhoDoUsuario as any).mockResolvedValueOnce([mockCarrinho]);

      await carrinhosController.concluirCompra(req, res);

      expect(service.concluiCompra).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("should return NO_CART when user has no cart", async () => {
      const req = { headers: { authorization: "Bearer token" } };
      const res = createMockResponse();

      vi.mocked(service.getCarrinhoDoUsuario as any).mockResolvedValueOnce([]);

      await carrinhosController.concluirCompra(req, res);

      expect(service.concluiCompra).not.toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});
```
