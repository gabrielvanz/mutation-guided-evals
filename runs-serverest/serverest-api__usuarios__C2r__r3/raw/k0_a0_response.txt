```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("usuarios API", () => {
  it("should get empty or list of users", async () => {
    const res = await request(app).get("/usuarios");
    expect(res.status).toBe(200);
    expect(res.body).toHaveProperty("quantidade");
    expect(res.body).toHaveProperty("usuarios");
    expect(Array.isArray(res.body.usuarios)).toBe(true);
  });

  it("should create a new user successfully", async () => {
    const uniqueEmail = `user_${Date.now()}_${Math.random()}@test.com`;
    const res = await request(app)
      .post("/usuarios")
      .send({
        nome: "Test User",
        email: uniqueEmail,
        password: "password123",
        administrador: "true"
      });

    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
    expect(res.body).toHaveProperty("_id");
  });

  it("should fail to create a user with duplicate email", async () => {
    const uniqueEmail = `dup_${Date.now()}_${Math.random()}@test.com`;
    const userData = {
      nome: "User One",
      email: uniqueEmail,
      password: "password123",
      administrador: "true"
    };

    const firstRes = await request(app).post("/usuarios").send(userData);
    expect(firstRes.status).toBe(201);

    const secondRes = await request(app).post("/usuarios").send({
      ...userData,
      nome: "User Two"
    });
    expect(secondRes.status).toBe(400);
    expect(secondRes.body).toEqual({ message: "Este email já está sendo usado" });
  });

  it("should fail to create a user with missing fields", async () => {
    const res = await request(app)
      .post("/usuarios")
      .send({
        nome: "Incomplete User"
      });

    expect(res.status).toBe(400);
  });

  it("should get a single user by id", async () => {
    const uniqueEmail = `getone_${Date.now()}_${Math.random()}@test.com`;
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Get One User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    expect(createRes.status).toBe(201);
    const userId = createRes.body._id;

    const getRes = await request(app).get(`/usuarios/${userId}`);
    expect(getRes.status).toBe(200);
    expect(getRes.body).toHaveProperty("_id", userId);
    expect(getRes.body).toHaveProperty("email", uniqueEmail);
  });

  it("should return user not found for non-existent valid 16-char id", async () => {
    const fakeId = "1234567890abcdef";
    const res = await request(app).get(`/usuarios/${fakeId}`);
    expect(res.status).toBe(400);
    expect(res.body).toEqual({ message: "Usuário não encontrado" });
  });

  it("should delete an existing user", async () => {
    const uniqueEmail = `del_${Date.now()}_${Math.random()}@test.com`;
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Delete User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    const userId = createRes.body._id;

    const delRes = await request(app).delete(`/usuarios/${userId}`);
    expect(delRes.status).toBe(200);
    expect(delRes.body).toEqual({ message: "Registro excluído com sucesso" });
  });

  it("should handle deletion of non-existent user", async () => {
    const delRes = await request(app).delete("/usuarios/nonexistentid123");
    expect(delRes.status).toBe(200);
    expect(delRes.body).toEqual({ message: "Nenhum registro excluído" });
  });

  it("should update an existing user via put", async () => {
    const uniqueEmail = `put_${Date.now()}_${Math.random()}@test.com`;
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Put User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    const userId = createRes.body._id;
    const updatedEmail = `updated_${uniqueEmail}`;

    const putRes = await request(app)
      .put(`/usuarios/${userId}`)
      .send({
        nome: "Updated Name",
        email: updatedEmail,
        password: "newpassword",
        administrador: "true"
      });

    expect(putRes.status).toBe(200);
    expect(putRes.body).toEqual({ message: "Registro alterado com sucesso" });
  });
});
```
