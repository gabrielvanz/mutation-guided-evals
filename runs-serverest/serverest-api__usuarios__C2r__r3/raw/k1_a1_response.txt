```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Usuarios API", () => {
  it("should get empty or list of users", async () => {
    const res = await request(app).get("/usuarios");
    expect(res.status).toBe(200);
    expect(res.body).toHaveProperty("quantidade");
    expect(res.body).toHaveProperty("usuarios");
    expect(Array.isArray(res.body.usuarios)).toBe(true);
  });

  it("should create a new user successfully", async () => {
    const uniqueEmail = "user_test_unique_001@test.com";
    const res = await request(app)
      .post("/usuarios")
      .send({
        nome: "Test User",
        email: uniqueEmail,
        password: "password123",
        administrador: "true"
      });

    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
    expect(res.body).toHaveProperty("_id");
  });

  it("should fail to create user with existing email", async () => {
    const uniqueEmail = "duplicate_test_unique_002@test.com";
    const userData = {
      nome: "User One",
      email: uniqueEmail,
      password: "password123",
      administrador: "false"
    };

    const firstRes = await request(app).post("/usuarios").send(userData);
    expect(firstRes.status).toBe(201);

    const secondRes = await request(app).post("/usuarios").send({
      ...userData,
      nome: "User Two"
    });

    expect(secondRes.status).toBe(400);
    expect(secondRes.body).toEqual({ message: "Este email já está sendo usado" });
  });

  it("should return 400 when getting non-existent user", async () => {
    const res = await request(app).get("/usuarios/abc123xyz7890123");
    expect(res.status).toBe(400);
    expect(res.body).toEqual({ message: "Usuário não encontrado" });
  });

  it("should get a specific user by id", async () => {
    const uniqueEmail = "getone_test_unique_003@test.com";
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Get One User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    const userId = createRes.body._id;

    const res = await request(app).get(`/usuarios/${userId}`);
    expect(res.status).toBe(200);
    expect(res.body).toHaveProperty("_id", userId);
    expect(res.body).toHaveProperty("email", uniqueEmail);
  });

  it("should delete an existing user", async () => {
    const uniqueEmail = "delete_test_unique_004@test.com";
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Delete User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    const userId = createRes.body._id;

    const res = await request(app).delete(`/usuarios/${userId}`);
    expect(res.status).toBe(200);
    expect(res.body).toEqual({ message: "Registro excluído com sucesso" });
  });

  it("should update an existing user via put", async () => {
    const uniqueEmail = "put_test_unique_005@test.com";
    const createRes = await request(app)
      .post("/usuarios")
      .send({
        nome: "Put User",
        email: uniqueEmail,
        password: "password123",
        administrador: "false"
      });

    const userId = createRes.body._id;

    const updateRes = await request(app)
      .put(`/usuarios/${userId}`)
      .send({
        nome: "Updated Name",
        email: uniqueEmail,
        password: "newpassword123",
        administrador: "true"
      });

    expect(updateRes.status).toBe(200);
    expect(updateRes.body).toEqual({ message: "Registro alterado com sucesso" });
  });
});
```
