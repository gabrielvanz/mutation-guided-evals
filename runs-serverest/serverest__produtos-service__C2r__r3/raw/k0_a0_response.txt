```ts
import { describe, it, expect } from "vitest";
import produtosService from "../../src/services/produtos-service";

describe("getAll", () => {
  it("should return all products matching query", async () => {
    const produtos = await produtosService.getAll({});
    expect(Array.isArray(produtos)).toBe(true);
  });
});

describe("getOne", () => {
  it("should return null when product does not exist", async () => {
    const produto = await produtosService.getOne("id-inexistente");
    expect(produto).toBeNull();
  });
});

describe("existeProduto", () => {
  it("should return null when product is not found by search criteria", async () => {
    const resultado = await produtosService.existeProduto({ nome: "Produto Inexistente XYZ" });
    expect(resultado).toBeNull();
  });
});

describe("criarProduto", () => {
  it("should create a product with formatted values and timestamp", async () => {
    const novoProduto = {
      nome: "  Produto Teste  ",
      preco: "199.99",
      quantidade: "10",
      descricao: "Teste",
      imagem: " imagem.jpg "
    };

    const criado = await produtosService.criarProduto(novoProduto);
    expect(criado).toHaveProperty("_id");
    expect(criado.nome).toBe("Produto Teste");
    expect(criado.preco).toBe(199.99);
    expect(criado.quantidade).toBe(10);
    expect(criado.imagem).toBe("imagem.jpg");
    expect(criado).toHaveProperty("alteradoEm");
  });
});

describe("getPrecosUnitariosPorLote", () => {
  it("should return error when product in batch does not exist", async () => {
    const lote = [
      { idProduto: "inexistente123", quantidade: 2 }
    ];
    const resultado = await produtosService.getPrecosUnitariosPorLote(lote);
    expect(resultado).toHaveProperty("error");
    expect(resultado.error.statusCode).toBe(400);
    expect(resultado.error.item.idProduto).toBe("inexistente123");
  });

  it("should return error when stock is insufficient for a product in batch", async () => {
    const criado = await produtosService.criarProduto({
      nome: "Produto Estoque",
      preco: "50",
      quantidade: "5"
    });

    const lote = [
      { idProduto: criado._id, quantidade: 10 }
    ];

    const resultado = await produtosService.getPrecosUnitariosPorLote(lote);
    expect(resultado).toHaveProperty("error");
    expect(resultado.error.statusCode).toBe(400);
    expect(resultado.error.item.quantidadeEstoque).toBe(5);
  });

  it("should return prices successfully for valid batch", async () => {
    const criado = await produtosService.criarProduto({
      nome: "Produto Lote Valido",
      preco: "25",
      quantidade: "10"
    });

    const lote = [
      { idProduto: criado._id, quantidade: 3 }
    ];

    const resultado = await produtosService.getPrecosUnitariosPorLote(lote);
    expect(resultado).toHaveProperty("produtosComPreco");
    expect(resultado.produtosComPreco[0].precoUnitario).toBe(25);
  });
});

describe("deleteById", () => {
  it("should remove product by id and return count of removed docs", async () => {
    const criado = await produtosService.criarProduto({
      nome: "Para Deletar",
      preco: "10",
      quantidade: "1"
    });

    const removidos = await produtosService.deleteById(criado._id);
    expect(removidos).toBe(1);
  });
});

describe("createOrUpdateById", () => {
  it("should create or update product by id with formatted values", async () => {
    const id = "id-customizado-put";
    const body = {
      nome: "  Produto PUT  ",
      preco: "15.5",
      quantidade: "8"
    };

    const atualizado = await produtosService.createOrUpdateById(id, body);
    expect(atualizado._id).toBe(id);
    expect(atualizado.nome).toBe("Produto PUT");
    expect(atualizado.preco).toBe(15.5);
    expect(atualizado.quantidade).toBe(8);
    expect(atualizado).toHaveProperty("alteradoEm");
  });
});

describe("updateById", () => {
  it("should update product directly with given body", async () => {
    const criado = await produtosService.criarProduto({
      nome: "Atualizar Direto",
      preco: "10",
      quantidade: "5"
    });

    const resultado = await produtosService.updateById(criado._id, { $set: { quantidade: 20 } });
    expect(resultado.quantidade).toBe(20);
  });
});
```
