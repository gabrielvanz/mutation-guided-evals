```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Produtos API", () => {
  async function getAdminToken() {
    const adminEmail = `admin_prod_${Date.now()}_${Math.random()}@qa.com`;
    await request(app)
      .post("/usuarios")
      .send({
        nome: "Admin Test",
        email: adminEmail,
        password: "teste",
        administrador: "true"
      });

    const loginRes = await request(app)
      .post("/login")
      .send({
        email: adminEmail,
        password: "teste"
      });

    return loginRes.body.authorization;
  }

  async function getRegularToken() {
    const userEmail = `user_prod_${Date.now()}_${Math.random()}@qa.com`;
    await request(app)
      .post("/usuarios")
      .send({
        nome: "Regular User",
        email: userEmail,
        password: "teste",
        administrador: "false"
      });

    const loginRes = await request(app)
      .post("/login")
      .send({
        email: userEmail,
        password: "teste"
      });

    return loginRes.body.authorization;
  }

  describe("GET /produtos", () => {
    it("should return all products", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto Teste ${Date.now()}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Descricao",
          quantidade: 10
        });

      const res = await request(app).get("/produtos");
      expect(res.status).toBe(200);
      expect(res.body).toHaveProperty("quantidade");
      expect(res.body).toHaveProperty("produtos");
      expect(Array.isArray(res.body.produtos)).toBe(true);
      expect(res.body.quantidade).toBeGreaterThanOrEqual(1);
    });

    it("should filter products by query parameters", async () => {
      const token = await getAdminToken();
      const uniqueName = `FiltroProd_${Date.now()}`;
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 250,
          descricao: "Descricao",
          quantidade: 5
        });

      const res = await request(app).get(`/produtos?nome=${uniqueName}`);
      expect(res.status).toBe(200);
      expect(res.body.quantidade).toBe(1);
      expect(res.body.produtos[0].nome).toBe(uniqueName);
    });
  });

  describe("GET /produtos/:id", () => {
    it("should return a product by id", async () => {
      const token = await getAdminToken();
      const uniqueName = `ProdGetOne_${Date.now()}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 50,
          descricao: "Descricao",
          quantidade: 10
        });

      const productId = postRes.body._id;

      const res = await request(app).get(`/produtos/${productId}`);
      expect(res.status).toBe(200);
      expect(res.body._id).toBe(productId);
      expect(res.body.nome).toBe(uniqueName);
    });

    it("should return 400 when product id does not exist", async () => {
      const res = await request(app).get("/produtos/abcdef1234567890");
      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Produto não encontrado" });
    });

    it("should return validation error when product id format is invalid", async () => {
      const res = await request(app).get("/produtos/invalid-id");
      expect(res.status).toBe(400);
    });
  });

  describe("POST /produtos", () => {
    it("should create a product when admin is authenticated and data is valid", async () => {
      const token = await getAdminToken();
      const uniqueName = `Novo Produto ${Date.now()}`;
      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 470,
          descricao: "Mouse Gamer",
          quantidade: 381,
          imagem: "http://imagem.com"
        });

      expect(res.status).toBe(201);
      expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
      expect(res.body).toHaveProperty("_id");
    });

    it("should return 400 when product name is already used", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto Duplicado ${Date.now()}`;
      const payload = {
        nome: uniqueName,
        preco: 100,
        descricao: "Desc",
        quantidade: 10
      };

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send(payload);

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send(payload);

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });

    it("should return 401 when token is missing", async () => {
      const res = await request(app)
        .post("/produtos")
        .send({
          nome: "Sem Token",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(401);
    });

    it("should return 403 when user is not admin", async () => {
      const token = await getRegularToken();
      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "User Comum",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(403);
    });

    it("should validate required body fields", async () => {
      const token = await getAdminToken();
      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          preco: 100
        });

      expect(res.status).toBe(400);
    });
  });

  describe("PUT /produtos/:id", () => {
    it("should update an existing product successfully", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto Para Editar ${Date.now()}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const productId = postRes.body._id;
      const updatedName = `Produto Editado ${Date.now()}`;

      const res = await request(app)
        .put(`/produtos/${productId}`)
        .set("authorization", token)
        .send({
          nome: updatedName,
          preco: 150,
          descricao: "Nova Desc",
          quantidade: 20
        });

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro alterado com sucesso" });
    });

    it("should create product via PUT if id does not exist", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto Put Upsert ${Date.now()}`;
      const randomId = "a1b2c3d4e5f6g7h8";

      const res = await request(app)
        .put(`/produtos/${randomId}`)
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 200,
          descricao: "Upsert Desc",
          quantidade: 5
        });

      expect(res.status).toBe(201);
      expect(res.body).toHaveProperty("message", "Cadastro realizado com sucesso");
      expect(res.body).toHaveProperty("_id", randomId);
    });

    it("should return 400 when updating with an already used name", async () => {
      const token = await getAdminToken();
      const name1 = `Prod A ${Date.now()}`;
      const name2 = `Prod B ${Date.now()}`;

      const p1 = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({ nome: name1, preco: 10, descricao: "D", quantidade: 1 });

      const p2 = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({ nome: name2, preco: 10, descricao: "D", quantidade: 1 });

      const res = await request(app)
        .put(`/produtos/${p2.body._id}`)
        .set("authorization", token)
        .send({ nome: name1, preco: 20, descricao: "D2", quantidade: 2 });

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });
  });

  describe("DELETE /produtos/:id", () => {
    it("should delete a product successfully", async () => {
      const token = await getAdminToken();
      const uniqueName = `Produto Para Deletar ${Date.now()}`;
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const productId = postRes.body._id;

      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro excluído com sucesso" });
    });

    it("should return DELETE_NONE when product does not exist", async () => {
      const token = await getAdminToken();
      const res = await request(app)
        .delete("/produtos/abcdef1234567890")
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Nenhum registro excluído" });
    });

    it("should not delete product if it is in a cart", async () => {
      const token = await getAdminToken();
      const regularToken = await getRegularToken();
      const uniqueName = `Produto No Carrinho ${Date.now()}`;

      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const productId = postRes.body._id;

      await request(app)
        .post("/carrinhos")
        .set("authorization", regularToken)
        .send({
          produtos: [{ idProduto: productId, quantidade: 2 }]
        });

      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(400);
      expect(res.body).toHaveProperty("message", "Não é permitido excluir produto que faz parte de carrinho");
      expect(res.body).toHaveProperty("idCarrinhos");
      expect(Array.isArray(res.body.idCarrinhos)).toBe(true);
    });
  });
});
```
