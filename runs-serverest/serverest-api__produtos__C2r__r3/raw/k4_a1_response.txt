```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Produtos API", () => {
  let counter = 0;
  function getUniqueString(prefix: string) {
    counter++;
    return `${prefix}${counter}`;
  }

  async function getAdminToken() {
    const adminEmail = `${getUniqueString("admin_prod")}@qa.com`;
    await request(app)
      .post("/usuarios")
      .send({
        nome: "Admin Test",
        email: adminEmail,
        password: "teste",
        administrador: "true"
      });

    const loginRes = await request(app)
      .post("/login")
      .send({
        email: adminEmail,
        password: "teste"
      });

    return loginRes.body.authorization;
  }

  describe("GET /produtos", () => {
    it("should list all products", async () => {
      const token = await getAdminToken();
      const uniqueName = getUniqueString("Produto List ");
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Descricao",
          quantidade: 10
        });

      const res = await request(app).get("/produtos");
      expect(res.status).toBe(200);
      expect(res.body).toHaveProperty("quantidade");
      expect(res.body).toHaveProperty("produtos");
      expect(Array.isArray(res.body.produtos)).toBe(true);
      expect(res.body.quantidade).toBeGreaterThanOrEqual(1);
    });

    it("should filter products by query parameters", async () => {
      const token = await getAdminToken();
      const uniqueName = getUniqueString("FilterUnique");
      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 250,
          descricao: "Filter test",
          quantidade: 5
        });

      const res = await request(app).get(`/produtos?nome=${uniqueName}`);
      expect(res.status).toBe(200);
      expect(res.body.quantidade).toBe(1);
      expect(res.body.produtos[0].nome).toBe(uniqueName);
    });
  });

  describe("GET /produtos/:id", () => {
    it("should return a product by valid id", async () => {
      const token = await getAdminToken();
      const uniqueName = getUniqueString("Produto GetOne ");
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 50,
          descricao: "Desc",
          quantidade: 20
        });

      const productId = postRes.body._id;

      const res = await request(app).get(`/produtos/${productId}`);
      expect(res.status).toBe(200);
      expect(res.body._id).toBe(productId);
      expect(res.body.nome).toBe(uniqueName);
    });

    it("should return 400 when product id does not exist", async () => {
      const res = await request(app).get("/produtos/abcdef0123456789");
      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Produto não encontrado" });
    });

    it("should return validation error for invalid id format", async () => {
      const res = await request(app).get("/produtos/invalid-id");
      expect(res.status).toBe(400);
    });
  });

  describe("POST /produtos", () => {
    it("should create a new product when user is admin and data is valid", async () => {
      const token = await getAdminToken();
      const uniqueName = getUniqueString("Novo Produto ");

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 499,
          descricao: "Mouse gamer",
          quantidade: 15,
          imagem: "mouse.jpg"
        });

      expect(res.status).toBe(201);
      expect(res.body).toHaveProperty("message", "Cadastro com sucesso");
      expect(res.body).toHaveProperty("_id");
    });

    it("should return 400 when product name is already used", async () => {
      const token = await getAdminToken();
      const uniqueName = getUniqueString("Duplicate Name ");

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 200,
          descricao: "Desc 2",
          quantidade: 5
        });

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });

    it("should return 401 when token is missing", async () => {
      const res = await request(app)
        .post("/produtos")
        .send({
          nome: "Sem Token",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(401);
    });

    it("should return validation error when required fields are missing", async () => {
      const token = await getAdminToken();
      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          preco: 100
        });

      expect(res.status).toBe(400);
    });
  });

  describe("PUT /produtos/:id", () => {
    it("should update an existing product", async () => {
      const token = await getAdminToken();
      const uniqueName = getUniqueString("Produto Put ");

      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Original",
          quantidade: 10
        });

      const productId = postRes.body._id;
      const updatedName = getUniqueString("Produto Updated ");

      const res = await request(app)
        .put(`/produtos/${productId}`)
        .set("authorization", token)
        .send({
          nome: updatedName,
          preco: 150,
          descricao: "Updated",
          quantidade: 20
        });

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro alterado com sucesso" });
    });

    it("should return 400 when updating with a name already used by another product", async () => {
      const token = await getAdminToken();
      const name1 = getUniqueString("Prod 1 ");
      const name2 = getUniqueString("Prod 2 ");

      await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: name1,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const postRes2 = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: name2,
          preco: 200,
          descricao: "Desc",
          quantidade: 5
        });

      const productId2 = postRes2.body._id;

      const res = await request(app)
        .put(`/produtos/${productId2}`)
        .set("authorization", token)
        .send({
          nome: name1,
          preco: 200,
          descricao: "Desc",
          quantidade: 5
        });

      expect(res.status).toBe(400);
      expect(res.body).toEqual({ message: "Já existe produto com esse nome" });
    });
  });

  describe("DELETE /produtos/:id", () => {
    it("should delete a product successfully when it is not in a cart", async () => {
      const token = await getAdminToken();
      const uniqueName = getUniqueString("Produto Delete ");

      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: uniqueName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const productId = postRes.body._id;

      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Registro excluído com sucesso" });
    });

    it("should return DELETE_NONE when deleting a non-existent product", async () => {
      const token = await getAdminToken();
      const res = await request(app)
        .delete("/produtos/abcdef0123456789")
        .set("authorization", token);

      expect(res.status).toBe(200);
      expect(res.body).toEqual({ message: "Nenhum registro excluído" });
    });

    it("should return 400 when attempting to delete a product present in a cart", async () => {
      const token = await getAdminToken();
      const userEmail = `${getUniqueString("user_cart")}@qa.com`;

      await request(app)
        .post("/usuarios")
        .send({
          nome: "Comum",
          email: userEmail,
          password: "teste",
          administrador: "false"
        });

      const userLogin = await request(app)
        .post("/login")
        .send({
          email: userEmail,
          password: "teste"
        });

      const userToken = userLogin.body.authorization;

      const prodName = getUniqueString("Prod Cart ");
      const postRes = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: prodName,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      const productId = postRes.body._id;

      await request(app)
        .post("/carrinhos")
        .set("authorization", userToken)
        .send({
          produtos: [
            { idProduto: productId, quantidade: 2 }
          ]
        });

      const res = await request(app)
        .delete(`/produtos/${productId}`)
        .set("authorization", token);

      expect(res.status).toBe(400);
      expect(res.body).toHaveProperty("message", "Não é permitido excluir produto que faz parte de carrinho");
      expect(res.body).toHaveProperty("idCarrinhos");
      expect(Array.isArray(res.body.idCarrinhos)).toBe(true);
    });
  });
});
```
