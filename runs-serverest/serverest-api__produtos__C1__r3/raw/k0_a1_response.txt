```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Produtos API", () => {
  it("should get all produtos", async () => {
    const res = await request(app).get("/produtos");
    expect(res.status).toBe(200);
    expect(res.body).toHaveProperty("quantidade");
    expect(res.body).toHaveProperty("produtos");
    expect(Array.isArray(res.body.produtos)).toBe(true);
  });

  it("should return product not found for non-existent id on getOne", async () => {
    const res = await request(app).get("/produtos/1234567890123456");
    expect(res.status).toBe(400);
    expect(res.body).toHaveProperty("message");
  });

  it("should create a produto successfully", async () => {
    const uniqueName = "Produto Teste Alfa";
    const res = await request(app)
      .post("/produtos")
      .send({
        nome: uniqueName,
        preco: 100,
        descricao: "Descricao do produto teste",
        quantidade: 10
      });

    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("_id");
    expect(res.body).toHaveProperty("message");
  });

  it("should fail to create a produto with duplicate name", async () => {
    const uniqueName = "Produto Duplicado Beta";
    const payload = {
      nome: uniqueName,
      preco: 150,
      descricao: "Descricao",
      quantidade: 5
    };

    const firstRes = await request(app).post("/produtos").send(payload);
    expect(firstRes.status).toBe(201);

    const secondRes = await request(app).post("/produtos").send(payload);
    expect(secondRes.status).toBe(400);
    expect(secondRes.body).toHaveProperty("message");
  });

  it("should delete a non-existent produto or return proper delete message", async () => {
    const res = await request(app).delete("/produtos/1234567890123456");
    expect(res.status).toBe(200);
    expect(res.body).toHaveProperty("message");
  });

  it("should update a produto successfully via put", async () => {
    const uniqueName = "Produto Original Gama";
    const createRes = await request(app)
      .post("/produtos")
      .send({
        nome: uniqueName,
        preco: 200,
        descricao: "Original",
        quantidade: 20
      });

    expect(createRes.status).toBe(201);
    const productId = createRes.body._id;

    const updatedName = "Produto Atualizado Delta";
    const putRes = await request(app)
      .put(`/produtos/${productId}`)
      .send({
        nome: updatedName,
        preco: 250,
        descricao: "Atualizado",
        quantidade: 15
      });

    expect(putRes.status).toBe(200);
    expect(putRes.body).toHaveProperty("message");
  });
});
```
