```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import * as logger from "../../src/utils/logger";

describe("errorHandler", () => {
  it("should handle headersSent when timedout is true", () => {
    const error = new Error("Test error");
    error.stack = "stack trace";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: { destroy: vi.fn() }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "warn",
      message: "Erro depois da resposta enviada: GET /test, timedout: true, Error: Test error, Stack: stack trace"
    });
    expect(req.socket.destroy).toHaveBeenCalled();
    spyLog.mockRestore();
  });

  it("should handle headersSent when timedout is false and writableEnded is true", () => {
    const error = new Error("Test error");
    error.code = "ERR_CODE";
    error.stack = "stack trace";
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/users",
      socket: { destroy: vi.fn() }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: "Erro depois da resposta enviada: POST /users, timedout: false, ERR_CODE: Test error, Stack: stack trace"
    });
    expect(req.socket.destroy).not.toHaveBeenCalled();
    spyLog.mockRestore();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "email" },
            type: "string.email",
            message: "email must be a valid email"
          }
        ]
      }
    };
    const req = {};
    const json = vi.fn().mockReturnThis();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    errorHandler(error, req, res, () => {});

    expect(status).toHaveBeenCalledWith(400);
    expect(json).toHaveBeenCalledWith({
      email: "email deve ser um email válido"
    });
  });

  it("should handle URIError", () => {
    const error = new URIError("malformed URI");
    const req = {};
    const json = vi.fn().mockReturnThis();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: "URI decode error: malformed URI" });
    expect(status).toHaveBeenCalledWith(400);
    expect(json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
    spyLog.mockRestore();
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = {};
    const json = vi.fn().mockReturnThis();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: "Entity parse error, user sending request without proper quotation marks." });
    expect(status).toHaveBeenCalledWith(400);
    expect(json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
    spyLog.mockRestore();
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large" };
    const req = {};
    const json = vi.fn().mockReturnThis();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: "Payload too large: entity.too.large" });
    expect(status).toHaveBeenCalledWith(413);
    expect(json).toHaveBeenCalledWith({ message: "Payload too large" });
    spyLog.mockRestore();
  });

  it("should handle parameters.too.many", () => {
    const error = { type: "parameters.too.many" };
    const req = {};
    const json = vi.fn().mockReturnThis();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: "Payload too large: parameters.too.many" });
    expect(status).toHaveBeenCalledWith(413);
    expect(json).toHaveBeenCalledWith({ message: "Payload too large" });
    spyLog.mockRestore();
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = {};
    const json = vi.fn().mockReturnThis();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    errorHandler(error, req, res, () => {});

    expect(status).toHaveBeenCalledWith(415);
    expect(json).toHaveBeenCalledWith({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = {};
    const json = vi.fn().mockReturnThis();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: "Unsupported content encoding: deflate" });
    expect(status).toHaveBeenCalledWith(415);
    expect(json).toHaveBeenCalledWith({ message: "Unsupported content encoding", encoding: "deflate" });
    spyLog.mockRestore();
  });

  it("should handle Z_ compressed body client error", () => {
    const error = new Error("zlib error");
    error.expose = true;
    error.statusCode = 400;
    error.code = "Z_DATA_ERROR";
    const req = {};
    const json = vi.fn().mockReturnThis();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: "Invalid compressed body: zlib error" });
    expect(status).toHaveBeenCalledWith(400);
    expect(json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
    spyLog.mockRestore();
  });

  it("should handle ETIMEDOUT error", () => {
    const error = { code: "ETIMEDOUT" };
    const req = {};
    const json = vi.fn().mockReturnThis();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    errorHandler(error, req, res, () => {});

    expect(status).toHaveBeenCalledWith(408);
    expect(json).toHaveBeenCalledWith({ message: "Timeout da requisição atingido" });
  });

  it("should handle general client error (ehErroDoCliente)", () => {
    const error = new Error("Some client error");
    error.expose = true;
    error.statusCode = 422;
    error.type = "some.type";
    const req = {};
    const json = vi.fn().mockReturnThis();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: "Client error 422: some.type" });
    expect(status).toHaveBeenCalledWith(422);
    expect(json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.type"
    });
    spyLog.mockRestore();
  });

  it("should handle internal server error with type", () => {
    const error = { type: "database.error", stack: "err-stack" };
    const req = { body: { foo: "bar" } };
    const json = vi.fn().mockReturnThis();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: "Error: database.error, Stack: err-stack, Request body: {\"foo\":\"bar\"}"
    });
    expect(status).toHaveBeenCalledWith(500);
    expect(json).toHaveBeenCalledWith(
      expect.objectContaining({
        message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
        error: "database.error"
      })
    );
    spyLog.mockRestore();
  });
});
```
