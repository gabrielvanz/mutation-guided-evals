```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import * as loggerObj from "../../src/utils/logger";

describe("errorHandler", () => {
  it("should handle headersSent when req is timedout", () => {
    const error = new Error("Timeout occurred");
    error.stack = "stack trace";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: {
        destroy: vi.fn()
      }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const spyLog = vi.spyOn(loggerObj, "log").mockImplementation(() => {});

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "warn",
      message: "Erro depois da resposta enviada: GET /test, timedout: true, undefined: Timeout occurred, Stack: stack trace"
    });
    expect(req.socket.destroy).toHaveBeenCalled();

    spyLog.mockRestore();
  });

  it("should handle headersSent when not timedout and response not ended", () => {
    const error = new Error("Some error");
    error.stack = "stack trace";
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/users",
      socket: {
        destroy: vi.fn()
      }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const spyLog = vi.spyOn(loggerObj, "log").mockImplementation(() => {});

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: "Erro depois da resposta enviada: POST /users, timedout: false, undefined: Some error, Stack: stack trace"
    });
    expect(req.socket.destroy).toHaveBeenCalled();

    spyLog.mockRestore();
  });

  it("should handle headersSent when response is already writableEnded", () => {
    const error = new Error("Some error");
    error.stack = "stack trace";
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/users",
      socket: {
        destroy: vi.fn()
      }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    const spyLog = vi.spyOn(loggerObj, "log").mockImplementation(() => {});

    errorHandler(error, req, res, () => {});

    expect(req.socket.destroy).not.toHaveBeenCalled();

    spyLog.mockRestore();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "nome" },
            type: "string.base",
            message: "nome must be a string"
          }
        ]
      }
    };
    const req = {};
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      nome: "nome deve ser uma string"
    });
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = {};
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };

    const spyLog = vi.spyOn(loggerObj, "log").mockImplementation(() => {});

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "URI decode error: Malformed URI"
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });

    spyLog.mockRestore();
  });

  it("should handle entity.parse.failed error", () => {
    const error = { type: "entity.parse.failed" };
    const req = {};
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };

    const spyLog = vi.spyOn(loggerObj, "log").mockImplementation(() => {});

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Entity parse error, user sending request without proper quotation marks."
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });

    spyLog.mockRestore();
  });

  it.each([
    "entity.too.large",
    "parameters.too.many"
  ])("should handle payload too large errors: %s", (errorType) => {
    const error = { type: errorType };
    const req = {};
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };

    const spyLog = vi.spyOn(loggerObj, "log").mockImplementation(() => {});

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: `Payload too large: ${errorType}`
    });
    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({
      message: "Payload too large"
    });

    spyLog.mockRestore();
  });

  it("should handle charset.unsupported error", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = {};
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({
      message: "Unsupported charset",
      charset: "utf-16"
    });
  });

  it("should handle encoding.unsupported error", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = {};
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };

    const spyLog = vi.spyOn(loggerObj, "log").mockImplementation(() => {});

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Unsupported content encoding: deflate"
    });
    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({
      message: "Unsupported content encoding",
      encoding: "deflate"
    });

    spyLog.mockRestore();
  });

  it("should handle Zlib client error (Z_DATA_ERROR)", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "invalid stored block lengths"
    };
    const req = {};
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };

    const spyLog = vi.spyOn(loggerObj, "log").mockImplementation(() => {});

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Invalid compressed body: invalid stored block lengths"
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });

    spyLog.mockRestore();
  });

  it("should handle ETIMEDOUT error", () => {
    const error = { code: "ETIMEDOUT" };
    const req = {};
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(408);
    expect(res.json).toHaveBeenCalledWith({
      message: "Timeout da requisição atingido"
    });
  });

  it("should handle general client error (ehErroDoCliente)", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "custom.client.error",
      message: "Bad request details"
    };
    const req = {};
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };

    const spyLog = vi.spyOn(loggerObj, "log").mockImplementation(() => {});

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Client error 422: custom.client.error"
    });
    expect(res.status).toHaveBeenCalledWith(422);
    expect(res.json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.client.error"
    });

    spyLog.mockRestore();
  });

  it("should handle internal server error with error type", () => {
    const error = {
      type: "database.error",
      stack: "db stack",
      message: "Connection failed"
    };
    const req = { body: { foo: "bar" } };
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };

    const spyLog = vi.spyOn(loggerObj, "log").mockImplementation(() => {});

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: 'Error: database.error, Stack: db stack, Request body: {"foo":"bar"}'
    });
    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith(
      expect.objectContaining({
        message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
        error: "database.error"
      })
    );

    spyLog.mockRestore();
  });
});
```
