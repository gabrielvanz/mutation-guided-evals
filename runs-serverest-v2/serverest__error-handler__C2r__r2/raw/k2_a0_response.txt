```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import logger from "../../src/utils/logger";

describe("errorHandler", () => {
  it("should handle ValidationError by returning status 400 with schema error messages", () => {
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
      headersSent: false
    };
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "email" },
            type: "string.email",
            message: "email must be a valid email"
          }
        ]
      }
    };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      email: "email deve ser um email válido"
    });
  });

  it("should handle URIError by returning status 400 and logging", () => {
    const spyLog = vi.spyOn(logger, "log");
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
      headersSent: false
    };
    const error = new URIError("Malformed URI");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "URI decode error: Malformed URI"
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
    spyLog.mockRestore();
  });

  it("should handle entity.parse.failed error by returning status 400", () => {
    const spyLog = vi.spyOn(logger, "log");
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
      headersSent: false
    };
    const error = { type: "entity.parse.failed" };

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Entity parse error, user sending request without proper quotation marks."
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
    spyLog.mockRestore();
  });

  it.each([
    { type: "entity.too.large" },
    { type: "parameters.too.many" }
  ])("should handle $type error by returning status 413", (error) => {
    const spyLog = vi.spyOn(logger, "log");
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
      headersSent: false
    };

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: `Payload too large: ${error.type}`
    });
    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
    spyLog.mockRestore();
  });

  it("should handle charset.unsupported error by returning status 415", () => {
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
      headersSent: false
    };
    const error = { type: "charset.unsupported", charset: "utf-16" };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({
      message: "Unsupported charset",
      charset: "utf-16"
    });
  });

  it("should handle encoding.unsupported error by returning status 415", () => {
    const spyLog = vi.spyOn(logger, "log");
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
      headersSent: false
    };
    const error = { type: "encoding.unsupported", encoding: "deflate" };

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Unsupported content encoding: deflate"
    });
    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({
      message: "Unsupported content encoding",
      encoding: "deflate"
    });
    spyLog.mockRestore();
  });

  it("should handle client error with Z_ prefix code by returning status 400", () => {
    const spyLog = vi.spyOn(logger, "log");
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
      headersSent: false
    };
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "Corpo corrompido"
    };

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Invalid compressed body: Corpo corrompido"
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
    spyLog.mockRestore();
  });

  it("should handle ETIMEDOUT error by returning status 408", () => {
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
      headersSent: false
    };
    const error = { code: "ETIMEDOUT" };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(408);
    expect(res.json).toHaveBeenCalledWith({
      message: "Timeout da requisição atingido"
    });
  });

  it("should handle general client error with expose true and 4xx status", () => {
    const spyLog = vi.spyOn(logger, "log");
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
      headersSent: false
    };
    const error = {
      expose: true,
      statusCode: 400,
      type: "entity.parse.something"
    };

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Client error 400: entity.parse.something"
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "entity.parse.something"
    });
    spyLog.mockRestore();
  });

  it("should handle general error with error.type by returning status 500", () => {
    const spyLog = vi.spyOn(logger, "log");
    const req = { body: {} };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
      headersSent: false
    };
    const error = { type: "database.error", stack: "Error stack" };

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith(
      expect.objectContaining({
        message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
        error: "database.error"
      })
    );
    spyLog.mockRestore();
  });

  it("should register error after response is sent if headersSent is true", () => {
    const spyLog = vi.spyOn(logger, "log");
    const socketDestroy = vi.fn();
    const req = {
      method: "GET",
      originalUrl: "/usuarios",
      timedout: false,
      socket: { destroy: socketDestroy }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };
    const error = {
      code: "ERR_ABORTED",
      message: "Aborted",
      stack: "stack"
    };

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: "Erro depois da resposta enviada: GET /usuarios, timedout: false, ERR_ABORTED: Aborted, Stack: stack"
    });
    expect(socketDestroy).toHaveBeenCalled();
    spyLog.mockRestore();
  });
});
```
