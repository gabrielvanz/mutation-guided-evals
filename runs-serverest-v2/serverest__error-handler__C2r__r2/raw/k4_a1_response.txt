```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import logger from "../../src/utils/logger";

describe("errorHandler", () => {
  it("should handle headersSent when request timed out", () => {
    const error = new Error("Timeout after response");
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/usuarios",
      socket: { destroy: vi.fn() }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      level: "warn",
      message: expect.stringContaining("timedout: true")
    });
    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("should handle headersSent when response is already writableEnded", () => {
    const error = new Error("Error after response");
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/login",
      socket: { destroy: vi.fn() }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      level: "error",
      message: expect.stringContaining("timedout: false")
    });
    expect(req.socket.destroy).not.toHaveBeenCalled();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "email" },
            type: "string.email",
            message: "email must be a valid email"
          }
        ]
      }
    };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      email: "email deve ser um email válido"
    });
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      message: "URI decode error: Malformed URI"
    });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      message: "Entity parse error, user sending request without proper quotation marks."
    });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it.each([
    ["entity.too.large"],
    ["parameters.too.many"]
  ])("should handle payload too large for type %s", (type) => {
    const error = { type };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      message: `Payload too large: ${type}`
    });
    expect(statusMock).toHaveBeenCalledWith(413);
    expect(jsonMock).toHaveBeenCalledWith({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "iso-8859-1" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(415);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Unsupported charset",
      charset: "iso-8859-1"
    });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      message: "Unsupported content encoding: deflate"
    });
    expect(statusMock).toHaveBeenCalledWith(415);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Unsupported content encoding",
      encoding: "deflate"
    });
  });

  it("should handle invalid compressed body with Z_ prefix and client error", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "invalid compressed format"
    };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      message: "Invalid compressed body: invalid compressed format"
    });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = { code: "ETIMEDOUT" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(408);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Timeout da requisição atingido"
    });
  });

  it("should handle general client error", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "some.client.error"
    };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      message: "Client error 422: some.client.error"
    });
    expect(statusMock).toHaveBeenCalledWith(422);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.client.error"
    });
  });

  it("should handle internal server error with error type", () => {
    const error = {
      type: "database.error",
      stack: "Error stack here"
    };
    const req = { body: { foo: "bar" } };
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      level: "error",
      message: "Error: database.error, Stack: Error stack here, Request body: {\"foo\":\"bar\"}"
    });
    expect(statusMock).toHaveBeenCalledWith(500);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
      error: "database.error",
      version: expect.any(String)
    });
  });
});
```
