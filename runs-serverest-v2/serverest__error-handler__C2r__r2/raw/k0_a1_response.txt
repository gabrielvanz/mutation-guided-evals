```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import loggerModule from "../../src/utils/logger";

describe("errorHandler", () => {
  function mockResponse() {
    const res: any = {};
    res.status = vi.fn().mockReturnValue(res);
    res.json = vi.fn().mockReturnValue(res);
    res.send = vi.fn().mockReturnValue(res);
    return res;
  }

  it("should handle headersSent when timedout is true", () => {
    const spyLog = vi.spyOn(loggerModule, "log").mockImplementation(() => {});
    const req: any = {
      timedout: true,
      method: "GET",
      originalUrl: "/usuarios",
      socket: { destroy: vi.fn() }
    };
    const res: any = {
      headersSent: true,
      writableEnded: false
    };
    const error = new Error("Timeout occurred");
    error.stack = "Error stack";

    errorHandler(error, req, res, vi.fn());

    expect(spyLog).toHaveBeenCalledWith({
      level: "warn",
      message: "Erro depois da resposta enviada: GET /usuarios, timedout: true, Error: Timeout occurred, Stack: Error stack"
    });
    expect(req.socket.destroy).toHaveBeenCalled();
    spyLog.mockRestore();
  });

  it("should handle headersSent when timedout is false and response is not writableEnded", () => {
    const spyLog = vi.spyOn(loggerModule, "log").mockImplementation(() => {});
    const req: any = {
      timedout: false,
      method: "POST",
      originalUrl: "/produtos",
      socket: { destroy: vi.fn() }
    };
    const res: any = {
      headersSent: true,
      writableEnded: false
    };
    const error = new Error("Some error");
    error.stack = "Error stack";

    errorHandler(error, req, res, vi.fn());

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: "Erro depois da resposta enviada: POST /produtos, timedout: false, Error: Some error, Stack: Error stack"
    });
    expect(req.socket.destroy).toHaveBeenCalled();
    spyLog.mockRestore();
  });

  it("should handle ValidationError", () => {
    const res = mockResponse();
    const req: any = { body: {} };
    const error: any = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "nome" },
            type: "any.required",
            message: '"nome" is required'
          }
        ]
      }
    };

    errorHandler(error, req, res, vi.fn());

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      nome: "nome é obrigatório"
    });
  });

  it("should handle URIError", () => {
    const spyLog = vi.spyOn(loggerModule, "log").mockImplementation(() => {});
    const res = mockResponse();
    const req: any = { body: {} };
    const error = new URIError("Malformed URI");

    errorHandler(error, req, res, vi.fn());

    expect(spyLog).toHaveBeenCalledWith({ message: "URI decode error: Malformed URI" });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
    spyLog.mockRestore();
  });

  it("should handle entity.parse.failed", () => {
    const spyLog = vi.spyOn(loggerModule, "log").mockImplementation(() => {});
    const res = mockResponse();
    const req: any = { body: {} };
    const error: any = { type: "entity.parse.failed" };

    errorHandler(error, req, res, vi.fn());

    expect(spyLog).toHaveBeenCalledWith({ message: "Entity parse error, user sending request without proper quotation marks." });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
    spyLog.mockRestore();
  });

  it("should handle entity.too.large", () => {
    const spyLog = vi.spyOn(loggerModule, "log").mockImplementation(() => {});
    const res = mockResponse();
    const req: any = { body: {} };
    const error: any = { type: "entity.too.large" };

    errorHandler(error, req, res, vi.fn());

    expect(spyLog).toHaveBeenCalledWith({ message: "Payload too large: entity.too.large" });
    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
    spyLog.mockRestore();
  });

  it("should handle parameters.too.many", () => {
    const spyLog = vi.spyOn(loggerModule, "log").mockImplementation(() => {});
    const res = mockResponse();
    const req: any = { body: {} };
    const error: any = { type: "parameters.too.many" };

    errorHandler(error, req, res, vi.fn());

    expect(spyLog).toHaveBeenCalledWith({ message: "Payload too large: parameters.too.many" });
    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
    spyLog.mockRestore();
  });

  it("should handle charset.unsupported", () => {
    const res = mockResponse();
    const req: any = { body: {} };
    const error: any = { type: "charset.unsupported", charset: "utf-16" };

    errorHandler(error, req, res, vi.fn());

    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const spyLog = vi.spyOn(loggerModule, "log").mockImplementation(() => {});
    const res = mockResponse();
    const req: any = { body: {} };
    const error: any = { type: "encoding.unsupported", encoding: "deflate" };

    errorHandler(error, req, res, vi.fn());

    expect(spyLog).toHaveBeenCalledWith({ message: "Unsupported content encoding: deflate" });
    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported content encoding", encoding: "deflate" });
    spyLog.mockRestore();
  });

  it("should handle client error with Z_ error code", () => {
    const spyLog = vi.spyOn(loggerModule, "log").mockImplementation(() => {});
    const res = mockResponse();
    const req: any = { body: {} };
    const error: any = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "Data error"
    };

    errorHandler(error, req, res, vi.fn());

    expect(spyLog).toHaveBeenCalledWith({ message: "Invalid compressed body: Data error" });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
    spyLog.mockRestore();
  });

  it("should handle ETIMEDOUT error code", () => {
    const res = mockResponse();
    const req: any = { body: {} };
    const error: any = { code: "ETIMEDOUT" };

    errorHandler(error, req, res, vi.fn());

    expect(res.status).toHaveBeenCalledWith(408);
    expect(res.json).toHaveBeenCalledWith({ message: "Timeout da requisição atingido" });
  });

  it("should handle general client error (ehErroDoCliente)", () => {
    const spyLog = vi.spyOn(loggerModule, "log").mockImplementation(() => {});
    const res = mockResponse();
    const req: any = { body: {} };
    const error: any = {
      expose: true,
      statusCode: 422,
      type: "some.type"
    };

    errorHandler(error, req, res, vi.fn());

    expect(spyLog).toHaveBeenCalledWith({ message: "Client error 422: some.type" });
    expect(res.status).toHaveBeenCalledWith(422);
    expect(res.json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.type"
    });
    spyLog.mockRestore();
  });

  it("should handle internal server error with error type", () => {
    const spyLog = vi.spyOn(loggerModule, "log").mockImplementation(() => {});
    const res = mockResponse();
    const req: any = { body: { test: true } };
    const error: any = { type: "DatabaseError", stack: "stack trace" };

    errorHandler(error, req, res, vi.fn());

    expect(spyLog).toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith(
      expect.objectContaining({
        message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
        error: "DatabaseError"
      })
    );
    spyLog.mockRestore();
  });
});
```
