```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import logger from "../../src/utils/logger";

describe("errorHandler", () => {
  it("should handle headersSent when timedout is true", () => {
    const error = new Error("Timeout after response");
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/usuarios",
      socket: {
        destroy: vi.fn()
      }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "warn",
      message: `Erro depois da resposta enviada: GET /usuarios, timedout: true, ${error.code || error.name}: ${error.message}, Stack: ${error.stack}`
    });
    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("should handle headersSent when timedout is false and res.writableEnded is true", () => {
    const error = new Error("Duplicate response");
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/login",
      socket: {
        destroy: vi.fn()
      }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: `Erro depois da resposta enviada: POST /login, timedout: false, ${error.code || error.name}: ${error.message}, Stack: ${error.stack}`
    });
    expect(req.socket.destroy).not.toHaveBeenCalled();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "nome" },
            type: "any.required",
            message: "nome is required"
          }
        ]
      }
    };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({ nome: "nome é obrigatório" });
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: "URI decode error: Malformed URI" });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: "Entity parse error, user sending request without proper quotation marks." });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it.each([
    ["entity.too.large"],
    ["parameters.too.many"]
  ])("should handle payload too large errors: %s", (type) => {
    const error = { type };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: `Payload too large: ${type}` });
    expect(statusMock).toHaveBeenCalledWith(413);
    expect(jsonMock).toHaveBeenCalledWith({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(415);
    expect(jsonMock).toHaveBeenCalledWith({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: "Unsupported content encoding: deflate" });
    expect(statusMock).toHaveBeenCalledWith(415);
    expect(jsonMock).toHaveBeenCalledWith({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle client error with Z_ prefix code", () => {
    const error = new Error("Z_DATA_ERROR compressed body invalid");
    error.expose = true;
    error.statusCode = 400;
    error.code = "Z_DATA_ERROR";

    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: `Invalid compressed body: ${error.message}` });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = { code: "ETIMEDOUT" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(408);
    expect(jsonMock).toHaveBeenCalledWith({ message: "Timeout da requisição atingido" });
  });

  it("should handle generic client error (ehErroDoCliente)", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "some.client.error",
      message: "Bad request details"
    };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({ message: "Client error 422: some.client.error" });
    expect(statusMock).toHaveBeenCalledWith(422);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.client.error"
    });
  });

  it("should handle internal server error with error type", () => {
    const error = {
      type: "internal.failure",
      message: "Something broke",
      stack: "Error stack"
    };
    const req = { body: { test: true } };
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: "Error: internal.failure, Stack: Error stack, Request body: {\"test\":true}"
    });
    expect(statusMock).toHaveBeenCalledWith(500);
    expect(jsonMock).toHaveBeenCalledWith(
      expect.objectContaining({
        message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
        error: "internal.failure"
      })
    );
  });
});
```
