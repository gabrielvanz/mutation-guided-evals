```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API tests", () => {
  it("should return 401 when accessing a protected route without authorization header", async () => {
    const response = await request(app).post("/carrinhos").send({
      produtos: []
    });

    expect(response.status).toBe(401);
    expect(response.body).toEqual({
      message: "Token de acesso ausente, inválido, expirado ou usuário do token não existe mais"
    });
  });

  it("should return 401 when accessing a protected route with a malformed token format", async () => {
    const response = await request(app)
      .post("/carrinhos")
      .set("authorization", "InvalidFormat token123")
      .send({
        produtos: []
      });

    expect(response.status).toBe(401);
    expect(response.body).toEqual({
      message: "Token de acesso ausente, inválido, expirado ou usuário do token não existe mais"
    });
  });

  it("should return 401 when accessing an admin route without authorization header", async () => {
    const response = await request(app).post("/produtos").send({
      nome: "Produto Teste",
      preco: 100,
      descricao: "Teste",
      quantidade: 10
    });

    expect(response.status).toBe(401);
    expect(response.body).toEqual({
      message: "Token de acesso ausente, inválido, expirado ou usuário do token não existe mais"
    });
  });

  it("should return 403 when a non-admin user tries to access an admin route", async () => {
    const userEmail = "user_non_admin_test@test.com";
    await request(app).post("/usuarios").send({
      nome: "Usuario Comum",
      email: userEmail,
      password: "teste",
      administrador: "false"
    });

    const loginRes = await request(app).post("/login").send({
      email: userEmail,
      password: "teste"
    });

    const token = loginRes.body.authorization;

    const response = await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send({
        nome: "Produto Teste",
        preco: 100,
        descricao: "Teste",
        quantidade: 10
      });

    expect(response.status).toBe(403);
    expect(response.body).toEqual({
      message: "Rota exclusiva para administradores"
    });
  });

  it("should handle URIError and return 400 with proper message", async () => {
    const response = await request(app).get("/usuarios/%E0%A2%A");

    expect(response.status).toBe(400);
    expect(response.body).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle invalid JSON body parse error and return 400", async () => {
    const response = await request(app)
      .post("/usuarios")
      .set("content-type", "application/json")
      .send("{invalid json");

    expect(response.status).toBe(400);
    expect(response.body).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });
});
```
