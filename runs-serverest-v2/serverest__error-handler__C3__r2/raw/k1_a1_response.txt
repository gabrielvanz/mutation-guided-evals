```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import * as logger from "../../src/utils/logger";

describe("errorHandler", () => {
  it("should handle headersSent when timedout is true", () => {
    const error = new Error("Test error");
    error.stack = "stack trace";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: { destroy: vi.fn() }
    };
    const socketDestroy = req.socket.destroy;
    const res = { headersSent: true, writableEnded: false };

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      level: "warn",
      message: "Erro depois da resposta enviada: GET /test, timedout: true, undefined: Test error, Stack: stack trace"
    });
    expect(socketDestroy).toHaveBeenCalled();
    logSpy.mockRestore();
  });

  it("should handle headersSent when timedout is false and writableEnded is true", () => {
    const error = new Error("Test error");
    error.stack = "stack trace";
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/test",
      socket: { destroy: vi.fn() }
    };
    const socketDestroy = req.socket.destroy;
    const res = { headersSent: true, writableEnded: true };

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      level: "error",
      message: "Erro depois da resposta enviada: POST /test, timedout: false, undefined: Test error, Stack: stack trace"
    });
    expect(socketDestroy).not.toHaveBeenCalled();
    logSpy.mockRestore();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "email" },
            type: "string.email",
            message: "email must be a valid email"
          }
        ]
      }
    };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      email: "email deve ser um email válido"
    });
  });

  it("should handle URIError", () => {
    const error = new URIError("malformed URI");
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({ message: "URI decode error: malformed URI" });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
    logSpy.mockRestore();
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({ message: "Entity parse error, user sending request without proper quotation marks." });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
    logSpy.mockRestore();
  });

  it.each([
    "entity.too.large",
    "parameters.too.many"
  ])("should handle payload too large types: %s", (type) => {
    const error = { type };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({ message: `Payload too large: ${type}` });
    expect(statusMock).toHaveBeenCalledWith(413);
    expect(jsonMock).toHaveBeenCalledWith({ message: "Payload too large" });
    logSpy.mockRestore();
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(415);
    expect(jsonMock).toHaveBeenCalledWith({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({ message: "Unsupported content encoding: deflate" });
    expect(statusMock).toHaveBeenCalledWith(415);
    expect(jsonMock).toHaveBeenCalledWith({ message: "Unsupported content encoding", encoding: "deflate" });
    logSpy.mockRestore();
  });

  it("should handle client error with Z_ prefix code", () => {
    const error = new Error("zlib error");
    error.expose = true;
    error.statusCode = 400;
    error.code = "Z_DATA_ERROR";
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({ message: "Invalid compressed body: zlib error" });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
    logSpy.mockRestore();
  });

  it("should handle ETIMEDOUT code", () => {
    const error = { code: "ETIMEDOUT" };
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(408);
    expect(jsonMock).toHaveBeenCalledWith({ message: "Timeout da requisição atingido" });
  });

  it("should handle generic client error (ehErroDoCliente)", () => {
    const error = new Error("Bad Request");
    error.expose = true;
    error.statusCode = 400;
    error.type = "custom.error.type";
    const req = {};
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({ message: "Client error 400: custom.error.type" });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.error.type"
    });
    logSpy.mockRestore();
  });

  it("should handle server error with error type", () => {
    const error = new Error("Internal Server Error");
    error.type = "some.server.error";
    error.stack = "some stack";
    const req = { body: { foo: "bar" } };
    const jsonMock = vi.fn();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = { headersSent: false, status: statusMock };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      level: "error",
      message: "Error: some.server.error, Stack: some stack, Request body: {\"foo\":\"bar\"}"
    });
    expect(statusMock).toHaveBeenCalledWith(500);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
      error: "some.server.error",
      version: expect.any(String)
    });
    logSpy.mockRestore();
  });
});
```
