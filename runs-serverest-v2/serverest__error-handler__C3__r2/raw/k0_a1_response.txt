```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import loggerObj from "../../src/utils/logger";

describe("errorHandler", () => {
  it("should handle ValidationError by returning status 400 with schema error message", () => {
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "email" },
            type: "string.email",
            message: "email must be a valid email"
          }
        ]
      }
    };
    const req = { body: {} };
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      email: "email deve ser um email válido"
    });
  });

  it("should handle URIError by returning status 400 and logging", () => {
    const spy = vi.spyOn(loggerObj, "log");
    const error = new URIError("Malformed URI");
    const req = { body: {} };
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, req, res, () => {});

    expect(spy).toHaveBeenCalledWith({ message: "URI decode error: Malformed URI" });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed error by returning status 400", () => {
    const spy = vi.spyOn(loggerObj, "log");
    const error = { type: "entity.parse.failed" };
    const req = { body: {} };
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, req, res, () => {});

    expect(spy).toHaveBeenCalledWith({ message: "Entity parse error, user sending request without proper quotation marks." });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it.each([
    "entity.too.large",
    "parameters.too.many"
  ])("should handle payload too large errors (%s) with status 413", (errorType) => {
    const spy = vi.spyOn(loggerObj, "log");
    const error = { type: errorType };
    const req = { body: {} };
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, req, res, () => {});

    expect(spy).toHaveBeenCalledWith({ message: `Payload too large: ${errorType}` });
    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
  });

  it("should handle charset.unsupported error with status 415", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = { body: {} };
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported error with status 415 and log", () => {
    const spy = vi.spyOn(loggerObj, "log");
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = { body: {} };
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, req, res, () => {});

    expect(spy).toHaveBeenCalledWith({ message: "Unsupported content encoding: deflate" });
    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle client error with Zlib code by returning status 400", () => {
    const spy = vi.spyOn(loggerObj, "log");
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "corrupt deflate stream"
    };
    const req = { body: {} };
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, req, res, () => {});

    expect(spy).toHaveBeenCalledWith({ message: "Invalid compressed body: corrupt deflate stream" });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error by returning status 408", () => {
    const error = { code: "ETIMEDOUT" };
    const req = { body: {} };
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(408);
    expect(res.json).toHaveBeenCalledWith({ message: "Timeout da requisição atingido" });
  });

  it("should handle general client error when expose is true and status is 4xx", () => {
    const spy = vi.spyOn(loggerObj, "log");
    const req = { body: {} };
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    const generalClientError = {
      expose: true,
      statusCode: 422,
      type: "custom.error.type"
    };

    errorHandler(generalClientError, req, res, () => {});

    expect(spy).toHaveBeenCalledWith({ message: "Client error 422: custom.error.type" });
    expect(res.status).toHaveBeenCalledWith(422);
    expect(res.json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.error.type"
    });
  });

  it("should handle internal server error (500) when error has a type", () => {
    const spy = vi.spyOn(loggerObj, "log");
    const error = { type: "some.server.error", stack: "stack-trace" };
    const req = { body: { foo: "bar" } };
    const res = {
      headersSent: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, req, res, () => {});

    expect(spy).toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith(
      expect.objectContaining({
        message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
        error: "some.server.error"
      })
    );
  });

  it("should handle error when headersSent is true (registrarErroDepoisDaResposta)", () => {
    const spy = vi.spyOn(loggerObj, "log");
    const socketDestroy = vi.fn();
    const error = {
      message: "some error",
      stack: "stack",
      code: "ERR_CODE"
    };
    const req = {
      timedout: true,
      method: "POST",
      originalUrl: "/usuarios",
      socket: { destroy: socketDestroy }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    errorHandler(error, req, res, () => {});

    expect(spy).toHaveBeenCalledWith(
      expect.objectContaining({
        level: "warn",
        message: expect.stringContaining("Erro depois da resposta enviada: POST /usuarios")
      })
    );
    expect(socketDestroy).toHaveBeenCalled();
  });
});
```
