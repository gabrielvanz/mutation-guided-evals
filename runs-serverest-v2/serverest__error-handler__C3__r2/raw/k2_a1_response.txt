```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import * as logger from "../../src/utils/logger";

describe("errorHandler", () => {
  it("should handle headersSent when timedout is true", () => {
    const error = new Error("Test error");
    error.stack = "stack trace";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: {
        destroy: vi.fn()
      }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "warn",
      message: "Erro depois da resposta enviada: GET /test, timedout: true, Error: Test error, Stack: stack trace"
    });
    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("should handle headersSent when timedout is false and writableEnded is true", () => {
    const error = new Error("Test error");
    error.stack = "stack trace";
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/users",
      socket: {
        destroy: vi.fn()
      }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: "Erro depois da resposta enviada: POST /users, timedout: false, Error: Test error, Stack: stack trace"
    });
    expect(req.socket.destroy).not.toHaveBeenCalled();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "nome" },
            type: "string.base",
            message: "nome deve ser uma string"
          }
        ]
      }
    };

    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, {}, res, () => {});

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      nome: "nome deve ser uma string"
    });
  });

  it("should handle URIError", () => {
    const error = new URIError("malformed URI");
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "URI decode error: malformed URI"
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed error", () => {
    const error = {
      type: "entity.parse.failed"
    };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Entity parse error, user sending request without proper quotation marks."
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it.each([
    ["entity.too.large"],
    ["parameters.too.many"]
  ])("should handle payload too large errors: %s", (type) => {
    const error = { type };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: `Payload too large: ${type}`
    });
    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({
      message: "Payload too large"
    });
  });

  it("should handle charset.unsupported error", () => {
    const error = {
      type: "charset.unsupported",
      charset: "utf-16"
    };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, {}, res, () => {});

    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({
      message: "Unsupported charset",
      charset: "utf-16"
    });
  });

  it("should handle encoding.unsupported error", () => {
    const error = {
      type: "encoding.unsupported",
      encoding: "deflate"
    };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Unsupported content encoding: deflate"
    });
    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({
      message: "Unsupported content encoding",
      encoding: "deflate"
    });
  });

  it("should handle client error with Z_ prefix in code", () => {
    const error = new Error("invalid compressed format");
    error.expose = true;
    error.statusCode = 400;
    error.code = "Z_DATA_ERROR";

    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Invalid compressed body: invalid compressed format"
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = {
      code: "ETIMEDOUT"
    };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, {}, res, () => {});

    expect(res.status).toHaveBeenCalledWith(408);
    expect(res.json).toHaveBeenCalledWith({
      message: "Timeout da requisição atingido"
    });
  });

  it("should handle generic client error (expose true, status 4xx)", () => {
    const error = new Error("Bad request");
    error.expose = true;
    error.statusCode = 422;
    error.type = "custom.type";

    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Client error 422: custom.type"
    });
    expect(res.status).toHaveBeenCalledWith(422);
    expect(res.json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.type"
    });
  });

  it("should handle internal server error with error type", () => {
    const error = {
      type: "database.error",
      stack: "db stack"
    };
    const req = {
      body: { foo: "bar" }
    };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: 'Error: database.error, Stack: db stack, Request body: {"foo":"bar"}'
    });
    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith(
      expect.objectContaining({
        message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
        error: "database.error"
      })
    );
  });
});
```
