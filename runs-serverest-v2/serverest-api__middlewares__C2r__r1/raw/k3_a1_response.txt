```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on protected routes", async () => {
      const response = await request(app).post("/carrinhos").send({
        produtos: []
      });

      expect(response.status).toBe(401);
      expect(response.body).toHaveProperty("message");
    });

    it("should return 401 when token format is invalid (missing Bearer)", async () => {
      const response = await request(app)
        .post("/carrinhos")
        .set("authorization", "invalidtoken")
        .send({ produtos: [] });

      expect(response.status).toBe(401);
      expect(response.body).toHaveProperty("message");
    });

    it("should return 401 when token is invalid/malformed JWT", async () => {
      const response = await request(app)
        .post("/carrinhos")
        .set("authorization", "Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.invalid.signature")
        .send({ produtos: [] });

      expect(response.status).toBe(401);
      expect(response.body).toHaveProperty("message");
    });

    it("should return 403 when user is not an administrator on admin-only routes", async () => {
      const email = "nonadmin@qa.com";
      const password = "teste";

      await request(app).post("/usuarios").send({
        nome: "Non Admin",
        email,
        password,
        administrador: "false"
      });

      const loginRes = await request(app).post("/login").send({
        email,
        password
      });

      const token = loginRes.body.authorization;

      const response = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Teste",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(response.status).toBe(403);
      expect(response.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 for JSON parse errors (entity.parse.failed)", async () => {
      const response = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send("{invalid-json");

      expect(response.status).toBe(400);
      expect(response.body).toEqual({
        message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
      });
    });

    it("should return 400 for invalid URI parameters", async () => {
      const response = await request(app).get("/usuarios/%E0%A4%A");

      expect(response.status).toBe(400);
      expect(response.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });
  });
});
```
