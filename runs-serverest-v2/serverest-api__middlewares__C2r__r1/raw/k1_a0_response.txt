```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on protected routes", async () => {
      const res = await request(app).post("/carrinhos").send({
        produtos: []
      });
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token format is invalid (missing Bearer prefix)", async () => {
      const res = await request(app)
        .post("/carrinhos")
        .set("authorization", "invalidtoken")
        .send({ produtos: [] });
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is malformed or invalid", async () => {
      const res = await request(app)
        .post("/carrinhos")
        .set("authorization", "Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.invalid.signature")
        .send({ produtos: [] });
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 403 when a non-admin tries to access an admin-only route", async () => {
      const uniqueSuffix = Date.now() + Math.random();
      const regularUser = {
        nome: `User ${uniqueSuffix}`,
        email: `user_${uniqueSuffix}@qa.com`,
        password: "teste",
        administrador: "false"
      };

      await request(app).post("/usuarios").send(regularUser);

      const loginRes = await request(app).post("/login").send({
        email: regularUser.email,
        password: regularUser.password
      });

      const token = loginRes.body.authorization;

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: `Produto ${uniqueSuffix}`,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(403);
      expect(res.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 with schema validation message when validation fails", async () => {
      const res = await request(app).post("/usuarios").send({
        nome: ""
      });

      expect(res.status).toBe(400);
      expect(res.body).toHaveProperty("message");
    });

    it("should return 400 when malformed JSON body with missing quotation marks is sent", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send('{"nome": unquoted}');

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
      });
    });

    it("should return 400 when URI decode error occurs via invalid percent-encoding in query params", async () => {
      const res = await request(app).get("/usuarios?_id=%E0%A1");

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });
  });
});
```
