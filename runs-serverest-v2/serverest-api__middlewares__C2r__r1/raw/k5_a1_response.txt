```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API Integration Tests", () => {
  describe("Authentication Middleware & Error Handler", () => {
    it("should return 401 when token is missing on protected route", async () => {
      const res = await request(app).post("/carrinhos").send({
        produtos: [{ id_produto: "any", quantidade: 1 }]
      });

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is malformed", async () => {
      const res = await request(app)
        .post("/carrinhos")
        .set("authorization", "InvalidFormat token123")
        .send({
          produtos: [{ id_produto: "any", quantidade: 1 }]
        });

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 401 when token is invalid/expired or user does not exist", async () => {
      const res = await request(app)
        .post("/carrinhos")
        .set("authorization", "Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.invalidtoken")
        .send({
          produtos: [{ id_produto: "any", quantidade: 1 }]
        });

      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado"
      });
    });

    it("should return 403 when non-admin tries to access admin route", async () => {
      const regularUser = {
        nome: "User Test",
        email: "user_test_middleware@qa.com",
        password: "teste",
        administrador: "false"
      };

      await request(app).post("/usuarios").send(regularUser);
      const loginRes = await request(app).post("/login").send({
        email: regularUser.email,
        password: regularUser.password
      });

      const token = loginRes.body.authorization;

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Test",
          preco: 10,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(403);
      expect(res.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });

    it("should handle URIError and return 400 with specific message", async () => {
      const res = await request(app).get("/usuarios/%E0%A4%A");

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });

    it("should handle invalid JSON payload and return 400 for entity parse failed", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send('{"nome": unquoted}');

      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
      });
    });
  });
});
```
