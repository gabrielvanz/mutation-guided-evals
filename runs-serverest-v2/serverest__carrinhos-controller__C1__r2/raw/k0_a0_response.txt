```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import carrinhosService from "../../src/services/carrinhos-service";
import constant from "../../src/utils/constants";

describe("carrinhos-controller", () => {
  let req: any;
  let res: any;

  beforeEach(() => {
    vi.restoreAllMocks();
    req = {
      query: {},
      params: {},
      headers: {},
      body: {}
    };
    res = {
      status: vi.fn().mockReturnThis(),
      send: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
      locals: {
        etapas: {
          regras: vi.fn(),
          regra: vi.fn(),
          abrir: vi.fn()
        }
      }
    };
  });

  describe("get", () => {
    it("should return 200 with carrinhos and quantidade", async () => {
      const mockCarrinhos = [{ _id: "1" }];
      vi.spyOn(carrinhosService, "getAll").mockResolvedValue(mockCarrinhos as any);

      await carrinhosController.get(req, res);

      expect(carrinhosService.getAll).toHaveBeenCalledWith(req.query);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: 1,
        carrinhos: mockCarrinhos
      });
    });
  });

  describe("getOne", () => {
    it("should return 400 when carrinho is not found", async () => {
      req.params.id = "invalid";
      vi.spyOn(carrinhosService, "getOne").mockResolvedValue(null);

      await carrinhosController.getOne(req, res);

      expect(carrinhosService.getOne).toHaveBeenCalledWith("invalid");
      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });

    it("should return 200 with carrinho when found", async () => {
      req.params.id = "valid";
      const mockCarrinho = { _id: "valid" };
      vi.spyOn(carrinhosService, "getOne").mockResolvedValue(mockCarrinho as any);

      await carrinhosController.getOne(req, res);

      expect(carrinhosService.getOne).toHaveBeenCalledWith("valid");
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });
  });

  describe("post", () => {
    it("should return 400 when user already has a cart", async () => {
      req.headers.authorization = "Bearer token";
      req.body.produtos = [{ idProduto: "p1", quantidade: "1" }];
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: { _id: "c1" }
      } as any);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return 400 when products are duplicated", async () => {
      req.headers.authorization = "Bearer token";
      req.body.produtos = [
        { idProduto: "p1", quantidade: "1" },
        { idProduto: "p1", quantidade: "2" }
      ];
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue(["p1"]);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["p1"]
      });
    });

    it("should return error status when getProdutosComPrecoUnitarioOuErro returns error for product not found", async () => {
      req.headers.authorization = "Bearer token";
      req.body.produtos = [{ idProduto: "p1", quantidade: "1" }];
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        error: {
          statusCode: 400,
          message: constant.PRODUCT_NOT_FOUND,
          item: { idProduto: "p1" },
          index: 0
        }
      } as any);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.PRODUCT_NOT_FOUND,
        item: { idProduto: "p1" }
      });
    });

    it("should return error status when getProdutosComPrecoUnitarioOuErro returns error for stock", async () => {
      req.headers.authorization = "Bearer token";
      req.body.produtos = [{ idProduto: "p1", quantidade: "10" }];
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        error: {
          statusCode: 400,
          message: constant.INSUFFICIENT_STOCK,
          item: { idProduto: "p1" },
          index: 0
        }
      } as any);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.INSUFFICIENT_STOCK,
        item: { idProduto: "p1" }
      });
    });

    it("should return 400 when totals are not safe", async () => {
      req.headers.authorization = "Bearer token";
      req.body.produtos = [{ idProduto: "p1", quantidade: "1" }];
      const produtosComPreco = [{ idProduto: "p1", quantidade: 1, precoUnitario: 10 }];
      
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: produtosComPreco
      } as any);
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(10);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(1);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(false);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });

    it("should successfully create cart when all validations pass", async () => {
      req.headers.authorization = "Bearer token";
      req.body.produtos = [{ idProduto: "p1", quantidade: "1" }];
      const produtosComPreco = [{ idProduto: "p1", quantidade: 1, precoUnitario: 10 }];

      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: produtosComPreco
      } as any);
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(10);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(1);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(true);
      vi.spyOn(carrinhosService, "baixarEstoque").mockResolvedValue(undefined);
      vi.spyOn(carrinhosService, "criarCarrinho").mockResolvedValue({ _id: "c_new" } as any);

      await carrinhosController.post(req, res);

      expect(carrinhosService.baixarEstoque).toHaveBeenCalledWith(produtosComPreco);
      expect(carrinhosService.criarCarrinho).toHaveBeenCalledWith({
        produtos: produtosComPreco,
        precoTotal: 10,
        quantidadeTotal: 1,
        idUsuario: "u1"
      });
      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.POST_SUCCESS,
        _id: "c_new"
      });
    });
  });

  describe("cancelarCompra", () => {
    it("should replenish stock and remove cart when user has a cart", async () => {
      req.headers.authorization = "Bearer token";
      const carrinho = { _id: "c1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([carrinho] as any);
      vi.spyOn(carrinhosService, "removeCarrinho").mockResolvedValue(undefined);

      await carrinhosController.cancelarCompra(req, res);

      expect(carrinhosService.removeCarrinho).toHaveBeenCalledWith(carrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}`
      });
    });

    it("should return NO_CART when user does not have a cart", async () => {
      req.headers.authorization = "Bearer token";
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([] as any);

      await carrinhosController.cancelarCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase and delete cart when user has a cart", async () => {
      req.headers.authorization = "Bearer token";
      const carrinho = { _id: "c1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([carrinho] as any);
      vi.spyOn(carrinhosService, "concluiCompra").mockResolvedValue(undefined);

      await carrinhosController.concluirCompra(req, res);

      expect(carrinhosService.concluiCompra).toHaveBeenCalledWith(carrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("should return NO_CART when user does not have a cart", async () => {
      req.headers.authorization = "Bearer token";
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([] as any);

      await carrinhosController.concluirCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});
```
