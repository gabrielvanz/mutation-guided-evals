```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import logger from "../../src/utils/logger";

describe("errorHandler middleware", () => {
  let req: any;
  let res: any;
  let next: any;

  beforeEach(() => {
    vi.clearAllMocks();
    req = {
      method: "GET",
      originalUrl: "/test",
      body: {},
      timedout: false,
      socket: {
        destroy: vi.fn(),
      },
    };
    res = {
      headersSent: false,
      writableEnded: false,
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis(),
    };
    next = vi.fn();
  });

  it("should handle headersSent true when response is not writable ended", () => {
    res.headersSent = true;
    res.writableEnded = false;
    const error = {
      name: "Error",
      message: "Test error",
      stack: "stacktrace",
    };

    vi.spyOn(logger, "log").mockImplementation(() => {});

    errorHandler(error, req, res, next);

    expect(logger.log).toHaveBeenCalled();
    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("should handle headersSent true when response is writable ended and timedout", () => {
    res.headersSent = true;
    res.writableEnded = true;
    req.timedout = true;
    const error = {
      name: "Error",
      message: "Test error",
      stack: "stacktrace",
    };

    vi.spyOn(logger, "log").mockImplementation(() => {});

    errorHandler(error, req, res, next);

    expect(logger.log).toHaveBeenCalled();
    expect(req.socket.destroy).not.toHaveBeenCalled();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "email" },
            type: "string.base",
            message: "email must be a string",
          },
        ],
      },
    };

    errorHandler(error, req, res, next);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      email: "email deve ser uma string",
    });
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    vi.spyOn(logger, "log").mockImplementation(() => {});

    errorHandler(error, req, res, next);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente.",
    });
    expect(logger.log).toHaveBeenCalled();
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    vi.spyOn(logger, "log").mockImplementation(() => {});

    errorHandler(error, req, res, next);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225",
    });
    expect(logger.log).toHaveBeenCalled();
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large" };
    vi.spyOn(logger, "log").mockImplementation(() => {});

    errorHandler(error, req, res, next);

    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
    expect(logger.log).toHaveBeenCalled();
  });

  it("should handle parameters.too.many", () => {
    const error = { type: "parameters.too.many" };
    vi.spyOn(logger, "log").mockImplementation(() => {});

    errorHandler(error, req, res, next);

    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
    expect(logger.log).toHaveBeenCalled();
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-8" };

    errorHandler(error, req, res, next);

    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported charset", charset: "utf-8" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    vi.spyOn(logger, "log").mockImplementation(() => {});

    errorHandler(error, req, res, next);

    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported content encoding", encoding: "deflate" });
    expect(logger.log).toHaveBeenCalled();
  });

  it("should handle client error with Z_ prefix and expose true", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "Compressed data error",
    };
    vi.spyOn(logger, "log").mockImplementation(() => {});

    errorHandler(error, req, res, next);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header.",
    });
    expect(logger.log).toHaveBeenCalled();
  });

  it("should handle ETIMEDOUT code", () => {
    const error = { code: "ETIMEDOUT" };

    errorHandler(error, req, res, next);

    expect(res.status).toHaveBeenCalledWith(408);
    expect(res.json).toHaveBeenCalledWith({ message: "Timeout da requisição atingido" });
  });

  it("should handle client error in general (ehErroDoCliente)", () => {
    const error = {
      expose: true,
      statusCode: 422,
      type: "some.client.error",
    };
    vi.spyOn(logger, "log").mockImplementation(() => {});

    errorHandler(error, req, res, next);

    expect(res.status).toHaveBeenCalledWith(422);
    expect(res.json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.client.error",
    });
    expect(logger.log).toHaveBeenCalled();
  });

  it("should handle general 500 error with type", () => {
    const error = {
      type: "some.internal.error",
      message: "Internal issue",
      stack: "some stack",
    };
    vi.spyOn(logger, "log").mockImplementation(() => {});

    errorHandler(error, req, res, next);

    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith(
      expect.objectContaining({
        message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
        error: "some.internal.error",
      })
    );
    expect(logger.log).toHaveBeenCalled();
  });

  it("should handle general 500 error without type", () => {
    const error = {
      message: "Unknown error",
      stack: "unknown stack",
    };
    vi.spyOn(logger, "log").mockImplementation(() => {});

    errorHandler(error, req, res, next);

    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith(
      expect.objectContaining({
        message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
        error: expect.any(Object),
      })
    );
    expect(logger.log).toHaveBeenCalled();
  });
});
```
