```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import logger from "../../src/utils/logger";

describe("errorHandler", () => {
  function mockResponse() {
    const res: any = {};
    res.status = vi.fn().mockReturnValue(res);
    res.json = vi.fn().mockReturnValue(res);
    res.send = vi.fn().mockReturnValue(res);
    res.headersSent = false;
    res.writableEnded = false;
    return res;
  }

  it("should handle headers already sent when timedout is true", () => {
    const req: any = {
      timedout: true,
      method: "GET",
      originalUrl: "/usuarios",
      socket: { destroy: vi.fn() }
    };
    const res: any = mockResponse();
    res.headersSent = true;
    const error: any = new Error("Timeout occurred");
    error.code = "ETIMEDOUT";

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith(
      expect.objectContaining({
        level: "warn",
        message: expect.stringContaining("timedout: true")
      })
    );
    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("should handle headers already sent when timedout is false", () => {
    const req: any = {
      timedout: false,
      method: "POST",
      originalUrl: "/produtos",
      socket: { destroy: vi.fn() }
    };
    const res: any = mockResponse();
    res.headersSent = true;
    const error: any = new Error("Response already sent");

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith(
      expect.objectContaining({
        level: "error",
        message: expect.stringContaining("timedout: false")
      })
    );
    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("should handle ValidationError", () => {
    const req: any = { body: {} };
    const res: any = mockResponse();
    const error: any = new Error("Validation Error");
    error.name = "ValidationError";
    error.details = {
      body: [
        {
          context: { label: "email" },
          type: "string.email",
          message: "email must be a valid email"
        }
      ]
    };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      email: "email deve ser um email válido"
    });
  });

  it("should handle URIError", () => {
    const req: any = { body: {} };
    const res: any = mockResponse();
    const error: any = new Error("URI malformed");
    error.name = "URIError";

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const req: any = { body: {} };
    const res: any = mockResponse();
    const error: any = new Error("Parse failed");
    error.type = "entity.parse.failed";

    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const req: any = { body: {} };
    const res: any = mockResponse();
    const error: any = new Error("Too large");
    error.type = "entity.too.large";

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
  });

  it("should handle parameters.too.many", () => {
    const req: any = { body: {} };
    const res: any = mockResponse();
    const error: any = new Error("Too many params");
    error.type = "parameters.too.many";

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const req: any = { body: {} };
    const res: any = mockResponse();
    const error: any = new Error("Unsupported charset");
    error.type = "charset.unsupported";
    error.charset = "utf-16";

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({
      message: "Unsupported charset",
      charset: "utf-16"
    });
  });

  it("should handle encoding.unsupported", () => {
    const req: any = { body: {} };
    const res: any = mockResponse();
    const error: any = new Error("Unsupported encoding");
    error.type = "encoding.unsupported";
    error.encoding = "br";

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({
      message: "Unsupported content encoding",
      encoding: "br"
    });
  });

  it("should handle client error with Z_ prefix code", () => {
    const req: any = { body: {} };
    const res: any = mockResponse();
    const error: any = new Error("Data error");
    error.expose = true;
    error.statusCode = 400;
    error.code = "Z_DATA_ERROR";

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const req: any = { body: {} };
    const res: any = mockResponse();
    const error: any = new Error("Connection timeout");
    error.code = "ETIMEDOUT";

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(408);
    expect(res.json).toHaveBeenCalledWith({
      message: "Timeout da requisição atingido"
    });
  });

  it("should handle generic client error (ehErroDoCliente)", () => {
    const req: any = { body: {} };
    const res: any = mockResponse();
    const error: any = new Error("Bad request");
    error.expose = true;
    error.statusCode = 400;
    error.type = "custom.client.error";

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.client.error"
    });
  });

  it("should handle server error with error type", () => {
    const req: any = { body: { test: true } };
    const res: any = mockResponse();
    const error: any = new Error("Internal server error");
    error.type = "database.error";
    error.stack = "Error stack";

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith({
      message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
      error: "database.error",
      version: expect.any(String)
    });
  });
});
```
