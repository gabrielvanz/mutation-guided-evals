```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import * as loggerModule from "../../src/utils/logger";

describe("errorHandler", () => {
  function mockResponse() {
    const res: any = {};
    res.status = vi.fn().mockReturnValue(res);
    res.json = vi.fn().mockReturnValue(res);
    res.send = vi.fn().mockReturnValue(res);
    return res;
  }

  it("should handle headersSent when timedout is true, logging warn level and message details", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = new Error("Timeout error details");
    (error as any).code = "ETIMEDOUT";
    error.stack = "ErrorStackTest";
    const req: any = { timedout: true, method: "GET", originalUrl: "/test-url", socket: { destroy: vi.fn() } };
    const res: any = { headersSent: true, writableEnded: false };
    
    errorHandler(error, req, res, () => {});
    expect(res.writableEnded).toBe(false);
    expect(req.socket.destroy).toHaveBeenCalled();
    expect(logSpy).toHaveBeenCalledWith({
      level: "warn",
      message: "Erro depois da resposta enviada: GET /test-url, timedout: true, ETIMEDOUT: Timeout error details, Stack: ErrorStackTest"
    });
    logSpy.mockRestore();
  });

  it("should handle headersSent when timedout is false and use error.name if code is missing", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = new Error("Normal error details");
    (error as any).name = "CustomErrorName";
    error.stack = "CustomStack";
    const req: any = { timedout: false, method: "POST", originalUrl: "/users", socket: { destroy: vi.fn() } };
    const res: any = { headersSent: true, writableEnded: false };
    
    errorHandler(error, req, res, () => {});
    expect(req.socket.destroy).toHaveBeenCalled();
    expect(logSpy).toHaveBeenCalledWith({
      level: "error",
      message: "Erro depois da resposta enviada: POST /users, timedout: false, CustomErrorName: Normal error details, Stack: CustomStack"
    });
    logSpy.mockRestore();
  });

  it("should handle headersSent when writableEnded is true without destroying socket", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = new Error("Done error");
    const req: any = { timedout: false, method: "GET", originalUrl: "/test", socket: { destroy: vi.fn() } };
    const res: any = { headersSent: true, writableEnded: true };
    
    errorHandler(error, req, res, () => {});
    expect(res.writableEnded).toBe(true);
    expect(req.socket.destroy).not.toHaveBeenCalled();
    logSpy.mockRestore();
  });

  it("should handle ValidationError", () => {
    const error = { name: "ValidationError", details: { body: [{ context: { label: "email" }, type: "string.email", message: "invalid" }] } };
    const req: any = {};
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({ email: "email deve ser um email válido" });
  });

  it("should handle URIError and log with exact message", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = new URIError("Malformed URI test");
    const req: any = {};
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
    expect(logSpy).toHaveBeenCalledWith({ message: "URI decode error: Malformed URI test" });
    logSpy.mockRestore();
  });

  it("should handle entity.parse.failed", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = { type: "entity.parse.failed" };
    const req: any = {};
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
    logSpy.mockRestore();
  });

  it.each([
    "entity.too.large",
    "parameters.too.many"
  ])("should handle payload too large: %s", (type) => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = { type };
    const req: any = {};
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
    expect(logSpy).toHaveBeenCalledWith({ message: `Payload too large: ${type}` });
    logSpy.mockRestore();
  });

  it("should handle charset.unsupported", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req: any = {};
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported charset", charset: "utf-16" });
    logSpy.mockRestore();
  });

  it("should handle encoding.unsupported and log encoding", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req: any = {};
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported content encoding", encoding: "deflate" });
    expect(logSpy).toHaveBeenCalledWith({ message: "Unsupported content encoding: deflate" });
    logSpy.mockRestore();
  });

  it("should handle Zlib client error with Z_ prefix", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = new Error("Zlib data corrupt");
    (error as any).expose = true;
    (error as any).statusCode = 400;
    (error as any).code = "Z_DATA_ERROR";
    const req: any = {};
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
    expect(logSpy).toHaveBeenCalledWith({ message: "Invalid compressed body: Zlib data corrupt" });
    logSpy.mockRestore();
  });

  it("should verify ehErroDoCliente strict boundary conditions (statusCode 499 vs 500)", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error499 = new Error("Edge 499");
    (error499 as any).expose = true;
    (error499 as any).statusCode = 499;
    (error499 as any).type = "edge.499";
    const req = { body: {} };
    const res1 = mockResponse();
    errorHandler(error499, req, res1, () => {});
    expect(res1.status).toHaveBeenCalledWith(499);

    const error500 = new Error("Edge 500");
    (error500 as any).expose = true;
    (error500 as any).statusCode = 500;
    (error500 as any).type = "edge.500";
    const res2 = mockResponse();
    errorHandler(error500, req, res2, () => {});
    expect(res2.status).toHaveBeenCalledWith(500);
    logSpy.mockRestore();
  });

  it("should not treat Zlib error as compressed body error if statusCode is outside 400-499 range", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = new Error("Other error");
    (error as any).expose = true;
    (error as any).statusCode = 500;
    (error as any).code = "Z_DATA_ERROR";
    (error as any).type = "some.type";
    const req: any = { body: {} };
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(500);
    logSpy.mockRestore();
  });

  it("should not treat error as client error if statusCode is 500 even if expose is true", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = new Error("Server error");
    (error as any).expose = true;
    (error as any).statusCode = 500;
    (error as any).type = "server.error";
    const req: any = { body: {} };
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith(expect.objectContaining({
      error: "server.error"
    }));
    logSpy.mockRestore();
  });

  it("should handle ETIMEDOUT error", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = { code: "ETIMEDOUT" };
    const req: any = {};
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(408);
    expect(res.json).toHaveBeenCalledWith({ message: "Timeout da requisição atingido" });
    logSpy.mockRestore();
  });

  it("should handle generic client error (ehErroDoCliente)", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = new Error("Bad request message");
    (error as any).expose = true;
    (error as any).statusCode = 400;
    const req: any = {};
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: undefined
    });
    expect(logSpy).toHaveBeenCalledWith({ message: "Client error 400: Bad request message" });
    logSpy.mockRestore();
  });

  it("should handle internal server error with error type", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error = new Error("Internal error message");
    error.stack = "ErrorStack123";
    (error as any).type = "database.error";
    const req: any = { body: { foo: "bar" } };
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith(expect.objectContaining({
      message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
      error: "database.error"
    }));
    expect(logSpy).toHaveBeenCalledWith({
      level: "error",
      message: "Error: database.error, Stack: ErrorStack123, Request body: {\"foo\":\"bar\"}"
    });
    logSpy.mockRestore();
  });

  it("should handle internal server error without error type", () => {
    const logSpy = vi.spyOn(loggerModule, "log");
    const error: any = new Error();
    error.stack = undefined;
    const req: any = { body: {} };
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith(expect.objectContaining({
      message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues"
    }));
    expect(logSpy).toHaveBeenCalled();
    logSpy.mockRestore();
  });
});
```
