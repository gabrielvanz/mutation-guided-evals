```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import logger from "../../src/utils/logger";

describe("errorHandler", () => {
  function mockResponse() {
    const res: any = {};
    res.status = vi.fn().mockReturnValue(res);
    res.json = vi.fn().mockReturnValue(res);
    res.send = vi.fn().mockReturnValue(res);
    return res;
  }

  it("should handle headersSent when timedout is true", () => {
    const error = new Error("Timeout error");
    error.stack = "Error stack trace";
    const req: any = { timedout: true, method: "GET", originalUrl: "/test", socket: { destroy: vi.fn() } };
    const res: any = { headersSent: true, writableEnded: false };
    const logSpy = vi.spyOn(logger, "log");
    
    errorHandler(error, req, res, () => {});
    expect(res.writableEnded).toBe(false);
    expect(req.socket.destroy).toHaveBeenCalled();
    expect(logSpy).toHaveBeenCalledWith({
      level: "warn",
      message: "Erro depois da resposta enviada: GET /test, timedout: true, Error: Timeout error, Stack: Error stack trace"
    });
    logSpy.mockRestore();
  });

  it("should handle headersSent when timedout is false and writableEnded is true", () => {
    const error = new Error("Done error");
    error.stack = "Error stack trace";
    const req: any = { timedout: false, method: "POST", originalUrl: "/users", socket: { destroy: vi.fn() } };
    const res: any = { headersSent: true, writableEnded: true };
    const logSpy = vi.spyOn(logger, "log");
    
    errorHandler(error, req, res, () => {});
    expect(res.writableEnded).toBe(true);
    expect(req.socket.destroy).not.toHaveBeenCalled();
    expect(logSpy).toHaveBeenCalledWith({
      level: "error",
      message: "Erro depois da resposta enviada: POST /users, timedout: false, Error: Done error, Stack: Error stack trace"
    });
    logSpy.mockRestore();
  });

  it("should handle headersSent with error.code when code is present", () => {
    const error = new Error("Custom error");
    (error as any).code = "ERR_CUSTOM";
    error.stack = "Stack trace";
    const req: any = { timedout: false, method: "PUT", originalUrl: "/item", socket: { destroy: vi.fn() } };
    const res: any = { headersSent: true, writableEnded: false };
    const logSpy = vi.spyOn(logger, "log");
    
    errorHandler(error, req, res, () => {});
    expect(logSpy).toHaveBeenCalledWith({
      level: "error",
      message: "Erro depois da resposta enviada: PUT /item, timedout: false, ERR_CUSTOM: Custom error, Stack: Stack trace"
    });
    logSpy.mockRestore();
  });

  it("should handle ValidationError", () => {
    const error = { name: "ValidationError", details: { body: [{ context: { label: "email" }, type: "string.email", message: "invalid" }] } };
    const req: any = {};
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({ email: "email deve ser um email válido" });
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req: any = {};
    const res = mockResponse();
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
    expect(logSpy).toHaveBeenCalledWith({ message: "URI decode error: Malformed URI" });
    logSpy.mockRestore();
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req: any = {};
    const res = mockResponse();
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
    expect(logSpy).toHaveBeenCalledWith({ message: "Entity parse error, user sending request without proper quotation marks." });
    logSpy.mockRestore();
  });

  it.each([
    "entity.too.large",
    "parameters.too.many"
  ])("should handle payload too large: %s", (type) => {
    const error = { type };
    const req: any = {};
    const res = mockResponse();
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
    expect(logSpy).toHaveBeenCalledWith({ message: `Payload too large: ${type}` });
    logSpy.mockRestore();
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req: any = {};
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req: any = {};
    const res = mockResponse();
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported content encoding", encoding: "deflate" });
    expect(logSpy).toHaveBeenCalledWith({ message: "Unsupported content encoding: deflate" });
    logSpy.mockRestore();
  });

  it("should handle Zlib client error starting with Z_", () => {
    const error = new Error("Zlib data error");
    (error as any).expose = true;
    (error as any).statusCode = 400;
    (error as any).code = "Z_DATA_ERROR";
    const req: any = {};
    const res = mockResponse();
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
    expect(logSpy).toHaveBeenCalledWith({ message: "Invalid compressed body: Zlib data error" });
    logSpy.mockRestore();
  });

  it("should not treat Zlib error as compressed body error if code does not start with Z_", () => {
    const error = new Error("Other error");
    (error as any).expose = true;
    (error as any).statusCode = 400;
    (error as any).code = "NOT_Z_ERROR";
    (error as any).type = "custom.type";
    const req: any = {};
    const res = mockResponse();
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "custom.type"
    });
    logSpy.mockRestore();
  });

  it("should handle ETIMEDOUT error", () => {
    const error = { code: "ETIMEDOUT" };
    const req: any = {};
    const res = mockResponse();

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(408);
    expect(res.json).toHaveBeenCalledWith({ message: "Timeout da requisição atingido" });
  });

  it("should handle generic client error with expose=true, status 400 and fallback to error.message when error.type is missing", () => {
    const error = new Error("Bad request message");
    (error as any).expose = true;
    (error as any).statusCode = 400;
    const req: any = {};
    const res = mockResponse();
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: undefined
    });
    expect(logSpy).toHaveBeenCalledWith({ message: "Client error 400: Bad request message" });
    logSpy.mockRestore();
  });

  it("should not treat client error if statusCode is outside 400-499 range", () => {
    const error = new Error("Server error style");
    (error as any).expose = true;
    (error as any).statusCode = 500;
    const req: any = { body: {} };
    const res = mockResponse();
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(500);
    logSpy.mockRestore();
  });

  it("should handle internal server error with error type and request body", () => {
    const error = new Error("Internal error");
    (error as any).type = "database.error";
    error.stack = "Error stack";
    const req: any = { body: { user: "test" } };
    const res = mockResponse();
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});
    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith(expect.objectContaining({
      message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
      error: "database.error"
    }));
    expect(logSpy).toHaveBeenCalledWith({
      level: "error",
      message: "Error: database.error, Stack: Error stack, Request body: {\"user\":\"test\"}"
    });
    logSpy.mockRestore();
  });
});
```
