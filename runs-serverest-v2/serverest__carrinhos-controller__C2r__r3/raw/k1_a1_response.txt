```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import carrinhosService from "../../src/services/carrinhos-service";
import constants from "../../src/utils/constants";

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.restoreAllMocks();
  });

  describe("get", () => {
    it("should return 200 with all carts", async () => {
      const mockCarrinhos = [{ _id: "1", precoTotal: 100 }];
      vi.spyOn(carrinhosService, "getAll").mockResolvedValue(mockCarrinhos as any);

      const req = { query: {} };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.get(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: 1,
        carrinhos: mockCarrinhos
      });
    });
  });

  describe("getOne", () => {
    it("should return 400 when cart is not found", async () => {
      vi.spyOn(carrinhosService, "getOne").mockResolvedValue(null as any);

      const req = { params: { id: "invalid-id" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        json: vi.fn(),
        send: vi.fn()
      };

      await carrinhosController.getOne(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constants.CART_NOT_FOUND });
    });

    it("should return 200 with cart when found", async () => {
      const mockCarrinho = { _id: "1", precoTotal: 50 };
      vi.spyOn(carrinhosService, "getOne").mockResolvedValue(mockCarrinho as any);

      const req = { params: { id: "1" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.getOne(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });
  });

  describe("post", () => {
    it("should return 400 when user already has a cart", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: { _id: "cart-1" } as any
      });

      const req = { headers: { authorization: "Bearer token" }, body: { produtos: [] } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn()
          }
        }
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constants.LIMIT_JUST_ONE_CART });
    });

    it("should return 400 when cart contains duplicate products", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: null
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue(["prod-1"]);

      const req = {
        headers: { authorization: "Bearer token" },
        body: {
          produtos: [
            { idProduto: "prod-1", quantidade: 1 },
            { idProduto: "prod-1", quantidade: 2 }
          ]
        }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn()
          }
        }
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constants.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["prod-1"]
      });
    });

    it("should return error status when product service returns an error", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: null
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        error: {
          statusCode: 400,
          message: constants.PRODUCT_NOT_FOUND,
          item: { idProduto: "prod-99" },
          index: 0
        }
      } as any);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod-99", quantidade: 1 }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn()
          }
        }
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constants.PRODUCT_NOT_FOUND,
        item: { idProduto: "prod-99" }
      });
    });

    it("should return 400 when totals are not safe", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: null
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "prod-1", precoUnitario: 10, quantidade: 1 }]
      });
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(100);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(1);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(false);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod-1", quantidade: 1 }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn()
          }
        }
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constants.CART_TOTALS_TOO_BIG });
    });

    it("should create cart successfully when all validations pass", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user-1",
        possuiCarrinho: null
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "prod-1", precoUnitario: 10, quantidade: 2 }]
      });
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(20);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(2);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(true);
      vi.spyOn(carrinhosService, "baixarEstoque").mockResolvedValue(undefined);
      vi.spyOn(carrinhosService, "criarCarrinho").mockResolvedValue({ _id: "new-cart-id" } as any);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod-1", quantidade: "2" }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn(),
            abrir: vi.fn()
          }
        }
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constants.POST_SUCCESS,
        _id: "new-cart-id"
      });
    });
  });

  describe("cancelarCompra", () => {
    it("should remove cart and replenish stock when user has a cart", async () => {
      const mockCarrinho = { _id: "cart-1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([mockCarrinho] as any);
      vi.spyOn(carrinhosService, "removeCarrinho").mockResolvedValue(undefined);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req as any, res as any);

      expect(carrinhosService.removeCarrinho).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constants.DELETE_SUCCESS}. ${constants.REPLENISHED_STOCK}`
      });
    });

    it("should return NO_CART when user does not have a cart", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([]);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constants.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase and delete cart when user has a cart", async () => {
      const mockCarrinho = { _id: "cart-1" };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([mockCarrinho] as any);
      vi.spyOn(carrinhosService, "concluiCompra").mockResolvedValue(undefined);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req as any, res as any);

      expect(carrinhosService.concluiCompra).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constants.DELETE_SUCCESS });
    });

    it("should return NO_CART when user does not have a cart", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([]);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constants.NO_CART });
    });
  });
});
```
