```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import carrinhosService from "../../src/services/carrinhos-service";
import constant from "../../src/utils/constants";

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.restoreAllMocks();
  });

  describe("get", () => {
    it("should return all carts with status 200", async () => {
      const mockCarrinhos = [{ _id: "1", precoTotal: 100, quantidadeTotal: 2, idUsuario: "u1", produtos: [] }];
      vi.spyOn(carrinhosService, "getAll").mockResolvedValue(mockCarrinhos);

      const req = { query: {} };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.get(req, res);

      expect(carrinhosService.getAll).toHaveBeenCalledWith(req.query);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: 1,
        carrinhos: mockCarrinhos
      });
    });
  });

  describe("getOne", () => {
    it("should return cart when found with status 200", async () => {
      const mockCarrinho = { _id: "1", precoTotal: 100, quantidadeTotal: 2, idUsuario: "u1", produtos: [] };
      vi.spyOn(carrinhosService, "getOne").mockResolvedValue(mockCarrinho);

      const req = { params: { id: "1" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        json: vi.fn(),
        send: vi.fn()
      };

      await carrinhosController.getOne(req, res);

      expect(carrinhosService.getOne).toHaveBeenCalledWith("1");
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });

    it("should return CART_NOT_FOUND when cart does not exist", async () => {
      vi.spyOn(carrinhosService, "getOne").mockResolvedValue(null);

      const req = { params: { id: "nao-existe" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        json: vi.fn()
      };

      await carrinhosController.getOne(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });
  });

  describe("post", () => {
    it("should create a cart successfully", async () => {
      const mockEtapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: {
          produtos: [
            { idProduto: "p1", quantidade: "2" }
          ]
        }
      };
      const res = {
        locals: { etapas: mockEtapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: false
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "p1", precoUnitario: 50, quantidade: 2 }]
      });
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(100);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(2);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(true);
      vi.spyOn(carrinhosService, "baixarEstoque").mockResolvedValue(undefined);
      vi.spyOn(carrinhosService, "criarCarrinho").mockResolvedValue({ _id: "c1" });

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.POST_SUCCESS,
        _id: "c1"
      });
    });

    it("should return LIMIT_JUST_ONE_CART if user already has a cart", async () => {
      const mockEtapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [] }
      };
      const res = {
        locals: { etapas: mockEtapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: { _id: "c-existente" }
      });

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return CART_WITH_DUPLICATE_PRODUCT if products have duplicates", async () => {
      const mockEtapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: {
          produtos: [
            { idProduto: "p1", quantidade: 1 },
            { idProduto: "p1", quantidade: 2 }
          ]
        }
      };
      const res = {
        locals: { etapas: mockEtapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: false
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue(["p1"]);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["p1"]
      });
    });

    it("should return error from getProdutosComPrecoUnitarioOuErro if product not found or stock insufficient", async () => {
      const mockEtapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: {
          produtos: [{ idProduto: "p1", quantidade: 1 }]
        }
      };
      const res = {
        locals: { etapas: mockEtapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: false
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        error: {
          statusCode: 400,
          message: constant.PRODUCT_NOT_FOUND,
          index: 0
        }
      });

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.PRODUCT_NOT_FOUND,
        index: undefined
      });
    });

    it("should return CART_TOTALS_TOO_BIG if totals are not safe", async () => {
      const mockEtapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: {
          produtos: [{ idProduto: "p1", quantidade: 1 }]
        }
      };
      const res = {
        locals: { etapas: mockEtapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: false
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "p1", precoUnitario: 10, quantidade: 1 }]
      });
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(Number.MAX_SAFE_INTEGER + 1);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(1);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(false);

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });
  });

  describe("cancelarCompra", () => {
    it("should cancel purchase and replenish stock when cart exists", async () => {
      const mockCarrinho = { _id: "c1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([mockCarrinho]);
      vi.spyOn(carrinhosService, "removeCarrinho").mockResolvedValue(undefined);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req, res);

      expect(carrinhosService.removeCarrinho).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}`
      });
    });

    it("should return NO_CART when user has no cart", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([]);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase successfully when cart exists", async () => {
      const mockCarrinho = { _id: "c1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([mockCarrinho]);
      vi.spyOn(carrinhosService, "concluiCompra").mockResolvedValue(undefined);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req, res);

      expect(carrinhosService.concluiCompra).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("should return NO_CART when user has no cart", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([]);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});
```
