```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import logger from "../../src/utils/logger";

describe("errorHandler", () => {
  it("should handle headersSent when request timed out", () => {
    const error = new Error("Timeout");
    error.stack = "stack";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: { destroy: vi.fn() }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "warn",
      message: "Erro depois da resposta enviada: GET /test, timedout: true, Error: Timeout, Stack: stack"
    });
    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("should handle headersSent when response is already writableEnded", () => {
    const error = new Error("Some error");
    error.stack = "stack";
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/test",
      socket: { destroy: vi.fn() }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: "Erro depois da resposta enviada: POST /test, timedout: false, Error: Some error, Stack: stack"
    });
    expect(req.socket.destroy).not.toHaveBeenCalled();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "nome" },
            type: "any.required",
            message: "nome is required"
          }
        ]
      }
    };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, {}, res, () => {});

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      nome: "nome é obrigatório"
    });
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };
    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "URI decode error: Malformed URI"
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed error", () => {
    const error = { type: "entity.parse.failed" };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };
    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Entity parse error, user sending request without proper quotation marks."
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large error", () => {
    const error = { type: "entity.too.large" };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };
    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Payload too large: entity.too.large"
    });
    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
  });

  it("should handle parameters.too.many error", () => {
    const error = { type: "parameters.too.many" };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };
    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Payload too large: parameters.too.many"
    });
    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
  });

  it("should handle charset.unsupported error", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, {}, res, () => {});

    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported error", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };
    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Unsupported content encoding: deflate"
    });
    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle client error with Z_ prefix code", () => {
    const error = new Error("Z_DATA_ERROR");
    error.expose = true;
    error.statusCode = 400;
    error.code = "Z_DATA_ERROR";
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };
    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Invalid compressed body: Z_DATA_ERROR"
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = { code: "ETIMEDOUT" };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };

    errorHandler(error, {}, res, () => {});

    expect(res.status).toHaveBeenCalledWith(408);
    expect(res.json).toHaveBeenCalledWith({ message: "Timeout da requisição atingido" });
  });

  it("should handle general client error (ehErroDoCliente)", () => {
    const error = new Error("Bad request");
    error.expose = true;
    error.statusCode = 400;
    error.type = "request.aborted";
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };
    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, {}, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Client error 400: request.aborted"
    });
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "request.aborted"
    });
  });

  it("should handle server error with error.type present", () => {
    const error = { type: "database.error", stack: "err-stack" };
    const req = { body: { foo: "bar" } };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn().mockReturnThis()
    };
    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: "Error: database.error, Stack: err-stack, Request body: {\"foo\":\"bar\"}"
    });
    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith({
      message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
      error: "database.error",
      version: expect.any(String)
    });
  });
});
```
