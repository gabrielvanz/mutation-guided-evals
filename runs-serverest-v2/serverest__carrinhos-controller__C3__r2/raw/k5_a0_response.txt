```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import carrinhosService from "../../src/services/carrinhos-service";
import constant from "../../src/utils/constants";

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.restoreAllMocks();
  });

  describe("get", () => {
    it("should return all carts with status 200", async () => {
      const carrinhosMock = [{ _id: "1", precoTotal: 100, quantidadeTotal: 2, produtos: [], idUsuario: "u1" }];
      vi.spyOn(carrinhosService, "getAll").mockResolvedValue(carrinhosMock as any);

      const req = { query: {} };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.get(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ quantidade: 1, carrinhos: carrinhosMock });
    });
  });

  describe("getOne", () => {
    it("should return 400 with CART_NOT_FOUND if cart does not exist", async () => {
      vi.spyOn(carrinhosService, "getOne").mockResolvedValue(null);

      const req = { params: { id: "nao-existe" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        json: vi.fn()
      };

      await carrinhosController.getOne(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });

    it("should return the cart with status 200 if it exists", async () => {
      const carrinhoMock = { _id: "1", precoTotal: 50, quantidadeTotal: 1, produtos: [], idUsuario: "u1" };
      vi.spyOn(carrinhosService, "getOne").mockResolvedValue(carrinhoMock as any);

      const req = { params: { id: "1" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.getOne(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(carrinhoMock);
    });
  });

  describe("post", () => {
    it("should return 400 with LIMIT_JUST_ONE_CART if user already has a cart", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: { _id: "c1" }
      } as any);

      const etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      const req = { headers: { authorization: "token" }, body: { produtos: [] } };
      const res = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return 400 with CART_WITH_DUPLICATE_PRODUCT if products have duplicates", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue(["p1"]);

      const etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      const req = {
        headers: { authorization: "token" },
        body: {
          produtos: [
            { idProduto: "p1", quantidade: 1 },
            { idProduto: "p1", quantidade: 2 }
          ]
        }
      };
      const res = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["p1"]
      });
    });

    it("should return error status and message if product check fails", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        error: {
          statusCode: 400,
          message: constant.PRODUCT_NOT_FOUND,
          item: { idProduto: "pX" },
          index: 0
        }
      } as any);

      const etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };
      const req = {
        headers: { authorization: "token" },
        body: { produtos: [{ idProduto: "pX", quantidade: 1 }] }
      };
      const res = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.PRODUCT_NOT_FOUND,
        item: { idProduto: "pX" }
      });
    });

    it("should return 400 if totals are not safe", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "p1", precoUnitario: 10, quantidade: 1 }]
      } as any);
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(10);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(1);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(false);

      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }] }
      };
      const res = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });

    it("should successfully create a cart and return 201", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "u1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      const produtosPreco = [{ idProduto: "p1", precoUnitario: 10, quantidade: 2 }];
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: produtosPreco
      } as any);
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(20);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(2);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(true);
      vi.spyOn(carrinhosService, "baixarEstoque").mockResolvedValue(undefined);
      vi.spyOn(carrinhosService, "criarCarrinho").mockResolvedValue({ _id: "c123" } as any);

      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const req = {
        headers: { authorization: "token" },
        body: { produtos: [{ idProduto: "p1", quantidade: "2" }] }
      };
      const res = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res as any);

      expect(carrinhosService.baixarEstoque).toHaveBeenCalledWith(produtosPreco);
      expect(carrinhosService.criarCarrinho).toHaveBeenCalledWith({
        produtos: produtosPreco,
        precoTotal: 20,
        quantidadeTotal: 2,
        idUsuario: "u1"
      });
      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({ message: constant.POST_SUCCESS, _id: "c123" });
    });
  });

  describe("cancelarCompra", () => {
    it("should remove cart and replenish stock if user has a cart", async () => {
      const carrinhoMock = { _id: "c1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([carrinhoMock] as any);
      vi.spyOn(carrinhosService, "removeCarrinho").mockResolvedValue(undefined);

      const req = { headers: { authorization: "token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req as any, res as any);

      expect(carrinhosService.removeCarrinho).toHaveBeenCalledWith(carrinhoMock);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}`
      });
    });

    it("should return NO_CART if user has no cart", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([] as any);

      const req = { headers: { authorization: "token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase and delete cart if user has a cart", async () => {
      const carrinhoMock = { _id: "c1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([carrinhoMock] as any);
      vi.spyOn(carrinhosService, "concluiCompra").mockResolvedValue(undefined);

      const req = { headers: { authorization: "token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req as any, res as any);

      expect(carrinhosService.concluiCompra).toHaveBeenCalledWith(carrinhoMock);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("should return NO_CART if user has no cart", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([] as any);

      const req = { headers: { authorization: "token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});
```
