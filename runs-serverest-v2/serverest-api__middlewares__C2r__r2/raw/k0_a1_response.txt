```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API tests", () => {
  it("should return 401 when accessing a protected route without authorization header", async () => {
    const res = await request(app).post("/carrinhos").send({
      produtos: []
    });

    expect(res.status).toBe(401);
    expect(res.body).toEqual({ message: "Token de acesso ausente, inválido ou expirado" });
  });

  it("should return 401 when accessing a protected route with a malformed token", async () => {
    const res = await request(app)
      .post("/carrinhos")
      .set("authorization", "InvalidFormat 12345")
      .send({ produtos: [] });

    expect(res.status).toBe(401);
    expect(res.body).toEqual({ message: "Token de acesso ausente, inválido ou expirado" });
  });

  it("should return 403 when accessing an admin-only route with a standard user token", async () => {
    const unique = Date.now();
    const email = `user_${unique}@serverest.test`;
    const password = "password123";

    await request(app).post("/usuarios").send({
      nome: "User Test",
      email,
      password,
      administrador: "false"
    });

    const loginRes = await request(app).post("/login").send({
      email,
      password
    });

    const token = loginRes.body.authorization;

    const res = await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send({
        nome: `Produto ${unique}`,
        preco: 10,
        descricao: "Teste",
        quantidade: 10
      });

    expect(res.status).toBe(403);
    expect(res.body).toEqual({ message: "Rota exclusiva para administradores" });
  });

  it("should return 400 when malformed JSON (entity.parse.failed) is sent", async () => {
    const res = await request(app)
      .post("/usuarios")
      .set("content-type", "application/json")
      .send("{ invalid json ");

    expect(res.status).toBe(400);
    expect(res.body).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should return 400 when a URI decode error occurs in the request path", async () => {
    const res = await request(app).get("/usuarios/%C0%C0");

    expect(res.status).toBe(400);
    expect(res.body).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });
});
```
