```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on a protected route", async () => {
      const response = await request(app).post("/carrinhos").send({
        produtos: []
      });

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido, expirado ou usuário do token não existe mais"
      });
    });

    it("should return 401 when token has bad format", async () => {
      const response = await request(app)
        .post("/carrinhos")
        .set("authorization", "InvalidTokenFormat")
        .send({
          produtos: []
        });

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido, expirado ou usuário do token não existe mais"
      });
    });

    it("should return 401 when token is invalid or user does not exist", async () => {
      const response = await request(app)
        .post("/carrinhos")
        .set("authorization", "Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJlbWFpbCI6Im5vbmV4aXN0ZW50QGVtYWlsLmNvbSIsInBhc3N3b3JkIjoidGVzdGUiLCJpYXQiOjE2MjAwMDAwMDB9.invalid_signature")
        .send({
          produtos: []
        });

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido, expirado ou usuário do token não existe mais"
      });
    });

    it("should return 403 when a non-admin user tries to access an admin-only route", async () => {
      const email = "user_common_admin_test@email.com";
      await request(app).post("/usuarios").send({
        nome: "User Comum",
        email,
        password: "teste",
        administrador: "false"
      });

      const loginResponse = await request(app).post("/login").send({
        email,
        password: "teste"
      });

      const token = loginResponse.body.authorization;

      const response = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Teste",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(response.status).toBe(403);
      expect(response.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 when malformed JSON entity is sent", async () => {
      const response = await request(app)
        .post("/usuarios")
        .set("Content-Type", "application/json")
        .send('{"nome": unquoted}');

      expect(response.status).toBe(400);
      expect(response.body).toEqual({
        message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
      });
    });

    it("should return 400 when URI decode error occurs in parameters", async () => {
      const response = await request(app).get("/usuarios/%");

      expect(response.status).toBe(400);
      expect(response.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });
  });
});
```
