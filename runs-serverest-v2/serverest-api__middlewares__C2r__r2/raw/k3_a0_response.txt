```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on a protected route", async () => {
      const res = await request(app).post("/carrinhos").send({
        produtos: []
      });
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado. Faça login novamente."
      });
    });

    it("should return 401 when token format is invalid (missing Bearer prefix)", async () => {
      const res = await request(app)
        .post("/carrinhos")
        .set("authorization", "invalidtoken")
        .send({
          produtos: []
        });
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado. Faça login novamente."
      });
    });

    it("should return 401 when token is completely invalid/malformed", async () => {
      const res = await request(app)
        .post("/carrinhos")
        .set("authorization", "Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.invalid.signature")
        .send({
          produtos: []
        });
      expect(res.status).toBe(401);
      expect(res.body).toEqual({
        message: "Token de acesso ausente, inválido ou expirado. Faça login novamente."
      });
    });

    it("should return 403 when accessing an admin-only route with a non-admin user token", async () => {
      const email = `user_${Date.now()}@qa.com`;
      await request(app).post("/usuarios").send({
        nome: "User Comum",
        email,
        password: "teste",
        administrador: "false"
      });

      const loginRes = await request(app).post("/login").send({
        email,
        password: "teste"
      });

      const token = loginRes.body.authorization;

      const res = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: `Produto_${Date.now()}`,
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(res.status).toBe(403);
      expect(res.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 when malformed JSON (entity.parse.failed) is sent", async () => {
      const res = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send('{"nome": "Teste", "email": unquoted_email@qa.com}');

      expect(res.status).toBe(400);
      expect(res.body.message).toContain("Adicione aspas em todos os valores");
    });

    it("should return 400 when URI is invalid (URIError)", async () => {
      const res = await request(app).get("/usuarios/%c0%af");
      expect(res.status).toBe(400);
      expect(res.body).toEqual({
        message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
      });
    });
  });
});
```
