```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import carrinhosService from "../../src/services/carrinhos-service";
import constant from "../../src/utils/constants";

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.restoreAllMocks();
  });

  describe("get", () => {
    it("should return all carts with status 200", async () => {
      const mockCarrinhos = [{ _id: "1", precoTotal: 100 }];
      vi.spyOn(carrinhosService, "getAll").mockResolvedValue(mockCarrinhos as any);

      const req = { query: {} };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.get(req as any, res as any);

      expect(carrinhosService.getAll).toHaveBeenCalledWith(req.query);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: 1,
        carrinhos: mockCarrinhos
      });
    });
  });

  describe("getOne", () => {
    it("should return cart when found with status 200", async () => {
      const mockCarrinho = { _id: "abc", precoTotal: 50 };
      vi.spyOn(carrinhosService, "getOne").mockResolvedValue(mockCarrinho as any);

      const req = { params: { id: "abc" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        json: vi.fn(),
        send: vi.fn()
      };

      await carrinhosController.getOne(req as any, res as any);

      expect(carrinhosService.getOne).toHaveBeenCalledWith("abc");
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });

    it("should return CART_NOT_FOUND when cart does not exist", async () => {
      vi.spyOn(carrinhosService, "getOne").mockResolvedValue(null as any);

      const req = { params: { id: "invalid" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        json: vi.fn(),
        send: vi.fn()
      };

      await carrinhosController.getOne(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });
  });

  describe("post", () => {
    it("should return LIMIT_JUST_ONE_CART when user already has a cart", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user123",
        possuiCarrinho: { _id: "cart1" }
      } as any);

      const req = { headers: { authorization: "Bearer token" }, body: { produtos: [] } };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const res = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return CART_WITH_DUPLICATE_PRODUCT when products are duplicated", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user123",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue(["prod1"]);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod1", quantidade: 1 }, { idProduto: "prod1", quantidade: 2 }] }
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const res = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["prod1"]
      });
    });

    it("should return product error when service returns one", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user123",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        error: {
          message: constant.PRODUCT_NOT_FOUND,
          statusCode: 400,
          index: 0,
          item: { idProduto: "bad" }
        }
      } as any);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "bad", quantidade: 1 }] }
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const res = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.PRODUCT_NOT_FOUND,
        item: { idProduto: "bad" }
      });
    });

    it("should return CART_TOTALS_TOO_BIG when totals are unsafe", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user123",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "prod1", precoUnitario: 10, quantidade: 5 }]
      } as any);
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(100);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(5);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(false);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod1", quantidade: 5 }] }
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const res = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });

    it("should create cart successfully with status 201", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user123",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "prod1", precoUnitario: 10, quantidade: 2 }]
      } as any);
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(20);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(2);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(true);
      vi.spyOn(carrinhosService, "baixarEstoque").mockResolvedValue(undefined);
      vi.spyOn(carrinhosService, "criarCarrinho").mockResolvedValue({ _id: "newCartId" } as any);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod1", quantidade: 2 }] }
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };
      const res = {
        locals: { etapas },
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req as any, res as any);

      expect(carrinhosService.baixarEstoque).toHaveBeenCalled();
      expect(carrinhosService.criarCarrinho).toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.POST_SUCCESS,
        _id: "newCartId"
      });
    });
  });

  describe("cancelarCompra", () => {
    it("should remove cart and replenish stock when user cart exists", async () => {
      const mockCarrinho = { _id: "cart1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([mockCarrinho] as any);
      vi.spyOn(carrinhosService, "removeCarrinho").mockResolvedValue(undefined as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req as any, res as any);

      expect(carrinhosService.removeCarrinho).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}`
      });
    });

    it("should return NO_CART when user has no cart", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase and delete cart when user cart exists", async () => {
      const mockCarrinho = { _id: "cart1" };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([mockCarrinho] as any);
      vi.spyOn(carrinhosService, "concluiCompra").mockResolvedValue(undefined as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req as any, res as any);

      expect(carrinhosService.concluiCompra).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("should return NO_CART when user has no cart", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});
```
