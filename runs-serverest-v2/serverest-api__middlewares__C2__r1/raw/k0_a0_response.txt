```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API tests", () => {
  it("should return 401 when accessing a protected route without authorization header", async () => {
    const response = await request(app).get("/carrinhos");

    expect(response.status).toBe(401);
    expect(response.body).toEqual({
      message: "Token de acesso ausente, inválido ou expirado. Faça login novamente."
    });
  });

  it("should return 401 when authorization header format is invalid", async () => {
    const response = await request(app)
      .get("/carrinhos")
      .set("authorization", "InvalidFormat token123");

    expect(response.status).toBe(401);
    expect(response.body).toEqual({
      message: "Token de acesso ausente, inválido ou expirado. Faça login novamente."
    });
  });

  it("should return 401 when token is invalid or does not belong to an existing user", async () => {
    const response = await request(app)
      .get("/carrinhos")
      .set("authorization", "Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJlbWFpbCI6Im5vbmV4aXN0ZW50QGVtYWlsLmNvbSIsInBhc3N3b3JkIjoidGVzdGUiLCJpYXQiOjE2MDAwMDAwMDB9.invalid");

    expect(response.status).toBe(401);
    expect(response.body).toEqual({
      message: "Token de acesso ausente, inválido ou expirado. Faça login novamente."
    });
  });

  it("should return 403 when a non-admin user tries to access an admin-only route", async () => {
    const uniqueEmail = `user_${Date.now()}_${Math.random()}@email.com`;
    await request(app)
      .post("/usuarios")
      .send({
        nome: "Usuario Comum",
        email: uniqueEmail,
        password: "teste",
        administrador: "false"
      });

    const loginResponse = await request(app)
      .post("/login")
      .send({
        email: uniqueEmail,
        password: "teste"
      });

    const token = loginResponse.body.authorization;

    const response = await request(app)
      .post("/produtos")
      .set("authorization", token)
      .send({
        nome: "Produto Teste",
        preco: 100,
        descricao: "Descricao",
        quantidade: 10
      });

    expect(response.status).toBe(403);
    expect(response.body).toEqual({
      message: "Rota exclusiva para administradores"
    });
  });

  it("should return 400 when malformed JSON body is sent (entity.parse.failed)", async () => {
    const response = await request(app)
      .post("/usuarios")
      .set("content-type", "application/json")
      .send("{ invalid json ");

    expect(response.status).toBe(400);
    expect(response.body).toEqual({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should return 400 when URI decode error occurs in route params", async () => {
    const response = await request(app).get("/usuarios/%E0%A2%C");

    expect(response.status).toBe(400);
    expect(response.body).toEqual({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });
});
```
