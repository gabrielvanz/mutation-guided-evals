```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API tests", () => {
  it("should return 401 when accessing a protected route without authorization header", async () => {
    const response = await request(app).post("/carrinhos").send({ produtos: [] });
    expect(response.status).toBe(401);
    expect(response.body).toHaveProperty("message");
  });

  it("should return 401 when accessing a protected route with malformed authorization header", async () => {
    const response = await request(app)
      .post("/carrinhos")
      .set("Authorization", "InvalidFormat")
      .send({ produtos: [] });
    expect(response.status).toBe(401);
    expect(response.body).toHaveProperty("message");
  });

  it("should return 401 when accessing a protected route with an invalid token", async () => {
    const response = await request(app)
      .post("/carrinhos")
      .set("Authorization", "Bearer invalid.jwt.token")
      .send({ produtos: [] });
    expect(response.status).toBe(401);
    expect(response.body).toHaveProperty("message");
  });

  it("should return 403 when a non-admin user tries to access an admin-only route", async () => {
    const email = "user_non_admin@test.com";
    const password = "password123";

    await request(app).post("/usuarios").send({
      nome: "Comum",
      email,
      password,
      administrador: "false",
    });

    const loginRes = await request(app).post("/login").send({
      email,
      password,
    });

    const token = loginRes.body.authorization;

    const response = await request(app)
      .post("/produtos")
      .set("Authorization", token)
      .send({
        nome: "Produto Teste",
        preco: 10,
        descricao: "Teste",
        quantidade: 5,
      });

    expect(response.status).toBe(403);
    expect(response.body).toHaveProperty("message");
  });

  it("should handle URIError and return 400 status through error handler", async () => {
    const response = await request(app).get("/usuarios/%E0%A2%A");
    expect(response.status).toBe(400);
    expect(response.body).toHaveProperty("message");
  });

  it("should handle entity.parse.failed and return 400 status through error handler", async () => {
    const response = await request(app)
      .post("/usuarios")
      .set("Content-Type", "application/json")
      .send("{invalid json");

    expect(response.status).toBe(400);
    expect(response.body).toHaveProperty("message");
  });
});
```
