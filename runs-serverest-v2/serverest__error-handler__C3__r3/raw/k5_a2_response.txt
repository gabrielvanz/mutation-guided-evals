```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import * as loggerObj from "../../src/utils/logger";

describe("errorHandler", () => {
  it("should handle headersSent when request timed out", () => {
    const error = new Error("Timeout occurred");
    error.stack = "stack trace";
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/users",
      socket: { destroy: vi.fn() }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const spyLog = vi.spyOn(loggerObj, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "warn",
      message: "Erro depois da resposta enviada: GET /users, timedout: true, Error: Timeout occurred, Stack: stack trace"
    });
    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("should handle headersSent when response already ended", () => {
    const error = new Error("Some error");
    error.stack = "stack trace";
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/login",
      socket: { destroy: vi.fn() }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    const spyLog = vi.spyOn(loggerObj, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: "Erro depois da resposta enviada: POST /login, timedout: false, Error: Some error, Stack: stack trace"
    });
    expect(req.socket.destroy).not.toHaveBeenCalled();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "email" },
            type: "string.email",
            message: "email must be a valid email"
          }
        ]
      }
    };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      email: "email deve ser um email válido"
    });
  });

  it("should handle URIError", () => {
    const error = new URIError("malformed URI");
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };
    const spyLog = vi.spyOn(loggerObj, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "URI decode error: malformed URI"
    });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };
    const spyLog = vi.spyOn(loggerObj, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Entity parse error, user sending request without proper quotation marks."
    });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };
    const spyLog = vi.spyOn(loggerObj, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Payload too large: entity.too.large"
    });
    expect(statusMock).toHaveBeenCalledWith(413);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Payload too large"
    });
  });

  it("should handle parameters.too.many", () => {
    const error = { type: "parameters.too.many" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };
    const spyLog = vi.spyOn(loggerObj, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Payload too large: parameters.too.many"
    });
    expect(statusMock).toHaveBeenCalledWith(413);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Payload too large"
    });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(415);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Unsupported charset",
      charset: "utf-16"
    });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };
    const spyLog = vi.spyOn(loggerObj, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Unsupported content encoding: deflate"
    });
    expect(statusMock).toHaveBeenCalledWith(415);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Unsupported content encoding",
      encoding: "deflate"
    });
  });

  it("should handle Zlib compression client error", () => {
    const error = new Error("zlib data error");
    error.expose = true;
    error.statusCode = 400;
    error.code = "Z_DATA_ERROR";
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };
    const spyLog = vi.spyOn(loggerObj, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Invalid compressed body: zlib data error"
    });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error", () => {
    const error = { code: "ETIMEDOUT" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(408);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Timeout da requisição atingido"
    });
  });

  it("should handle generic client error (ehErroDoCliente)", () => {
    const error = new Error("Bad request");
    error.expose = true;
    error.statusCode = 400;
    error.type = "some.client.error";
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };
    const spyLog = vi.spyOn(loggerObj, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      message: "Client error 400: some.client.error"
    });
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.client.error"
    });
  });

  it("should handle server error with error type", () => {
    const error = new Error("Internal Server Error");
    error.type = "database.error";
    error.stack = "some stack";
    const req = { body: { foo: "bar" } };
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };
    const spyLog = vi.spyOn(loggerObj, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith({
      level: "error",
      message: "Error: database.error, Stack: some stack, Request body: {\"foo\":\"bar\"}"
    });
    expect(statusMock).toHaveBeenCalledWith(500);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
      error: "database.error",
      version: expect.any(String)
    });
  });
});
```
