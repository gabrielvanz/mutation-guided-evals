```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import logger from "../../src/utils/logger";

describe("errorHandler", () => {
  it("should handle headersSent when req.timedout is true", () => {
    const error = new Error("Timeout after response");
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/usuarios",
      socket: {
        destroy: vi.fn()
      }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith(
      expect.objectContaining({
        level: "warn"
      })
    );
    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("should handle headersSent when req.timedout is false", () => {
    const error = new Error("Error after response");
    const req = {
      timedout: false,
      method: "POST",
      originalUrl: "/produtos",
      socket: {
        destroy: vi.fn()
      }
    };
    const res = {
      headersSent: true,
      writableEnded: true
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith(
      expect.objectContaining({
        level: "error"
      })
    );
    expect(req.socket.destroy).not.toHaveBeenCalled();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "nome" },
            type: "string.base",
            message: "nome must be a string"
          }
        ]
      }
    };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      nome: "nome deve ser uma string"
    });
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalled();
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed error type", () => {
    const error = { type: "entity.parse.failed" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalled();
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large error type", () => {
    const error = { type: "entity.too.large" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalled();
    expect(statusMock).toHaveBeenCalledWith(413);
    expect(jsonMock).toHaveBeenCalledWith({ message: "Payload too large" });
  });

  it("should handle parameters.too.many error type", () => {
    const error = { type: "parameters.too.many" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalled();
    expect(statusMock).toHaveBeenCalledWith(413);
    expect(jsonMock).toHaveBeenCalledWith({ message: "Payload too large" });
  });

  it("should handle charset.unsupported error type", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(415);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Unsupported charset",
      charset: "utf-16"
    });
  });

  it("should handle encoding.unsupported error type", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalled();
    expect(statusMock).toHaveBeenCalledWith(415);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Unsupported content encoding",
      encoding: "deflate"
    });
  });

  it("should handle client error with Z_ prefix in code", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "Invalid data"
    };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalled();
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT error code", () => {
    const error = { code: "ETIMEDOUT" };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    errorHandler(error, req, res, () => {});

    expect(statusMock).toHaveBeenCalledWith(408);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Timeout da requisição atingido"
    });
  });

  it("should handle generic client error (ehErroDoCliente)", () => {
    const error = {
      expose: true,
      statusCode: 400,
      type: "entity.parse.failed.something"
    };
    const req = {};
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalled();
    expect(statusMock).toHaveBeenCalledWith(400);
    expect(jsonMock).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "entity.parse.failed.something"
    });
  });

  it("should handle fallback server error with type", () => {
    const error = {
      type: "some.internal.error",
      stack: "at stack"
    };
    const req = { body: {} };
    const jsonMock = vi.fn().mockReturnThis();
    const statusMock = vi.fn().mockReturnValue({ json: jsonMock });
    const res = {
      headersSent: false,
      status: statusMock
    };

    const spyLog = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(spyLog).toHaveBeenCalledWith(
      expect.objectContaining({
        level: "error"
      })
    );
    expect(statusMock).toHaveBeenCalledWith(500);
    expect(jsonMock).toHaveBeenCalledWith(
      expect.objectContaining({
        message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
        error: "some.internal.error"
      })
    );
  });
});
```
