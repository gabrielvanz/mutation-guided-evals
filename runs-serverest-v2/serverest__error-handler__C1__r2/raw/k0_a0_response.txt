```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import logger from "../../src/utils/logger";

describe("error-handler middleware", () => {
  beforeEach(() => {
    vi.restoreAllMocks();
  });

  it("should handle headersSent and close socket if not writableEnded", () => {
    const logSpy = vi.spyOn(logger, "log").mockImplementation(() => {});
    const req = {
      timedout: true,
      method: "GET",
      originalUrl: "/test",
      socket: { destroy: vi.fn() }
    };
    const res = {
      headersSent: true,
      writableEnded: false
    };
    const error = { code: "ERR_TEST", name: "TestError", message: "test message", stack: "stack" };

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("should handle ValidationError and return status 400 with schema error message", () => {
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };
    const error = {
      name: "ValidationError",
      details: { body: [{ context: { label: "email" }, type: "string.base", message: "must be string" }] }
    };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({ email: "email deve ser uma string" });
  });

  it("should handle URIError and return status 400", () => {
    const logSpy = vi.spyOn(logger, "log").mockImplementation(() => {});
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };
    const error = { name: "URIError", message: "malformed URI" };

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed and return status 400", () => {
    const logSpy = vi.spyOn(logger, "log").mockImplementation(() => {});
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };
    const error = { type: "entity.parse.failed" };

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large and return status 413", () => {
    const logSpy = vi.spyOn(logger, "log").mockImplementation(() => {});
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };
    const error = { type: "entity.too.large" };

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(413);
    expect(res.json).toHaveBeenCalledWith({ message: "Payload too large" });
  });

  it("should handle charset.unsupported and return status 415", () => {
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };
    const error = { type: "charset.unsupported", charset: "utf-16" };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported and return status 415", () => {
    const logSpy = vi.spyOn(logger, "log").mockImplementation(() => {});
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };
    const error = { type: "encoding.unsupported", encoding: "deflate" };

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(415);
    expect(res.json).toHaveBeenCalledWith({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle client error with Z_ prefix and return status 400", () => {
    const logSpy = vi.spyOn(logger, "log").mockImplementation(() => {});
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };
    const error = { expose: true, statusCode: 400, code: "Z_DATA_ERROR", message: "data error" };

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT code and return status 408", () => {
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };
    const error = { code: "ETIMEDOUT" };

    errorHandler(error, req, res, () => {});

    expect(res.status).toHaveBeenCalledWith(408);
    expect(res.json).toHaveBeenCalledWith({ message: "Timeout da requisição atingido" });
  });

  it("should handle general client error and return status with error code", () => {
    const logSpy = vi.spyOn(logger, "log").mockImplementation(() => {});
    const req = {};
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };
    const error = { expose: true, statusCode: 422, type: "some.type" };

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(422);
    expect(res.json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.type"
    });
  });

  it("should handle internal error with error type and return status 500", () => {
    const logSpy = vi.spyOn(logger, "log").mockImplementation(() => {});
    const req = { body: {} };
    const res = {
      status: vi.fn().mockReturnThis(),
      json: vi.fn()
    };
    const error = { type: "database.error", stack: "some-stack" };

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(500);
    expect(res.json).toHaveBeenCalledWith(
      expect.objectContaining({
        message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
        error: "database.error"
      })
    );
  });
});
```
