```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import carrinhosService from "../../src/services/carrinhos-service";
import constants from "../../src/utils/constants";

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.restoreAllMocks();
  });

  describe("get", () => {
    it("should return all carts with status 200", async () => {
      const mockCarrinhos = [{ _id: "1", precoTotal: 10 }];
      vi.spyOn(carrinhosService, "getAll").mockResolvedValueOnce(mockCarrinhos as any);

      const req = { query: undefined };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.get(req as any, res as any);

      expect(carrinhosService.getAll).toHaveBeenCalledWith(undefined);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ quantidade: 1, carrinhos: mockCarrinhos });
    });
  });

  describe("getOne", () => {
    it("should return cart when found with status 200", async () => {
      const mockCarrinho = { _id: "1", precoTotal: 10 };
      vi.spyOn(carrinhosService, "getOne").mockResolvedValueOnce(mockCarrinho as any);

      const req = { params: { id: "1" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        json: vi.fn()
      };

      await carrinhosController.getOne(req as any, res as any);

      expect(carrinhosService.getOne).toHaveBeenCalledWith("1");
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });

    it("should return CART_NOT_FOUND when cart does not exist", async () => {
      vi.spyOn(carrinhosService, "getOne").mockResolvedValueOnce(null);

      const req = { params: { id: "999" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        json: vi.fn()
      };

      await carrinhosController.getOne(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constants.CART_NOT_FOUND });
    });
  });

  describe("post", () => {
    it("should return LIMIT_JUST_ONE_CART when user already has a cart", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValueOnce({
        idUsuario: "user1",
        possuiCarrinho: { _id: "cart1" }
      } as any);

      const req = { headers: { authorization: "Bearer token" }, body: { produtos: [] } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn()
          }
        }
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constants.LIMIT_JUST_ONE_CART });
    });

    it("should return CART_WITH_DUPLICATE_PRODUCT when products have duplicates", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValueOnce({
        idUsuario: "user1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValueOnce(["prod1"]);

      const req = {
        headers: { authorization: "Bearer token" },
        body: {
          produtos: [
            { idProduto: "prod1", quantidade: 1 },
            { idProduto: "prod1", quantidade: 2 }
          ]
        }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn()
          }
        }
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constants.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["prod1"]
      });
    });

    it("should return product service error when fetching prices fails", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValueOnce({
        idUsuario: "user1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValueOnce([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValueOnce({
        error: {
          statusCode: 400,
          message: constants.PRODUCT_NOT_FOUND,
          index: 0,
          item: { idProduto: "prod1" }
        }
      } as any);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod1", quantidade: 1 }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn()
          }
        }
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constants.PRODUCT_NOT_FOUND,
        item: { idProduto: "prod1" }
      });
    });

    it("should return CART_TOTALS_TOO_BIG when totals are unsafe", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValueOnce({
        idUsuario: "user1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValueOnce([]);
      const produtosComPreco = [{ idProduto: "prod1", precoUnitario: 10, quantidade: 1 }];
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValueOnce({
        produtosComPrecoUnitario: produtosComPreco
      } as any);
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValueOnce(10);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValueOnce(1);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValueOnce(false);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod1", quantidade: 1 }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn(),
            abrir: vi.fn()
          }
        }
      };

      await carrinhosController.post(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constants.CART_TOTALS_TOO_BIG });
    });

    it("should create cart successfully with status 201", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValueOnce({
        idUsuario: "user1",
        possuiCarrinho: null
      } as any);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValueOnce([]);
      const produtosComPreco = [{ idProduto: "prod1", precoUnitario: 10, quantidade: 2 }];
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValueOnce({
        produtosComPrecoUnitario: produtosComPreco
      } as any);
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValueOnce(20);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValueOnce(2);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValueOnce(true);
      vi.spyOn(carrinhosService, "baixarEstoque").mockResolvedValueOnce(undefined);
      vi.spyOn(carrinhosService, "criarCarrinho").mockResolvedValueOnce({ _id: "cart123" } as any);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod1", quantidade: "2" }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn(),
            abrir: vi.fn()
          }
        }
      };

      await carrinhosController.post(req as any, res as any);

      expect(carrinhosService.baixarEstoque).toHaveBeenCalledWith(produtosComPreco);
      expect(carrinhosService.criarCarrinho).toHaveBeenCalledWith({
        produtos: produtosComPreco,
        precoTotal: 20,
        quantidadeTotal: 2,
        idUsuario: "user1"
      });
      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({ message: constants.POST_SUCCESS, _id: "cart123" });
    });
  });

  describe("cancelarCompra", () => {
    it("should remove cart and replenish stock when user has a cart", async () => {
      const carrinho = { _id: "cart1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValueOnce([carrinho] as any);
      vi.spyOn(carrinhosService, "removeCarrinho").mockResolvedValueOnce(undefined);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req as any, res as any);

      expect(carrinhosService.removeCarrinho).toHaveBeenCalledWith(carrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constants.DELETE_SUCCESS}. ${constants.REPLENISHED_STOCK}`
      });
    });

    it("should return NO_CART when user does not have a cart", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValueOnce([] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constants.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase and delete cart when user has a cart", async () => {
      const carrinho = { _id: "cart1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValueOnce([carrinho] as any);
      vi.spyOn(carrinhosService, "concluiCompra").mockResolvedValueOnce(undefined);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req as any, res as any);

      expect(carrinhosService.concluiCompra).toHaveBeenCalledWith(carrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constants.DELETE_SUCCESS });
    });

    it("should return NO_CART when user does not have a cart", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValueOnce([] as any);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req as any, res as any);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constants.NO_CART });
    });
  });
});
```
