```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import carrinhosService from "../../src/services/carrinhos-service";
import constants from "../../src/utils/constants";

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.restoreAllMocks();
  });

  describe("get", () => {
    it("should return all carrinhos with status 200", async () => {
      const mockCarrinhos = [
        {
          _id: "qbMqntef4iTOwWfg",
          alteradoEm: undefined,
          idUsuario: "0uxuPY0cbmQhpEz1",
          precoTotal: 6180,
          produtos: [
            {
              idProduto: "BeeJh5lz3k6kSIzA",
              precoUnitario: 470,
              quantidade: 2
            },
            {
              idProduto: "K6leHdftCeOJj8BJ",
              precoUnitario: 5240,
              quantidade: 1
            }
          ],
          quantidadeTotal: 3
        }
      ];
      vi.spyOn(carrinhosService, "getAll").mockResolvedValueOnce(mockCarrinhos);

      const req = { query: {} };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.get(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: 1,
        carrinhos: mockCarrinhos
      });
    });
  });

  describe("getOne", () => {
    it("should return CART_NOT_FOUND when carrinho does not exist", async () => {
      vi.spyOn(carrinhosService, "getOne").mockResolvedValueOnce(null);

      const req = { params: { id: "invalid-id" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        json: vi.fn()
      };

      await carrinhosController.getOne(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constants.CART_NOT_FOUND });
    });

    it("should return the carrinho when found", async () => {
      const mockCarrinho = { _id: "1", precoTotal: 10, quantidadeTotal: 2, idUsuario: "u1", produtos: [] };
      vi.spyOn(carrinhosService, "getOne").mockResolvedValueOnce(mockCarrinho);

      const req = { params: { id: "1" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.getOne(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });
  });

  describe("post", () => {
    it("should return LIMIT_JUST_ONE_CART when user already has a cart", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValueOnce({
        idUsuario: "u1",
        possuiCarrinho: { _id: "c1" }
      });

      const resLocals = {
        etapas: {
          regras: vi.fn(),
          regra: vi.fn()
        }
      };
      const req = { headers: { authorization: "Bearer token" }, body: { produtos: [] } };
      const res = {
        locals: resLocals,
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constants.LIMIT_JUST_ONE_CART });
    });

    it("should return CART_WITH_DUPLICATE_PRODUCT when duplicate products exist", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValueOnce({
        idUsuario: "u1",
        possuiCarrinho: null
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValueOnce(["prod1"]);

      const resLocals = {
        etapas: {
          regras: vi.fn(),
          regra: vi.fn()
        }
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod1", quantidade: 1 }] }
      };
      const res = {
        locals: resLocals,
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constants.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["prod1"]
      });
    });

    it("should return error when getProdutosComPrecoUnitarioOuErro fails", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValueOnce({
        idUsuario: "u1",
        possuiCarrinho: null
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValueOnce([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValueOnce({
        error: { message: constants.PRODUCT_NOT_FOUND, statusCode: 400, index: 0, item: { idProduto: "p1" } }
      });

      const resLocals = {
        etapas: {
          regras: vi.fn(),
          regra: vi.fn()
        }
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }] }
      };
      const res = {
        locals: resLocals,
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constants.PRODUCT_NOT_FOUND,
        item: { idProduto: "p1", index: 0 }
      });
    });

    it("should return CART_TOTALS_TOO_BIG when totals are not safe", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValueOnce({
        idUsuario: "u1",
        possuiCarrinho: null
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValueOnce([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValueOnce({
        produtosComPrecoUnitario: [{ idProduto: "p1", precoUnitario: 10, quantidade: 1 }]
      });
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(10);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(1);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(false);

      const resLocals = {
        etapas: {
          regras: vi.fn(),
          regra: vi.fn()
        }
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }] }
      };
      const res = {
        locals: resLocals,
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constants.CART_TOTALS_TOO_BIG });
    });

    it("should create cart successfully when all validations pass", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValueOnce({
        idUsuario: "u1",
        possuiCarrinho: null
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValueOnce([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValueOnce({
        produtosComPrecoUnitario: [{ idProduto: "p1", precoUnitario: 10, quantidade: 2 }]
      });
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(20);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(2);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(true);
      vi.spyOn(carrinhosService, "baixarEstoque").mockResolvedValueOnce(undefined);
      vi.spyOn(carrinhosService, "criarCarrinho").mockResolvedValueOnce({ _id: "c123" });

      const resLocals = {
        etapas: {
          regras: vi.fn(),
          regra: vi.fn(),
          abrir: vi.fn()
        }
      };
      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: "2" }] }
      };
      const res = {
        locals: resLocals,
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constants.POST_SUCCESS,
        _id: "c123"
      });
    });
  });

  describe("cancelarCompra", () => {
    it("should return NO_CART when user has no cart", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValueOnce([]);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constants.NO_CART });
    });

    it("should remove cart and replenish stock when user has a cart", async () => {
      const mockCarrinho = { _id: "c1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValueOnce([mockCarrinho]);
      vi.spyOn(carrinhosService, "removeCarrinho").mockResolvedValueOnce(undefined);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req, res);

      expect(carrinhosService.removeCarrinho).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constants.DELETE_SUCCESS}. ${constants.REPLENISHED_STOCK}`
      });
    });
  });

  describe("concluirCompra", () => {
    it("should return NO_CART when user has no cart", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValueOnce([]);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constants.NO_CART });
    });

    it("should conclude purchase and delete cart when user has a cart", async () => {
      const mockCarrinho = { _id: "c1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValueOnce([mockCarrinho]);
      vi.spyOn(carrinhosService, "concluiCompra").mockResolvedValueOnce(undefined);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req, res);

      expect(carrinhosService.concluiCompra).toHaveBeenCalledWith(mockCarrinho);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constants.DELETE_SUCCESS });
    });
  });
});
```
