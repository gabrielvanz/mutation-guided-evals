```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import carrinhosService from "../../src/services/carrinhos-service";
import constant from "../../src/utils/constants";

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.restoreAllMocks();
  });

  describe("get", () => {
    it("should return status 200 and list of carts", async () => {
      const mockCarrinhos = await carrinhosService.getAll({});

      const req = { query: {} };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.get(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        quantidade: mockCarrinhos.length,
        carrinhos: mockCarrinhos
      });
    });
  });

  describe("getOne", () => {
    it("should return status 400 when cart is not found", async () => {
      vi.spyOn(carrinhosService, "getOne").mockResolvedValue(null);

      const req = { params: { id: "invalidId" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.getOne(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });

    it("should return status 200 and cart data when cart exists", async () => {
      const mockCarrinho = { _id: "cart1", precoTotal: 50, quantidadeTotal: 5, idUsuario: "user1", produtos: [] };
      vi.spyOn(carrinhosService, "getOne").mockResolvedValue(mockCarrinho);

      const req = { params: { id: "cart1" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.getOne(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });
  });

  describe("post", () => {
    it("should return status 400 when user already has a cart", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user1",
        possuiCarrinho: { _id: "existingCart" }
      });

      const req = { headers: { authorization: "Bearer token" }, body: { produtos: [] } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn(),
            abrir: vi.fn()
          }
        }
      };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return status 400 when product list contains duplicates", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user1",
        possuiCarrinho: null
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue(["prod1"]);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod1", quantidade: 1 }, { idProduto: "prod1", quantidade: 2 }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn(),
            abrir: vi.fn()
          }
        }
      };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["prod1"]
      });
    });

    it("should return error status when getting product unit price fails", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user1",
        possuiCarrinho: null
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        error: {
          statusCode: 400,
          message: constant.PRODUCT_NOT_FOUND,
          index: 0,
          item: { idProduto: "missing" }
        }
      });

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "missing", quantidade: 1 }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn(),
            abrir: vi.fn()
          }
        }
      };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.PRODUCT_NOT_FOUND,
        item: { idProduto: "missing" }
      });
    });

    it("should return status 400 when cart totals are not safe", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user1",
        possuiCarrinho: null
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "prod1", precoUnitario: 10, quantidade: 1 }]
      });
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(10);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(1);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(false);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod1", quantidade: "1" }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn(),
            abrir: vi.fn()
          }
        }
      };

      await carrinhosController.post(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });

    it("should successfully create a cart when all rules pass", async () => {
      vi.spyOn(carrinhosService, "usuarioJaPossuiCarrinho").mockResolvedValue({
        idUsuario: "user1",
        possuiCarrinho: null
      });
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValue([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValue({
        produtosComPrecoUnitario: [{ idProduto: "prod1", precoUnitario: 10, quantidade: 2 }]
      });
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValue(20);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValue(2);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValue(true);
      vi.spyOn(carrinhosService, "baixarEstoque").mockResolvedValue(undefined);
      vi.spyOn(carrinhosService, "criarCarrinho").mockResolvedValue({ _id: "newCartId" });

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod1", quantidade: "2" }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        locals: {
          etapas: {
            regras: vi.fn(),
            regra: vi.fn(),
            abrir: vi.fn()
          }
        }
      };

      await carrinhosController.post(req, res);

      expect(carrinhosService.baixarEstoque).toHaveBeenCalled();
      expect(carrinhosService.criarCarrinho).toHaveBeenCalled();
      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.POST_SUCCESS,
        _id: "newCartId"
      });
    });
  });

  describe("cancelarCompra", () => {
    it("should remove cart and replenish stock when user cart exists", async () => {
      const mockCart = { _id: "cart1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([mockCart]);
      vi.spyOn(carrinhosService, "removeCarrinho").mockResolvedValue(undefined);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req, res);

      expect(carrinhosService.removeCarrinho).toHaveBeenCalledWith(mockCart);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}`
      });
    });

    it("should return NO_CART message when user cart does not exist", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([]);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });

  describe("concluirCompra", () => {
    it("should conclude purchase and delete cart when user cart exists", async () => {
      const mockCart = { _id: "cart1", produtos: [] };
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([mockCart]);
      vi.spyOn(carrinhosService, "concluiCompra").mockResolvedValue(undefined);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req, res);

      expect(carrinhosService.concluiCompra).toHaveBeenCalledWith(mockCart);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });

    it("should return NO_CART message when user cart does not exist on conclude", async () => {
      vi.spyOn(carrinhosService, "getCarrinhoDoUsuario").mockResolvedValue([]);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });
  });
});
```
