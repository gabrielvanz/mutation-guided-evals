```ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import carrinhosController from "../../src/controllers/carrinhos-controller";
import carrinhosService from "../../src/services/carrinhos-service";
import authService from "../../src/services/auth-service";
import usuariosService from "../../src/services/usuarios-service";
import constant from "../../src/utils/constants";

describe("carrinhos-controller", () => {
  beforeEach(() => {
    vi.restoreAllMocks();
  });

  describe("get", () => {
    it("should return status 200 and list of carts", async () => {
      const mockCarrinhos = [
        {
          _id: "qbMqntef4iTOwWfg",
          alteradoEm: undefined,
          idUsuario: "0uxuPY0cbmQhpEz1",
          precoTotal: 6180,
          produtos: [
            {
              idProduto: "BeeJh5lz3k6kSIzA",
              precoUnitario: 470,
              quantidade: 2
            },
            {
              idProduto: "K6leHdftCeOJj8BJ",
              precoUnitario: 5240,
              quantidade: 1
            }
          ],
          quantidadeTotal: 3
        }
      ];
      vi.spyOn(carrinhosService, "getAll").mockResolvedValueOnce(mockCarrinhos);

      const req = { query: {} };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.get(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ quantidade: 1, carrinhos: mockCarrinhos });
    });
  });

  describe("getOne", () => {
    it("should return status 400 when cart is not found", async () => {
      vi.spyOn(carrinhosService, "getOne").mockResolvedValueOnce(null);

      const req = { params: { id: "invalid-id" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        json: vi.fn()
      };

      await carrinhosController.getOne(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({ message: constant.CART_NOT_FOUND });
    });

    it("should return status 200 and cart when found", async () => {
      const mockCarrinho = { _id: "1", precoTotal: 10, quantidadeTotal: 1 };
      vi.spyOn(carrinhosService, "getOne").mockResolvedValueOnce(mockCarrinho);

      const req = { params: { id: "1" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn(),
        json: vi.fn()
      };

      await carrinhosController.getOne(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith(mockCarrinho);
    });
  });

  describe("post", () => {
    it("should return status 400 when user already has a cart", async () => {
      vi.spyOn(authService, "verifyToken").mockReturnValueOnce({ email: "a@a.com", password: "123" });
      vi.spyOn(usuariosService, "getDadosDoUsuario").mockResolvedValueOnce({ _id: "user-1" });
      vi.spyOn(carrinhosService, "existeCarrinho").mockResolvedValueOnce({ _id: "cart-1" });

      const req = { headers: { authorization: "Bearer token" }, body: { produtos: [] } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };

      await carrinhosController.post(req, { ...res, locals: { etapas } });

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.LIMIT_JUST_ONE_CART });
    });

    it("should return status 400 when products have duplicates", async () => {
      vi.spyOn(authService, "verifyToken").mockReturnValueOnce({ email: "a@a.com", password: "123" });
      vi.spyOn(usuariosService, "getDadosDoUsuario").mockResolvedValueOnce({ _id: "user-1" });
      vi.spyOn(carrinhosService, "existeCarrinho").mockResolvedValueOnce(null);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValueOnce(["prod-1"]);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "prod-1", quantidade: 1 }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };

      await carrinhosController.post(req, { ...res, locals: { etapas } });

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.CART_WITH_DUPLICATE_PRODUCT,
        idProdutosDuplicados: ["prod-1"]
      });
    });

    it("should return status from error when product retrieval fails", async () => {
      vi.spyOn(authService, "verifyToken").mockReturnValueOnce({ email: "a@a.com", password: "123" });
      vi.spyOn(usuariosService, "getDadosDoUsuario").mockResolvedValueOnce({ _id: "user-1" });
      vi.spyOn(carrinhosService, "existeCarrinho").mockResolvedValueOnce(null);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValueOnce([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValueOnce({
        error: { statusCode: 400, message: constant.PRODUCT_NOT_FOUND, index: 0, item: { idProduto: "p1" } }
      });

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn()
      };

      await carrinhosController.post(req, { ...res, locals: { etapas } });

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.PRODUCT_NOT_FOUND,
        item: { idProduto: "p1" }
      });
    });

    it("should return status 400 when totals are not safe", async () => {
      vi.spyOn(authService, "verifyToken").mockReturnValueOnce({ email: "a@a.com", password: "123" });
      vi.spyOn(usuariosService, "getDadosDoUsuario").mockResolvedValueOnce({ _id: "user-1" });
      vi.spyOn(carrinhosService, "existeCarrinho").mockResolvedValueOnce(null);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValueOnce([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValueOnce({
        produtosComPrecoUnitario: [{ idProduto: "p1", precoUnitario: 10, quantidade: 1 }]
      });
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValueOnce(10);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValueOnce(1);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValueOnce(false);

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: 1 }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };

      await carrinhosController.post(req, { ...res, locals: { etapas } });

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.send).toHaveBeenCalledWith({ message: constant.CART_TOTALS_TOO_BIG });
    });

    it("should create cart successfully when all validations pass", async () => {
      vi.spyOn(authService, "verifyToken").mockReturnValueOnce({ email: "a@a.com", password: "123" });
      vi.spyOn(usuariosService, "getDadosDoUsuario").mockResolvedValueOnce({ _id: "user-1" });
      vi.spyOn(carrinhosService, "existeCarrinho").mockResolvedValueOnce(null);
      vi.spyOn(carrinhosService, "extrairProdutosDuplicados").mockReturnValueOnce([]);
      vi.spyOn(carrinhosService, "getProdutosComPrecoUnitarioOuErro").mockResolvedValueOnce({
        produtosComPrecoUnitario: [{ idProduto: "p1", precoUnitario: 10, quantidade: 2 }]
      });
      vi.spyOn(carrinhosService, "precoTotal").mockReturnValueOnce(20);
      vi.spyOn(carrinhosService, "quantidadeTotal").mockReturnValueOnce(2);
      vi.spyOn(carrinhosService, "totaisSaoSeguros").mockReturnValueOnce(true);
      vi.spyOn(carrinhosService, "baixarEstoque").mockResolvedValueOnce(undefined);
      vi.spyOn(carrinhosService, "criarCarrinho").mockResolvedValueOnce({ _id: "new-cart-id" });

      const req = {
        headers: { authorization: "Bearer token" },
        body: { produtos: [{ idProduto: "p1", quantidade: "2" }] }
      };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };
      const etapas = {
        regras: vi.fn(),
        regra: vi.fn(),
        abrir: vi.fn()
      };

      await carrinhosController.post(req, { ...res, locals: { etapas } });

      expect(res.status).toHaveBeenCalledWith(201);
      expect(res.send).toHaveBeenCalledWith({
        message: constant.POST_SUCCESS,
        _id: "new-cart-id"
      });
    });
  });

  describe("cancelarCompra", () => {
    it("should return NO_CART message when user has no cart", async () => {
      vi.spyOn(authService, "verifyToken").mockReturnValueOnce({ email: "a@a.com", password: "123" });
      vi.spyOn(usuariosService, "getDadosDoUsuario").mockResolvedValueOnce({ _id: "user-1" });
      vi.spyOn(carrinhosService, "getAll").mockResolvedValueOnce([]);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });

    it("should remove cart and replenish stock when user has a cart", async () => {
      const mockCart = { _id: "cart-1", produtos: [] };
      vi.spyOn(authService, "verifyToken").mockReturnValueOnce({ email: "a@a.com", password: "123" });
      vi.spyOn(usuariosService, "getDadosDoUsuario").mockResolvedValueOnce({ _id: "user-1" });
      vi.spyOn(carrinhosService, "getAll").mockResolvedValueOnce([mockCart]);
      vi.spyOn(carrinhosService, "removeCarrinho").mockResolvedValueOnce(undefined);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.cancelarCompra(req, res);

      expect(carrinhosService.removeCarrinho).toHaveBeenCalledWith(mockCart);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({
        message: `${constant.DELETE_SUCCESS}. ${constant.REPLENISHED_STOCK}`
      });
    });
  });

  describe("concluirCompra", () => {
    it("should return NO_CART message when user has no cart", async () => {
      vi.spyOn(authService, "verifyToken").mockReturnValueOnce({ email: "a@a.com", password: "123" });
      vi.spyOn(usuariosService, "getDadosDoUsuario").mockResolvedValueOnce({ _id: "user-1" });
      vi.spyOn(carrinhosService, "getAll").mockResolvedValueOnce([]);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.NO_CART });
    });

    it("should conclude purchase successfully when user has a cart", async () => {
      const mockCart = { _id: "cart-1", produtos: [] };
      vi.spyOn(authService, "verifyToken").mockReturnValueOnce({ email: "a@a.com", password: "123" });
      vi.spyOn(usuariosService, "getDadosDoUsuario").mockResolvedValueOnce({ _id: "user-1" });
      vi.spyOn(carrinhosService, "getAll").mockResolvedValueOnce([mockCart]);
      vi.spyOn(carrinhosService, "concluiCompra").mockResolvedValueOnce(undefined);

      const req = { headers: { authorization: "Bearer token" } };
      const res = {
        status: vi.fn().mockReturnThis(),
        send: vi.fn()
      };

      await carrinhosController.concluirCompra(req, res);

      expect(carrinhosService.concluiCompra).toHaveBeenCalledWith(mockCart);
      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.send).toHaveBeenCalledWith({ message: constant.DELETE_SUCCESS });
    });
  });
});
```
