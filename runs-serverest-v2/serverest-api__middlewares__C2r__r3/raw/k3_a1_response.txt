```ts
import { describe, it, expect } from "vitest";
import request from "supertest";
import app from "../../src/app";

describe("Middlewares API", () => {
  describe("Authentication Middleware", () => {
    it("should return 401 when token is missing on protected routes", async () => {
      const response = await request(app).post("/carrinhos").send({
        produtos: []
      });

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido, expirado ou usuário não encontrado"
      });
    });

    it("should return 401 when token format is bad", async () => {
      const response = await request(app)
        .post("/carrinhos")
        .set("authorization", "InvalidFormat")
        .send({ produtos: [] });

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido, expirado ou usuário não encontrado"
      });
    });

    it("should return 401 when token is invalid or user does not exist", async () => {
      const response = await request(app)
        .post("/carrinhos")
        .set("authorization", "Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJlbWFpbCI6Im5vbmVzdGlzdEBlbWFpbC5jb20iLCJwYXNzd29yZCI6InBhc3N3b3JkIiwiaWF0IjoxNTE2MjM5MDIyfQ.invalidSignatureMock")
        .send({ produtos: [] });

      expect(response.status).toBe(401);
      expect(response.body).toEqual({
        message: "Token de acesso ausente, inválido, expirado ou usuário não encontrado"
      });
    });

    it("should return 403 when user is not an administrator on admin-only routes", async () => {
      const email = "user_common_test_auth@email.com";
      const password = "teste";

      await request(app).post("/usuarios").send({
        nome: "Usuario Comum",
        email,
        password,
        administrador: "false"
      });

      const loginRes = await request(app).post("/login").send({
        email,
        password
      });

      const token = loginRes.body.authorization;

      const response = await request(app)
        .post("/produtos")
        .set("authorization", token)
        .send({
          nome: "Produto Teste",
          preco: 100,
          descricao: "Desc",
          quantidade: 10
        });

      expect(response.status).toBe(403);
      expect(response.body).toEqual({
        message: "Rota exclusiva para administradores"
      });
    });
  });

  describe("Error Handler Middleware", () => {
    it("should return 400 when sending invalid JSON syntax", async () => {
      const response = await request(app)
        .post("/usuarios")
        .set("content-type", "application/json")
        .send("{\"nome\": 'invalid quote'}");

      expect(response.status).toBe(400);
      expect(response.body.message).toContain("Adicione aspas em todos os valores");
    });
  });
});
```
