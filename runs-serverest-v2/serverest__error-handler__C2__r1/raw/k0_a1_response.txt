```ts
import { describe, it, expect, vi } from "vitest";
import errorHandler from "../../src/middlewares/error-handler";
import logger from "../../src/utils/logger";

describe("errorHandler", () => {
  it("should handle headersSent when request timed out", () => {
    const error = new Error("Timeout");
    error.stack = "stack";
    const req = { timedout: true, method: "GET", originalUrl: "/test", socket: { destroy: vi.fn() } };
    const res = { headersSent: true, writableEnded: false };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      level: "warn",
      message: "Erro depois da resposta enviada: GET /test, timedout: true, undefined: Timeout, Stack: stack"
    });
    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("should handle headersSent when request did not timeout and response is not writableEnded", () => {
    const error = new Error("Some error");
    error.stack = "stack";
    const req = { timedout: false, method: "POST", originalUrl: "/test", socket: { destroy: vi.fn() } };
    const res = { headersSent: true, writableEnded: false };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalledWith({
      level: "error",
      message: "Erro depois da resposta enviada: POST /test, timedout: false, undefined: Some error, Stack: stack"
    });
    expect(req.socket.destroy).toHaveBeenCalled();
  });

  it("should handle headersSent when writableEnded is true", () => {
    const error = new Error("Some error");
    error.stack = "stack";
    const req = { timedout: false, method: "POST", originalUrl: "/test", socket: { destroy: vi.fn() } };
    const res = { headersSent: true, writableEnded: true };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(req.socket.destroy).not.toHaveBeenCalled();
  });

  it("should handle ValidationError", () => {
    const error = {
      name: "ValidationError",
      details: {
        body: [
          {
            context: { label: "nome" },
            type: "any.required",
            message: "\"nome\" is required"
          }
        ]
      }
    };
    const req = {};
    const json = vi.fn();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    errorHandler(error, req, res, () => {});

    expect(status).toHaveBeenCalledWith(400);
    expect(json).toHaveBeenCalledWith({ nome: "nome é obrigatório" });
  });

  it("should handle URIError", () => {
    const error = new URIError("Malformed URI");
    const req = {};
    const json = vi.fn();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(status).toHaveBeenCalledWith(400);
    expect(json).toHaveBeenCalledWith({
      message: "Parâmetro de URL inválido. Verifique os caracteres especiais e tente novamente."
    });
  });

  it("should handle entity.parse.failed", () => {
    const error = { type: "entity.parse.failed" };
    const req = {};
    const json = vi.fn();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(status).toHaveBeenCalledWith(400);
    expect(json).toHaveBeenCalledWith({
      message: "Adicione aspas em todos os valores. Para mais informações acesse a issue https://github.com/ServeRest/ServeRest/issues/225"
    });
  });

  it("should handle entity.too.large", () => {
    const error = { type: "entity.too.large" };
    const req = {};
    const json = vi.fn();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(status).toHaveBeenCalledWith(413);
    expect(json).toHaveBeenCalledWith({ message: "Payload too large" });
  });

  it("should handle parameters.too.many", () => {
    const error = { type: "parameters.too.many" };
    const req = {};
    const json = vi.fn();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(status).toHaveBeenCalledWith(413);
    expect(json).toHaveBeenCalledWith({ message: "Payload too large" });
  });

  it("should handle charset.unsupported", () => {
    const error = { type: "charset.unsupported", charset: "utf-16" };
    const req = {};
    const json = vi.fn();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    errorHandler(error, req, res, () => {});

    expect(status).toHaveBeenCalledWith(415);
    expect(json).toHaveBeenCalledWith({ message: "Unsupported charset", charset: "utf-16" });
  });

  it("should handle encoding.unsupported", () => {
    const error = { type: "encoding.unsupported", encoding: "deflate" };
    const req = {};
    const json = vi.fn();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(status).toHaveBeenCalledWith(415);
    expect(json).toHaveBeenCalledWith({ message: "Unsupported content encoding", encoding: "deflate" });
  });

  it("should handle client error with Z_ prefix in code", () => {
    const error = {
      expose: true,
      statusCode: 400,
      code: "Z_DATA_ERROR",
      message: "invalid block type"
    };
    const req = {};
    const json = vi.fn();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(status).toHaveBeenCalledWith(400);
    expect(json).toHaveBeenCalledWith({
      message: "Corpo comprimido inválido. Envie o corpo de acordo com o header Content-Encoding ou remova o header."
    });
  });

  it("should handle ETIMEDOUT code", () => {
    const error = { code: "ETIMEDOUT" };
    const req = {};
    const json = vi.fn();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };

    errorHandler(error, req, res, () => {});

    expect(status).toHaveBeenCalledWith(408);
    expect(json).toHaveBeenCalledWith({ message: "Timeout da requisição atingido" });
  });

  it("should handle general client error", () => {
    const error = {
      expose: true,
      statusCode: 400,
      type: "some.type",
      message: "some message"
    };
    const req = {};
    const json = vi.fn();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(status).toHaveBeenCalledWith(400);
    expect(json).toHaveBeenCalledWith({
      message: "Não foi possível ler a requisição. Confira o corpo e os headers enviados.",
      error: "some.type"
    });
  });

  it("should handle server error with error type", () => {
    const error = { type: "server.error", stack: "err-stack" };
    const req = { body: { foo: "bar" } };
    const json = vi.fn();
    const status = vi.fn().mockReturnValue({ json });
    const res = { headersSent: false, status };
    const logSpy = vi.spyOn(logger, "log");

    errorHandler(error, req, res, () => {});

    expect(logSpy).toHaveBeenCalled();
    expect(status).toHaveBeenCalledWith(500);
    expect(json).toHaveBeenCalledWith(
      expect.objectContaining({
        message: "Abra uma issue informando essa resposta. https://github.com/ServeRest/ServeRest/issues",
        error: "server.error"
      })
    );
  });
});
```
